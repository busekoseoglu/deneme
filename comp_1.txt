import pandas as pd
XLS = "vardiya_plani_kitle_YYYYMMDD_HHMM.xlsx"          # <-- mip_ihtiyac run'ının Excel'i

g = pd.read_excel(XLS, sheet_name="COVERAGE_GUNLUK")
o = pd.read_excel(XLS, sheet_name="AGENT_OZET")

N = len(o)                                   # modeldeki kadro
ort_gun = o["toplam_calisma"].mean()         # kişi başı aylık çalışma günü
F = g["fazla_toplam"].sum()                  # ihtiyacın üstü kişi-gün
E = g["eksik_toplam"].sum()                  # ihtiyacın altı kişi-gün (0'a yakın olmalı)

# 1) ortalama yöntemi
fazla_1 = (F - E) / ort_gun
# 2) en yoğun gün yöntemi
g["oran"] = g["toplam_talep"] / g["toplam_atanan"]
k = g["oran"].max()
gerekli_2 = N * k

print(f"kadro N                         : {N}")
print(f"kişi başı çalışma günü          : {ort_gun:.1f}")
print(f"ihtiyaç toplamı (kişi-gün)      : {g['toplam_talep'].sum()}")
print(f"atanan toplamı (kişi-gün)       : {g['toplam_atanan'].sum()}")
print(f"fazla / eksik (kişi-gün)        : {F} / {E}")
print(f"1) ortalama  -> gereken ≈ {N - fazla_1:.0f}  (fazla ≈ {fazla_1:.0f} kişi)")
print(f"2) en yoğun  -> gereken ≈ {gerekli_2:.0f}  (fazla ≈ {N - gerekli_2:.0f} kişi)")
for _ad, _alt in (("hafta içi", g[g["hafta_sonu"] == 0]), ("hafta sonu", g[g["hafta_sonu"] == 1])):
    print(f"   {_ad:10s} en yüksek oran {_alt['oran'].max():.3f} -> {N * _alt['oran'].max():.0f} kişi")
print("\nkadroyu belirleyen 5 gün:")
print(g.sort_values("oran", ascending=False)[["tarih", "gun", "toplam_talep", "toplam_atanan",
                                              "eksik_toplam", "fazla_toplam", "oran"]].head(5).to_string(index=False))
