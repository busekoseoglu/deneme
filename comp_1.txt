_TR = str.maketrans("çğıöşüÇĞİÖŞÜ", "cgiosuCGIOSU")
_ARALIK_RE = re.compile(r"^(\d{1,2}):(\d{2})\s*[-–]\s*(\d{1,2}):(\d{2})$")
_OFF_KELIMELERI = {"off", "tatil", "haftatatili", "hafta_tatili"}
_IZIN_KELIMELERI = {"izin", "yillikizin", "yillik_izin", "ucretsizizin", "ucretsiz_izin",
                    "idariizin", "idari_izin", "dogumizni", "dogum_izni", "sutizni",

def _parse_vardiya(v):
    """-> ('calisma', '08:00', '17:00') | ('off', None, None) | ('izin', None, None) | (None, ...)"""
    try:
        if v is None or pd.isna(v):
            return (None, None, None)
    except (TypeError, ValueError):
        pass
    s = str(v).strip()
    if s == "" or s.lower() in ("nan", "nat", "none"):
        return (None, None, None)

    m = _ARALIK_RE.match(s)
    if m:
        # "24:00" gecerli bir saat degil (0-23). Gece yarisi -> "00:00";
        # make_shift_datetime bitis <= baslangic oldugunda gunu 1 artiriyor,
        # yani 16:00-24:00 dogru sekilde ertesi gun 00:00'a denk geliyor.
        _bh, _bm = int(m.group(1)), m.group(2)
        _eh, _em = int(m.group(3)), m.group(4)
        if _bh >= 24 or _eh >= 24:
            _stat["saat_24_duzeltildi"] = _stat.get("saat_24_duzeltildi", 0) + 1
        bas = f"{_bh % 24:02d}:{_bm}"
        bit = f"{_eh % 24:02d}:{_em}"
        return ("calisma", bas, bit)

    k = re.sub(r"[\s\-]+", "", s.translate(_TR).lower())
    if k in _OFF_KELIMELERI:
        return ("off", None, None)
    if k in _IZIN_KELIMELERI:
        return ("izin", None, None)
    return (None, None, None)

# %% [KISIT] - KURAL 3: W31 HAFTALIK OFF DENGESİ (TEMMUZ - AĞUSTOS SINIRI)
# Sadece W31'e (planın ilk gününü içeren ISO hafta) uygulanır.
# Ağustos'un TAM haftalarına uygulanmaz: agent bir hafta 3 off talep edip başka haftada
# telafi edebilir; oradaki bağlayıcı kısıt HÜCRE 37'deki aylık minimumdur.
#
#   temmuz_off = W31'in Temmuz günlerinde 'off' olan gün sayısı
#                yıllık izin ve *_izinli_flg (süt izni düzeni) günleri SAYILMAZ
#   kalan_off  = clamp(HAFTALIK_OFF_KOTASI - temmuz_off, 0, Ağustos gün sayısı)
#   fazla_off  = max(0, verilen off - kalan_off)   -> agent'ın o haftaki ÇALIŞMA BORCU
#
# ---------------------------------------------------------------------------------
# DURUM: BORÇ MEKANİZMASI HENÜZ KAPALI (iş biriminden cevap bekleniyor)
#
#   HARD tavan (sum(off) <= kalan_off) KALDIRILDI.
#   Sebep: 1-2 Ağustos'ta off_t/off_t2 talebi olan agentta gercek_off 1'e sabitlenir;
#          kalan_off=1 iken "2 <= 1" çıkıp model INFEASIBLE oluyordu.
#
#   Şu an bu hücre modele SADECE soft ceza ekliyor (hak edilen off verilsin diye).
#   fazla_off değişkeni kuruluyor ama HİÇBİR kısıtta kullanılmıyor - yalnızca
#   çözümden sonra "kim kaç gün borçlandı" diye okunabilsin diye duruyor.
#
#   İş birimi "telafi edilecek" derse tek yapılacak: aşağıdaki BORÇ bloğunu aç.
# ---------------------------------------------------------------------------------
#
# gercek_off (HÜCRE 41'de tanımlı) semantiği tam istediğimiz gibi:
#   izin günü      -> gercek_off = 0  (izin off yerine geçmez)
#   off_t / off_t2 -> gercek_off = 1  (talep edilen off, off sayılır)
#   normal gün     -> gercek_off = 1 - work
from datetime import timedelta

W31_BORC_AKTIF = False        # <-- iş biriminden "telafi edilecek" cevabı gelince True yap

if "PLAN_START_DATE" not in globals():

w31_eksik_off = {}      # agent -> IntVar : kotadan EKSİK verilen off (soft ceza)

w31_debug_rows = []

if not W31_AGUSTOS_DS:
    print("KURAL 3 atlandı: W31'in Ağustos tarafında plan günü yok.")
else:
    _n_agu = len(W31_AGUSTOS_DS)
    for a_raw in AGENTS:
        a = str(a_raw).strip()

        # Temmuz verisi olmayan agent için varsayım yapmıyoruz (DEBUG [1] listeler).
        if not (gecmis_kapsam.get(a, set()) & W31_TEMMUZ_SET):
            w31_atlanan_veri_yok += 1
            continue

        temmuz_off  = len(gecmis_off_gunleri.get(a, set()) & W31_TEMMUZ_SET)
        temmuz_izin = len(gecmis_izin_gunleri.get(a, set()) & W31_TEMMUZ_SET)
        temmuz_haftalik_izin = len(gecmis_haftalik_izin_gunleri.get(a, set()) & W31_TEMMUZ_SET)
        temmuz_cal  = len(gecmis_calisma_gunleri.get(a, set()) & W31_TEMMUZ_SET)

            continue

        _off_toplam = model.NewIntVar(0, len(off_vars), f"w31_off_toplam_{a}")

        # EKSİK off -> soft ceza (hak ettiği off'u alsın)

        w31_fazla_off[a] = _fazla

        # ---- BORÇ BLOĞU (kapalı) ----
        # İş birimi onay verince W31_BORC_AKTIF = True yap.

            _tip_dates = tip_map.get(a, {})

                                 if str(t).strip().lower() == "izin" and d in aylik_is_gunu_dates})

            model.Add(sum(_ay_vars) <= _baz + _fazla + MAX_OVERTIME_PER_MONTH)

        # off_t/off_t2 zorunlu off olduğu için borcu şimdiden kesin olanlar

            in ("off_t", "off_t2")
        )
        if _zorunlu_off > kalan_off:
            w31_kesin_borclu.append((a, temmuz_off, kalan_off, _zorunlu_off, _zorunlu_off - kalan_off))

        w31_debug_rows.append({
            "agent_user_code": a,
            "temmuz_calisma": temmuz_cal,
            "temmuz_off": temmuz_off,
            "temmuz_izin": temmuz_izin,
            "temmuz_haftalik_izin": temmuz_haftalik_izin,
            "kalan_off": kalan_off,
            "agustos_zorunlu_off": _zorunlu_off,
        })

    w31_debug_df = pd.DataFrame(w31_debug_rows)

    print(f"KURAL 3 - W31 haftalık off dengesi ({W31_GUNLER[0]} .. {W31_GUNLER[-1]})")
    print(f"  Ağustos tarafı: {W31_AGUSTOS_DS} | off kotası: {HAFTALIK_OFF_KOTASI}")
    print(f"  BORÇ MEKANİZMASI: {'AÇIK' if W31_BORC_AKTIF else 'KAPALI (iş birimi cevabı bekleniyor)'}")
    print(f"  hard tavan: YOK (kaldırıldı - off_t ile çakışıp infeasible yapıyordu)")
    print(f"  soft ceza (eksik off): {len(w31_eksik_off)} agent | W31_EKSIK_OFF_W = {W31_EKSIK_OFF_W}")
    print(f"  gözlem değişkeni (fazla off/borç): {len(w31_fazla_off)} agent")
    print(f"  Temmuz verisi olmadığı için atlanan agent: {w31_atlanan_veri_yok}")

    if len(w31_debug_df):
        print("\n  kalan_off dağılımı (1-2 Ağustos'ta hak ettiği off):")

# Ağustos içi   : pairwise yasak -> x[sh1] + x[sh2] <= 1
# Temmuz köprüsü: geçmiş vardiya SABİT=1 olduğu için doğrudan x == 0 (hard)
#
# Temmuz bitiş saati artık VERİDEN geliyor ("08:00-17:00" -> bitiş 17:00).
# 8/9 saat türetme kuralı KALDIRILDI.
# Temmuz'da off/izin olan gün vardiya üretmez -> 11h hesabına hiç girmez.

def make_shift_datetime(ds, baslangic, bitis):
    start_dt = pd.to_datetime(f"{ds} {baslangic}")
    end_dt = pd.to_datetime(f"{ds} {bitis}")
    if end_dt <= start_dt:
        end_dt = end_dt + pd.Timedelta(days=1)
    return start_dt, end_dt

shift_dt_map = {}
for ds in PLAN_GUNLER:
    for v in gun_vardiyalari.get(ds, []):
        baslangic, bitis = saat[(ds, v)]
        start_dt, end_dt = make_shift_datetime(ds, baslangic, bitis)
        shift_dt_map[(ds, v)] = {"start_dt": start_dt, "end_dt": end_dt}

# --- Ağustos içi (mevcut mantık, değişmedi) ---
min_rest_constraints = 0
for a in AGENTS:
    agent_shift_options = []
    for ds in PLAN_GUNLER:
        for v in gun_vardiyalari.get(ds, []):
            if (a, ds, v) not in x:
                continue
            agent_shift_options.append({
                "ds": ds, "v": v,
                "start_dt": shift_dt_map[(ds, v)]["start_dt"],
                "end_dt": shift_dt_map[(ds, v)]["end_dt"],
            })
    agent_shift_options = sorted(agent_shift_options, key=lambda r: r["start_dt"])
    for i in range(len(agent_shift_options)):
        sh1 = agent_shift_options[i]
        for j in range(i + 1, len(agent_shift_options)):
            sh2 = agent_shift_options[j]
            if sh2["start_dt"] <= sh1["end_dt"]:
                rest_minutes = -1
            else:
                rest_minutes = int((sh2["start_dt"] - sh1["end_dt"]).total_seconds() / 60)
            if rest_minutes < MIN_REST_MINUTES:
                model.Add(x[(a, sh1["ds"], sh1["v"])] + x[(a, sh2["ds"], sh2["v"])] <= 1)
                min_rest_constraints += 1
            else:
                break

# --- Temmuz geçmişi (SABİT önceki vardiya) ---
# rest = a_start - g_end olduğundan, EN GEÇ biten geçmiş vardiya her Ağustos vardiyası için
# en küçük dinlenmeyi verir. Tüm geçmiş vardiyaları taramak yerine max(g_end) yeterli.

        if g_end_max is None or g_end > g_end_max:
            g_end_max = g_end
    if g_end_max is None:

    for ds in PLAN_GUNLER:
        if pd.to_datetime(ds) >= esik:

def _linear(cc):
    try:
        lin = cc.linear
        if lin is None or len(lin.vars) == 0:
            return None
        return lin
    except Exception:
        return None


def _enf(cc):
    try:
        return list(cc.enforcement_literal or [])
    except Exception:
        return []


def _ad(idx):
    if idx < 0:
        return f"NOT({_ad(-idx - 1)})"
    nm = proto.variables[idx].name or f"var{idx}"
    return f"{nm}#{idx}"


# ---------- 1) Sabitlenmiş değişkenleri topla ----------
sabit = {}
for i in range(len(proto.variables)):
    d = list(proto.variables[i].domain)
    if len(d) == 2 and d[0] == d[1]:
        sabit[i] = d[0]

for k in range(len(proto.constraints)):
    cc = proto.constraints[k]
    if _enf(cc):
        continue                      # koşullu kısıt - kesin sabitleme değil
    L = _linear(cc)
    if L is None:
        continue
    vs, co, dom = list(L.vars), list(L.coeffs), list(L.domain)
    if len(vs) == 1 and len(dom) == 2 and dom[0] == dom[1] and co[0] != 0:
        if dom[0] % co[0] == 0:
            sabit[vs[0]] = dom[0] // co[0]

print(f"sabit değere kilitlenmiş değişken: {len(sabit)}")


def _alt_ust(v):
    if v in sabit:
        return sabit[v], sabit[v]
    d = list(proto.variables[v].domain)
    return d[0], d[-1]


def _domain_kesisir(lo, hi, dom):
    for i in range(0, len(dom), 2):
        if not (hi < dom[i] or lo > dom[i + 1]):
            return True
    return False


def _incele(k, ayrintili=False):
    cc = proto.constraints[k]
    L = _linear(cc)
    if L is None:
        return None
    vs, co, dom = list(L.vars), list(L.coeffs), list(L.domain)
    lo = hi = 0
    for v, c in zip(vs, co):
        a, b = _alt_ust(v)
        if c >= 0:
            lo += c * a; hi += c * b
        else:
            lo += c * b; hi += c * a
    if _domain_kesisir(lo, hi, dom):
        return None
    return (k, vs, co, dom, lo, hi, _enf(cc), all(v in sabit for v in vs))

def _dp(available, seed):
    cur = {seed: 0}
    for i in range(_n):
        nxt = {}
        for st, val in cur.items():
            if available[i] and bin(st).count("1") + 1 <= MAX_CONSECUTIVE_WORK_DAYS:
                ns = ((st << 1) | 1) & _MASK
                if nxt.get(ns, -1) < val + 1:
                    nxt[ns] = val + 1
            ns = (st << 1) & _MASK
            if nxt.get(ns, -1) < val:
                nxt[ns] = val
        cur = nxt
    return max(cur.values()) if cur else 0

# %% [TEŞHİS 3] - YAYILIM (PROPAGATION) İLE ÇELİŞKİ ZİNCİRİNİ BUL
# Presolve'un yaptığını yapar: linear kısıtlarda sınır yayılımını sabit noktaya kadar çalıştırır.
# Bir kısıt sağlanamaz hâle gelir ya da bir değişkenin aralığı boşalırsa durur ve sebebini yazar.
#
# Model kurulduktan SONRA (OBJECTIVE dahil) çalıştır. Solve GEREKMEZ. 1-3 dk sürebilir.

MAX_TUR = 60
MAX_RAPOR = 12
BUYUK = 10 ** 15          # bunun ustu "sinirsiz" sayilir

proto = model.Proto() if hasattr(model, "Proto") else model.proto
NV, NC = len(proto.variables), len(proto.constraints)
print(f"model: {NV} değişken | {NC} kısıt")

lo = [0] * NV
hi = [0] * NV
for i in range(NV):
    d = list(proto.variables[i].domain)
    lo[i], hi[i] = d[0], d[-1]
adlar = [proto.variables[i].name or f"var{i}" for i in range(NV)]

kis = []
for k in range(NC):
    cc = proto.constraints[k]
    try:
        L = cc.linear
        vs = list(L.vars)
        if not vs:
            continue
    except Exception:
        continue
    try:
        enf = list(cc.enforcement_literal)
    except Exception:
        enf = []
    dom = list(L.domain)
    kis.append((k, vs, list(L.coeffs), dom[0], dom[-1], enf))

NK = len(kis)
print(f"yayılıma girecek linear kısıt: {NK}")

var_kis = {}
for idx in range(NK):
    for v in kis[idx][1]:
        var_kis.setdefault(v, []).append(idx)


def _lit(l):
    """enforcement literal durumu: 1 açık, 0 kapalı, None belirsiz"""
    v = l if l >= 0 else -l - 1
    if lo[v] != hi[v]:
        return None
    return lo[v] if l >= 0 else 1 - lo[v]


def _aktif(enf):
    for l in enf:
        d = _lit(l)
        if d == 0:
            return False
        if d is None:
            return None
    return True


def _yaz(idx):
    k, vs, co, dmn, dmx, enf = kis[idx]
    ifade = "  +  ".join(f"{c}*{adlar[v]}[{lo[v]},{hi[v]}]" for v, c in zip(vs, co))
    d1 = "-inf" if dmn < -BUYUK else dmn
    d2 = "+inf" if dmx > BUYUK else dmx
    s = f"kısıt #{k}:  {ifade}   ∈ [{d1}, {d2}]"
    if enf:
        s += "\n     enforcement: " + str([("NOT " if l < 0 else "") + adlar[l if l >= 0 else -l - 1]
                                           for l in enf])
    return s


kuyruk = list(range(NK))
kuyrukta = [True] * NK
suclu = None
daraltma = 0

for tur in range(MAX_TUR):
    if not kuyruk or suclu:
        break
    yeni = []
    for idx in kuyruk:
        kuyrukta[idx] = False
        k, vs, co, dmn, dmx, enf = kis[idx]
        if enf and _aktif(enf) is not True:
            continue

        amin = amax = 0
        for v, c in zip(vs, co):
            if c >= 0:
                amin += c * lo[v]; amax += c * hi[v]
            else:
                amin += c * hi[v]; amax += c * lo[v]

        if amin > dmx or amax < dmn:
            suclu = (idx, amin, amax, "kısıt sağlanamaz")
            break

        if amin < -BUYUK or amax > BUYUK:
            continue                      # sınırsız - daraltma anlamsız

        for v, c in zip(vs, co):
            if c == 0:
                continue
            if c > 0:
                kalan_min = amin - c * lo[v]
                kalan_max = amax - c * hi[v]
            else:
                kalan_min = amin - c * hi[v]
                kalan_max = amax - c * lo[v]
            alt = dmn - kalan_max          # c*x >= alt
            ust = dmx - kalan_min          # c*x <= ust
            if alt < -BUYUK and ust > BUYUK:
                continue
            # Python: floor(a/b) = a//b ,  ceil(a/b) = -((-a)//b)   (b'nin isareti farketmez)
            if c > 0:
                y_lo = -((-alt) // c) if alt > -BUYUK else lo[v]
                y_hi = (ust // c) if ust < BUYUK else hi[v]
            else:
                y_lo = -((-ust) // c) if ust < BUYUK else lo[v]
                y_hi = (alt // c) if alt > -BUYUK else hi[v]

            degisti = False
            if y_lo > lo[v]:
                lo[v] = y_lo; daraltma += 1; degisti = True
            if y_hi < hi[v]:
                hi[v] = y_hi; daraltma += 1; degisti = True
            if lo[v] > hi[v]:
                suclu = (idx, amin, amax, f"{adlar[v]} aralığı boşaldı -> [{lo[v]}, {hi[v]}]")
                break
            if degisti:
                for j in var_kis.get(v, ()):
                    if not kuyrukta[j]:
                        kuyrukta[j] = True; yeni.append(j)
        if suclu:
            break
    if suclu:
        break
    kuyruk = yeni
    print(f"  tur {tur + 1}: kuyruk {len(kuyruk)} | toplam daraltma {daraltma}")

print("\n" + "=" * 78)
if suclu is None:
    print("Yayılım çelişki BULAMADI (sınır yayılımı seviyesinde tutarlı).")
    print("-> Çelişki mantıksal/kombinatoryal seviyede; grup bazlı izolasyon gerekir.")
else:
    idx, amin, amax, sebep = suclu
    k, vs, co, dmn, dmx, enf = kis[idx]
    print(f"ÇELİŞKİ BULUNDU  ({sebep})")
    print("=" * 78)
    print(_yaz(idx))
    print(f"\n  ifadenin mümkün aralığı [{amin}, {amax}] | olması gereken [{dmn}, {dmx}]")
    print(f"\n  Bu değişkenleri sıkıştıran diğer kısıtlar:")
    for v in vs:
        print(f"\n  >>> {adlar[v]}  ->  [{lo[v]}, {hi[v]}]   (model tanımı {list(proto.variables[v].domain)})")
        n = 0
        for j in var_kis.get(v, ()):
            if j == idx or len(kis[j][1]) > 4:
                continue
            print("      " + _yaz(j).replace("\n", "\n      "))
            n += 1
            if n >= MAX_RAPOR:
                print("      ... (kesildi)")
                break

    import re as _re
    _GUN = ["Pzt", "Sal", "Çrş", "Prş", "Cum", "Cmt", "Paz"]
    _ag = set()
    for v in vs:
        m = _re.match(r"^(?:x|work|gercek_off)_(.+?)_(\d{4}-\d{2}-\d{2}|\d{8})", adlar[v])
        if m:
            _ag.add((m.group(1), m.group(2)))
    if _ag:
        print(f"\n  ETKİLENEN AGENT / GÜN:")
        for a, g in sorted(_ag):
            _d = pd.to_datetime(g).date()
            _hs = a in globals().get("hamile_sut_agents", set())
            _sec = [v for v in gun_vardiyalari.get(_d.isoformat(), []) if (a, _d.isoformat(), v) in x]
            print(f"    {a} @ {_d} ({_GUN[_d.weekday()]}) hamile/süt={_hs} "
                  f"tip={tip_map.get(a, {}).get(_d, '-')} tanımlı vardiya={len(_sec)}")
print("=" * 78)

    """New* çağrılarını gerçek modele geçirir, Add*/Minimize/Maximize çağrılarını yutar."""

def _dp(available, seed):
    cur = {seed: 0}
    for i in range(_n):
        nxt = {}
        for st, val in cur.items():
            if available[i] and bin(st).count("1") + 1 <= MAX_CONSECUTIVE_WORK_DAYS:
                ns = ((st << 1) | 1) & _MASK
                if nxt.get(ns, -1) < val + 1:
                    nxt[ns] = val + 1
            ns = (st << 1) & _MASK
            if nxt.get(ns, -1) < val:
                nxt[ns] = val
        cur = nxt
    return max(cur.values()) if cur else 0

def _dp_prefix(available, seed):
    """best[i] = ilk i günde çalışılabilecek MAKSİMUM gün sayısı (i = 0..n)"""
    cur = {seed: 0}
    best = [0] * (_n + 1)
    for i in range(_n):
        nxt = {}
        for st, val in cur.items():
            if available[i] and bin(st).count("1") + 1 <= MAX_CONSECUTIVE_WORK_DAYS:
                ns = ((st << 1) | 1) & _MASK
                if nxt.get(ns, -1) < val + 1:
                    nxt[ns] = val + 1
            ns = (st << 1) & _MASK
            if nxt.get(ns, -1) < val:
                nxt[ns] = val
        cur = nxt
        best[i + 1] = max(cur.values()) if cur else 0
    return best

# Temmuz geçmişi SADECE bu dilimdeki agentlara uygulanır. Diğerleri Temmuz'da hiç
# çalışmamış sayılır. Amaç: hangi agent(lar) infeasible yapıyor, ikili aramayla bulmak.
#   (0, 0)            -> Temmuz hiç uygulanmaz  (feasible bekleniyor)
#   (0, len(AGENTS))  -> hepsine uygulanır      (infeasible bekleniyor)
# Dilimi yarıya bölerek daralt: infeasible veren yarıda devam et.
_agents_sirali = sorted({str(a).strip() for a in AGENTS})
MAX6_TEMMUZ_DILIM = (0, len(_agents_sirali))

_bas, _bit = MAX6_TEMMUZ_DILIM
_temmuz_uygula = set(_agents_sirali[_bas:_bit])
print(f"max-6 Temmuz baskısı uygulanan agent dilimi: [{_bas}:{_bit}] "
      f"-> {len(_temmuz_uygula)} / {len(_agents_sirali)} agent")

plan_date_set_mc = {pd.to_datetime(ds).date() for ds in PLAN_GUNLER}
date_to_ds = {pd.to_datetime(ds).date(): ds for ds in PLAN_GUNLER}

window_starts = []
d = PLAN_START_DATE - timedelta(days=WINDOW_DAYS - 1)
son_start = PLAN_END_DATE - timedelta(days=WINDOW_DAYS - 1)
while d <= son_start:
    window_starts.append(d)
    d += timedelta(days=1)

max_consecutive_constraints = 0
gecmis_pencere_katki = 0
atlanan_baglayici_degil = 0
max6_sifir_limit = []   # (agent, pencere_baslangic, kapatilan_agustos_gun_sayisi)

for a_raw in AGENTS:
    a = str(a_raw).strip()
    worked_a = gecmis_calisma_gunleri.get(a, set()) if a in _temmuz_uygula else set()

    for ws in window_starts:
        window_dates = [ws + timedelta(days=k) for k in range(WINDOW_DAYS)]

        aug_vars = [
            work[(a, date_to_ds[dd])]
            for dd in window_dates
            if dd in plan_date_set_mc and (a, date_to_ds[dd]) in work
        ]
        if not aug_vars:
            continue   # tamamen geçmişte kalan pencere - kısıtlanacak değişken yok

        gecmis_calismis = sum(1 for dd in window_dates if dd < PLAN_START_DATE and dd in worked_a)
        limit = MAX_CONSECUTIVE_WORK_DAYS - gecmis_calismis

        # Bağlayıcı olmayan pencereyi modele hiç ekleme (limit >= mevcut değişken sayısı).
        # Semantik olarak fark yok, sadece gereksiz kısıt üretilmiyor.
        if limit >= len(aug_vars):
            atlanan_baglayici_degil += 1
            continue
        if limit < 0:
            limit = 0

        model.Add(sum(aug_vars) <= limit)
        max_consecutive_constraints += 1

        if gecmis_calismis > 0:
            gecmis_pencere_katki += 1
        if limit == 0:
            max6_sifir_limit.append((a, str(ws), len(aug_vars)))

print(f"max 6 gün üst üste kısıtı: {max_consecutive_constraints} adet")
print(f"  - bağlayıcı olmadığı için atlanan pencere : {atlanan_baglayici_degil}")
print(f"  - Temmuz çalışması limiti düşüren pencere : {gecmis_pencere_katki}")
print(f"  - limit=0 (Ağustos günleri tamamen kapanan) pencere: {len(max6_sifir_limit)}")
for r in max6_sifir_limit[:15]:
    print(f"      agent={r[0]}  pencere={r[1]}  kapanan Ağustos günü={r[2]}")
