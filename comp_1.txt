WITH AYLAR AS (
    SELECT DISTINCT last_day_of_month AS ay
    FROM   edw.cmn_calendar
    WHERE  last_day_of_month BETWEEN DATE '2026-01-31' AND DATE '2026-10-31'
)
SELECT a.ay,
       COUNT(DISTINCT d.party_id) AS kumulatif_aktif_kullanici
FROM   aylar a
JOIN   dm_cad.cad_customer_dgtl_sess_daily d
  ON   d.data_date BETWEEN DATE '2026-01-31' AND a.ay
 AND   d.data_date IN (SELECT ay FROM aylar)
WHERE  d.main_channel_id = 7
  AND  d.time_period_id  = <1_AYLIK_PERIYOT_ID>
  AND  d.total_transaction_count > 0
GROUP BY a.ay
ORDER BY a.ay;
