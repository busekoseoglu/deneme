# ============ 2026-09-07 - BİRİM DEĞİŞTİ: TAKIM-GÜN -> KİŞİ BAŞI GÜN ============
# ESKİ tablo haftalık BASE mantığıyla üretiliyordu: her (ISO hafta, takım) için
# en çok kişinin çalıştığı vardiya o haftanın base'i sayılır, base günleri
# toplanırdı. Birim TAKIM-GÜN'dü, takım büyüklüğünden bağımsızdı.
#
# YENİ tablo base'e hiç bakmıyor. Takımın bütün agentlarının o vardiya altında
# çalıştığı TOPLAM AGENT-GÜN alınıp TAKIM MEVCUDUNA BÖLÜNÜYOR. Yani:
#   "ben 10 gün 15:00'ta çalıştım, sen 5 gün" -> takım toplamı 15 agent-gün
#   takım 15 kişilikse -> 1.0 kişi başı gün
# Bölme İŞ BİRİMİ tarafında, veride yapılıyor; model bölmüyor.
#
# Bunun sonucu: "takımın base'i değildi ama ben orada çalıştım" artık sayılıyor.
# Muafiyetli ve devir alan agentlar da geçmişe dahil.
#
# BEKLENEN ŞEMA:
#   yil | rapor_bas | teamleader_user_code | working_main_group | shifts_adj |
#   total_gun_sayisi | agent_sayisi | gun_agent_ratio
#
#   total_gun_sayisi = HAM agent-gün (bölünmemiş)
#   agent_sayisi     = takımın agent sayısı (bölen)
#   gun_agent_ratio  = total_gun_sayisi / agent_sayisi  -> KİŞİ BAŞI GÜN
#
#   DENGE HESABI gun_agent_ratio'yu KULLANIR. total_gun_sayisi sadece tutarlılık
#   kontrolü ve raporlama için okunur; modele girmez.
#
#   rapor_bas = planlanan ay (Eylül planlanıyorsa 9). Tablo birden çok kesit
#   taşıyabilir; hangisinin kullanılacağı CONFIG'deki TAKIM_VARDIYA_GECMIS_KESIT
#   ile seçilir, None ise EN SON kesit otomatik alınır.
#
#   agent_sayisi = kişi başı hesabında kullanılan BÖLEN. Model bu ayın katkısını
#   da AYNI sayıya böler; iki taraf farklı bölen kullanırsa denge sessizce
#   yanlış ölçer. Bu yüzden bölen veriden gelir, notebook kendi saymaz.

def _tvg_bas_saat(s):
    """'15:00-00:00' -> '15:00'. Ayrıştıramazsa None döner (sessizce yutmaz, sayılır)."""
    t = str(s).strip()
    if not t or t.lower() in ("nan", "none", "null"):
        return None
    t = t.split("-")[0].strip().replace(".", ":")
    m = re.match(r"^(\d{1,2}):?(\d{2})$", t)
    if not m:
        return None
    _h, _dk = int(m.group(1)), int(m.group(2))
    if _h > 23 or _dk > 59:
        return None
    return f"{_h:02d}:{_dk:02d}"

# %% [KONTROL VERİ] - TAKIM LİDERİ ID TUTARLILIĞI (bütün tablolarda)
# 2026-09-14 EKLENEN. Takım anahtarı teamleader_name'den teamleader_user_code'a
# geçti. Bu hücre tek bir soruyu cevaplıyor:
#
#     "Bir ID bütün verilerde AYNI kişiyi temsil ediyor mu?"
#
# Neden gerekli: isimler güvenilmez (kızlık / evlilik soyadı farkı), ID'ler elle
# eşleştirildi. Eşleşme koparsa MODEL HATA VERMEZ - takım geçmiş tabloda
# bulunamaz, denge kuralı o takım için sessizce kurulmaz. Bu yüzden kontrol
# solve'dan ÇOK ÖNCE, veri hazırlığının sonunda duruyor.
#
# SABİT TABLO LİSTESİ YOK: globals() taranır, teamleader_user_code kolonu olan
# HER DataFrame kontrole girer. Sonradan veri eklenirse kendiliğinden kapsanır.
#
# Üç ayrı soruya bakılıyor:
#   1) BİR ID -> BİRDEN ÇOK İSİM   : kızlık soyadı ise NORMAL; iki farklı kişiye
#                                     aynı ID verilmişse CİDDİ. Gözle bakılmalı.
#   2) BİR İSİM -> BİRDEN ÇOK ID   : aynı kişi ikiye bölünmüş ya da iki kişinin
#                                     adı aynı. Takım anahtarını ikiye böler.
#   3) BİR ID -> BİRDEN ÇOK GRUP   : takım anahtarı (grup | id) ikiye bölünür,
#                                     geçmiş eşleşmesi yarısını kaybeder.
#
# Hiçbir durumda DURMAZ - rapor eder. Karar veriyi hazırlayanda.

_TL_KOD = "teamleader_user_code"
_TL_ISIM = "teamleader_name"
_TL_GRUP = "working_main_group"

tl_tablo_ozet_df = pd.DataFrame()
tl_id_isim_df = pd.DataFrame()
tl_isim_id_df = pd.DataFrame()
tl_id_grup_df = pd.DataFrame()
tl_kapsam_df = pd.DataFrame()


def _tl_norm(_s):
    """[HÜCRE 6] ve [HÜCRE 3D] ile BİREBİR aynı normalizasyon."""
    return _s.astype(str).str.strip()


_TL_BOS = {"", "nan", "None", "NaT", "<NA>", "NULL", "null"}

# --- tabloları bul: globals()'ta teamleader_user_code kolonu olan her df ---
_tl_tablolar = {}
for _ad, _obj in sorted(globals().items()):
    if _ad.startswith("_") or not isinstance(_obj, pd.DataFrame):
        continue
    if _TL_KOD in _obj.columns:
        _tl_tablolar[_ad] = _obj

if not _tl_tablolar:
    print(f"!!! Hiçbir tabloda '{_TL_KOD}' kolonu YOK.")
    print("    SQL hücrelerine kolonu ekleyip yeniden çalıştır. Kontrol atlandı.")
else:
    print(f"=== TAKIM LİDERİ ID TUTARLILIĞI ===")
    print(f"Kontrole giren tablo: {len(_tl_tablolar)} -> {', '.join(_tl_tablolar)}")

    # kolonu OLMAYAN ama teamleader_name olan tablolar - atlanmış olabilir
    _tl_eksik_tablo = [
        _ad for _ad, _o in sorted(globals().items())
        if not _ad.startswith("_") and isinstance(_o, pd.DataFrame)
        and _TL_ISIM in _o.columns and _TL_KOD not in _o.columns
    ]
    if _tl_eksik_tablo:
        print(f"\n!!! DİKKAT: {_TL_ISIM} olup {_TL_KOD} OLMAYAN tablo: "
              f"{', '.join(_tl_eksik_tablo)}")
        print(f"    Bu tablolar kontrol dışı kaldı - kolonu onlara da ekle.")

    # ---------------- tablo bazında özet ----------------
    _ozet, _uzun = [], []
    for _ad, _df in _tl_tablolar.items():
        _kod = _tl_norm(_df[_TL_KOD])
        _bos = int(_kod.isin(_TL_BOS).sum())
        _gecerli = ~_kod.isin(_TL_BOS)
        _ozet.append({
            "tablo": _ad,
            "satir": len(_df),
            "benzersiz_id": int(_kod[_gecerli].nunique()),
            "bos_id_satiri": _bos,
            "isim_kolonu_var": _TL_ISIM in _df.columns,
            "grup_kolonu_var": _TL_GRUP in _df.columns,
        })
        # uzun format: (tablo, id, isim, grup)
        _alt = pd.DataFrame({"tablo": _ad, "id": _kod})
        _alt[_TL_ISIM] = (_tl_norm(_df[_TL_ISIM]) if _TL_ISIM in _df.columns else None)
        _alt[_TL_GRUP] = (_tl_norm(_df[_TL_GRUP]) if _TL_GRUP in _df.columns else None)
        _uzun.append(_alt[_gecerli.values])

    tl_tablo_ozet_df = pd.DataFrame(_ozet)
    print("\n--- tablo özeti ---")
    print(tl_tablo_ozet_df.to_string(index=False))
    _bos_top = int(tl_tablo_ozet_df["bos_id_satiri"].sum())
    if _bos_top:
        print(f"\n!!! {_bos_top} satırda ID BOŞ - bu satırların takımı "
              f"'... | nan' olur ve geçerli takım gibi davranır.")

    _u = pd.concat(_uzun, ignore_index=True).drop_duplicates()

    # ---------------- 1) BİR ID -> BİRDEN ÇOK İSİM ----------------
    _isimli = _u[_u[_TL_ISIM].notna() & ~_u[_TL_ISIM].isin(_TL_BOS)]
    if len(_isimli):
        _g = _isimli.groupby("id")[_TL_ISIM].agg(["nunique", lambda s: sorted(set(s))])
        _g.columns = ["isim_sayisi", "isimler"]
        _g["tablolar"] = _isimli.groupby("id")["tablo"].agg(lambda s: sorted(set(s)))
        tl_id_isim_df = _g[_g["isim_sayisi"] > 1].reset_index().sort_values(
            "isim_sayisi", ascending=False)
        print(f"\n--- 1) BİR ID -> BİRDEN ÇOK İSİM ---")
        print(f"  etkilenen ID: {len(tl_id_isim_df)}")
        if len(tl_id_isim_df):
            print("  Kızlık/evlilik soyadı farkıysa NORMAL - eşleştirme zaten bunun için.")
            print("  AMA iki FARKLI kişiye aynı ID verilmişse ciddi hata. Gözle bak:")
            print(tl_id_isim_df.head(15).to_string(index=False))
    else:
        print(f"\n--- 1) BİR ID -> BİRDEN ÇOK İSİM ---")
        print(f"  Hiçbir tabloda {_TL_ISIM} kolonu yok - bu kontrol yapılamadı.")

    # ---------------- 2) BİR İSİM -> BİRDEN ÇOK ID ----------------
    if len(_isimli):
        _g2 = _isimli.groupby(_TL_ISIM)["id"].agg(["nunique", lambda s: sorted(set(s))])
        _g2.columns = ["id_sayisi", "idler"]
        _g2["tablolar"] = _isimli.groupby(_TL_ISIM)["tablo"].agg(lambda s: sorted(set(s)))
        tl_isim_id_df = _g2[_g2["id_sayisi"] > 1].reset_index().sort_values(
            "id_sayisi", ascending=False)
        print(f"\n--- 2) BİR İSİM -> BİRDEN ÇOK ID ---")
        print(f"  etkilenen isim: {len(tl_isim_id_df)}")
        if len(tl_isim_id_df):
            print("  !!! Aynı kişi ikiye bölünmüş ya da iki kişinin adı aynı olabilir.")
            print("  Bölünmüşse o takımın geçmişi de ikiye bölünür, denge yanlış ölçer:")
            print(tl_isim_id_df.head(15).to_string(index=False))
        else:
            print("  Temiz.")

    # ---------------- 3) BİR ID -> BİRDEN ÇOK GRUP ----------------
    _gruplu = _u[_u[_TL_GRUP].notna() & ~_u[_TL_GRUP].isin(_TL_BOS)]
    print(f"\n--- 3) BİR ID -> BİRDEN ÇOK working_main_group ---")
    if len(_gruplu):
        _g3 = _gruplu.groupby("id")[_TL_GRUP].agg(["nunique", lambda s: sorted(set(s))])
        _g3.columns = ["grup_sayisi", "gruplar"]
        _g3["tablolar"] = _gruplu.groupby("id")["tablo"].agg(lambda s: sorted(set(s)))
        tl_id_grup_df = _g3[_g3["grup_sayisi"] > 1].reset_index().sort_values(
            "grup_sayisi", ascending=False)
        print(f"  etkilenen ID: {len(tl_id_grup_df)}")
        if len(tl_id_grup_df):
            print("  !!! Takım anahtarı 'grup | id' olduğu için bu ID'ler İKİ AYRI")
            print("      takıma bölünür; geçmiş eşleşmesi yarısını kaybeder.")
            print(tl_id_grup_df.head(15).to_string(index=False))
        else:
            print("  Temiz.")
    else:
        print(f"  Hiçbir tabloda {_TL_GRUP} kolonu yok - bu kontrol yapılamadı.")

    # ---------------- 4) ID KAPSAMI: hangi ID hangi tabloda var ----------------
    _kaps = (_u.assign(_v=1)
             .pivot_table(index="id", columns="tablo", values="_v",
                          aggfunc="max", fill_value=0))
    _kaps["tablo_sayisi"] = _kaps.sum(axis=1)
    tl_kapsam_df = _kaps.reset_index()
    print(f"\n--- 4) ID KAPSAMI ---")
    print(f"  toplam benzersiz ID: {len(tl_kapsam_df)}")
    print("  kaç tabloda görünüyor (dağılım):")
    print(tl_kapsam_df["tablo_sayisi"].value_counts().sort_index()
          .rename_axis("tablo_sayisi").to_frame("id_sayisi").to_string())

    # plan tarafı (df_tam) ile geçmiş tarafı arasındaki fark - asıl kritik olan
    if "df_tam" in _tl_tablolar and "df_takim_vardiya_gecmis" in _tl_tablolar:
        _plan_id = set(_tl_norm(_tl_tablolar["df_tam"][_TL_KOD])) - _TL_BOS
        _gec_id = set(_tl_norm(_tl_tablolar["df_takim_vardiya_gecmis"][_TL_KOD])) - _TL_BOS
        _sadece_plan = sorted(_plan_id - _gec_id)
        _sadece_gec = sorted(_gec_id - _plan_id)
        print(f"\n  df_tam (plan) ID sayısı          : {len(_plan_id)}")
        print(f"  geçmiş tablo ID sayısı           : {len(_gec_id)}")
        print(f"  SADECE planda olan (geçmişi yok) : {len(_sadece_plan)}")
        if _sadece_plan:
            print("    !!! Bu takımlar denge kuralında 'geçmişi 0' sayılır ve")
            print("        haksız yere 'en geride' görünürler:")
            for _i in _sadece_plan[:10]:
                print(f"        {_i}")
        print(f"  SADECE geçmişte olan (planda yok): {len(_sadece_gec)}"
              f"   (kapanmış/değişmiş ekipler - normal)")

    print("\nÜretilen tablolar: tl_tablo_ozet_df, tl_id_isim_df, tl_isim_id_df, "
          "tl_id_grup_df, tl_kapsam_df")
