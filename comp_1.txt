objective_terms = []

# 2026-09-29: hafta sonu coverage cezasi CARPAN ile ucuzlatilir. Gerekcesi CONFIG'de.
# Katsayilar tamsayi olmali (CP-SAT), int() ile yuvarlaniyor.
if "HAFTA_SONU_COVERAGE_CARPANI" not in globals():
    raise SystemExit("CONFIG'de tanımlı olmalı: HAFTA_SONU_COVERAGE_CARPANI")
_hs_under_w = int(round(UNDER_BUFFER_W * HAFTA_SONU_COVERAGE_CARPANI))
_hs_missing_w = int(round(MISSING_TO_REQUIRED_W * HAFTA_SONU_COVERAGE_CARPANI))
print(f"coverage ağırlığı -> hafta içi {UNDER_BUFFER_W}/{MISSING_TO_REQUIRED_W} | "
      f"hafta sonu {_hs_under_w}/{_hs_missing_w} (çarpan {HAFTA_SONU_COVERAGE_CARPANI})")

for ds in PLAN_GUNLER:
    _hafta_sonu_mu = pd.to_datetime(ds).weekday() >= 5
    _u_w = _hs_under_w if _hafta_sonu_mu else UNDER_BUFFER_W
    _m_w = _hs_missing_w if _hafta_sonu_mu else MISSING_TO_REQUIRED_W
    for v in gun_vardiyalari.get(ds, []):
        objective_terms.append(_u_w * under_buffer[(ds, v)])
        objective_terms.append(_m_w * missing_to_required[(ds, v)])

        if gece_aksam_vardiyasi_mi[(ds, v)]:
            excess_weight = EXCESS_GECE_AKSAM_W
        elif mola_destek_vardiyasi_mi[(ds, v)]:
            excess_weight = EXCESS_MOLA_DESTEK_W
        else:
            excess_weight = EXCESS_NORMAL_W

        objective_terms.append(excess_weight * excess_to_required[(ds, v)])
        objective_terms.append(OVER_BUFFER_W * over_buffer[(ds, v)])

# İkinci mesai ekstra cezası
for a in AGENTS:
    objective_terms.append(IKINCI_MESAI_W * ikinci_mesai_aylik[a])

# Hafta sonu çalışma adil dağıtım cezası
for a in hafta_sonu_calisma_sayisi.keys():

# 2026-09-28: hafta sonu cezası 2. kademe (dışbükey). Değişken yoksa eski davranış.
if "fazla_hafta_sonu_kademe1" in globals() and fazla_hafta_sonu_kademe1:
    for _a_hk, _var_hk in fazla_hafta_sonu_kademe1.items():
        objective_terms.append(HAFTA_SONU_KADEME1_W * _var_hk)
    print("hafta sonu 1. kademe terimi:", len(fazla_hafta_sonu_kademe1),
          "| eşik:", HAFTA_SONU_KADEME1_GUN, "| W:", HAFTA_SONU_KADEME1_W)

# Arife / resmi tatil mesai ve ihlal cezaları
if "arife_mesai" in globals() and ARIFE_MESAI_W != 0:
    for (a, ds, v), var in arife_mesai.items():
        objective_terms.append(ARIFE_MESAI_W * var)

if "resmi_tatil_mesai" in globals() and RESMI_TATIL_MESAI_W != 0:
    for (a, ds, v), var in resmi_tatil_mesai.items():
        objective_terms.append(RESMI_TATIL_MESAI_W * var)

if "resmi_tatil_kisitli_ihlal" in globals():
    for (a, ds, v), var in resmi_tatil_kisitli_ihlal.items():

print("ARIFE_MESAI_W:", ARIFE_MESAI_W)
print("RESMI_TATIL_MESAI_W:", RESMI_TATIL_MESAI_W)
print("RESMI_TATIL_KISITLI_IHLAL_W:", RESMI_TATIL_KISITLI_IHLAL_W)

# Haftalık hedef sapma cezası - ÖLÜ BLOK (2026-08-10 tespit)

    for (a, wk), var in weekly_under.items():
        objective_terms.append(WEEKLY_UNDER_W * var)
if "weekly_over" in globals():

print("WEEKLY UNDER W:", WEEKLY_UNDER_W)
print("WEEKLY OVER W:", WEEKLY_OVER_W)

# Günlük fazla atama yığılma cezası
if "daily_excess_over" in globals():
    for ds, var in daily_excess_over.items():
        objective_terms.append(DAILY_EXCESS_OVER_W * var)
print("DAILY EXCESS SOFT LIMIT:", DAILY_EXCESS_SOFT_LIMIT)
print("DAILY_EXCESS_OVER_W:", DAILY_EXCESS_OVER_W)

# ===== 2026-08-12 EKLENEN - TAKIM VARDIYA GECMIS DENGESI =====
