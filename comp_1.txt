# Tek başına çalışır. Zorunlu gereksinim: [KONTROL] - ÇÖZÜMÜ ÇIKAR + ORTAK YARDIMCILAR
# hücresi (cozum_calisma / cozum_atama). Diğer df'ler yoksa o sheet atlanır, hücre patlamaz.
#
# NORM / TELAFİ / MESAİ tanımı:
#
# 1) HAFTALIK ADIM - hangi hafta sonu günleri "fazla"?
#      hafta_kota    = o haftadaki hafta içi (Pzt-Cum) günlerden İZİN ve RESMİ TATİL
#                      OLMAYANLARIN sayısı. Aydan sarkan haftalarda (W31) Temmuz'un
#                      hafta içi günleri de sayılır.
#                      SABİT 5 DEĞİL: izin bir iş gününün yerine geçer, o hafta
#                      agentin normal çalışması 5 değil 4'tür.
#      hafta_calisma = o ISO haftada çalışılan gün sayısı
#      fazla         = max(0, hafta_calisma - hafta_kota)
#      o haftanın çalışılan hafta sonu günleri tarih sırasına dizilir;
#      son `fazla` tanesi ADAY, kalanı NORM çalışma.
#      W31 gibi aydan sarkan haftalarda Temmuz'da çalışılan günler de sayılır.
#
# 2) AYLIK ADIM - adaylardan hangileri gerçekten ÜCRETLİ mesai?
#      aylik_fazla = max(0, sayilan_calisma - baz_gereken_calisma)   (resmi tatil hariç)
#      adaylar kronolojik sırada; son `aylik_fazla` tanesi -> MESAİ
#                                  kalanı                  -> TELAFİ
#
# Neden: agent bir hafta 3 gün off alıp (normali 2) 4 gün çalışırsa 1 gün borçlanır;
# başka bir hafta 6 gün çalışıp borcu kapatır. Aylık toplam yine baz'dır, yani
# ÜCRETLİ MESAİ YOKTUR - o gün TELAFİ'dir. Ama agent borcunu kapatıp ÜSTÜNE de
# fazladan çalışırsa aylik_fazla > 0 olur ve o günler MESAİ olarak kalır.
#
# Sonuç: MESAİ gün sayısı = AGENT_OZET sayfasındaki kota_farki ile BİREBİR aynıdır.

if "cozum_calisma" not in globals() or "cozum_atama" not in globals():

    )

_NORMAL_WORK_DAYS = globals().get("NORMAL_WORK_DAYS", 5)
_saat = globals().get("saat", {})
_tip_map = globals().get("tip_map", {})
_gecmis_gunleri_x = globals().get("gecmis_calisma_gunleri", {})
_g_calisma   = globals().get("gecmis_calisma", {})            # agent -> {date: (bas, bit)}
_g_off       = globals().get("gecmis_off_gunleri", {})
_g_izin      = globals().get("gecmis_izin_gunleri", {})
_g_h_izin    = globals().get("gecmis_haftalik_izin_gunleri", {})
_g_kapsam    = globals().get("gecmis_kapsam", {})

_plan_dates = sorted(pd.to_datetime(ds).date() for ds in PLAN_GUNLER)
_PLAN_BAS = min(_plan_dates)
_PLAN_BIT = max(_plan_dates)
_plan_date_set_x = set(_plan_dates)

_resmi_tatil_set = set()
if "resmi_tatil_plan_gunleri" in globals():
    _resmi_tatil_set = {pd.to_datetime(d).date() for d in resmi_tatil_plan_gunleri}
elif "RESMI_TATIL_GUNLERI" in globals():
    _resmi_tatil_set = {pd.to_datetime(d).date() for d in RESMI_TATIL_GUNLERI}

_GUN_ADI = ["Pzt", "Sal", "Çar", "Per", "Cum", "Cmt", "Paz"]


def _vardiya_metni(ds, v):
    """Vardiya kodunu okunur saat aralığına çevirir."""
    if v is None:
        return ""
    st = _saat.get((ds, v))
    if st:
        return f"{str(st[0]).strip()[:5]}-{str(st[1]).strip()[:5]}"
    return str(v)


# ---------------------------------- AYLIK KOTA (baz_gereken_calisma) - agent
_baz_map_x = {}
if "aylik_calisma_debug_df" in globals() and not aylik_calisma_debug_df.empty:
    _baz_map_x = dict(zip(
        aylik_calisma_debug_df["agent_user_code"].astype(str).str.strip(),
        aylik_calisma_debug_df["baz_gereken_calisma"].astype(int),
    ))
    _izin_map_x = dict(zip(
        aylik_calisma_debug_df["agent_user_code"].astype(str).str.strip(),
        aylik_calisma_debug_df["izin_is_gunu_sayisi"].astype(int),
    ))
    _aylik_is_gunu_x = int(aylik_calisma_debug_df["aylik_is_gunu"].iloc[0])
else:
    _is_gunu_d = {
        pd.to_datetime(ds).date() for ds in PLAN_GUNLER
        if pd.to_datetime(ds).weekday() in (0, 1, 2, 3, 4)
        and pd.to_datetime(ds).date() not in _resmi_tatil_set
    }
    _aylik_is_gunu_x = len(_is_gunu_d)
    _baz_map_x, _izin_map_x = {}, {}
    for a_raw in AGENTS:
        a = str(a_raw).strip()
        _izn = {d for d, t in _tip_map.get(a, {}).items()
                if str(t).strip().lower() == "izin" and d in _is_gunu_d}
        _izin_map_x[a] = len(_izn)
        _baz_map_x[a] = _aylik_is_gunu_x - len(_izn)

# --------------------------- HAFTALIK NORM / TELAFİ / MESAİ HESABI (agent)
_hafta_gunleri = defaultdict(list)
for ds in PLAN_GUNLER:
    d = pd.to_datetime(ds).date()
    _pzt = d - _td(days=d.weekday())
    _hafta_gunleri[_pzt].append(ds)
for _k in _hafta_gunleri:
    _hafta_gunleri[_k] = sorted(_hafta_gunleri[_k], key=lambda s: pd.to_datetime(s))

mesai_gunleri_map = {}   # agent -> {ds}  ücretli mesai
telafi_gunleri_map = {}  # agent -> {ds}  borç kapatma
norm_gunleri_map = {}    # agent -> {ds}  kota içi hafta sonu
hafta_calisma_debug_rows = []

for a_raw in AGENTS:
    a = str(a_raw).strip()
    _gecmis_a = _gecmis_gunleri_x.get(a, set())
    _aday_a, _norm_a = [], set()   # _aday_a: haftalık kotayı aşan hafta sonu günleri (kronolojik)

    for _pzt, _wk_gunler in sorted(_hafta_gunleri.items()):
        _calisilan = [ds for ds in _wk_gunler if cozum_calisma.get((a, ds), 0) == 1]

        # aydan sarkan hafta: plan öncesi (Temmuz) çalışılan günleri de say
        _hafta_araligi = {_pzt + _td(days=k) for k in range(7)}
        _temmuz_ek = len({d for d in _gecmis_a if d in _hafta_araligi and d < _PLAN_BAS})

        _hafta_calisma = len(_calisilan) + _temmuz_ek

        # --- haftalık kota: izin ve resmi tatil günleri DÜŞÜLÜR ---
        _kota = 0
        for _k2 in range(5):                       # Pzt..Cum
            _d2 = _pzt + _td(days=_k2)
            if _d2 < _PLAN_BAS:
                # aydan sarkan hafta (W31): Temmuz'un hafta içi günleri
                if _d2 not in _g_izin.get(a, set()):
                    _kota += 1
            elif _d2 in _plan_date_set_x:
                if _d2 in _resmi_tatil_set:
                    continue
                if str(_tip_map.get(a, {}).get(_d2, "")).strip().lower() == "izin":
                    continue
                _kota += 1
        _kota = min(_NORMAL_WORK_DAYS, _kota)

        _fazla = max(0, _hafta_calisma - _kota)

        _hs_calisilan = [ds for ds in _calisilan if pd.to_datetime(ds).weekday() in (5, 6)]
        if _fazla > 0 and _hs_calisilan:
            _aday_bu_hafta = list(_hs_calisilan[-_fazla:])
        else:
            _aday_bu_hafta = []
        _norm_bu_hafta = set(_hs_calisilan) - set(_aday_bu_hafta)

        _aday_a.extend(_aday_bu_hafta)
        _norm_a |= _norm_bu_hafta

        hafta_calisma_debug_rows.append({
            "agent_user_code": a,
            "hafta_pzt": str(_pzt),
            "plan_ici_calisma": len(_calisilan),
            "temmuzdan_gelen_calisma": _temmuz_ek,
            "hafta_toplam_calisma": _hafta_calisma,
            "kota": _kota,
            "fazla": _fazla,
            "hafta_sonu_calisilan": len(_hs_calisilan),
            "norm": len(_norm_bu_hafta),
            "kota_asan_aday": len(_aday_bu_hafta),
        })

    # ---- AYLIK ADIM: adaylardan kaçı gerçekten ÜCRETLİ mesai? ----
    _aday_a = sorted(set(_aday_a), key=lambda d: pd.to_datetime(d))
    _sayilan_a = sum(
        1 for ds in PLAN_GUNLER
        if cozum_calisma.get((a, ds), 0) == 1
        and pd.to_datetime(ds).date() not in _resmi_tatil_set
    )
    _aylik_fazla_a = max(0, _sayilan_a - int(_baz_map_x.get(a, 0)))
    _mesai_sayisi = min(_aylik_fazla_a, len(_aday_a))

    mesai_gunleri_map[a] = set(_aday_a[len(_aday_a) - _mesai_sayisi:]) if _mesai_sayisi else set()
    telafi_gunleri_map[a] = set(_aday_a) - mesai_gunleri_map[a]
    norm_gunleri_map[a] = _norm_a

hafta_calisma_debug_df = pd.DataFrame(hafta_calisma_debug_rows)

# --------------------------------------------------- TAKVİM TABLOSU KURULUMU
_kimlik_kaynak = ["agent_user_code", "agent_name",
                  "working_main_group", "takim", "lokasyon", "mesaiye_kalamaz_flg"]
_kimlik_kolonlari = [c for c in _kimlik_kaynak if c in df_tam.columns]
_kimlik_df = df_tam[_kimlik_kolonlari].copy()
_kimlik_df["agent_user_code"] = _kimlik_df["agent_user_code"].astype(str).str.strip()
_kimlik_lookup = _kimlik_df.set_index("agent_user_code").to_dict("index")

_ozet_kolonlari = ["toplam_calisma", "hafta_ici_calisma", "norm_hafta_sonu",
                   "telafi_hafta_sonu", "mesai_hafta_sonu", "resmi_tatil_calisma", "izin_gun",
                   "off_gun", "off_t_talep", "off_t_karsilanmayan", "off_t2_talep",
                   "temmuz_calisma"]

# --- TEMMUZ kolonları (26 Temmuz -> plan başlangıcına kadar, salt okunur geçmiş) ---
TEMMUZ_GUN_SAYISI = 6            # 26 Temmuz'dan itibaren
_temmuz_tarihleri = [_PLAN_BAS - _td(days=k) for k in range(TEMMUZ_GUN_SAYISI, 0, -1)]
_temmuz_kolonlari = []
_temmuz_kolon_date = {}
for d in _temmuz_tarihleri:
    _kol = f"{d.day} Tem {_GUN_ADI[d.weekday()]}"
    _temmuz_kolonlari.append(_kol)
    _temmuz_kolon_date[_kol] = d

# --- AĞUSTOS (plan) kolonları ---
_gun_kolonlari = []
_gun_kolon_ds = {}
for ds in PLAN_GUNLER:
    d = pd.to_datetime(ds).date()
    _kol = f"{d.day} {_GUN_ADI[d.weekday()]}"
    _gun_kolonlari.append(_kol)
    _gun_kolon_ds[_kol] = ds

takvim_rows = []
hucre_renk = {}   # (satir_no_0bazli, kolon_adi) -> RENK anahtari

for _i, a_raw in enumerate(AGENTS):
    a = str(a_raw).strip()
    _k = _kimlik_lookup.get(a, {})
    row = {c: _k.get(c, "") for c in _kimlik_kolonlari}
    row["agent_user_code"] = a

    _tip_dates = _tip_map.get(a, {})
    _mesai_a = mesai_gunleri_map.get(a, set())
    _telafi_a = telafi_gunleri_map.get(a, set())
    _norm_a = norm_gunleri_map.get(a, set())

    _n_calisma = _n_hafta_ici = _n_norm = _n_mesai = _n_telafi = _n_rt = 0
    _n_izin = _n_off = _n_off_t = _n_off_t_ihlal = _n_off_t2 = 0
    _n_temmuz_calisma = 0

    # ---- TEMMUZ (geçmiş, plan dışı) ----
    _gc_a = _g_calisma.get(a, {})
    for _kol in _temmuz_kolonlari:
        d = _temmuz_kolon_date[_kol]
        if d in _gc_a:
            _bas, _bit = _gc_a[d]
            row[_kol] = f"{str(_bas).strip()[:5]}-{str(_bit).strip()[:5]}"
            hucre_renk[(_i, _kol)] = "g_calisma"
            _n_temmuz_calisma += 1
        elif d in _g_izin.get(a, set()):
            row[_kol] = "İZİN"
            hucre_renk[(_i, _kol)] = "g_izin"
        elif d in _g_h_izin.get(a, set()):
            # süt izni düzeni: off DEĞİL, ayrı kategori
            row[_kol] = "H.İZİN"
            hucre_renk[(_i, _kol)] = "g_h_izin"
        elif d in _g_off.get(a, set()):
            row[_kol] = "OFF"
            hucre_renk[(_i, _kol)] = "g_off"
        else:
            row[_kol] = ""   # kayıt yok

    # ---- AĞUSTOS (plan) ----
    for _kol in _gun_kolonlari:
        ds = _gun_kolon_ds[_kol]
        d = pd.to_datetime(ds).date()
        tip = str(_tip_dates.get(d, "")).strip().lower()
        calisiyor = cozum_calisma.get((a, ds), 0) == 1
        vardiya = _vardiya_metni(ds, cozum_atama.get((a, ds))) if calisiyor else ""

        if tip == "izin":
            _n_izin += 1
            row[_kol] = "İZİN"
            hucre_renk[(_i, _kol)] = "izin"
            continue

        if tip == "off_t":
            _n_off_t += 1
        elif tip == "off_t2":
            _n_off_t2 += 1

        if not calisiyor:
            if tip == "off_t":
                row[_kol] = "OFF_T"
                hucre_renk[(_i, _kol)] = "off_t"
            elif tip == "off_t2":
                row[_kol] = "OFF_T2"
                hucre_renk[(_i, _kol)] = "off_t2"
            else:
                _n_off += 1
                row[_kol] = "OFF"
                hucre_renk[(_i, _kol)] = "off"
            continue

        # --- buradan sonrası: agent o gün ÇALIŞIYOR ---
        _n_calisma += 1
        if d in _resmi_tatil_set:
            _n_rt += 1
            row[_kol] = f"RESMİ TATİL MESAİ\n{vardiya}"
            _renk = "resmi_tatil"
        elif ds in _mesai_a:
            _n_mesai += 1
            row[_kol] = f"MESAİ\n{vardiya}"
            _renk = "mesai"
        elif ds in _telafi_a:
            _n_telafi += 1
            row[_kol] = f"TELAFİ\n{vardiya}"
            _renk = "telafi"
        elif ds in _norm_a:
            _n_norm += 1
            row[_kol] = f"NORM\n{vardiya}"
            _renk = "norm"
        else:
            _n_hafta_ici += 1
            row[_kol] = vardiya
            _renk = None

        # off_t talebi vardı ama karşılanmadı -> rengi ezer
        if tip == "off_t":
            _n_off_t_ihlal += 1
            _renk = "off_t_ihlal"

        if _renk:
            hucre_renk[(_i, _kol)] = _renk

    row["toplam_calisma"] = _n_calisma
    row["hafta_ici_calisma"] = _n_hafta_ici
    row["norm_hafta_sonu"] = _n_norm
    row["mesai_hafta_sonu"] = _n_mesai
    row["telafi_hafta_sonu"] = _n_telafi
    row["resmi_tatil_calisma"] = _n_rt
    row["izin_gun"] = _n_izin
    row["off_gun"] = _n_off
    row["off_t_talep"] = _n_off_t
    row["off_t_karsilanmayan"] = _n_off_t_ihlal
    row["off_t2_talep"] = _n_off_t2
    row["temmuz_calisma"] = _n_temmuz_calisma

    takvim_rows.append(row)

takvim_df = pd.DataFrame(
    takvim_rows,
    columns=_kimlik_kolonlari + _ozet_kolonlari + _temmuz_kolonlari + _gun_kolonlari,
)

print("TAKVİM tablosu hazır:", takvim_df.shape)
print("  toplam çalışma       :", int(takvim_df["toplam_calisma"].sum()))
print("  NORM hafta sonu      :", int(takvim_df["norm_hafta_sonu"].sum()))
print("  TELAFİ hafta sonu    :", int(takvim_df["telafi_hafta_sonu"].sum()))
print("  MESAİ hafta sonu     :", int(takvim_df["mesai_hafta_sonu"].sum()))
print("  resmi tatil çalışma  :", int(takvim_df["resmi_tatil_calisma"].sum()))
print("  karşılanmayan off_t  :", int(takvim_df["off_t_karsilanmayan"].sum()))

# ------------------------------------------------------- AGENT ÖZET TABLOSU
# "21 iş günü kotasını doldurdu mu?" sorusunun cevabı.
# Kota = baz_gereken_calisma = aylık iş günü - izinli iş günü.
# Karşılaştırma resmi tatil HARİÇ çalışma ile yapılır ([HAZIRLIK/KISIT] - AYLIK
# ÇALIŞMA DENGESİ hücresi de resmi tatil günlerini toplama saymıyor).

_ozet_rows = []
for _r in takvim_rows:
    a = _r["agent_user_code"]
    _baz = int(_baz_map_x.get(a, 0))
    _sayilan = int(_r["toplam_calisma"]) - int(_r["resmi_tatil_calisma"])
    _fark = _sayilan - _baz
    if _fark == 0:
        _durum = "TAM"
    elif _fark > 0:
        _durum = f"FAZLA (+{_fark} mesai günü)"
    else:
        _durum = f"EKSİK ({_fark})"
    _sat = {c: _r.get(c, "") for c in _kimlik_kolonlari}
    _sat.update({
        "aylik_is_gunu": _aylik_is_gunu_x,
        "izin_is_gunu": int(_izin_map_x.get(a, 0)),
        "kota_baz_gereken_calisma": _baz,
        "sayilan_calisma": _sayilan,
        "kota_farki": _fark,
        "kota_durumu": _durum,
        "toplam_calisma": int(_r["toplam_calisma"]),
        "hafta_ici_calisma": int(_r["hafta_ici_calisma"]),
        "norm_hafta_sonu": int(_r["norm_hafta_sonu"]),
        "telafi_hafta_sonu": int(_r["telafi_hafta_sonu"]),

_atanan_sayac = {}
for (a, ds), v in cozum_atama.items():
    if cozum_calisma.get((a, ds), 0) == 1:
        _atanan_sayac[(ds, v)] = _atanan_sayac.get((ds, v), 0) + 1

_cov_lower = globals().get("coverage_lower", {})
_cov_upper = globals().get("coverage_upper", {})
_gece_mi = globals().get("gece_aksam_vardiyasi_mi", {})
_ust_limit = globals().get("fazla_atama_ust_limiti", {})
_day_week_x = globals().get("day_week", {})

_cov_rows = []
for ds in PLAN_GUNLER:
    d = pd.to_datetime(ds).date()
    for v in gun_vardiyalari.get(ds, []):
        _req = int(talep[(ds, v)])
        _asg = int(_atanan_sayac.get((ds, v), 0))
        _st = _saat.get((ds, v), ("", ""))
        _cov_rows.append({
            "tarih": d.strftime("%Y-%m-%d"),
            "gun": _GUN_ADI[d.weekday()],
            "hafta": _day_week_x.get(ds, ""),
            "vardiya": v,
            "baslangic": str(_st[0]).strip()[:5],
            "bitis": str(_st[1]).strip()[:5],
            "gece_aksam_mi": bool(_gece_mi.get((ds, v), False)),
            "talep": _req,
            "atanan": _asg,
            "fark": _asg - _req,
            "karsilanma_yuzde": round(100 * _asg / _req, 1) if _req else None,
            "buffer_alt": _cov_lower.get((ds, v)),
            "buffer_ust": _cov_upper.get((ds, v)),
            "buffer_altinda": (_asg < _cov_lower[(ds, v)]) if (ds, v) in _cov_lower else None,
            "buffer_ustunde": (_asg > _cov_upper[(ds, v)]) if (ds, v) in _cov_upper else None,
            "fazla_atama_ust_limiti": _ust_limit.get((ds, v)),
        })
