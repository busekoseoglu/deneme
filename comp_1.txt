# %% [KONTROL 12] - HAFTA İÇİ TAKIM BASE VARDİYASI (HARD)
# Hafta içi (Pzt-Cuma), arife/resmi tatil hariç: agent yalnızca takımının seçtiği base'de çalışır.
# 2026-08-12: takim_muaf[(a, wk)] = 1 olan agent-haftalar İHLAL DEĞİL - base o agente
# fiziksel olarak yasak olduğu için muaf tutulmuştur (sabah çalışan / kıdemsiz).
#
# 2026-08-31: SARKAN HAFTADA KİŞİ BAZLI DEVİR ALAN AGENTLAR DA İHLAL DEĞİL.
# [HÜCRE] - TAKIM HAFTALIK BASE VARDİYA içinde bu agentlar için takım kilidi
# HİÇ KURULMUYOR; agent kendi devraldığı vardiyaya bağlanıyor ve o vardiya
# tanımı gereği takım base'inden FARKLI. Bu hücre kuralı tanımadığı için onları
# "GERÇEK İHLAL" sayıyordu; kaydet() varsayılanı HARD olduğundan Excel
# KURAL_KONTROL sayfasında kırmızı, olmayan bir ihlal görünüyordu.
# Çıktı özet; detay Excel'de.
ozel_tatil = set(globals().get("ozel_tatil_plan_gunleri", []))
_muaf_map = globals().get("takim_muaf", {})
_muaf_aktif = {(a, wk) for (a, wk), var in _muaf_map.items() if V(var) == 1}

# Devir sözlüğü yoksa (kural kapalı / hücre çalışmamış) boş kalır ve bu hücre
# 31 Ağustos öncesiyle birebir aynı davranır.
_devir_map = dict(globals().get("SARKAN_DEVIR_AGENT", {}) or {})
_SWK = globals().get("SARKAN_WK")

secilen_base = {}
for (t, wk, v), var in team_week_base.items():
    if V(var) == 1:
        secilen_base[(t, wk)] = v

_ihlal_rows, _muaf_rows, _devir_rows = [], [], []
for a_raw in AGENTS:
    a = str(a_raw).strip()
    t = agent_team.get(a)
    if t is None: continue
    t = str(t).strip()
    for ds in PLAN_GUNLER:
        if ds in ozel_tatil: continue
        if pd.to_datetime(ds).weekday() >= 5: continue
        v = cozum_atama.get((a, ds))
        if not v: continue
        wk = day_week[ds]
        base = secilen_base.get((t, wk))
        if base is None or v == base: continue
        _row = {"takim": t, "hafta": wk, "agent_user_code": a,
                "tarih": pd.to_datetime(ds).strftime("%Y-%m-%d"),
                "atanan_vardiya": v, "takim_base": base}
        if _SWK is not None and wk == _SWK and a in _devir_map:
            # ay geçişi devri: takım kilidi bu agente uygulanmıyor -> ihlal DEĞİL
            _dv = _devir_map[a]
            _row["devralinan_vardiya"] = _dv
            _row["devir_durumu"] = ("devraldığı vardiyada"
                                    if v == _dv else "SAPTI (sarkan_kisi_sapma cezası)")
            _devir_rows.append(_row)
        elif (a, wk) in _muaf_aktif:
            _muaf_rows.append(_row)
        else:
            _ihlal_rows.append(_row)

takim_base_ihlal_df = pd.DataFrame(_ihlal_rows)
takim_base_muaf_df = pd.DataFrame(_muaf_rows)
takim_base_devir_df = pd.DataFrame(_devir_rows)

kaydet("Hafta içi takım base vardiyası", len(_ihlal_rows))
print(f"   muafiyet kullanılan agent-hafta : {len(_muaf_aktif)}")
print(f"   muafiyetli sapma (agent-gün)    : {len(_muaf_rows)}")
print(f"   ay geçişi devri sapma (agent-gün): {len(_devir_rows)}  <- İHLAL DEĞİL")

# ===== 2026-08-31 - AY GEÇİŞİ: DEVİR AGENTLARI UYUM RAPORU =====
# Devir agentı sarkan hafta boyunca devraldığı TEK vardiyada kalmalı. Kural SOFT:
# başka kural zorlarsa sapar, sarkan_kisi_sapma cezası ödenir, model DURMAZ.
# Bu yüzden RAPOR - HARD sayacına girmez.
_devir_uyum_rows = []
if _devir_map and _SWK is not None:
    for _a5, _dv5 in sorted(_devir_map.items()):
        _vs5, _gun5 = set(), 0
        for ds in PLAN_GUNLER:
            if ds in ozel_tatil:
                continue
            if pd.to_datetime(ds).weekday() >= 5:
                continue
            if day_week.get(ds) != _SWK:
                continue
            _v5 = cozum_atama.get((_a5, ds))
            if not _v5 or cozum_calisma.get((_a5, ds), 0) != 1:
                continue
            _gun5 += 1
            _vs5.add(_v5)
        if _gun5 == 0:
            _durum5 = "sarkan haftada hiç çalışmamış"
        elif _vs5 == {_dv5}:
            _durum5 = "UYDU - hafta boyu devraldığı vardiyada"
        elif _dv5 in _vs5:
            _durum5 = "KISMEN - bazı günler başka vardiyada"
        else:
            _durum5 = "SAPTI - devraldığı vardiyada hiç çalışmamış"
        _devir_uyum_rows.append({
            "agent_user_code": _a5,
            "takim": str(agent_team.get(_a5, "")).strip(),
            "devralinan_vardiya": _dv5,
            "calisilan_gun": _gun5,
            "farkli_vardiya": len(_vs5),
            "vardiyalar": ", ".join(sorted(str(_x) for _x in _vs5)),
            "durum": _durum5,
        })

sarkan_devir_uyum_df = pd.DataFrame(_devir_uyum_rows)
_devir_sapan = int(sum(1 for _r5 in _devir_uyum_rows
                       if _r5["durum"].startswith(("SAPTI", "KISMEN"))))
kaydet("Ay geçişi - kişi devri sapması [SOFT]", _devir_sapan, tur="RAPOR")
if _devir_uyum_rows:
    print(f"   devir alan agent                : {len(_devir_uyum_rows)}"
          f" | sarkan hafta: {_SWK}")
    print(sarkan_devir_uyum_df["durum"].value_counts().to_string())
    if _devir_sapan:
        display(sarkan_devir_uyum_df[
            sarkan_devir_uyum_df["durum"].str.startswith(("SAPTI", "KISMEN"))].head(20))
elif _devir_map:
    print("   devir sözlüğü dolu ama sarkan hafta bulunamadı - SARKAN_WK yok.")
else:
    print("   ay geçişi kişi devri YOK (kural kapalı ya da devir alan agent çıkmadı)")

# ===== 2026-08-19 - MUAF AGENT HAFTA BOYU TEK VARDIYADA MI? =====

if _ihlal_rows:
    print("\n   !!! GERÇEK İHLAL - takım x vardiya (adet):")

# %% [KONTROL] - TAKIM VARDİYA DAĞILIMI (HAFTA İÇİ / HAFTA SONU) [RAPOR]
# Her takım hafta içi ve hafta sonu kaç FARKLI vardiyada çalışmış?
#
# Beklenti: hafta içi takım-hafta başına TEK vardiya olmalı
# ([HÜCRE] - TAKIM HAFTALIK BASE VARDİYA: HAFTA İÇİ HARD / HAFTA SONU SERBEST).
# Hafta sonu kural yok, dağılma normal.
#
# Hafta içi bölünme varsa sebebi de yazılır:
#   - arife / resmi tatil günü      -> takım base kuralı o günleri zaten atlıyor
#   - sabah çalışan agent           -> takımın base vardiyasına giremiyor
#   - ay geçişi kişi bazlı devri    -> sarkan haftada takım kilidi HİÇ kurulmuyor
#   - takım-hafta base seçilmemiş   -> o hafta base yok
# Bu hücre modele DOKUNMAZ.

# ---------------------------------------------------------------- yardımcılar
_ozel_tatil_t = set(globals().get("ozel_tatil_plan_gunleri", []))
_ozel_tatil_date_t = {pd.to_datetime(d).date() for d in _ozel_tatil_t}

_dw_t = globals().get("day_week")
if _dw_t is None:
    _dw_t = {}
    for ds in PLAN_GUNLER:
        _iso = pd.to_datetime(ds).isocalendar()
        _dw_t[ds] = f"{int(_iso.year)}-W{int(_iso.week):02d}"

_at_t = globals().get("agent_team")
if _at_t is None:
    _at_t = dict(zip(
        df_tam["agent_user_code"].astype(str).str.strip(),
        df_tam["takim"].astype(str).str.strip(),
    ))


def _vt(ds, v):
    """Vardiyayı okunur saat aralığına çevirir."""
    st = globals().get("saat", {}).get((ds, v))
    if st:
        return f"{str(st[0]).strip()[:5]}-{str(st[1]).strip()[:5]}"
    return str(v)


# --- agent özel bayrakları ---
_FLAG_KOL = ["sabah_calisir_flg", "hamile_flg", "sut_izni_flg",
             "mesaiye_kalamaz_flg", "kismi_calisan_flg"]
_flag_var = [c for c in _FLAG_KOL if c in df_tam.columns]
_agent_flag = {}
for _, _r in df_tam.iterrows():
    _a = str(_r["agent_user_code"]).strip()
    _agent_flag[_a] = {
        c: int(pd.to_numeric(_r.get(c, 0), errors="coerce") or 0) for c in _flag_var
    }


def _ozel_etiket(a):
    _f = _agent_flag.get(a, {})
    _et = [c.replace("_flg", "") for c in _flag_var if _f.get(c, 0) == 1]
    return ", ".join(_et)


# --- sabah çalışan için vardiya uygunluğu (hücre içinde yeniden türetilir) ---
_S_BITIS = globals().get("SABAH_CALISIR_MAX_BITIS_SAATI")
_S_BASLANGIC = globals().get("SABAH_CALISIR_MAX_BASLANGIC_SAATI")
_S_GECE_YASAK = globals().get("SABAH_CALISIR_GECE_VARDIYA_YASAK", False)
_S_GECE_SET = globals().get("GECE_AKSAM_VARDIYA_SET", set())


def _t_dk(t):
    h, m = str(t).strip()[:5].split(":")
    return int(h) * 60 + int(m)


def _sabah_uygun(ds, v):
    """Sabah çalışan agent bu vardiyada çalışabilir mi? Bilinmiyorsa None."""
    _st = globals().get("saat", {}).get((ds, v))
    if _st is None or _S_BITIS is None:
        return None
    b, e = str(_st[0]).strip()[:5], str(_st[1]).strip()[:5]
    bd, ed = _t_dk(b), _t_dk(e)
    if ed < bd:
        ed += 24 * 60
    if _S_GECE_YASAK and (b, e) in _S_GECE_SET:
        return False
    if _S_BASLANGIC is not None and bd > _t_dk(_S_BASLANGIC):
        return False
    return ed <= _t_dk(_S_BITIS)


# --- kıdemsiz agentlar (gece vardiyasına giremezler) ---
_kd_map_t = globals().get("kidem_ay_map", {})
_kd_min_t = globals().get("KIDEM_MIN_AY")
_kd_vars_t = float(globals().get("KIDEM_EKSIK_VARSAYILAN_AY", 0))
_kd_bas_t = {str(b).strip()[:5] for b in globals().get("KIDEM_YASAK_VARDIYA_BASLANGIC", ())}
_kidemsiz_t = set()
if _kd_min_t is not None:
    for _a0 in AGENTS:
        _a0 = str(_a0).strip()
        _v0 = _kd_map_t.get(_a0)
        _v0 = _kd_vars_t if (_v0 is None or pd.isna(_v0)) else float(_v0)
        if _v0 < float(_kd_min_t):
            _kidemsiz_t.add(_a0)

# --- takım base muafiyeti kullanılan agent-haftalar ---
_muaf_aktif_t = set()
_muaf_map_t = globals().get("takim_muaf", {})
if _muaf_map_t and "solver" in globals():
    try:
        _muaf_aktif_t = {k for k, var in _muaf_map_t.items() if solver.Value(var) == 1}
    except Exception as _e:
        print("takim_muaf okunamadı:", _e)


# --- 2026-08-30 EKLENEN - AY GEÇİŞİ: kişi bazlı vardiya devri (28 Ağustos kuralı) ---
# Sarkan haftada devir alan agent için takım kilidi HİÇ kurulmaz; agent kendi
# devraldığı vardiyaya bağlanır ([HÜCRE] - TAKIM HAFTALIK BASE VARDİYA, SOFT).
# Devir listesine ancak takımından FARKLI vardiya devredenler girdiği için bu
# agentlar tanımı gereği base'den sapmış görünür. Bu dal olmadan hepsi
# "sebep bulunamadı" kovasına düşüyordu (rapor 19 Ağu, kural 28 Ağu tarihli).
_sarkan_devir_t = dict(globals().get("SARKAN_DEVIR_AGENT", {}) or {})
_SARKAN_WK_T = globals().get("SARKAN_WK")


def _kidem_yasak_mi(ds, v):
    """Kıdemsiz agent bu vardiyada çalışamaz mı?"""
    _st = globals().get("saat", {}).get((ds, v))
    if _st is None or not _kd_bas_t:
        return False
    return str(_st[0]).strip()[:5] in _kd_bas_t


# --- takım-hafta seçilen base vardiya ---
_secilen_base = {}
if "team_week_base" in globals() and "solver" in globals():
    try:
        for (_t, _wk, _v), _var in team_week_base.items():
            if solver.Value(_var) == 1:
                _secilen_base[(_t, _wk)] = _v
    except Exception as _e:
        print("team_week_base okunamadı, base en sık vardiyadan türetilecek:", _e)

# ------------------------------------------------- takım-hafta bazında toplama
_th = {}   # (takim, hafta) -> dict
for a_raw in AGENTS:
    a = str(a_raw).strip()
    t = _at_t.get(a)
    if t is None or (isinstance(t, float) and pd.isna(t)):
        continue
    t = str(t).strip()
    for ds in PLAN_GUNLER:
        v = cozum_atama.get((a, ds))
        if not v or cozum_calisma.get((a, ds), 0) != 1:
            continue
        _wk = _dw_t[ds]
        _key = (t, _wk)
        _d = _th.setdefault(_key, {
            "hi_vardiya": {}, "hs_vardiya": {}, "hi_agent": set(), "hs_agent": set(),
            "hi_satir": [],
        })
        _hafta_ici = pd.to_datetime(ds).weekday() in (0, 1, 2, 3, 4)
        _vk = _vt(ds, v)
        if _hafta_ici:
            _d["hi_vardiya"][_vk] = _d["hi_vardiya"].get(_vk, 0) + 1
            _d["hi_agent"].add(a)
            _d["hi_satir"].append((a, ds, v, _vk))
        else:
            _d["hs_vardiya"][_vk] = _d["hs_vardiya"].get(_vk, 0) + 1
            _d["hs_agent"].add(a)

# ------------------------------------------------------------ 1) HAFTA TABLOSU
_hafta_rows = []
_sapma_rows = []
for (t, wk), d in sorted(_th.items()):
    _base_v = _secilen_base.get((t, wk))
    if _base_v is None and d["hi_vardiya"]:
        _base_v = max(d["hi_vardiya"].items(), key=lambda kv: kv[1])[0]
        _base_kaynak = "en sık (base okunamadı)"
        _base_kod = None
    else:
        _base_kaynak = "team_week_base" if _secilen_base else "yok"
        _base_kod = _base_v

    _base_metni = _base_v
    if _base_kod is not None:
        for (a2, ds2, v2, vk2) in d["hi_satir"]:
            if v2 == _base_kod:
                _base_metni = vk2
                break

    _hafta_rows.append({
        "takim": t, "hafta": wk,
        "hafta_ici_farkli_vardiya": len(d["hi_vardiya"]),
        "hafta_ici_base": _base_metni,
        "hafta_ici_vardiyalar": ", ".join(
            f"{k} ({v})" for k, v in sorted(d["hi_vardiya"].items(), key=lambda kv: -kv[1])
        ),
        "hafta_ici_calisan_agent": len(d["hi_agent"]),
        "hafta_sonu_farkli_vardiya": len(d["hs_vardiya"]),
        "hafta_sonu_vardiyalar": ", ".join(
            f"{k} ({v})" for k, v in sorted(d["hs_vardiya"].items(), key=lambda kv: -kv[1])
        ),
        "hafta_sonu_calisan_agent": len(d["hs_agent"]),
        "base_kaynagi": _base_kaynak,
    })

    # hafta içi base'den sapanlar
    if len(d["hi_vardiya"]) > 1:
        for (a2, ds2, v2, vk2) in d["hi_satir"]:
            if vk2 == _base_metni:
                continue
            _ds_date = pd.to_datetime(ds2).date()
            _sabah_engel = (
                _agent_flag.get(a2, {}).get("sabah_calisir_flg", 0) == 1
                and _base_kod is not None and _sabah_uygun(ds2, _base_kod) is False
            )
            _kidem_engel = (
                a2 in _kidemsiz_t and _base_kod is not None and _kidem_yasak_mi(ds2, _base_kod)
            )
            # ay geçişi: bu agent sarkan haftada kendi devraldığı vardiyaya bağlı mı?
            _devir_v = (_sarkan_devir_t.get(a2)
                        if (_SARKAN_WK_T is not None and wk == _SARKAN_WK_T) else None)
            _devir_vk = _vt(ds2, _devir_v) if _devir_v is not None else None
            if ds2 in _ozel_tatil_t or _ds_date in _ozel_tatil_date_t:
                _sebep = "arife / resmi tatil günü (takım kuralı dışı)"
            elif _sabah_engel and _kidem_engel:
                _sebep = "MUAFİYET - sabah çalışan + kıdemsiz, base yasak"
            elif _sabah_engel:
                _sebep = "MUAFİYET - sabah çalışan, base vardiyasına giremiyor"
            elif _kidem_engel:
                _sebep = "MUAFİYET - kıdemsiz, base gece vardiyası"
            elif _devir_v is not None and vk2 == _devir_vk:
                _sebep = "AY GEÇİŞİ - kişi bazlı devir (devraldığı vardiya)"
            elif _devir_v is not None:
                _sebep = "AY GEÇİŞİ - kişi devri SAPTI (ceza ödendi)"
            elif (a2, wk) in _muaf_aktif_t:
                _sebep = "MUAFİYET - takim_muaf aktif (sebep eşleştirilemedi)"
            elif _base_kod is None:
                _sebep = "takım-hafta base seçilmemiş"
            elif _ozel_etiket(a2):
                _sebep = f"özel agent ({_ozel_etiket(a2)})"
            else:
                _sebep = "sebep bulunamadı - incele"
            _sapma_rows.append({
                "takim": t, "hafta": wk,
                "agent_user_code": a2,
                "tarih": pd.to_datetime(ds2).strftime("%Y-%m-%d"),
                "gun": ["Pzt", "Sal", "Çar", "Per", "Cum", "Cmt", "Paz"][pd.to_datetime(ds2).weekday()],
                "atanan_vardiya": vk2,
                "takim_base": _base_metni,
                "ozel_bayraklar": _ozel_etiket(a2),
                "kidemsiz": a2 in _kidemsiz_t,
                "takim_muaf_aktif": (a2, wk) in _muaf_aktif_t,
                "sarkan_devir": _devir_vk or "",
                "sebep": _sebep,
            })

takim_vardiya_hafta_df = pd.DataFrame(_hafta_rows)
takim_vardiya_sapma_df = pd.DataFrame(_sapma_rows)

# ------------------------------------------------------------- 2) TAKIM ÖZETİ
_takim_agent = {}
for a_raw in AGENTS:
    a = str(a_raw).strip()
    t = _at_t.get(a)
    if t is None or (isinstance(t, float) and pd.isna(t)):
        continue
    _takim_agent.setdefault(str(t).strip(), []).append(a)

_ozet_rows = []
for t, agentlar in sorted(_takim_agent.items()):
    _hi_set, _hs_set, _hafta_sayisi, _bolunen_hafta = set(), set(), 0, 0
    for (t2, wk), d in _th.items():
        if t2 != t:
            continue
        _hafta_sayisi += 1
        _hi_set |= set(d["hi_vardiya"].keys())
        _hs_set |= set(d["hs_vardiya"].keys())
        if len(d["hi_vardiya"]) > 1:
            _bolunen_hafta += 1
    _row = {

        _kural_df.to_excel(writer, sheet_name="KURAL_KONTROL", index=False)
    for _ad, _df in _ham_sheetler:
        if isinstance(_df, pd.DataFrame) and not _df.empty:
            _df.to_excel(writer, sheet_name=_ad[:31], index=False)
        else:
            print(f"  ! {_ad} yok/boş - sheet atlandı")

    # ---------------- TAKVIM biçimlendirme
    ws = writer.sheets["TAKVIM"]
    _kol_idx = {c: i + 1 for i, c in enumerate(takvim_df.columns)}
    _ince = Side(style="thin", color="BBBBBB")
    _kenar = Border(left=_ince, right=_ince, top=_ince, bottom=_ince)

    for c in takvim_df.columns:
        _h = ws.cell(row=1, column=_kol_idx[c])
        _h.fill = PatternFill("solid", fgColor=RENK["baslik"])
        _h.font = Font(bold=True, size=9)
        _h.alignment = Alignment(horizontal="center", vertical="center", wrap_text=True)

    for (_i, _kol), _renk_key in hucre_renk.items():
        _c = ws.cell(row=_i + 2, column=_kol_idx[_kol])
        _c.fill = PatternFill("solid", fgColor=RENK[_renk_key])

    for _kol in _gecmis_kolonlari:
        _h = ws.cell(row=1, column=_kol_idx[_kol])
        _h.fill = PatternFill("solid", fgColor=RENK["baslik_gec"])

    for _r in range(2, len(takvim_df) + 2):
        for _kol in _gecmis_kolonlari + _gun_kolonlari:
            _c = ws.cell(row=_r, column=_kol_idx[_kol])
            _c.alignment = Alignment(horizontal="center", vertical="center", wrap_text=True)
            _c.font = Font(size=8)
            _c.border = _kenar

    for c in _kimlik_kolonlari:
        ws.column_dimensions[get_column_letter(_kol_idx[c])].width = 26
    for c in _ozet_kolonlari:
        ws.column_dimensions[get_column_letter(_kol_idx[c])].width = 11
    for c in _gecmis_kolonlari + _gun_kolonlari:
        ws.column_dimensions[get_column_letter(_kol_idx[c])].width = 13
    ws.row_dimensions[1].height = 32

    # Sadece bu kolonlar sabit kalsın; kalan kimlik/bayrak kolonları kayabilsin.
    _SABIT_KOLONLAR = ["agent_user_code", "agent_name", "working_main_group", "takim"]
    _dondurulacak = 0
    for _c in takvim_df.columns:
        if _c in _SABIT_KOLONLAR:
            _dondurulacak += 1
        else:
            break
    ws.freeze_panes = ws.cell(row=2, column=_dondurulacak + 1)
    print(f"TAKVIM dondurulan kolon: {_dondurulacak} "
          f"({', '.join(list(takvim_df.columns)[:_dondurulacak])})")
    ws.auto_filter.ref = f"A1:{get_column_letter(len(takvim_df.columns))}1"

    # ---------------- LEJANT renk örnekleri
    ws_l = writer.sheets["LEJANT"]
    ws_l.column_dimensions["A"].width = 24
    ws_l.column_dimensions["B"].width = 70
    ws_l.cell(row=1, column=1).font = Font(bold=True)
    ws_l.cell(row=1, column=2).font = Font(bold=True)
    for _j, (_g, _renk_key, _ac) in enumerate(LEJANT):
        _c = ws_l.cell(row=_j + 2, column=1)
        if _renk_key:
            _c.fill = PatternFill("solid", fgColor=RENK[_renk_key])
        _c.alignment = Alignment(horizontal="center")

    # ---------------- AGENT_OZET / COVERAGE biçimlendirme
    for _sh, _df_sh in (("AGENT_OZET", agent_ozet_df), ("COVERAGE", coverage_df),
                        ("COVERAGE_GUNLUK", coverage_gunluk_df)):
        _w = writer.sheets[_sh]
        for _ci, _cn in enumerate(_df_sh.columns, start=1):
            _hc = _w.cell(row=1, column=_ci)
            _hc.fill = PatternFill("solid", fgColor=RENK["baslik"])
            _hc.font = Font(bold=True, size=9)
            _hc.alignment = Alignment(horizontal="center", vertical="center", wrap_text=True)
            _w.column_dimensions[get_column_letter(_ci)].width = max(12, min(30, len(str(_cn)) + 4))
        _w.freeze_panes = "A2"
        _w.auto_filter.ref = f"A1:{get_column_letter(len(_df_sh.columns))}1"
        _w.row_dimensions[1].height = 30

    # AGENT_OZET: kotayı tutturmayanları boya
    _wo = writer.sheets["AGENT_OZET"]
    _kf_idx = list(agent_ozet_df.columns).index("kota_farki") + 1
    _kd_idx = list(agent_ozet_df.columns).index("kota_durumu") + 1
    for _j, _kf in enumerate(agent_ozet_df["kota_farki"].tolist()):
        if _kf != 0:
            _fill = PatternFill("solid", fgColor=RENK["mesai"] if _kf > 0 else RENK["off_t_ihlal"])
            _wo.cell(row=_j + 2, column=_kf_idx).fill = _fill
            _wo.cell(row=_j + 2, column=_kd_idx).fill = _fill

    # COVERAGE: eksik/fazla atanan kovaları boya
    _wc = writer.sheets["COVERAGE"]
    _fk_idx = list(coverage_df.columns).index("fark") + 1
    for _j, _fk in enumerate(coverage_df["fark"].tolist()):
        if _fk < 0:
            _wc.cell(row=_j + 2, column=_fk_idx).fill = PatternFill("solid", fgColor=RENK["off_t_ihlal"])
        elif _fk > 0:
            _wc.cell(row=_j + 2, column=_fk_idx).fill = PatternFill("solid", fgColor=RENK["mesai"])

    # COVERAGE_GUNLUK: eksiği olan günleri boya
    _wg = writer.sheets["COVERAGE_GUNLUK"]
    _eg_idx = list(coverage_gunluk_df.columns).index("eksik_toplam") + 1
    _bk_idx = list(coverage_gunluk_df.columns).index("bos_kova") + 1
    for _j in range(len(coverage_gunluk_df)):
        if int(coverage_gunluk_df["eksik_toplam"].iloc[_j]) > 0:
            _wg.cell(row=_j + 2, column=_eg_idx).fill = PatternFill(
                "solid", fgColor=RENK["off_t_ihlal"])
        if int(coverage_gunluk_df["bos_kova"].iloc[_j]) > 0:
            _wg.cell(row=_j + 2, column=_bk_idx).fill = PatternFill(
                "solid", fgColor=RENK["off_t_ihlal"])

    # ---------------- KURAL_KONTROL renklendirme
    if "KURAL_KONTROL" in writer.sheets:
        ws_k = writer.sheets["KURAL_KONTROL"]
        ws_k.column_dimensions["A"].width = 55
        ws_k.column_dimensions["B"].width = 10
        ws_k.column_dimensions["C"].width = 10
        _kk = _kural_df.reset_index(drop=True)
        for _j in range(len(_kk)):
            _hard = str(_kk.loc[_j, "tur"]).strip().upper() == "HARD"
            _ihl = int(_kk.loc[_j, "ihlal"]) > 0
            if _hard and _ihl:
                for _cc in range(1, len(_kk.columns) + 1):
                    ws_k.cell(row=_j + 2, column=_cc).fill = PatternFill(
                        "solid", fgColor=RENK["hard_ihlal"]
                    )

print(f"\nExcel yazıldı: {EXCEL_DOSYA}")
print("Sheet'ler:", ", ".join(["TAKVIM", "AGENT_OZET", "COVERAGE",
                                 "COVERAGE_GUNLUK", "LEJANT"] +
      (["KURAL_KONTROL"] if isinstance(_kural_df, pd.DataFrame) and not _kural_df.empty else []) +
      [ad for ad, d in _ham_sheetler if isinstance(d, pd.DataFrame) and not d.empty]))
