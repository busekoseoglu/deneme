# %% [AYARLAR] - İNGİLİZCE BİRLEŞİK RAPOR (KİTLE + GOLD)
# 2026-11-13. Kitle ve gold Excel çıktılarını okur, ingilizceciler için birleşik 7/24 raporu üretir.
# Modellere DOKUNMAZ; iki run bittikten sonra çalıştırılır. Dosya yollarını elle ver.
KITLE_EXCEL = "/BURAYA/vardiya_plani_kitle_YYYYAAGG_SSDD.xlsx"
GOLD_EXCEL = "/BURAYA/vardiya_plani_gold_YYYYAAGG_SSDD.xlsx"
CIKTI_KLASOR = "."
INGILIZCE_MIN_KISI = 1          # yarım saatte en az kaç ingilizceci olmalı (modellerdeki CONFIG ile aynı)
GECE_PENCERE = ("00:00", "06:00")   # bu aralığı kapsayan vardiya "gece" sayılır (AGENT_LISTE gece haftası)




# %% [OKU] - TAKVIM + INGILIZCE_7_24 + COVERAGE
import re
import pandas as pd
from datetime import date, timedelta

_GUN = ["Pzt", "Sal", "Çar", "Per", "Cum", "Cmt", "Paz"]
_AY = ["Oca", "Şub", "Mar", "Nis", "May", "Haz", "Tem", "Ağu", "Eyl", "Eki", "Kas", "Ara"]
_SAAT_RE = re.compile(r"(\d{2}:\d{2})-(\d{2}:\d{2})")

def _oku(yol, kuyruk):
    xl = pd.ExcelFile(yol)
    for sh in ("TAKVIM", "COVERAGE"):
        if sh not in xl.sheet_names:
            raise SystemExit(f"{kuyruk}: '{sh}' sayfası yok -> {yol}")
    tk = pd.read_excel(xl, "TAKVIM")
    if "english_flg" not in tk.columns:
        raise SystemExit(f"{kuyruk}: TAKVIM'de english_flg kolonu yok -> [EXCEL] hücresinin güncel hâli alınmalı.")
    cov = pd.read_excel(xl, "COVERAGE")
    i724 = pd.read_excel(xl, "INGILIZCE_7_24") if "INGILIZCE_7_24" in xl.sheet_names else None
    tarih = pd.to_datetime(cov["tarih"]).dt.date
    bas = tarih.min()
    tk = tk[pd.to_numeric(tk["english_flg"], errors="coerce").fillna(0).astype(int) == 1].copy()
    tk["kuyruk"] = kuyruk
    print(f"{kuyruk}: {yol}\n   ay {bas:%Y-%m} | ingilizceci {len(tk)} | INGILIZCE_7_24 sayfası "
          f"{'var' if i724 is not None else 'YOK'}")
    return tk, (bas.year, bas.month), i724

_tk_k, _ay_k, _i724_k = _oku(KITLE_EXCEL, "kitle")
_tk_g, _ay_g, _i724_g = _oku(GOLD_EXCEL, "gold")
if _ay_k != _ay_g:
    raise SystemExit(f"Kitle ({_ay_k}) ve gold ({_ay_g}) farklı ayın planı - aynı ayın son run'larını ver.")
YIL, AY = _ay_k

def _gun_kolonlari(df):
    """Kolon adı -> tarih. Plan günü '5 Per', geçmiş gün '31 Eki Cmt'."""
    m = {}
    for c in df.columns:
        s = str(c).strip()
        p = re.fullmatch(r"(\d{1,2}) (" + "|".join(_GUN) + r")", s)
        if p:
            m[c] = date(YIL, AY, int(p.group(1)))
            continue
        g = re.fullmatch(r"(\d{1,2}) (" + "|".join(_AY) + r") (" + "|".join(_GUN) + r")", s)
        if g:
            _ay = _AY.index(g.group(2)) + 1
            m[c] = date(YIL - (1 if _ay > AY else 0), _ay, int(g.group(1)))
    return m

ing = pd.concat([_tk_k, _tk_g], ignore_index=True)
_kol_tarih = {**_gun_kolonlari(_tk_k), **_gun_kolonlari(_tk_g)}
PLAN_TARIH = sorted({d for d in _kol_tarih.values() if d.year == YIL and d.month == AY})
print(f"plan günü: {len(PLAN_TARIH)} | toplam ingilizceci: {len(ing)} "
      f"(kitle {len(_tk_k)}, gold {len(_tk_g)})")





# %% [HESAP] - YARIM SAAT KAPSAMA (kuyruk + lokasyon kırılımı)
def _dk(t):
    return int(t[:2]) * 60 + int(t[3:5])

def _aralik(hucre):
    """TAKVIM hücresinden (bas, bit) dakika; çalışmıyorsa None. 'MESAİ\\n09:00-18:00' de olur."""
    bul = _SAAT_RE.findall(str(hucre)) if pd.notna(hucre) else []
    if not bul:
        return None
    b, e = _dk(bul[-1][0]), _dk(bul[-1][1])
    return b, (e + 1440 if e <= b else e)

_ilk = PLAN_TARIH[0]
_idx = {d: i for i, d in enumerate(PLAN_TARIH)}
_lok = "lokasyon" if "lokasyon" in ing.columns else None
SLOT = [f"{k // 2:02d}:{30 * (k % 2):02d}" for k in range(48)]

kapsama = []        # (tarih, slot, kuyruk, lokasyon, agent)
plan_vardiya = []   # (agent, kuyruk, tarih, bas, bit)
for _, r in ing.iterrows():
    a, q = str(r["agent_user_code"]).strip(), r["kuyruk"]
    lk = str(r[_lok]).strip() if _lok else ""
    for c, d in _kol_tarih.items():
        if c not in r.index:
            continue
        ar = _aralik(r[c])
        if ar is None:
            continue
        b, e = ar
        if d in _idx:
            plan_vardiya.append((a, q, d, b, e))
        for gun_ofset in (0, 1):              # aynı gün + gece yarısını geçen kısım ertesi gün
            hedef = d + timedelta(days=gun_ofset)
            if hedef not in _idx:
                continue
            for k in range(48):
                s0 = k * 30 + 1440 * gun_ofset
                if b <= s0 and e >= s0 + 30:
                    kapsama.append((hedef, k, q, lk, a))

kap = pd.DataFrame(kapsama, columns=["tarih", "slot", "kuyruk", "lokasyon", "agent"]).drop_duplicates()
_say = kap.groupby(["tarih", "slot", "kuyruk"]).size().unstack("kuyruk", fill_value=0)
_tam = pd.MultiIndex.from_product([PLAN_TARIH, range(48)], names=["tarih", "slot"])
_say = _say.reindex(_tam, fill_value=0)
for q in ("kitle", "gold"):
    if q not in _say.columns:
        _say[q] = 0
_say["toplam"] = _say["kitle"] + _say["gold"]
print(f"kapsama satırı: {len(kap)} | boş yarım saat (toplam < {INGILIZCE_MIN_KISI}): "
      f"{int((_say['toplam'] < INGILIZCE_MIN_KISI).sum())}")

# --- sağlama: modelin kendi INGILIZCE_7_24 sayfasıyla aynı mı? ---
for q, i724 in (("kitle", _i724_k), ("gold", _i724_g)):
    if i724 is None:
        print(f"! {q}: INGILIZCE_7_24 sayfası yok, sağlama atlandı")
        continue
    i724 = i724.copy(); i724["tarih"] = pd.to_datetime(i724["tarih"]).dt.date
    _fark = 0
    for _, r in i724.iterrows():
        for k, s in enumerate(SLOT):
            if s in i724.columns and (r["tarih"], k) in _say.index:
                _fark += int(r[s]) != int(_say.loc[(r["tarih"], k), q])
    print(f"   sağlama {q}: model sayfasıyla farklı hücre {_fark} (0 olmalı)")







# %% [RAPOR] - SAYFALAR + EXCEL
from openpyxl.styles import PatternFill, Font, Alignment
from openpyxl.utils import get_column_letter
import os

def _pivot(seri):
    return seri.unstack("slot").reindex(columns=range(48)).set_axis(SLOT, axis=1)

def _araliklar(bos_k):
    """[3,4,5,9] -> '01:30-03:00, 04:30-05:00'"""
    out, i = [], 0
    while i < len(bos_k):
        j = i
        while j + 1 < len(bos_k) and bos_k[j + 1] == bos_k[j] + 1:
            j += 1
        e = bos_k[j] + 1
        out.append(f"{SLOT[bos_k[i]]}-{'24:00' if e == 48 else SLOT[e]}")
        i = j + 1
    return ", ".join(out)

# 1) BIRLESIK_7_24 - toplam ingilizceci
birlesik = _pivot(_say["toplam"])
birlesik.insert(0, "bos_saat", (birlesik < INGILIZCE_MIN_KISI).sum(axis=1) / 2)
birlesik.insert(1, "fazla_saat", (birlesik[SLOT] > INGILIZCE_MIN_KISI).sum(axis=1) / 2)

# 2) KAYNAK - kitle/gold
_kit, _gol = _pivot(_say["kitle"]), _pivot(_say["gold"])
kaynak = _kit.astype(str) + "/" + _gol.astype(str)

# 3) LOKASYON - her yarım saatte lokasyon -> kişi
_lk = (kap.groupby(["tarih", "slot", "lokasyon"]).size().reset_index(name="n")
       .assign(t=lambda d: d["lokasyon"] + ":" + d["n"].astype(str))
       .groupby(["tarih", "slot"])["t"].apply(", ".join))
lokasyon_df = _lk.reindex(_tam, fill_value="").pipe(_pivot)
lokasyon_ozet = (kap.groupby(["lokasyon", "kuyruk"]).size().div(2)
                 .rename("ingilizceci_kisi_saat").reset_index())

# 4) GUNLUK_OZET
_g = []
for d in PLAN_TARIH:
    t, k, g = (_say.loc[d, c].to_numpy() for c in ("toplam", "kitle", "gold"))
    bos = [i for i in range(48) if t[i] < INGILIZCE_MIN_KISI]
    _g.append({"tarih": d, "gun": _GUN[d.weekday()], "hafta_sonu": int(d.weekday() >= 5),
               "bos_saat": len(bos) / 2,
               "sadece_kitle_saat": int(((k >= INGILIZCE_MIN_KISI) & (g == 0)).sum()) / 2,
               "sadece_gold_saat": int(((g >= INGILIZCE_MIN_KISI) & (k == 0)).sum()) / 2,
               "ikisi_birlikte_saat": int(((k > 0) & (g > 0)).sum()) / 2,
               "fazla_saat": int((t > INGILIZCE_MIN_KISI).sum()) / 2,
               "max_ayni_anda": int(t.max()),
               "bos_araliklar": _araliklar(bos)})
gunluk_ozet = pd.DataFrame(_g)

# 5) SAAT_OZET
_s = _say.reset_index()
saat_ozet = (_s.groupby("slot").agg(bos_gun=("toplam", lambda x: int((x < INGILIZCE_MIN_KISI).sum())),
                                    ort_toplam=("toplam", "mean"), ort_kitle=("kitle", "mean"),
                                    ort_gold=("gold", "mean"), min_toplam=("toplam", "min"))
             .round(2).reset_index())
saat_ozet.insert(0, "saat", [SLOT[k] for k in saat_ozet["slot"]]); saat_ozet = saat_ozet.drop(columns="slot")

# 6) AGENT_LISTE
_pv = pd.DataFrame(plan_vardiya, columns=["agent", "kuyruk", "tarih", "bas", "bit"])
_pv["hafta"] = _pv["tarih"].map(lambda d: f"{d.isocalendar()[0]}-W{d.isocalendar()[1]:02d}")
_g0, _g1 = _dk(GECE_PENCERE[0]), _dk(GECE_PENCERE[1])
_pv["gece"] = _pv.apply(lambda r: (r["bas"] <= _g0 + 1440 and r["bit"] >= _g1 + 1440)
                        or (r["bas"] <= _g0 and r["bit"] >= _g1), axis=1)
_pv["saat"] = _pv.apply(lambda r: f"{r['bas'] // 60:02d}:{r['bas'] % 60:02d}", axis=1)
# takım çoğunluğu: takımın İNGİLİZCECİ OLMAYAN üyelerinin hafta içi en sık başlangıç saati (= takım base'i)
_tum = pd.concat([pd.read_excel(KITLE_EXCEL, "TAKVIM").assign(kuyruk="kitle"),
                  pd.read_excel(GOLD_EXCEL, "TAKVIM").assign(kuyruk="gold")], ignore_index=True)

for _, r in _tum[pd.to_numeric(_tum["english_flg"], errors="coerce").fillna(0).astype(int) != 1].iterrows():
    for c, d in _kol_tarih.items():
        if d in _idx and d.weekday() < 5 and c in r.index and _aralik(r[c]):
            b = _aralik(r[c])[0]
            _tc.append((r.get("takim"), f"{d.isocalendar()[0]}-W{d.isocalendar()[1]:02d}", f"{b // 60:02d}:{b % 60:02d}"))

takim_cogunluk = _tcd.groupby(["takim", "hafta"])["saat"].agg(lambda s: s.value_counts().idxmax()).to_dict()

_al = []
for _, r in ing.iterrows():

    _al.append({"kuyruk": r["kuyruk"], "agent_user_code": a, "agent_name": r.get("agent_name", ""),

                "mesaiye_kalamaz_flg": r.get("mesaiye_kalamaz_flg", ""), "sabah_calisir_flg": r.get("sabah_calisir_flg", ""),
                "calisma_gun": len(p), "hafta_sonu_gun": int(sum(d.weekday() >= 5 for d in p["tarih"])),
                "gece_gun": int(p["gece"].sum()), "gece_hafta": int(p.loc[p["gece"], "hafta"].nunique()),
                "takim_cogunlugundan_farkli_hafta": _farkli,
                "hafta_ici_vardiyalar": ", ".join(f"{wk[-3:]}:{st}" for wk, st in _hs.items())})
agent_liste = pd.DataFrame(_al)
