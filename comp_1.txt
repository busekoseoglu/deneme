def _tb_dk_fark(_t1, _t2):
    """İki saat arasındaki DAİRESEL dakika farkı: 23:30 ile 00:00 arası 30'dur."""
    _d = abs(_tb_dk(_t1) - _tb_dk(_t2))
    return min(_d, 1440 - _d)


def _saat_eslestir(_hedef, _aday_map, _tol):
    """(bas, bit) -> aday_map içindeki EN YAKIN vardiya kodu.

    Birebir eşleşme varsa o döner. Yoksa başlangıç ve bitiş sapmalarının
    ikisi de _tol içinde olan adaylar arasından toplam sapması en küçük
    olan seçilir; eşitlikte vardiya kodu alfabetik olarak küçük olan
    (çözüm runlar arası kaymasın diye). Aday yoksa (None, None) döner.
    Dönüş: (vardiya_kodu, toplam_sapma_dk)  - birebirse sapma 0.
    """
    if _hedef in _aday_map:
        return _aday_map[_hedef], 0
    if not _tol:
        return None, None
    _en_v, _en_s = None, None
    for (_ab, _ae), _av in _aday_map.items():
        _d1 = _tb_dk_fark(_ab, _hedef[0])
        _d2 = _tb_dk_fark(_ae, _hedef[1])
        if _d1 > _tol or _d2 > _tol:
            continue
        _s = _d1 + _d2
        if _en_s is None or _s < _en_s or (_s == _en_s and str(_av) < str(_en_v)):
            _en_v, _en_s = _av, _s
    return _en_v, _en_s

    """Agent a, wk haftasındaki v vardiyasında ÇALIŞAMAZ mı?"""
    _st = _tb_wk_saat.get((wk, v))
    if _st is None:
        return False
    b, e = _st
    if a in _tb_kidemsiz and b in _tb_kidem_bas:
        return True
    if a in _tb_sabah and _tb_sabah_bitis is not None:
        bd, ed = _tb_dk(b), _tb_dk(e)
        if ed < bd:
            ed += 24 * 60
        if _tb_gece_yasak and (b, e) in _tb_gece_set:
            return True
        if _tb_sabah_baslangic is not None and bd > _tb_dk(_tb_sabah_baslangic):
            return True
        if ed > _tb_dk(_tb_sabah_bitis):
            return True
    return False

# ===== 2026-08-31 - DEVIR AGENTI DA HAFTA BOYU TEK VARDIYADA =====
# 28 Agustos'taki hata: devir agenti icin takim kilidi kaldirildi ve yerine
# `x[(a,ds,v)] <= sarkan_kisi_sapma[a]` konuldu. Sapma degiskeni AGENT BASINA
# TEK oldugu icin bir kez 1 olunca kisit `x <= 1`e donusuyor, yani sarkan hafta
# boyunca agent icin HICBIR kisit kalmiyordu - her gun baska vardiyada

#   x[(a, ds, v)] <= devir_week_base[(a, v)]    -> hafta ici gunlerde
#   devir_week_base[(a, devraldigi)] + sapma >= 1 -> SOFT, tercihen devraldigi
#
# "Hafta boyu tek vardiya" HARD, "hangi vardiya" SOFT. Ay gecisi kurallarinin
# HARD olmama ilkesi korunur: model mecbur kalirsa sapar, ama TEK bir baska
# vardiyaya sapar - her gun baskasina degil.

        if not _uygun7 or _dv7 not in _uygun7:
            # havuz bos ya da devraldigi vardiya havuzda yok -> Sigma==1 kisiti
            # agenti calisamaz hale getirebilir. Infeasible olmasin diye bu agent
            # devir kapsamindan CIKARILIR, normal takim kilidine tabi olur.
            _devir_uygun_yok.append({

            model.Add(devir_week_base[(_a7, _dv7)] + sarkan_kisi_sapma[_a7] >= 1)
        _devir_base_kurulan.add(_a7)
