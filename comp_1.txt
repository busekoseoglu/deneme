# ============ 2026-09-07 - BİRİM DEĞİŞTİ: TAKIM-GÜN -> KİŞİ BAŞI GÜN ============
# ESKİ tablo haftalık BASE mantığıyla üretiliyordu: her (ISO hafta, takım) için
# en çok kişinin çalıştığı vardiya o haftanın base'i sayılır, base günleri
# toplanırdı. Birim TAKIM-GÜN'dü, takım büyüklüğünden bağımsızdı.
#
# YENİ tablo base'e hiç bakmıyor. Takımın bütün agentlarının o vardiya altında
# çalıştığı TOPLAM AGENT-GÜN alınıp TAKIM MEVCUDUNA BÖLÜNÜYOR. Yani:
#   "ben 10 gün 15:00'ta çalıştım, sen 5 gün" -> takım toplamı 15 agent-gün
#   takım 15 kişilikse -> 1.0 kişi başı gün
# Bölme İŞ BİRİMİ tarafında, veride yapılıyor; model bölmüyor.
#
# Bunun sonucu: "takımın base'i değildi ama ben orada çalıştım" artık sayılıyor.
# Muafiyetli ve devir alan agentlar da geçmişe dahil.
#
# BEKLENEN ŞEMA:
#   yil | rapor_bas | teamleader_name | working_main_group | shifts_adj |
#   total_gun_sayisi | agent_sayisi | gun_agent_ratio
#
#   total_gun_sayisi = HAM agent-gün (bölünmemiş)
#   agent_sayisi     = takımın agent sayısı (bölen)
#   gun_agent_ratio  = total_gun_sayisi / agent_sayisi  -> KİŞİ BAŞI GÜN
#
#   DENGE HESABI gun_agent_ratio'yu KULLANIR. total_gun_sayisi sadece tutarlılık
#   kontrolü ve raporlama için okunur; modele girmez.
#
#   rapor_bas = planlanan ay (Eylül planlanıyorsa 9). Tablo birden çok kesit
#   taşıyabilir; hangisinin kullanılacağı CONFIG'deki TAKIM_VARDIYA_GECMIS_KESIT
#   ile seçilir, None ise EN SON kesit otomatik alınır.
#
#   agent_sayisi = kişi başı hesabında kullanılan BÖLEN. Model bu ayın katkısını
#   da AYNI sayıya böler; iki taraf farklı bölen kullanırsa denge sessizce
#   yanlış ölçer. Bu yüzden bölen veriden gelir, notebook kendi saymaz.
# ================================================================================

def _tvg_bas_saat(s):
    """'15:00-00:00' -> '15:00'. Ayrıştıramazsa None döner (sessizce yutmaz, sayılır)."""
    t = str(s).strip()
    if not t or t.lower() in ("nan", "none", "null"):
        return None
    t = t.split("-")[0].strip().replace(".", ":")
    m = re.match(r"^(\d{1,2}):?(\d{2})$", t)
    if not m:
        return None
    _h, _dk = int(m.group(1)), int(m.group(2))
    if _h > 23 or _dk > 59:
        return None
    return f"{_h:02d}:{_dk:02d}"

# --- (takım, başlangıç saati) -> KİŞİ BAŞI gün (gun_agent_ratio, float) ---

# Oranları toplamak doğrudur: bir takımın bütün satırlarında bölen (agent_sayisi)
# aynı olduğu için Σ(gün_i / n) = (Σ gün_i) / n. Bölenin takım içinde sabit

# Ondalık korunur; tamsayıya çevirme [KISIT] hücresinde ölçekle yapılır.

# %% [KISIT] - TAKIM VARDİYA GEÇMİŞ DENGESİ (SOFT)
# Takımların geçmişte belirli bir vardiyada KİŞİ BAŞI kaç gün çalıştığına bakıp
# bu ayın dağıtımını dengeler. Amaç: yıl sonunda tüm takımların o vardiyadaki
# kişi başı çalışma günü birbirine yakın olsun - aynı kişiler hep aynı vardiyada
# kalmasın.
#
# ÖN KOŞUL: [HÜCRE 3D] - TAKIM VARDİYA GEÇMİŞİ (SQL)
#             -> takim_vardiya_gecmis_map (gun_agent_ratio toplamı, float)
#             -> takim_mevcut_map         (agent_sayisi = kişi başı bölen)
#           [HÜCRE 9] - CP-SAT MODEL      -> x
#           [HÜCRE] - TAKIM HAFTALIK BASE VARDİYA -> team_week_base (sadece tavan için)
#
# ============ 2026-09-07 - BİRİM DEĞİŞTİ: TAKIM-GÜN -> KİŞİ BAŞI GÜN ============
# ESKİ: gecmis ve bu_ay'ın ikisi de takım-gündü.
#         bu_ay[t] = Σ_hafta team_week_base[(t,wk,v)] × hafta içi gün
#       Takım büyüklüğü hesaba girmiyordu, normalizasyon yapılmıyordu.
#
# YENİ: geçmiş tablo gun_agent_ratio taşıyor = total_gun_sayisi / agent_sayisi,
#       yani kişi başı gün (bkz. [HÜCRE 3D]). Bu ayın katkısı
#       team_week_base'ten TÜRETİLEMEZ - base takım-gün üretir, kişi-gün değil.
#       Bu yüzden bu ay doğrudan x'ten sayılır:
#
#         agent_gun[t] = Σ_{a ∈ takım t} Σ_{hafta içi gün} Σ_{v: başlangıç = v*} x[(a,ds,v)]
#         bu_ay_kb[t]  = agent_gun[t] / agent_sayisi[t]        (kişi başı)
#
#       Bölen VERİDEN gelir (agent_sayisi kolonu). Geçmişteki gun_agent_ratio
#       hangi bölenle hesaplandıysa model de aynısını kullanır; notebook kendi
#       saymaz. [HÜCRE 3D] ratio == total / agent_sayisi tutuyor mu diye bakar.
#       team_size_map'e düşülürse iki taraf sessizce ayrışabilir - o yüzden
#       fallback olsa bile UYARI basılır.
#
# BUNUN İKİ SONUCU VAR:
#   1) Muafiyetli ve devir alan agentlar artık sayıya GİRİYOR. x'ten saydığımız
#      için takım base'ini takip etmeyenler de dahil - yeni verinin semantiği bu.
#   2) Denge artık x'i DOĞRUDAN itiyor (eskiden sadece base'i seçiyordu), yani
#      coverage ile daha sert yarışıyor. Ağırlık kalibrasyonu kritik.
# ================================================================================
#
# MATEMATİK (her denge vardiyası v* için ayrı ayrı):
#
#   kum[t] = gecmis_kb[t] + bu_ay_kb[t]          (ikisi de ×OLCEK ölçekli tamsayı)
#   N × kum[t] − Σ_t kum[t]  =  sapma_arti[t] − sapma_eksi[t]
#   ceza = TAKIM_VARDIYA_DENGE_W × Σ (sapma_arti + sapma_eksi)      [OBJECTIVE'de]
#
# N ile çarpım BÖLME YAPMAMAK içindir: ortalamadan sapmanın N katı, tamsayı kalır.
# Ceza ortalamadan UZAKLIKLA ORANTILI - sert bir eşik/kesme yok. Geçmişte kişi
# başı az almış takımın sapma_eksi'si büyük olduğu için model o vardiyayı ona
# vermeye meyleder; çok almışa vermemeye. Kimse sıfırlanmaz, herkes ortalamaya
# doğru itilir ("insafsız olmasın").
#
# ÖLÇEK: kişi başı gün ondalıklı, CP-SAT tamsayı ister. Her şey
# TAKIM_VARDIYA_DENGE_OLCEK ile çarpılır (100 -> 0.01 gün hassasiyet).
# Bölme taban bölmedir; hata takım başına en fazla 1/OLCEK gün.
#
# HAFTA İÇİ: geçmiş veri sadece hafta içi sayıyor, bu ay da öyle sayılır
# (TAKIM_VARDIYA_DENGE_HAFTA_SONU_DAHIL). Resmi tatil / arife hafta içiyse
# geçmişe girmiş -> bu ayda da SAYILIR. (Eski hücre onları hariç tutuyordu.)
#
# AYLIK TAVAN AYNI KALDI: hâlâ team_week_base üzerinden, HAFTA bazında.
# Tavan başka bir iş yapıyor - "takım ay boyu aynı vardiyada park etmesin",
# yani gündüz rotasyonunun kardeşi. Denge ise "yıl içinde kişi başı adalet".
# İkisi aynı değişkene bağlanırsa ikisi de bulanıklaşır.
#
# Vardiya BAŞLANGIÇ SAATİNDEN eşleşir, koddan değil (kodlar aydan aya değişir).
# Yeni vardiya dengelemek için CONFIG'deki tuple'a eklemek yeterli:
#     TAKIM_VARDIYA_DENGE_BASLANGICLARI = ("15:00", "00:00")
#
# 00:00 NOTU: AYDA MAX 2 HAFTA GECE/AKŞAM (HARD) kuralına dokunulmaz. Denge onun
# ALTINDA çalışır - tavanı zorlamaz, 2 haftalık hakkın hangi takımlara gideceğini
# belirler. Denge SOFT olduğu için çakışmada HARD her zaman kazanır.
#
# TAVAN NEDEN SOFT: [HÜCRE] - HAFTALIK TAKIM BASE KAPASİTE KORUMASI her hafta o
# vardiyaya yeterli takım seçilmesini ZORUNLU kılıyor. Tavan HARD olsaydı ikisi
# çelişip INFEASIBLE verebilirdi.

_tvd_gerekli = [
    "ENABLE_TAKIM_VARDIYA_DENGE", "TAKIM_VARDIYA_DENGE_BASLANGICLARI",
    "TAKIM_VARDIYA_DENGE_W", "TAKIM_VARDIYA_DENGE_MAX_HAFTA",
    "TAKIM_VARDIYA_DENGE_MAX_HAFTA_ASIM_W", "TAKIM_VARDIYA_GECMIS_EKSIK_VARSAYILAN",
    "TAKIM_VARDIYA_DENGE_OLCEK", "TAKIM_VARDIYA_DENGE_HAFTA_SONU_DAHIL",
]
_tvd_eksik = [k for k in _tvd_gerekli if k not in globals()]
if _tvd_eksik:
    raise SystemExit(
        f"CONFIG'de tanımlı olmalı: {_tvd_eksik} -> [CONFIG] hücresini çalıştır."
    )

takim_vardiya_denge_agentgun = {}  # (takim, baslangic) -> IntVar   bu ayki AGENT-gün
takim_vardiya_denge_buay = {}      # (takim, baslangic) -> IntVar   bu ayki kişi başı (ölçekli)
takim_vardiya_denge_hafta = {}     # (takim, baslangic) -> IntVar   bu ayki hafta (tavan için)
takim_vardiya_denge_sapma = {}     # (takim, baslangic) -> (arti, eksi)  OBJECTIVE okur
takim_vardiya_denge_asim = {}      # (takim, baslangic) -> IntVar   tavan aşımı, OBJECTIVE okur
takim_vardiya_denge_gecmis = {}    # (takim, baslangic) -> int      sabit geçmiş (ölçekli)
takim_vardiya_denge_mevcut = {}    # takim -> int                   kullanılan bölen
takim_vardiya_denge_rows = []

if not ENABLE_TAKIM_VARDIYA_DENGE:
    print("ENABLE_TAKIM_VARDIYA_DENGE = False -> takım vardiya geçmiş dengesi KURULMADI.")
    print("Hiçbir değişken yaratılmadı, objective'e ceza girmeyecek.")
else:
    _tvd_gecmis_map = globals().get("takim_vardiya_gecmis_map")
    if not _tvd_gecmis_map:
        raise SystemExit(
            "takim_vardiya_gecmis_map yok/boş. Önce [HÜCRE 3D] - TAKIM VARDİYA GEÇMİŞİ (SQL) "
            "hücresini çalıştır. (Dengeyi bilinçli olarak kapatmak için "
            "CONFIG'de ENABLE_TAKIM_VARDIYA_DENGE = False yap.)"
        )

    _tvd_olcek = int(TAKIM_VARDIYA_DENGE_OLCEK)
    _tvd_hs_dahil = bool(TAKIM_VARDIYA_DENGE_HAFTA_SONU_DAHIL)
    _tvd_arife_kod = globals().get("arife_ozel_vardiya_kodlari", set())
    _tvd_eksik_vars = float(TAKIM_VARDIYA_GECMIS_EKSIK_VARSAYILAN)

    # --- BÖLEN: veriden gelen agent_sayisi; yoksa team_size_map'e düşer ---
    _tvd_mevcut_map = globals().get("takim_mevcut_map") or {}
    _tvd_team_size = globals().get("team_size_map") or {}
    if not _tvd_mevcut_map:
        print("!!! UYARI: takim_mevcut_map YOK - kişi başı bölen team_size_map'ten alınacak.")
        print("    (takim_mevcut_map, [HÜCRE 3D]'de agent_sayisi kolonundan üretilir.)")
        print("    Bu, iş biriminin veride kullandığı bölenden FARKLI olabilir; farkı")
        print("    kimse görmez, denge sessizce yanlış ölçer. [HÜCRE 3D]'yi çalıştır.")

    # --- sayılan plan günleri: hafta içi (resmi tatil / arife DAHİL) ---
    _tvd_gunler = [
        ds for ds in PLAN_GUNLER
        if _tvd_hs_dahil or pd.to_datetime(ds).weekday() in (0, 1, 2, 3, 4)
    ]

    # --- (gün, vardiya) -> başlangıç saati ---
    _tvd_gun_bas = {}
    for _ds in _tvd_gunler:
        for _v in gun_vardiyalari.get(_ds, []):
            if _v in _tvd_arife_kod:
                continue
            if (_ds, _v) in saat:
                _tvd_gun_bas[(_ds, _v)] = str(saat[(_ds, _v)][0]).strip()[:5]

    # --- takım -> agent listesi ---
    _tvd_uyeler = {}
    for _a in AGENTS:
        _a = str(_a).strip()
        _t = agent_team.get(_a)
        if _t is None or pd.isna(_t):
            continue
        _tvd_uyeler.setdefault(str(_t).strip(), []).append(_a)

    _tvd_takimlar = sorted({str(t).strip() for t in TAKIMLAR})

    # --- geçmiş veri ile plan takımlarının eşleşmesi ---
    _tvd_gecmis_takimlar = {str(k[0]).strip() for k in _tvd_gecmis_map}
    _tvd_eslesmeyen = [t for t in _tvd_takimlar if t not in _tvd_gecmis_takimlar]
    _tvd_fazla = sorted(_tvd_gecmis_takimlar - set(_tvd_takimlar))
    if _tvd_eslesmeyen:
        print(f"!!! UYARI: plandaki {len(_tvd_eslesmeyen)} takımın geçmiş tabloda HİÇ kaydı yok "
              f"-> {_tvd_eksik_vars} kişi başı gün sayılacak "
              f"(TAKIM_VARDIYA_GECMIS_EKSIK_VARSAYILAN).")
        for _t in _tvd_eslesmeyen[:10]:
            print(f"      {_t}")
        if len(_tvd_eslesmeyen) > 10:
            print(f"      ... ve {len(_tvd_eslesmeyen) - 10} takım daha")
        print("    Takım adı 'working_main_group | teamleader_name' formatında eşleşir;")
        print("    eşleşmeyen çoksa geçmiş tablodaki yazımı kontrol et.")
    if _tvd_fazla:
        print(f"NOT: geçmiş tabloda olup planda OLMAYAN {len(_tvd_fazla)} takım var "
              f"(kapanmış/değişmiş ekipler) - denge hesabına girmiyorlar.")

    # --- her takım için BÖLEN belirle, iki kaynak arasındaki farkı raporla ---
    _tvd_bolen_fark = []
    _tvd_bolensiz = []
    for _t in _tvd_takimlar:
        _n_veri = _tvd_mevcut_map.get(_t)
        _n_model = _tvd_team_size.get(_t)
        if _n_model is None:
            _n_model = len(_tvd_uyeler.get(_t, []))
        _n = _n_veri if _n_veri else _n_model
        if not _n or int(_n) <= 0:
            _tvd_bolensiz.append(_t)
            continue
        takim_vardiya_denge_mevcut[_t] = int(_n)
        if _n_veri and _n_model and int(_n_veri) != int(_n_model):
            _tvd_bolen_fark.append({
                "takim": _t, "veriden_agent_sayisi": int(_n_veri),
                "modeldeki_mevcut": int(_n_model),
                "fark": int(_n_veri) - int(_n_model),
            })

    takim_vardiya_denge_bolen_fark_df = pd.DataFrame(_tvd_bolen_fark)
    if _tvd_bolen_fark:
        print(f"\n!!! DİKKAT: {len(_tvd_bolen_fark)} takımda veriden gelen agent_sayisi ile "
              f"modeldeki mevcut FARKLI.")
        print("    VERİDEN gelen kullanıldı (geçmiş kişi başı günler onunla hesaplandığı için).")
        print("    Fark büyükse geçmiş tablo eski bir kadro kesitiyle üretilmiş olabilir.")
        print(takim_vardiya_denge_bolen_fark_df
              .reindex(takim_vardiya_denge_bolen_fark_df["fark"].abs()
                       .sort_values(ascending=False).index)
              .head(10).to_string(index=False))
    if _tvd_bolensiz:
        print(f"\n!!! UYARI: {len(_tvd_bolensiz)} takımın mevcudu bulunamadı/0 "
              f"-> denge hesabına girmiyorlar: {_tvd_bolensiz[:5]}")

    print(f"\nSayılan plan günü: {len(_tvd_gunler)} "
          f"({'hafta sonu DAHİL' if _tvd_hs_dahil else 'sadece hafta içi'}, "
          f"resmi tatil/arife dahil)")
    print(f"Ölçek: ×{_tvd_olcek}  (kişi başı gün -> tamsayı, {1 / _tvd_olcek:g} gün hassasiyet)")

    # ================== HER DENGE VARDİYASI İÇİN AYRI HESAP ==================
    for _bas in TAKIM_VARDIYA_DENGE_BASLANGICLARI:
        _bas = str(_bas).strip()[:5]

        # bu başlangıçlı vardiyanın açık olduğu (gün, vardiya) çiftleri
        _tvd_gun_v = {}
        for (_ds, _v), _b in _tvd_gun_bas.items():
            if _b == _bas:
                _tvd_gun_v.setdefault(_ds, []).append(_v)

        if not _tvd_gun_v:
            print(f"\n!!! UYARI: '{_bas}' başlangıçlı vardiya sayılan günlerin HİÇBİRİNDE yok "
                  f"-> denge kurulmadı.")
            print(f"    Plandaki başlangıç saatleri: {sorted(set(_tvd_gun_bas.values()))}")
            continue

        # ---- takım başına agent-gün ve kişi başı değişkenleri ----
        _kum_ifadeler, _gecerli_t = [], []
        for _t in _tvd_takimlar:
            _n = takim_vardiya_denge_mevcut.get(_t)
            if not _n:
                continue
            _terms = []
            for _ds, _vler in _tvd_gun_v.items():
                for _a in _tvd_uyeler.get(_t, []):
                    for _v in _vler:
                        if (_a, _ds, _v) in x:
                            _terms.append(x[(_a, _ds, _v)])
            if not _terms:
                continue      # bu takımın hiçbir agentı o vardiyada çalışamıyor

            _ag_ust = len(_terms)
            _ag = model.NewIntVar(0, _ag_ust, f"tvd_agentgun_{_bas}_{_t}")
            model.Add(_ag == sum(_terms))
            takim_vardiya_denge_agentgun[(_t, _bas)] = _ag

            # kişi başı (ölçekli) = OLCEK * agent_gun / mevcut   -- TABAN BÖLME
            # _n sabit olduğu için iki kısıt da lineer kalır.
            _kb_ust = (_tvd_olcek * _ag_ust) // _n + 1
            _kb = model.NewIntVar(0, _kb_ust, f"tvd_buay_kb_{_bas}_{_t}")
            model.Add(_kb * _n <= _tvd_olcek * _ag)
            model.Add(_kb * _n > _tvd_olcek * _ag - _n)
            takim_vardiya_denge_buay[(_t, _bas)] = _kb

            _gec_ham = float(_tvd_gecmis_map.get((_t, _bas), _tvd_eksik_vars))
            _gec = int(round(_tvd_olcek * _gec_ham))
            takim_vardiya_denge_gecmis[(_t, _bas)] = _gec
            _kum_ifadeler.append(_gec + _kb)
            _gecerli_t.append(_t)

        if len(_gecerli_t) < 2:
            print(f"\n!!! UYARI: '{_bas}' için denge kurulacak takım sayısı {len(_gecerli_t)} "
                  f"-> denge anlamsız, atlandı.")
            continue

        # ---- AYLIK TAVAN: hâlâ team_week_base üzerinden, HAFTA bazında ----
        # (gün bazlı denge ile aynı değişkeni paylaşmaz - bilinçli, bkz. başlık yorumu)
        _tvd_hafta_v = {}
        _twb = globals().get("team_week_base", {})
        for _ds in _tvd_gun_v:
            _wk = day_week[_ds]
            for _v in _tvd_gun_v[_ds]:
                _tvd_hafta_v.setdefault(_wk, set()).add(_v)

        # ---- ölçekli L1 sapma: N*kumulatif[t] - Σ kumulatif ----
        _N = len(_gecerli_t)
        _toplam_kum = sum(_kum_ifadeler)
        _max_gec = max(takim_vardiya_denge_gecmis[(t, _bas)] for t in _gecerli_t)
        _max_buay = max(takim_vardiya_denge_buay[(t, _bas)].Proto().domain[-1]
                        for t in _gecerli_t)
        _lim = _N * (_max_gec + _max_buay) + 1

        for _t in _gecerli_t:
            _gec = takim_vardiya_denge_gecmis[(_t, _bas)]
            _kb = takim_vardiya_denge_buay[(_t, _bas)]

            _arti = model.NewIntVar(0, _lim, f"tvd_sapma_arti_{_bas}_{_t}")
            _eksi = model.NewIntVar(0, _lim, f"tvd_sapma_eksi_{_bas}_{_t}")
            model.Add(_N * (_gec + _kb) - _toplam_kum == _arti - _eksi)
            takim_vardiya_denge_sapma[(_t, _bas)] = (_arti, _eksi)

            # ---- aylık tavan (SOFT): en fazla TAKIM_VARDIYA_DENGE_MAX_HAFTA hafta ----
            _hafta_terms = [
                _twb[(_t, _wk, _v)]
                for _wk, _vset in _tvd_hafta_v.items()
                for _v in _vset
                if (_t, _wk, _v) in _twb
            ]
            if _hafta_terms:
                _hft = model.NewIntVar(0, len(_tvd_hafta_v), f"tvd_hafta_{_bas}_{_t}")
                model.Add(_hft == sum(_hafta_terms))
                takim_vardiya_denge_hafta[(_t, _bas)] = _hft
                _asim = model.NewIntVar(0, len(_tvd_hafta_v), f"tvd_tavan_asim_{_bas}_{_t}")
                model.Add(_hft <= int(TAKIM_VARDIYA_DENGE_MAX_HAFTA) + _asim)
                takim_vardiya_denge_asim[(_t, _bas)] = _asim

            takim_vardiya_denge_rows.append({
                "takim": _t,
                "baslangic": _bas,
                "agent_sayisi": takim_vardiya_denge_mevcut[_t],
                "gecmis_kisi_basi_gun": round(_gec / _tvd_olcek, 2),
                "bu_ay_agentgun_ust_limit": int(
                    takim_vardiya_denge_agentgun[(_t, _bas)].Proto().domain[-1]),
                "kapsanan_gun": len(_tvd_gun_v),
                "kapsanan_hafta": len(_tvd_hafta_v),
            })

        # ---- ÖN KONTROL ÇIKTISI (solve'dan ÖNCE bakılır) ----
        _gec_seri = pd.Series({
            t: takim_vardiya_denge_gecmis[(t, _bas)] / _tvd_olcek for t in _gecerli_t
        })
        # bu ay hiç dağıtılmasa objective'e giren sapma ne kadar olurdu?
        # AĞIRLIK KALİBRASYONU BURADAN YAPILIR: bu sayı × W, denge kaleminin
        # objective'deki büyüklüğüdür. coverage eksiği 300.000/slot ile kıyasla.
        _kum0 = pd.Series({t: takim_vardiya_denge_gecmis[(t, _bas)] for t in _gecerli_t})
        _sapma0 = int((_N * _kum0 - _kum0.sum()).abs().sum())
        print(f"\n=== DENGE KURULDU: '{_bas}' başlangıçlı vardiya ===")
        print(f"  denge takım sayısı        : {_N}")
        print(f"  kapsanan gün / hafta      : {len(_tvd_gun_v)} gün, {len(_tvd_hafta_v)} hafta")
        print(f"  aylık tavan               : {int(TAKIM_VARDIYA_DENGE_MAX_HAFTA)} hafta (SOFT, "
              f"aşım cezası {TAKIM_VARDIYA_DENGE_MAX_HAFTA_ASIM_W}, team_week_base üzerinden)")
        print(f"  GEÇMİŞ kişi başı min/ort/max : {_gec_seri.min():.1f} / "
              f"{_gec_seri.mean():.1f} / {_gec_seri.max():.1f} gün")
        print(f"  geçmişte hiç almamış takım   : {int((_gec_seri == 0).sum())}")
        print(f"  mevcut yayılım (max-min)     : {(_gec_seri.max() - _gec_seri.min()):.1f} kişi başı gün")
        print(f"  >> AĞIRLIK KALİBRASYONU: bu ay hiç dağıtılmazsa Σ sapma = {_sapma0:,}")
        print(f"     objective katkısı ≈ {_sapma0:,} × W({TAKIM_VARDIYA_DENGE_W}) "
              f"= {_sapma0 * int(TAKIM_VARDIYA_DENGE_W):,}")
        print(f"     (coverage eksiği 300.000/slot ile kıyasla; büyükse W'yi düşür)")
        print(f"  EN AZ alan 5 takım (bu ay öncelikli):")
        for _t, _g in _gec_seri.sort_values().head(5).items():
            print(f"      {_g:>7.1f} gün   {_t}")
        print(f"  EN ÇOK alan 5 takım (bu ay geri planda):")
        for _t, _g in _gec_seri.sort_values().tail(5).items():
            print(f"      {_g:>7.1f} gün   {_t}")
