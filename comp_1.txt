# %% [KISIT] - HAFTA SONU EN AZ 1 OFF (SOFT)
# 2026-10-14. İş birimi NORM'u sevmiyor: hafta içi 2 off + hafta sonu iki gün çalışma
# görmek istemiyor, her hafta en az 1 off hafta sonunda olsun. Ceza (agent-hafta başına):
#   Cmt VE Paz çalışılmış  VE  o hafta MODELİN verdiği en az bir hafta içi off var
# Hafta içi off kişinin TALEBİYSE (off_t / off_t2) ceza yok. İzin / resmi tatil / arife off
# sayılmaz: izinli hafta + tam hafta sonu NORM değil MESAİ'dir, bu kural dokunmaz.
# SOFT: hafta sonu kapasite açığı (kişi başı 150.000) bundan pahalı -> gerçekten eksik varsa
# model yine iki günü de çalıştırabilir. Sadece Cmt ve Paz'ı planda olan haftalar.
# Mesaiye kalamaz (mk_hafta_sonu_agents) HARİÇ - onlar hafta sonu çalışır. Ceza OBJECTIVE'de.
for _p in ("ENABLE_HS_EN_AZ_1_OFF", "HS_EN_AZ_1_OFF_W"):
    if _p not in globals():
        raise SystemExit(f"CONFIG'de tanımlı olmalı: {_p} -> [CONFIG] hücresini çalıştır.")

hs_tam_calisma = {}     # (a, hafta) -> BoolVar: 1 = model kaynaklı tam hafta sonu çalışma
if ENABLE_HS_EN_AZ_1_OFF:
    _hs1_haric = globals().get("mk_hafta_sonu_agents", set())
    for a_raw in AGENTS:
        a = str(a_raw).strip()
        if a in _hs1_haric:
            continue
        _tip = tip_map.get(a, {})
        for wk, _gunler in week_days.items():
            _hs = [ds for ds in _gunler if pd.to_datetime(ds).weekday() >= 5]
            if len(_hs) < 2 or not all((a, ds) in work for ds in _hs):
                continue
            # modelin off verebileceği hafta içi günler: talep / izin / tatil / arife değil
            _hi = [ds for ds in _gunler
                   if pd.to_datetime(ds).weekday() < 5
                   and pd.to_datetime(ds).date() not in calisma_disi_date_set
                   and str(_tip.get(pd.to_datetime(ds).date(), "")).strip().lower()
                       not in ("izin", "off_t", "off_t2")]
            if not _hi:
                continue
            _tam = model.NewBoolVar(f"hs_tam_{a}_{wk}")
            model.Add(_tam >= work[(a, _hs[0])] + work[(a, _hs[1])] - 1)
            _z = model.NewBoolVar(f"hs_tam_model_off_{a}_{wk}")
            for ds in _hi:
                model.Add(_z >= _tam - work[(a, ds)])     # tam hafta sonu + o gün off
            hs_tam_calisma[(a, wk)] = _z

print("ENABLE_HS_EN_AZ_1_OFF:", ENABLE_HS_EN_AZ_1_OFF, "| W:", HS_EN_AZ_1_OFF_W)
print("izlenen agent-hafta:", len(hs_tam_calisma))
