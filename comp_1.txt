# %% [KONTROL] - AY GEÇİŞİ: SARKAN HAFTA SÜREKLİLİĞİ
# Ön koşul: [KONTROL] - ÇÖZÜMÜ ÇIKAR + ORTAK YARDIMCILAR
# İki şeyi doğrular:
#   1) sarkan haftada takım base'i geçmişteki vardiyayla AYNI mı (süreklilik)
#   2) sarkan haftada agentin GEÇMİŞ + PLAN toplam off'u tavanı aşıyor mu
# Sarkan hafta yoksa (ay Pazartesi başlıyorsa) hücre sessizce atlar.

from datetime import timedelta

sarkan_base_ihlal_df = pd.DataFrame()
sarkan_off_ihlal_df = pd.DataFrame()

_sk_plan_d = sorted(pd.to_datetime(d).date() for d in PLAN_GUNLER)
_sk_ilk = _sk_plan_d[0]
_sk_sarkan = sorted(_sk_ilk - timedelta(days=k) for k in range(_sk_ilk.weekday(), 0, -1))
_sk_wk = day_week[min(PLAN_GUNLER, key=lambda s: pd.to_datetime(s))]

if not _sk_sarkan:
    print(f"Sarkan hafta yok - plan {_sk_ilk} Pazartesi başlıyor. Kontrol atlandı (normal).")
elif "cozum_atama" not in globals():
    print("cozum_atama yok - önce [KONTROL] - ÇÖZÜMÜ ÇIKAR hücresini çalıştır.")
else:
    _sk_gecmis = globals().get("gecmis_calisma", {})
    _sk_gecmis_off = globals().get("gecmis_off_gunleri", {})
    print(f"Sarkan hafta: {_sk_wk} | sarkan günler: {[str(d) for d in _sk_sarkan]}")

    # ---------- 1) TAKIM BASE SÜREKLİLİĞİ ----------
    # Sabitlenemeyen takımlar (geçmişte off, saat eşleşmedi vb.) İHLAL DEĞİLDİR -
    # onlarda model zaten serbest seçti. Sadece SABİTLENDİ olanlar kontrol edilir.
    _sk_sabit = set()
    _sdf = globals().get("sarkan_hafta_base_df")
    if _sdf is not None and not _sdf.empty:
        _sk_sabit = {str(r["takim"]).strip() for _, r in _sdf.iterrows()
                     if str(r["durum"]).strip() == "SABİTLENDİ"}

    _sk_secilen = {}
    if "team_week_base" in globals():
        for (_t, _wk, _v), _var in team_week_base.items():
            if _wk == _sk_wk and V(_var) == 1:
                _sk_secilen[str(_t).strip()] = _v

    _sk_rows = []
    for _t in sorted(_sk_sabit):
        _v_sec = _sk_secilen.get(_t)
        _st_sec = None
        if _v_sec is not None:
            for _ds in PLAN_GUNLER:
                if day_week[_ds] == _sk_wk and (_ds, _v_sec) in saat:
                    _st_sec = (str(saat[(_ds, _v_sec)][0]).strip()[:5],
                               str(saat[(_ds, _v_sec)][1]).strip()[:5])
                    break
        _beklenen = None
        if _sdf is not None and not _sdf.empty:
            for _, _r in _sdf.iterrows():
                if str(_r["takim"]).strip() == _t and str(_r["durum"]).strip() == "SABİTLENDİ":
                    _beklenen = _r["gecmis_saat"]
                    break
        _sec_metni = "-".join(_st_sec) if _st_sec else None
        if _beklenen != _sec_metni:
            _sk_rows.append({"takim": _t, "hafta": _sk_wk,
                             "gecmis_base": _beklenen, "secilen_base": _sec_metni})

    sarkan_base_ihlal_df = pd.DataFrame(_sk_rows)
    kaydet("Ay geçişi: sarkan haftada takım base sürekliliği", len(_sk_rows))
    print(f"   sabitlenen takım                : {len(_sk_sabit)}")
    print(f"   sabitlenemeyen (model serbest)  : "
          f"{0 if _sdf is None or _sdf.empty else len(_sdf) - len(_sk_sabit)}")
    if len(_sk_rows):
        display(sarkan_base_ihlal_df.head(20))

    # ---------- 2) SARKAN HAFTADA TOPLAM OFF ----------
    # geçmişteki off + plandaki off, haftalık tavanı aşmamalı.
    # DİKKAT: kısıt hücresi düşümü SADECE plan tarafında hafta içi günü olan
    # sarkan haftada uyguluyor (ay Cmt/Paz başlarsa uygulanmıyor - o haftanın
    # iş günleri tamamen önceki ayda). Kontrol de aynı koşula uymalı, yoksa
    # uygulanmamış bir kuralı ihlal diye raporlar.
    _tavan = int(globals().get("MODEL_OFF_HAFTALIK_MAX", 2))
    _sk_plan_gun = [ds for ds in PLAN_GUNLER if day_week[ds] == _sk_wk]
    _sk_plan_hi = [ds for ds in _sk_plan_gun if pd.to_datetime(ds).weekday() in (0, 1, 2, 3, 4)]
    _sk_off_uygulandi = bool(_sk_plan_hi) and bool(globals().get("ENABLE_SARKAN_HAFTA_SUREKLILIK"))
    _tip = globals().get("tip_map", {})
    _sayilmayan = set(globals().get("calisma_disi_date_set", set()))
    _sayilmayan = {d for d in _sayilmayan if d.weekday() in (0, 1, 2, 3, 4)}

    _off_rows = []
    for _a_raw in AGENTS:
        _a = str(_a_raw).strip()
        _tip_a = _tip.get(_a, {})
        # plan tarafı: çalışmayan ve izin olmayan günler
        _plan_off = 0
        _sayilan = 0
        for _ds in _sk_plan_gun:
            _d = pd.to_datetime(_ds).date()
            if str(_tip_a.get(_d, "")).strip().lower() == "izin" or _d in _sayilmayan:
                continue
            _sayilan += 1
            if cozum_calisma.get((_a, _ds), 0) != 1:
                _plan_off += 1
        _g_off = len(set(_sk_sarkan) & set(_sk_gecmis_off.get(_a, set())))
        # zorunlu off (talep) tavanı yükseltebilir - kısıt hücresiyle aynı mantık
        _zorunlu = sum(1 for _ds in _sk_plan_gun
                       if str(_tip_a.get(pd.to_datetime(_ds).date(), "")).strip().lower()
                       in ("off_t", "off_t2"))
        _tavan_a = max(_tavan, _zorunlu)
        if _g_off + _plan_off > _tavan_a:
            _off_rows.append({"agent_user_code": _a, "hafta": _sk_wk,
                              "gecmis_off": _g_off, "plan_off": _plan_off,
                              "toplam_off": _g_off + _plan_off, "tavan": _tavan_a,
                              "sayilan_plan_gunu": _sayilan})

    sarkan_off_ihlal_df = pd.DataFrame(_off_rows)
    if _sk_off_uygulandi:
        kaydet("Ay geçişi: sarkan haftada toplam off tavanı", len(_off_rows),
               tur="HARD" if globals().get("MODEL_OFF_HAFTALIK_HARD") else "SOFT")
    else:
        # kural bu ay UYGULANMADI -> ihlal değil, sadece gözlem
        kaydet("Ay geçişi: sarkan haftada toplam off [kural bu ay uygulanmadı]",
               len(_off_rows), tur="RAPOR")
        print("   NOT: plan tarafında hafta içi gün yok (ay hafta sonu başlıyor) ->")
        print("        geçmiş off düşümü UYGULANMADI. Aşağıdaki sayı ihlal DEĞİL,")
        print("        o haftanın iş günleri zaten önceki ayın planındaydı.")
    print(f"   geçmişte off almış agent        : "
          f"{sum(1 for a in AGENTS if set(_sk_sarkan) & set(_sk_gecmis_off.get(str(a).strip(), set())))}")
    if len(_off_rows):
        print(f"   {len(_off_rows)} agent sarkan haftada tavanın üstünde:")
        display(sarkan_off_ihlal_df.sort_values("toplam_off", ascending=False).head(20))
