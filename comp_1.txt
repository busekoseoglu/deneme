# %% [KISIT] - HAFTALIK OFF TAVANI (MODEL 3. OFF'U KENDİ VEREMEZ)
# Bir agent bir haftada en fazla MODEL_OFF_HAFTALIK_MAX (2) gün off alır.
# off_t / off_t2 talepleri bu tavanın İÇİNDE sayılır: 2 gün talep karşılandıysa
# model üstüne ekstra off VEREMEZ.
#
# Tavan sabit 2 değil, TABAN 2:
#     hafta_tavani = max(MODEL_OFF_HAFTALIK_MAX, zorunlu_off_sayisi)
#     zorunlu_off  = (off_t ∪ off_t2 ∪ agentin o gün hiç çalışamayacağı günler) ∩ hafta
# Böylece:
#   - 3 gün off talebi olan agent 3'ünü de alır, 4. off olmaz
#   - yapısal olarak çalışamayan günleri olan agent (hamile/süt hafta sonu) infeasible olmaz
#
# izin günleri sayıma HİÇ girmez (off değil, ayrı kategori) - o günler kotayı da düşürüyor.
# Kısmi haftalarda (W31: 2 gün, W36: 1 gün) tavan zaten bağlayıcı olmaz, kısıt kurulmaz.
#
# Bu hücre [CONFIG], [HÜCRE 9] ve [HAZIRLIK / KISIT] - AYLIK ÇALIŞMA DENGESİ'nden
# SONRA çalışmalıdır. Ayarlanabilir parametrelerin tamamı CONFIG'de.

#   MODEL_OFF_HAFTALIK_MAX   haftalık off tavanı (2)
#   MODEL_OFF_HAFTALIK_HARD  False -> soft + objective cezası
#   MODEL_OFF_TAVAN_ASIM_W   soft moddaki ceza ağırlığı

    if _dw.get(ds) == _ot_sarkan_wk and pd.to_datetime(ds).weekday() in (0, 1, 2, 3, 4)
]
_ot_hafta_ici_var = bool(_ot_sarkan_plan_hi)

# hamile / süt izinli agentlar hafta sonu çalışamaz ([HÜCRE 14]). O hücre x'i
# YARATIP == 0 zorladığı için aşağıdaki "_var_mi" kontrolü bunu yakalayamıyor;
# burada df_tam'dan doğrudan türetilir.
_ot_hs_calisamaz = set()
for _c1, _c2 in (("hamile_flg", "sut_izni_flg"),):
    _m = None
    for _c in (_c1, _c2):
        if _c in df_tam.columns:
            _s = pd.to_numeric(df_tam[_c], errors="coerce").fillna(0).astype(int) == 1
            _m = _s if _m is None else (_m | _s)
    if _m is not None:
        _ot_hs_calisamaz = set(df_tam.loc[_m, "agent_user_code"].astype(str).str.strip())

if not _ot_sarkan_gunler:
    print("Sarkan hafta YOK (plan Pazartesi başlıyor) - geçmiş off düşümü uygulanmayacak.")
elif not _ot_aktif:
    print("ENABLE_SARKAN_HAFTA_SUREKLILIK = False -> geçmiş off düşümü UYGULANMIYOR.")
elif not _ot_hafta_ici_var:
    print(f"Sarkan hafta {_ot_sarkan_wk}: plan tarafında HAFTA İÇİ GÜN YOK "
          f"(ay hafta sonu başlıyor) -> geçmiş off düşümü UYGULANMIYOR.")
    print("   Sebep: o haftanın iş günleri tamamen önceki ayda; düşüm agentı "
          "hafta sonu çalışmaya zorlardı.")
    _ot_aktif = False

    print(f"Sarkan hafta: {_ot_sarkan_wk} | sarkan gün: "
          f"{sorted(str(d) for d in _ot_sarkan_gunler)}")
    print(f"   plan tarafındaki hafta içi gün: {len(_ot_sarkan_plan_hi)} -> düşüm UYGULANACAK")
    print(f"   hafta sonu çalışamayan agent (hamile/süt): {len(_ot_hs_calisamaz)}")
    if not _ot_gecmis_off:
        print("!!! UYARI: gecmis_off_gunleri yok/boş - geçmiş off düşümü YAPILAMIYOR. "
              "Önce geçmiş işleme hücresini çalıştır.")
        _ot_aktif = False
_ot_dusulen_toplam = 0

model_off_tavan_asim = {}          # (agent, hafta) -> IntVar   (sadece soft modda)
off_tavan_kisit_sayisi = 0
off_tavan_atlanan = 0
off_tavan_debug_rows = []
off_tavan_marj_rows = []

for a_raw in AGENTS:
    a = str(a_raw).strip()
    _tip_a = tip_map.get(a, {})
    _zorunlu_calisma_toplam = 0

    for _wk, _wk_gunler in _hafta_plan_gunleri.items():
        # izin günleri sayıma girmez
        _sayilan = []
        for ds in _wk_gunler:
            _d_ot = pd.to_datetime(ds).date()
            _tip = str(_tip_a.get(_d_ot, "")).strip().lower()
            if _tip == "izin" or _d_ot in _sayilmayan_ot:
                continue
            _sayilan.append(ds)
        if not _sayilan:
            continue

        # zorunlu off: talep edilmiş (off_t/off_t2) VEYA agentin hiç çalışamayacağı gün
        _zorunlu = []
        for ds in _sayilan:
            _tip = str(_tip_a.get(pd.to_datetime(ds).date(), "")).strip().lower()
            _var_mi = any((a, ds, v) in x for v in gun_vardiyalari.get(ds, []))
            if _tip in ("off_t", "off_t2") or not _var_mi:
                _zorunlu.append(ds)

        _tavan = max(MODEL_OFF_HAFTALIK_MAX, len(_zorunlu))

        # --- sarkan haftada geçmişte alınan off'lar tavandan düşülür ---
        # Sadece OFF sayılır; geçmişteki izin / haftalık izin sayılmaz (plan
        # tarafında da izin sayıma girmiyor). Kaydı olmayan gün de sayılmaz.
        _gecmis_off_sayisi = 0
        _tavan_kalan = _tavan          # diğer tüm haftalarda davranış DEĞİŞMEZ
        if _ot_aktif and _wk == _ot_sarkan_wk:
            _gecmis_off_sayisi = len(_ot_sarkan_gunler & set(_ot_gecmis_off.get(a, set())))
            _ot_dusulen_toplam += _gecmis_off_sayisi
            # Agentin FİİLEN çalışamayacağı günler düşümden KORUNUR: o günler için
            # off "harcamak" zorunda değil. Hamile/süt izinli agentın hafta sonu
            # günleri de buraya dahil (yukarıdaki _zorunlu bunları yakalayamıyor).
            _korunan = set(_zorunlu)
            if a in _ot_hs_calisamaz:
                _korunan |= {ds for ds in _sayilan
                             if pd.to_datetime(ds).weekday() in (5, 6)}
            _tavan_kalan = max(len(_korunan), _tavan - _gecmis_off_sayisi)

        _min_calisma = len(_sayilan) - _tavan_kalan
        _zorunlu_calisma_toplam += max(0, _min_calisma)

        off_tavan_debug_rows.append({
            "agent_user_code": a, "hafta": _wk,
            "sayilan_gun": len(_sayilan), "zorunlu_off": len(_zorunlu),
            "tavan": _tavan, "gecmis_off": _gecmis_off_sayisi, "tavan_kalan": _tavan_kalan,
            "min_calisma": max(0, _min_calisma),
            "kisit_kuruldu": _min_calisma > 0,
        })

        if _min_calisma <= 0:
            off_tavan_atlanan += 1      # tavan bağlayıcı değil
            continue

        _work_vars = [work[(a, ds)] for ds in _sayilan if (a, ds) in work]
        if len(_work_vars) < len(_sayilan):
            _min_calisma = len(_work_vars) - _tavan_kalan
            if _min_calisma <= 0:
                off_tavan_atlanan += 1
                continue

        if MODEL_OFF_HAFTALIK_HARD:
            model.Add(sum(_work_vars) >= _min_calisma).WithName(f"off_tavan_{a}_{_wk}")
        else:
            _asim = model.NewIntVar(0, len(_work_vars), f"off_tavan_asim_{a}_{_wk}")
            model_off_tavan_asim[(a, _wk)] = _asim
            model.Add(sum(_work_vars) + _asim >= _min_calisma).WithName(f"off_tavan_soft_{a}_{_wk}")
        off_tavan_kisit_sayisi += 1

    # --- solve öncesi marj: haftalık zorunlu minimum vs aylık tavan ---
    _baz = _baz_map_t.get(a)
    if _baz is not None:
        _aylik_tavan = _baz if a in _mk_agents_t else _baz + MAX_OVERTIME_PER_MONTH
        off_tavan_marj_rows.append({
            "agent_user_code": a,
            "mesaiye_kalamaz": a in _mk_agents_t,
            "haftalik_zorunlu_calisma_toplami": _zorunlu_calisma_toplam,
            "baz_gereken_calisma": _baz,
            "aylik_tavan": _aylik_tavan,
            "marj": _aylik_tavan - _zorunlu_calisma_toplam,
        })

off_tavan_debug_df = pd.DataFrame(off_tavan_debug_rows)
off_tavan_marj_df = pd.DataFrame(off_tavan_marj_rows)

print(f"MODEL_OFF_HAFTALIK_MAX : {MODEL_OFF_HAFTALIK_MAX}")
print(f"mod                    : {'HARD' if MODEL_OFF_HAFTALIK_HARD else 'SOFT (ceza=' + str(MODEL_OFF_TAVAN_ASIM_W) + ')'}")
print(f"kurulan kısıt          : {off_tavan_kisit_sayisi} agent-hafta")
print(f"bağlayıcı olmayan      : {off_tavan_atlanan} agent-hafta (kısıt kurulmadı)")

if not off_tavan_debug_df.empty:

    print(f"  marjı SIFIR agent   : {len(_sifir)}")
    if len(_negatif):

        display(_negatif.sort_values("marj").head(30))
else:

    _sh_takim_uye = {}
    for _a in AGENTS:

    # ============ 1) TAKIM BASE SÜREKLİLİĞİ ============
    for _t, _uyeler in sorted(_sh_takim_uye.items()):
        # sarkan günlerde takımın çalıştığı saat aralıkları (agent-gün sayısıyla)

            sarkan_hafta_base_rows.append({
                "takim": _t, "hafta": _sh_wk, "gecmis_saat": None, "vardiya": None,
                "agent_gun": 0, "durum": "GEÇMİŞ ÇALIŞMA YOK - model serbest seçer"})
            continue

        _gecmis_st = max(_sayac.items(), key=lambda kv: (kv[1], kv[0]))[0]
        _agent_gun = _sayac[_gecmis_st]
        _v = _sh_saat_v.get(_gecmis_st)

        if _v is None:
            sarkan_hafta_base_rows.append({
                "takim": _t, "hafta": _sh_wk, "gecmis_saat": "-".join(_gecmis_st),
                "vardiya": None, "agent_gun": _agent_gun,
                "durum": "SAAT EŞLEŞMEDİ - bu ayın vardiyaları arasında yok"})
            continue
        if (_t, _sh_wk, _v) not in team_week_base:
            sarkan_hafta_base_rows.append({
                "takim": _t, "hafta": _sh_wk, "gecmis_saat": "-".join(_gecmis_st),
                "vardiya": _v, "agent_gun": _agent_gun,
                "durum": "TAKIM İÇİN BASE SEÇENEĞİ YOK"})
            continue
        if (_t, _sh_wk, _v) in _sh_yasak:
            # bütünlük kuralı bu base'i kapatmış -> == 1 dersek infeasible olur
            sarkan_hafta_base_rows.append({
                "takim": _t, "hafta": _sh_wk, "gecmis_saat": "-".join(_gecmis_st),
                "vardiya": _v, "agent_gun": _agent_gun,
                "durum": "BÜTÜNLÜK KURALI KAPATMIŞ - kadro/kıdem değişmiş olabilir"})
            continue

        model.Add(team_week_base[(_t, _sh_wk, _v)] == 1)
        sarkan_takim_base_kisiti += 1
        sarkan_hafta_base_rows.append({
            "takim": _t, "hafta": _sh_wk, "gecmis_saat": "-".join(_gecmis_st),
            "vardiya": _v, "agent_gun": _agent_gun, "durum": "SABİTLENDİ"})

    # ============ 2) MUAF AGENT BİREYSEL BASE SÜREKLİLİĞİ ============
    # agent_week_base sadece muaf OLABİLECEK agent-haftalar için üretilir.
    # Sabitleme KOŞULLU yapılır:  agent_week_base[v] == takim_muaf
    #   muaf = 1 -> agent geçmişteki v'sinde kalır
    #   muaf = 0 -> takım base'i artık ona uygun (kıdemi dolmuş olabilir),
    #               bireysel base sıfırlanır ve agent TAKIMA KATILIR
    # Böylece "== 1" ile sabitleyip infeasible üretme riski yok.
    _awb = globals().get("agent_week_base", {})
    _tmuaf = globals().get("takim_muaf", {})
    if _awb:
        for _a in sorted({a for (a, wk, v) in _awb if wk == _sh_wk}):
            _gc = _sh_gecmis.get(_a, {})
            _sayac_a = {}
            for _d in _sh_sarkan:
                if _d not in _gc:
                    continue
                _st = (str(_gc[_d][0]).strip()[:5], str(_gc[_d][1]).strip()[:5])
                _sayac_a[_st] = _sayac_a.get(_st, 0) + 1
            if not _sayac_a:
                # EDGE CASE: agent sarkan günlerde off/izinde -> base yok, model seçer
                sarkan_agent_base_rows.append({
                    "agent_user_code": _a, "hafta": _sh_wk, "gecmis_saat": None,
                    "vardiya": None, "durum": "GEÇMİŞ ÇALIŞMA YOK - model serbest seçer"})
                continue
            _st_a = max(_sayac_a.items(), key=lambda kv: (kv[1], kv[0]))[0]
            _v_a = _sh_saat_v.get(_st_a)
            if _v_a is None or (_a, _sh_wk, _v_a) not in _awb:
                sarkan_agent_base_rows.append({
                    "agent_user_code": _a, "hafta": _sh_wk, "gecmis_saat": "-".join(_st_a),
                    "vardiya": _v_a,
                    "durum": "EŞLEŞME YOK ya da agent o vardiyada çalışamıyor"})
                continue
            model.Add(_awb[(_a, _sh_wk, _v_a)] == _tmuaf[(_a, _sh_wk)])
            sarkan_agent_base_kisiti += 1
            sarkan_agent_base_rows.append({
                "agent_user_code": _a, "hafta": _sh_wk, "gecmis_saat": "-".join(_st_a),
                "vardiya": _v_a, "durum": "KOŞULLU SABİTLENDİ (muaf ise geçmiş vardiyası)"})

    for a_raw in AGENTS:
        a = str(a_raw).strip()
        _izn = {d for d, t in _tip_map.get(a, {}).items()
                if str(t).strip().lower() == "izin" and d in _is_gunu_d}
        _izin_map_x[a] = len(_izn)
        _baz_map_x[a] = _aylik_is_gunu_x - len(_izn)

# --------------------------- HAFTALIK NORM / TELAFİ / MESAİ HESABI (agent)
_hafta_gunleri = defaultdict(list)
for ds in PLAN_GUNLER:
    d = pd.to_datetime(ds).date()
    _pzt = d - _td(days=d.weekday())
    _hafta_gunleri[_pzt].append(ds)
for _k in _hafta_gunleri:
    _hafta_gunleri[_k] = sorted(_hafta_gunleri[_k], key=lambda s: pd.to_datetime(s))

mesai_gunleri_map = {}   # agent -> {ds}  ücretli mesai
telafi_gunleri_map = {}  # agent -> {ds}  borç kapatma
norm_gunleri_map = {}    # agent -> {ds}  kota içi hafta sonu
hafta_calisma_debug_rows = []

for a_raw in AGENTS:
    a = str(a_raw).strip()
    _gecmis_a = _gecmis_gunleri_x.get(a, set())
    _aday_a, _norm_a = [], set()   # _aday_a: haftalık kotayı aşan hafta sonu günleri (kronolojik)

    for _pzt, _wk_gunler in sorted(_hafta_gunleri.items()):
        # RESMİ TATİL günleri haftalık hesaba GİRMEZ: ayrı kategori (RESMİ TATİL MESAİ),
        # aylık toplama da (sayilan_calisma) girmiyor. Haftalık ve aylık adım aynı
        # tabanı kullanmazsa resmi tatil günü aday listesinde yer tutup MESAİ slotunu
        # çalar ve gerçek mesai günü yanlışlıkla TELAFİ olarak etiketlenir.
        _calisilan = [
            ds for ds in _wk_gunler
            if cozum_calisma.get((a, ds), 0) == 1
            and pd.to_datetime(ds).date() not in _sayilmayan_set
        ]

        # aydan sarkan hafta: plan öncesi (Temmuz) çalışılan günleri de say
        _hafta_araligi = {_pzt + _td(days=k) for k in range(7)}
        _temmuz_ek = len({d for d in _gecmis_a if d in _hafta_araligi and d < _PLAN_BAS})

        _hafta_calisma = len(_calisilan) + _temmuz_ek

        # --- haftalık kota: izin ve resmi tatil günleri DÜŞÜLÜR ---
        _kota = 0
        for _k2 in range(5):                       # Pzt..Cum
            _d2 = _pzt + _td(days=_k2)
            if _d2 < _PLAN_BAS:
                # aydan sarkan hafta (W31): Temmuz'un hafta içi günleri
                if _d2 not in _g_izin.get(a, set()):
                    _kota += 1
            elif _d2 in _plan_date_set_x:
                if _d2 in _sayilmayan_set:
                    continue
                if str(_tip_map.get(a, {}).get(_d2, "")).strip().lower() == "izin":
                    continue
                _kota += 1
        _kota = min(_NORMAL_WORK_DAYS, _kota)

        _fazla = max(0, _hafta_calisma - _kota)

        _hs_calisilan = [ds for ds in _calisilan if pd.to_datetime(ds).weekday() in (5, 6)]
        if _fazla > 0 and _hs_calisilan:
            _aday_bu_hafta = list(_hs_calisilan[-_fazla:])
        else:
            _aday_bu_hafta = []
        _norm_bu_hafta = set(_hs_calisilan) - set(_aday_bu_hafta)

        _aday_a.extend(_aday_bu_hafta)
        _norm_a |= _norm_bu_hafta

        hafta_calisma_debug_rows.append({
            "agent_user_code": a,
            "hafta_pzt": str(_pzt),
            "plan_ici_calisma": len(_calisilan),
            "temmuzdan_gelen_calisma": _temmuz_ek,
            "hafta_toplam_calisma": _hafta_calisma,
            "kota": _kota,
            "fazla": _fazla,
            "hafta_sonu_calisilan": len(_hs_calisilan),
            "norm": len(_norm_bu_hafta),
            "kota_asan_aday": len(_aday_bu_hafta),
        })

    # ---- AYLIK ADIM: adaylardan kaçı gerçekten ÜCRETLİ mesai? ----
    _aday_a = sorted(set(_aday_a), key=lambda d: pd.to_datetime(d))
    _sayilan_a = sum(
        1 for ds in PLAN_GUNLER
        if cozum_calisma.get((a, ds), 0) == 1
        and pd.to_datetime(ds).date() not in _sayilmayan_set
    )
    _aylik_fazla_a = max(0, _sayilan_a - int(_baz_map_x.get(a, 0)))
    _mesai_sayisi = min(_aylik_fazla_a, len(_aday_a))

    mesai_gunleri_map[a] = set(_aday_a[len(_aday_a) - _mesai_sayisi:]) if _mesai_sayisi else set()
    telafi_gunleri_map[a] = set(_aday_a) - mesai_gunleri_map[a]
    norm_gunleri_map[a] = _norm_a

        if d in _resmi_tatil_set:

            row[_kol] = f"RESMİ TATİL MESAİ\n{vardiya}"
            _renk = "resmi_tatil"

                _renk = "arife"
            else:
                _n_arife_mesai += 1
                row[_kol] = f"ARİFE MESAİ\n{vardiya}"
                _renk = "arife_mesai"
        elif ds in _mesai_a:
            _n_mesai += 1
            row[_kol] = f"MESAİ\n{vardiya}"
            _renk = "mesai"
        elif ds in _telafi_a:
            _n_telafi += 1
            row[_kol] = f"TELAFİ\n{vardiya}"
            _renk = "telafi"
        elif ds in _norm_a:

    row["arife_mesai"] = _n_arife_mesai
    row["izin_gun"] = _n_izin

    row["off_t_talep"] = _n_off_t
    row["off_t_karsilanmayan"] = _n_off_t_ihlal

with pd.ExcelWriter(EXCEL_DOSYA, engine="openpyxl") as writer:
    takvim_df.to_excel(writer, sheet_name="TAKVIM", index=False)
    agent_ozet_df.to_excel(writer, sheet_name="AGENT_OZET", index=False)
    coverage_df.to_excel(writer, sheet_name="COVERAGE", index=False)
    _lejant_df.to_excel(writer, sheet_name="LEJANT", index=False)
    if isinstance(_kural_df, pd.DataFrame) and not _kural_df.empty:
        _kural_df.to_excel(writer, sheet_name="KURAL_KONTROL", index=False)
    for _ad, _df in _ham_sheetler:
        if isinstance(_df, pd.DataFrame) and not _df.empty:
            _df.to_excel(writer, sheet_name=_ad[:31], index=False)
        else:
            print(f"  ! {_ad} yok/boş - sheet atlandı")

    # ---------------- TAKVIM biçimlendirme
    ws = writer.sheets["TAKVIM"]
    _kol_idx = {c: i + 1 for i, c in enumerate(takvim_df.columns)}
    _ince = Side(style="thin", color="BBBBBB")
    _kenar = Border(left=_ince, right=_ince, top=_ince, bottom=_ince)

    for c in takvim_df.columns:
        _h = ws.cell(row=1, column=_kol_idx[c])
        _h.fill = PatternFill("solid", fgColor=RENK["baslik"])
        _h.font = Font(bold=True, size=9)
        _h.alignment = Alignment(horizontal="center", vertical="center", wrap_text=True)

    for (_i, _kol), _renk_key in hucre_renk.items():
        _c = ws.cell(row=_i + 2, column=_kol_idx[_kol])
        _c.fill = PatternFill("solid", fgColor=RENK[_renk_key])

    for _kol in _temmuz_kolonlari:
        _h = ws.cell(row=1, column=_kol_idx[_kol])
        _h.fill = PatternFill("solid", fgColor=RENK["baslik_gec"])

    for _r in range(2, len(takvim_df) + 2):
        for _kol in _temmuz_kolonlari + _gun_kolonlari:
            _c = ws.cell(row=_r, column=_kol_idx[_kol])
            _c.alignment = Alignment(horizontal="center", vertical="center", wrap_text=True)
            _c.font = Font(size=8)
            _c.border = _kenar

    for c in _kimlik_kolonlari:
        ws.column_dimensions[get_column_letter(_kol_idx[c])].width = 26
    for c in _ozet_kolonlari:
        ws.column_dimensions[get_column_letter(_kol_idx[c])].width = 11
    for c in _temmuz_kolonlari + _gun_kolonlari:
        ws.column_dimensions[get_column_letter(_kol_idx[c])].width = 13
    ws.row_dimensions[1].height = 32

    # Sadece bu kolonlar sabit kalsın; kalan kimlik/bayrak kolonları kayabilsin.
    _SABIT_KOLONLAR = ["agent_user_code", "agent_name", "working_main_group", "takim"]
    _dondurulacak = 0
    for _c in takvim_df.columns:
        if _c in _SABIT_KOLONLAR:
            _dondurulacak += 1
        else:
            break
    ws.freeze_panes = ws.cell(row=2, column=_dondurulacak + 1)
    print(f"TAKVIM dondurulan kolon: {_dondurulacak} "
          f"({', '.join(list(takvim_df.columns)[:_dondurulacak])})")
    ws.auto_filter.ref = f"A1:{get_column_letter(len(takvim_df.columns))}1"

    # ---------------- LEJANT renk örnekleri
    ws_l = writer.sheets["LEJANT"]
    ws_l.column_dimensions["A"].width = 24
    ws_l.column_dimensions["B"].width = 70
    ws_l.cell(row=1, column=1).font = Font(bold=True)
    ws_l.cell(row=1, column=2).font = Font(bold=True)
    for _j, (_g, _renk_key, _ac) in enumerate(LEJANT):

        elif _fk > 0:
            _wc.cell(row=_j + 2, column=_fk_idx).fill = PatternFill("solid", fgColor=RENK["mesai"])

    # ---------------- KURAL_KONTROL renklendirme
    if "KURAL_KONTROL" in writer.sheets:
        ws_k = writer.sheets["KURAL_KONTROL"]
        ws_k.column_dimensions["A"].width = 55
        ws_k.column_dimensions["B"].width = 10
        ws_k.column_dimensions["C"].width = 10
        _kk = _kural_df.reset_index(drop=True)

            if _hard and _ihl:
                for _cc in range(1, len(_kk.columns) + 1):
                    ws_k.cell(row=_j + 2, column=_cc).fill = PatternFill(
                        "solid", fgColor=RENK["hard_ihlal"]
                    )
