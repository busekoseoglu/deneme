# %% [KISIT] - PROVİZYON: HAFTA SONU BÖLÜNMÜŞ VARDİYA
# 2026-10-19 KURUMSAL. Sözde vardiya: [HÜCRE] - PROVİZYON HAFTA SONU VARDİYASINI EKLE
#   (a) provizyoncu hafta sonu sadece provizyon vardiyasında çalışır
#   (b) provizyoncu olmayan provizyon vardiyasını alamaz
#   (c) her hafta sonu günü >= 1 provizyoncu, kaçış PROVIZYON_GUN_MUAF_W
#   (d) aynı provizyoncu aynı hafta sonunun Cmt+Pzr'ını birlikte çalışmaz,
#       kaçış PROVIZYON_CIFT_GUN_W (< GUN_MUAF_W: diğerleri yoksa gün boş kalmaz)
# Çift off kuralı provizyonculara da uygulanır; hafta sonu adil dağıtımından muaflar.
# Cezalar OBJECTIVE'de.
from datetime import timedelta

for _p in ("PROVIZYON_GUN_MUAF_W", "PROVIZYON_CIFT_GUN_W"):
    if _p not in globals():
        raise SystemExit(f"CONFIG'de tanımlı olmalı: {_p} -> [CONFIG] hücresini çalıştır.")

prov_gun_muaf = {}       # ds -> BoolVar (o gün provizyoncu yok)
prov_cift_kacis = {}     # (a, Cmt ds) -> BoolVar (Cmt+Pzr aynı kişi)
_pv = PROVIZYON_VARDIYA_KODU
_pv_tarih = {pd.to_datetime(ds).date(): ds for ds, _ in prov_vardiyalar}
_pv_bos = []

for d, ds in sorted(_pv_tarih.items()):
    for a in AGENTS:
        for v in gun_vardiyalari.get(ds, []):
            if (a, ds, v) in x and (a in prov_agents) != (v == _pv):
                model.Add(x[(a, ds, v)] == 0)                       # (a) + (b)
    _gv = [x[(a, ds, _pv)] for a in sorted(prov_agents) if (a, ds, _pv) in x]
    if not _gv:
        _pv_bos.append(ds)
    prov_gun_muaf[ds] = model.NewBoolVar(f"prov_gun_muaf_{ds}")
    model.Add(sum(_gv) + prov_gun_muaf[ds] >= 1)                     # (c)
    ds2 = _pv_tarih.get(d + timedelta(days=1)) if d.weekday() == 5 else None
    if ds2 is None:
        continue
    for a in sorted(prov_agents):
        if (a, ds, _pv) in x and (a, ds2, _pv) in x:
            prov_cift_kacis[(a, ds)] = model.NewBoolVar(f"prov_cift_{a}_{ds}")
            model.Add(x[(a, ds, _pv)] + x[(a, ds2, _pv)] <= 1 + prov_cift_kacis[(a, ds)])   # (d)

print("provizyoncu:", len(prov_agents), "| hafta sonu günü:", len(prov_gun_muaf),
      "| Cmt+Pzr kısıtı (agent-hafta sonu):", len(prov_cift_kacis))
if _pv_bos:
    print("!!! müsait provizyoncu OLMAYAN gün (izin/off_t2) -> kaçışa düşer:", _pv_bos)

# 2026-11-12: HAFTA İÇİ - provizyoncu sadece PROVIZYON_HI_VARDIYALAR'da (HARD), hafta boyu AYNI
# vardiyada (HARD), her vardiyada hedef kadar kişi (SOFT: FEDA vardiyası ucuz), AKSAM vardiyası
# ayda en fazla MAX_NIGHT_WEEKS_PER_MONTH hafta (HARD). Sözde vardiyayı sadece provizyoncu alır.
for _p in ("PROVIZYON_HI_FEDA", "PROVIZYON_HI_AKSAM", "PROVIZYON_HI_BOS_W",
           "PROVIZYON_HI_FEDA_BOS_W", "PROVIZYON_HI_FAZLA_W"):
    if _p not in globals():
        raise SystemExit(f"CONFIG'de tanımlı olmalı: {_p} -> [CONFIG] hücresini çalıştır.")
prov_hi_eksik = {}      # (ds, vardiya saati) -> IntVar (hedefin altı)
prov_hi_fazla = {}      # (ds, vardiya saati) -> IntVar (hedefin üstü)
_ph_hafta = {}          # (a, hafta, vardiya saati) -> BoolVar
_cd_ph = set(globals().get("calisma_disi_date_set", set()))
_ph_kod = {}
for (ds, _sl), v in prov_hi_slot_v.items():
    _ph_kod.setdefault(ds, {})[v] = _sl
for ds, _kod in sorted(_ph_kod.items()):
    wk, _tatil = day_week[ds], pd.to_datetime(ds).date() in _cd_ph
    for a in AGENTS:
        _pa = a in prov_agents
        for v in gun_vardiyalari.get(ds, []):
            if (a, ds, v) not in x:
                continue
            if _pa and v in _kod:
                _b = _ph_hafta.get((a, wk, _kod[v]))
                if _b is None:
                    _b = _ph_hafta[(a, wk, _kod[v])] = model.NewBoolVar(f"prov_hi_{a}_{wk}_{_kod[v][0]}")
                model.Add(x[(a, ds, v)] <= _b)                       # hafta boyu aynı vardiya
            elif (_pa and not _tatil) or (not _pa and (ds, v) in prov_hi_sozde):
                model.Add(x[(a, ds, v)] == 0)                       # sadece provizyon vardiyası / sözde sadece provizyoncu
    if _tatil:
        continue
    for v, _sl in _kod.items():
        _n = sum(x[(a, ds, v)] for a in sorted(prov_agents) if (a, ds, v) in x)
        _h = int(PROVIZYON_HI_VARDIYALAR[_sl])
        prov_hi_eksik[(ds, _sl)] = model.NewIntVar(0, _h, f"prov_hi_eksik_{ds}_{_sl[0]}")
        prov_hi_fazla[(ds, _sl)] = model.NewIntVar(0, len(prov_agents), f"prov_hi_fazla_{ds}_{_sl[0]}")
        model.Add(_n + prov_hi_eksik[(ds, _sl)] >= _h)
        model.Add(_n - prov_hi_fazla[(ds, _sl)] <= _h)
for a in sorted(prov_agents):
    for wk in WEEKS:
        _bs = [b for (a2, w2, s), b in _ph_hafta.items() if a2 == a and w2 == wk]
        if len(_bs) > 1:
            model.Add(sum(_bs) <= 1)
    _ak = [b for (a2, w2, s), b in _ph_hafta.items() if a2 == a and s in PROVIZYON_HI_AKSAM]
    if _ak:
        model.Add(sum(_ak) <= MAX_NIGHT_WEEKS_PER_MONTH)            # akşam vardiyası ayda max hafta
print("provizyon hafta içi -> gün x vardiya hedefi:", len(prov_hi_eksik),
      "| agent-hafta-vardiya:", len(_ph_hafta), "| akşam max hafta:", MAX_NIGHT_WEEKS_PER_MONTH)

# 2026-11-12: NORM DENGESİ - provizyoncunun NORM haftası = hafta sonu çalıştığı + hafta içi off'u olan
# hafta (Excel haftalık adımının model karşılığı). Ayda PROVIZYON_NORM_HEDEF'in üstündeki her NORM
# PROVIZYON_NORM_FAZLA_W -> 0/1/2 yerine 1/1/1. Sarkan / ay sonu yarım haftalar (hafta sonu ya da hafta
# içi planda yok) sayılmaz. W = 0 kuralı kapatır.
for _p in ("PROVIZYON_NORM_HEDEF", "PROVIZYON_NORM_FAZLA_W"):
    if _p not in globals():
        raise SystemExit(f"CONFIG'de tanımlı olmalı: {_p} -> [CONFIG] hücresini çalıştır.")
prov_norm_hafta = {}    # (a, hafta) -> BoolVar
prov_norm_fazla = {}    # a -> IntVar (hedefin üstü)
if prov_agents and PROVIZYON_NORM_FAZLA_W:
    _cd_pn = set(globals().get("calisma_disi_date_set", set()))
    for a in sorted(prov_agents):
        for wk, _gunler in week_days.items():
            _hs = [work[(a, ds)] for ds in _gunler if pd.to_datetime(ds).weekday() >= 5]
            _hi = [work[(a, ds)] for ds in _gunler if pd.to_datetime(ds).weekday() < 5
                   and pd.to_datetime(ds).date() not in _cd_pn
                   and str(tip_map.get(a, {}).get(pd.to_datetime(ds).date(), "")).strip().lower() != "izin"]
            if not _hs or not _hi:
                continue
            _hsw = model.NewBoolVar(f"prov_hs_{a}_{wk}")
            model.Add(_hsw <= sum(_hs))
            _off = model.NewBoolVar(f"prov_hi_off_{a}_{wk}")
            for _w in _hi:
                model.Add(_off >= 1 - _w)
            _n = model.NewBoolVar(f"prov_norm_{a}_{wk}")
            for _w in _hs:
                model.Add(_n >= _w + _off - 1)        # hafta sonu çalıştı + hafta içi off -> NORM haftası
            prov_norm_hafta[(a, wk)] = _n
        _ns = [v for (a2, _), v in prov_norm_hafta.items() if a2 == a]
        if _ns:
            prov_norm_fazla[a] = model.NewIntVar(0, len(_ns), f"prov_norm_fazla_{a}")
            model.Add(prov_norm_fazla[a] >= sum(_ns) - PROVIZYON_NORM_HEDEF)
print("provizyon NORM dengesi -> hedef:", PROVIZYON_NORM_HEDEF, "| W:", PROVIZYON_NORM_FAZLA_W,
      "| izlenen agent-hafta:", len(prov_norm_hafta))
