def dakika(t):
    s = str(t).strip()[:5]
    h, m = s.split(":")
    return int(h) * 60 + int(m)


def _sbh_norm(t):
    return str(t).strip()[:5]


SABAH_BITIS_LIMIT = dakika(SABAH_CALISIR_MAX_BITIS_SAATI)
SABAH_BASLANGIC_LIMIT = dakika(SABAH_CALISIR_MAX_BASLANGIC_SAATI)


def sabah_vardiyasi_uygun_mu(bas, bit):
    """(uygun_mu, red_sebebi) döner. Sabah agent bu vardiyada çalışabilir mi?"""
    b, e = _sbh_norm(bas), _sbh_norm(bit)
    bas_dk, bit_dk = dakika(b), dakika(e)
    if bit_dk < bas_dk:
        bit_dk += 24 * 60
    if SABAH_CALISIR_GECE_VARDIYA_YASAK and (b, e) in GECE_AKSAM_VARDIYA_SET:
        return False, "gece/akşam vardiyası"
    if bas_dk > SABAH_BASLANGIC_LIMIT:
        return False, f"başlangıç > {SABAH_CALISIR_MAX_BASLANGIC_SAATI}"
    if bit_dk > SABAH_BITIS_LIMIT:
        return False, f"bitiş > {SABAH_CALISIR_MAX_BITIS_SAATI}"
    return True, ""

def _k_sabah_sebep(bas, bit):
    """İhlal sebebini döner, ihlal yoksa None."""
    b, e = str(bas).strip()[:5], str(bit).strip()[:5]
    bas_dk, bit_dk = _dk(b), _dk(e)
    if bit_dk < bas_dk:
        bit_dk += 24 * 60
    if _K_GECE_YASAK and (b, e) in _K_GECE_SET:
        return "gece/akşam vardiyası"
    if bas_dk > _K_BASLANGIC:
        return f"başlangıç > {SABAH_CALISIR_MAX_BASLANGIC_SAATI}"
    if bit_dk > _K_BITIS:
        return f"bitiş > {SABAH_CALISIR_MAX_BITIS_SAATI}"
    return None
