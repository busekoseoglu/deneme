# %% [KISIT] - KIDEMSİZ AGENT GECE VARDİYASI YASAĞI (HARD)
# Kıdemi KIDEM_MIN_AY aydan az olan agent, KIDEM_YASAK_VARDIYA_BASLANGIC saatlerinde
# başlayan vardiyalara (00:00 gece vardiyası) atanamaz. Hafta içi / hafta sonu fark etmez.
#
# TAKIM KİLİDİYLE ÇAKIŞMA YOK: team_week_base bir KARAR değişkeni, veri değil.
# Takımda kıdemsiz agent varsa model o takıma gece vardiyasını base olarak SEÇMEZ,
# uygun bir base seçer. Kilit hiç zorlanmaz, gevşetmeye gerek yok.
#
# Parametreler [CONFIG] hücresinden gelir, burada TANIMLANMAZ.
# Veri [HÜCRE 3C] - KIDEM VERİSİ (SQL) hücresinden gelir.

_eksik_cfg = [k for k in ("KIDEM_MIN_AY", "KIDEM_YASAK_VARDIYA_BASLANGIC",
                          "KIDEM_EKSIK_VARSAYILAN_AY") if k not in globals()]
if _eksik_cfg:
    raise SystemExit(f"CONFIG'de tanımlı olmalı, eksik: {_eksik_cfg} -> [CONFIG] hücresini çalıştır.")
if "kidem_ay_map" not in globals():
    raise SystemExit("kidem_ay_map yok. Önce [HÜCRE 3C] - KIDEM VERİSİ (SQL) hücresini çalıştır.")

_yasak_bas = {str(b).strip()[:5] for b in KIDEM_YASAK_VARDIYA_BASLANGIC}

# ---------------- 1) VERİ DENETİMİ ----------------
# Kıdem tablosunda olmayan agent KIDEM_EKSIK_VARSAYILAN_AY (varsayılan 0) kabul edilir,
# yani KIDEMSİZ sayılır ve gece vardiyasına giremez. Güvenli taraf budur.
# Model DURMAZ, sadece uyarır - eksik listesi kidem_eksik_df içinde kalır.
_agents_x = [str(a).strip() for a in AGENTS]


def _kidem_of(a):
    _v = kidem_ay_map.get(a)
    if _v is None or pd.isna(_v):
        return float(KIDEM_EKSIK_VARSAYILAN_AY)
    return float(_v)


_kidem_yok = [a for a in _agents_x if a not in kidem_ay_map or pd.isna(kidem_ay_map.get(a))]

_kimlik_kd = ["agent_user_code", "agent_name", "takim"]
_kimlik_kd = [c for c in _kimlik_kd if c in df_tam.columns]
_kd_lookup = (
    df_tam[_kimlik_kd].assign(
        agent_user_code=df_tam["agent_user_code"].astype(str).str.strip()
    ).set_index("agent_user_code").to_dict("index")
)
kidem_eksik_df = pd.DataFrame([
    dict({"agent_user_code": a}, **{c: _kd_lookup.get(a, {}).get(c, "") for c in _kimlik_kd
                                   if c != "agent_user_code"},
         kidem_varsayilan=KIDEM_EKSIK_VARSAYILAN_AY)
    for a in _kidem_yok
])

if _kidem_yok:
    print("!" * 74)
    print(f"!!! KIDEM VERİSİ EKSİK: {len(_kidem_yok)} / {len(_agents_x)} agentin kıdem kaydı yok.")
    print(f"!!! Bu agentlar {KIDEM_EKSIK_VARSAYILAN_AY} ay kabul edildi -> KIDEMSİZ sayılıyorlar,")
    print("!!! gece vardiyasına atanamayacaklar. Veriyi düzeltince tekrar çalıştır.")
    print("!!! Tam liste: kidem_eksik_df")
    print("!" * 74)
    display(kidem_eksik_df.head(50))
else:
    print(f"Kıdem verisi tam: {len(_agents_x)}/{len(_agents_x)} agent.")

kidemsiz_agents = {a for a in _agents_x if _kidem_of(a) < float(KIDEM_MIN_AY)}
print(f"  bunun {len(_kidem_yok)} tanesi kıdem kaydı olmadığı için kıdemsiz sayıldı"
      if _kidem_yok else "")

# ---------------- 2) YASAK VARDİYALAR ----------------
_yasak_gun_vardiya = [
    (ds, v)
    for ds in PLAN_GUNLER
    for v in gun_vardiyalari.get(ds, [])
    if str(saat[(ds, v)][0]).strip()[:5] in _yasak_bas
]

kidem_gece_yasak_kisiti = 0
for a in kidemsiz_agents:
    for ds, v in _yasak_gun_vardiya:
        if (a, ds, v) in x:
            model.Add(x[(a, ds, v)] == 0).WithName(f"kidem_gece_{a}_{ds}_{v}")
            kidem_gece_yasak_kisiti += 1

print(f"KIDEM_MIN_AY: {KIDEM_MIN_AY} ay | yasak başlangıç saatleri: {sorted(_yasak_bas)}")
print(f"kıdemsiz agent: {len(kidemsiz_agents)} / {len(_agents_x)}")
print(f"yasak (gün, vardiya) kovası: {len(set(_yasak_gun_vardiya))}")
print(f"kurulan kısıt: {kidem_gece_yasak_kisiti}")

# ---------------- 3) ÖN KONTROL: gece vardiyasını kim karşılayacak? ----------------
# Gece vardiyasına ancak base'i gece olan takımlar girebilir (takım kilidi).
#
# ENABLE_TAKIM_BASE_MUAFIYET = True  -> kıdemsizi olan takım da gece base alabilir;
#     kıdemsiz agent o hafta kilitten MUAF olup başka vardiyada çalışır, takımın
#     geri kalanı geceye girer. Kapasite = TÜM agentlar - kıdemsizler.
# ENABLE_TAKIM_BASE_MUAFIYET = False -> kıdemsizi olan takım gece base alamaz
#     (alırsa o kişiler o hafta hiç çalışamaz). Kapasite = kıdemsizi OLMAYAN takımlar.
_agent_team_k = globals().get("agent_team", {})
_takim_kidemsiz = {}
_takim_boyut = {}
for a in _agents_x:
    t = str(_agent_team_k.get(a, "")).strip()
    if not t:
        continue
    _takim_boyut[t] = _takim_boyut.get(t, 0) + 1
    if a in kidemsiz_agents:
        _takim_kidemsiz[t] = _takim_kidemsiz.get(t, 0) + 1

_muafiyet_acik = bool(globals().get("ENABLE_TAKIM_BASE_MUAFIYET", False))
if _muafiyet_acik:
    # her takım gece base alabilir; sadece kıdemsizler o vardiyaya giremez
    _temiz_kapasite = sum(_takim_boyut.values()) - sum(_takim_kidemsiz.values())
    _kapasite_aciklama = "tüm takımlar (muafiyet açık), kıdemsizler hariç"
else:
    _temiz_kapasite = sum(v for t, v in _takim_boyut.items() if _takim_kidemsiz.get(t, 0) == 0)
    _kapasite_aciklama = "sadece kıdemsizi OLMAYAN takımlar (muafiyet kapalı)"
_cov_low_k = globals().get("coverage_lower", {})
_gece_max_talep = 0
for ds, v in _yasak_gun_vardiya:
    if pd.to_datetime(ds).weekday() in (0, 1, 2, 3, 4):
        _gece_max_talep = max(_gece_max_talep, int(_cov_low_k.get((ds, v), talep.get((ds, v), 0))))

kidem_takim_df = pd.DataFrame([
    {"takim": t, "agent_sayisi": _takim_boyut[t], "kidemsiz": _takim_kidemsiz.get(t, 0),
     "gece_base_alabilir": _muafiyet_acik or _takim_kidemsiz.get(t, 0) == 0}
    for t in sorted(_takim_boyut)
]).sort_values(["gece_base_alabilir", "kidemsiz"], ascending=[True, False])

print(f"\nÖN KONTROL - gece vardiyası kapasitesi (hafta içi):")
print(f"  takım base muafiyeti          : {'AÇIK' if _muafiyet_acik else 'KAPALI'}")
print(f"  kapasite kaynağı              : {_kapasite_aciklama}")
print(f"  gece vardiyasına girebilecek  : {_temiz_kapasite} agent")
print(f"  gece vardiyası en yüksek hafta içi talebi : {_gece_max_talep}")
if _gece_max_talep > _temiz_kapasite:
    print("  !!! UYARI: talep kapasiteyi AŞIYOR -> takım base kapasite guard'ı sıkışabilir.")
    print("  !!! INFEASIBLE riski. Muafiyet kapalıysa açmayı dene.")
else:
    print("  marj:", _temiz_kapasite - _gece_max_talep)
if len(kidemsiz_agents):
    print("\nKıdemsizi olan takımlar (muafiyet açıkken bunlar da gece base alabilir):")
    display(kidem_takim_df[kidem_takim_df["kidemsiz"] > 0])
