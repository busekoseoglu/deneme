# Off / izin günleri YAZILMAZ. Arife özel vardiyası (09:00-13:00) kendi saatiyle ayrı satır.
# Model sadece inhouse agent planlıyor. Kuyruk ve ay dosya adına plandan gelir (sabit yok).
import pickle

if "cozum_calisma" not in globals() or "cozum_atama" not in globals():
    raise SystemExit("cozum_calisma / cozum_atama yok. Önce [KONTROL] - ÇÖZÜMÜ ÇIKAR hücresini çalıştır.")

_ap_rows = []
for (a, ds), v in cozum_atama.items():
    if cozum_calisma.get((a, ds), 0) != 1:
        continue
    b, e = saat[(ds, v)]
    _ap_rows.append({"date": pd.to_datetime(ds).strftime("%Y-%m-%d"),
                     "vardiya": f"{str(b).strip()[:5]}-{str(e).strip()[:5]}"})
df_plan = (pd.DataFrame(_ap_rows, columns=["date", "vardiya"])
           .sort_values(["date", "vardiya"]).reset_index(drop=True))

_ap_bas = pd.to_datetime(min(PLAN_GUNLER))
AGENT_PLAN_PKL = f"agent_cov_{QUEUE}_{_ap_bas.year}_{_ap_bas.month:02d}_ark.pkl"
with open(AGENT_PLAN_PKL, "wb") as f:
    pickle.dump({"agent_plan": df_plan[["date", "vardiya"]]}, f)

print(f"agent_plan: {len(df_plan)} satır | {df_plan['date'].nunique()} gün | "
      f"çalışan agent-gün: {sum(cozum_calisma.values())} (eşit olmalı)")
print(f"yazıldı: {AGENT_PLAN_PKL}")
display(df_plan.head())
