# %% [KONTROL] - TAKIM VARDİYA GEÇMİŞ DENGESİ [RAPOR/SOFT]
# Ön koşul: [KONTROL] - ÇÖZÜMÜ ÇIKAR (cozum_calisma / cozum_atama)
#           [KISIT] - TAKIM VARDİYA GEÇMİŞ DENGESİ
# Denge SOFT'tur: burada "ihlal" yoktur, dağıtımın geçmişe UYUP UYMADIĞI ölçülür.
#
# ======= 2026-09-11 - YENİ FORMÜLASYONA UYARLANDI =======
# Kural artık kişi başı kümülatifi ortalamaya yaklaştırmıyor; geçmişten türetilmiş
# bir CEZA KATSAYISI ile "geçmişte çok almış takım bu ay o vardiyaya girmesin"
# diyor (bkz. kısıt hücresinin başlığı).
#
# Bunun raporlamaya yansıması: bu ayın katkısı artık modelde bir değişken DEĞİL.
# Burada doğrudan ÇÖZÜMDEN sayılıyor (cozum_atama), yani modele hiçbir yük
# bindirmeden. Geçmişle kıyaslanabilir olsun diye SADECE HAFTA İÇİ günler
# sayılıyor - geçmiş tablo da hafta içi sayıyor.
#
# İki soru, iki sütun grubu:
#   "önceden ne kadar vardı" -> gecmis_ham_agent_gun / gecmis_kisi_basi_gun
#   "bu ay ne verildi"       -> bu_ay_agent_gun / bu_ay_kisi_basi_gun / bu_ay_hafta
#
# Ham geçmişin satır satır hâli Excel'de ayrıca var: TAKIM_VARDIYA_GECMIS_HAM.
# ========================================================

takim_vardiya_denge_sonuc_df = pd.DataFrame()

_kd_katsayi = globals().get("takim_vardiya_denge_katsayi")
_kd_gecmis = globals().get("takim_vardiya_denge_gecmis", {})
_kd_hedef = globals().get("takim_vardiya_denge_hedef", {})
_kd_hafta = globals().get("takim_vardiya_denge_hafta", {})
_kd_asim = globals().get("takim_vardiya_denge_asim", {})
_kd_ceza = globals().get("takim_vardiya_denge_ceza", {})
_kd_ham_map = globals().get("takim_vardiya_gecmis_ham_map", {})
_kd_mevcut_map = globals().get("takim_mevcut_map", {})
_kd_solver = globals().get("solver")

if not _kd_katsayi:
    print("Takım vardiya denge değişkeni yok "
          "(ENABLE_TAKIM_VARDIYA_DENGE = False ya da kısıt hücresi çalışmadı) - kontrol atlandı.")
elif _kd_solver is None:
    print("solver yok - önce [KONTROL 0] - SOLVER hücresini çalıştır.")
elif "cozum_atama" not in globals():
    print("cozum_atama yok - önce [KONTROL] - ÇÖZÜMÜ ÇIKAR hücresini çalıştır.")
else:
    _kd_w = int(globals().get("TAKIM_VARDIYA_DENGE_W", 0))
    _kd_saat = globals().get("saat", {})

    # --- BU AY: çözümden agent-gün sayımı (hafta içi, başlangıç saatine göre) ---
    # Modele değişken eklemeden, sadece çözüm okunarak yapılıyor.
    # 2026-09-14: agent-gün BASE / BASE DIŞI diye AYRIŞTIRILIYOR.
    # Kural yalnızca team_week_base'e ceza yazıyor, yani sadece takımın base'ini
    # etkileyebiliyor. Muafiyetli agentlar ve sarkan haftada devir alanlar takım
    # base'ini TAKİP ETMİYOR - onların 15:00 çalışması kuralın erişemediği bir
    # atama. İkisini tek sayıda toplamak korelasyonu ölçülemez hâle getiriyordu:
    # W artıyor, metrik oynamıyor, çünkü metriğin bir kısmı kuralın dışında.
    _kd_twb = globals().get("team_week_base", {})
    _kd_dw = globals().get("day_week", {})

    # (takim, hafta) -> o haftanın base başlangıç saati (çözümden)
    _kd_base_bas = {}
    if _kd_twb:
        for (_t_, _wk_, _v_), _var_ in _kd_twb.items():
            try:
                if _kd_solver.Value(_var_) != 1:
                    continue
            except Exception:
                continue
            # bu vardiyanın başlangıcı: o haftanın herhangi bir gününden
            for _ds_ in PLAN_GUNLER:
                if _kd_dw.get(_ds_) == _wk_ and (_ds_, _v_) in _kd_saat:
                    _kd_base_bas[(str(_t_).strip(), _wk_)] = \
                        str(_kd_saat[(_ds_, _v_)][0]).strip()[:5]
                    break

    _kd_buay_ag = {}          # (takim, baslangic) -> agent-gün (TOPLAM)
    _kd_buay_base = {}        # (takim, baslangic) -> base haftalarında olan kısım
    for (_a, _ds), _v in cozum_atama.items():
        if cozum_calisma.get((_a, _ds), 0) != 1 or _v is None:
            continue
        if pd.to_datetime(_ds).weekday() > 4:
            continue          # geçmiş sadece hafta içi sayıyor
        _st = _kd_saat.get((_ds, _v))
        if not _st:
            continue
        _b = str(_st[0]).strip()[:5]
        _t = agent_team.get(str(_a).strip())
        if _t is None or pd.isna(_t):
            continue
        _t = str(_t).strip()
        _key = (_t, _b)
        _kd_buay_ag[_key] = _kd_buay_ag.get(_key, 0) + 1
        # takımın O HAFTAKİ base'i bu başlangıç mı?
        if _kd_base_bas.get((_t, _kd_dw.get(_ds))) == _b:
            _kd_buay_base[_key] = _kd_buay_base.get(_key, 0) + 1

    # --- BÖLEN TUTARLILIĞI: veriden gelen agent_sayisi vs MODELİN üyeleri ---
    # 2026-09-14: İki taraf aynı kişileri saymayabilir. df_tam, [HÜCRE 4] ve
    # [HÜCRE 5]'ten geçmiş hâli: PT'liler (kismi_calisan_flg=1) ve eğitim
    # ekipleri ELENMİŞ. Veriden gelen agent_sayisi kaynak sistemin mevcudu ise
    # onları İÇERİYOR olabilir. O zaman kişi başı sayılar iki tarafta farklı
    # tabana bölünür ve korelasyon yanlış ölçülür.
    _kd_model_uye = {}
    for _a_ in AGENTS:
        _t_ = agent_team.get(str(_a_).strip())
        if _t_ is None or pd.isna(_t_):
            continue
        _t_ = str(_t_).strip()
        _kd_model_uye[_t_] = _kd_model_uye.get(_t_, 0) + 1

    _kd_bolen_fark = []
    for _t_, _nm in sorted(_kd_model_uye.items()):
        _nv = _kd_mevcut_map.get(_t_)
        if _nv and int(_nv) != int(_nm):
            _kd_bolen_fark.append({
                "takim": _t_, "veriden_agent_sayisi": int(_nv),
                "modelin_uye_sayisi": int(_nm),
                "fark": int(_nv) - int(_nm),
                "fark_yuzde": round(100 * (int(_nv) - int(_nm)) / int(_nv), 1),
            })
    takim_denge_bolen_fark_df = pd.DataFrame(_kd_bolen_fark)
    # 2026-09-14: BU BİR HATA DEĞİL, BEKLENEN DURUM. Fark normal çünkü df_tam
    # PT'lileri (kismi_calisan_flg=1) ve eğitim ekiplerini eliyor; veriden gelen
    # agent_sayisi ise önceki ayın tam mevcudu. İş birimi kararı: bütün kişi başı
    # hesaplarda VERİDEN GELEN kullanılır - geçmişteki gun_agent_ratio de onunla
    # hesaplandığı için iki taraf aynı tabana bölünmüş olur.
    # Burada sadece farkın BÜYÜKLÜĞÜ izleniyor: bir takımda anormal büyük fark
    # varsa o takımın mevcudu ay içinde ciddi değişmiş ya da veri eski demektir.
    if _kd_bolen_fark:
        _bf = takim_denge_bolen_fark_df
        print(f"\nBölen: veriden gelen agent_sayisi kullanılıyor (önceki ay mevcudu).")
        print(f"  modelin üye sayısından farklı olan takım: "
              f"{len(_bf)} / {len(_kd_model_uye)}  -> NORMAL (PT + eğitim elenmiş)")
        print(f"  ortalama fark %{_bf['fark_yuzde'].abs().mean():.1f}, "
              f"en büyük %{_bf['fark_yuzde'].abs().max():.1f}")
        _bf_buyuk = _bf[_bf["fark_yuzde"].abs() > 40]
        if len(_bf_buyuk):
            print(f"  !!! %40'tan fazla sapan {len(_bf_buyuk)} takım - mevcut ay içinde "
                  f"ciddi değişmiş ya da veri eski olabilir:")
            print(_bf_buyuk.reindex(_bf_buyuk["fark_yuzde"].abs()
                  .sort_values(ascending=False).index).head(10).to_string(index=False))
        print("  (tablo: takim_denge_bolen_fark_df)")
    else:
        print("\nBölen: veriden gelen agent_sayisi = modelin üye sayısı ✓")

    _kd_rows = []
    for (_t, _b), _w_t in sorted(_kd_katsayi.items()):
        _n = _kd_mevcut_map.get(_t)
        _n_model = _kd_model_uye.get(_t)
        _ag = int(_kd_buay_ag.get((_t, _b), 0))
        _hf = (int(_kd_solver.Value(_kd_hafta[(_t, _b)]))
               if (_t, _b) in _kd_hafta else None)
        _as = (int(_kd_solver.Value(_kd_asim[(_t, _b)]))
               if (_t, _b) in _kd_asim else 0)
        _gec_kb = float(_kd_gecmis.get((_t, _b), 0.0))
        _kd_rows.append({
            "takim": _t,
            "baslangic": _b,
            "agent_sayisi": _n,
            "modelin_uye_sayisi": _n_model,
            # --- GEÇMİŞ (Ocak - önceki ay sonu) ---
            "gecmis_ham_agent_gun": round(float(_kd_ham_map.get((_t, _b), 0.0)), 1),
            "gecmis_kisi_basi_gun": round(_gec_kb, 2),
            "ortalama": round(float(_kd_hedef.get(_b, 0.0)), 2),
            "ortalamadan_fark": round(_gec_kb - float(_kd_hedef.get(_b, 0.0)), 2),
            "ceza_katsayisi": int(_w_t),
            "hafta_basina_ceza": int(_w_t) * _kd_w,
            # --- BU AY (çözümden) ---
            "bu_ay_agent_gun": _ag,
            # base haftalarında olan kısım = kuralın ETKİLEYEBİLDİĞİ atama
            "bu_ay_agent_gun_base": int(_kd_buay_base.get((_t, _b), 0)),
            # kalanı muafiyetli / devir agentlar = kuralın DIŞI
            "bu_ay_agent_gun_base_disi": _ag - int(_kd_buay_base.get((_t, _b), 0)),
            # 2026-09-14: KİŞİ BAŞI HESAP HER YERDE VERİDEN GELEN agent_sayisi
            # İLE YAPILIR (iş birimi kararı). Sebebi: geçmişteki gun_agent_ratio
            # o bölenle hesaplandı; modelin kendi üye sayısını kullanmak
            # geçmişle KIYASLANAMAZ bir sayı üretir.
            # agent_sayisi önceki ayın mevcudu - geçmişle aynı taban olduğu için
            # doğru olan bu, bu ayın mevcudu değil.
            "bu_ay_kisi_basi_gun": (round(_ag / _n, 2) if _n else None),
            "bu_ay_hafta": _hf,
            "tavan_asimi_hafta": _as,
            "odenen_ceza": (int(_kd_solver.Value(_kd_ceza[(_t, _b)])) * _kd_w
                            if (_t, _b) in _kd_ceza else 0),
        })

    takim_vardiya_denge_sonuc_df = pd.DataFrame(_kd_rows).sort_values(
        ["baslangic", "gecmis_kisi_basi_gun"]
    ).reset_index(drop=True)

    _toplam_asan = 0
    print("=== TAKIM VARDİYA GEÇMİŞ DENGESİ (katsayı cezası formülasyonu) ===")
    for _b, _g in takim_vardiya_denge_sonuc_df.groupby("baslangic"):
        _asan = _g[_g["tavan_asimi_hafta"] > 0]
        _toplam_asan += len(_asan)
        _alan = _g[_g["bu_ay_agent_gun"] > 0]
        _bedava = _g[_g["ceza_katsayisi"] == 0]
        _cezali = _g[_g["ceza_katsayisi"] > 0]

        print(f"\n--- '{_b}' başlangıçlı vardiya | {len(_g)} takım ---")
        print(f"  bu ay o vardiyada çalışan takım : {len(_alan)} / {len(_g)}")
        print(f"  bu ay dağıtılan AGENT-GÜN       : {int(_g['bu_ay_agent_gun'].sum()):,}")
        print(f"  katsayısı 0 (bedava) takım      : {len(_bedava)}")
        print(f"  cezalı takım                    : {len(_cezali)}")

        # --- KURALIN ÇALIŞTIĞININ KANITI ---
        # Katsayısı 0 olanlar o vardiyayı ALMALI, cezalılar ALMAMALI.
        if len(_bedava) and len(_cezali):
            _b_ort = _bedava["bu_ay_kisi_basi_gun"].mean()
            _c_ort = _cezali["bu_ay_kisi_basi_gun"].mean()
            print(f"  bu ay kişi başı ortalama:")
            print(f"      katsayısı 0 olanlar         : {_b_ort:.2f} gün")
            print(f"      cezalı olanlar              : {_c_ort:.2f} gün")
            if pd.notna(_b_ort) and pd.notna(_c_ort):
                print(f"      fark                        : {_b_ort - _c_ort:+.2f} gün  "
                      f"({'DOĞRU YÖN (az almışa verildi)' if _b_ort > _c_ort else 'DİKKAT: çok almışa veriliyor'})")

        # --- KURALIN GÜCÜ NEREYE YETİYOR ---
        # Kural yalnızca base'i etkiliyor; agent-günün base DIŞI kısmı
        # (muafiyetli + devir agentlar) onun erişemediği atama.
        _ag_top = int(_g["bu_ay_agent_gun"].sum())
        _ag_base = int(_g["bu_ay_agent_gun_base"].sum())
        _ag_dis = _ag_top - _ag_base
        print(f"  agent-gün dağılımı:")
        print(f"      takım base'inden        : {_ag_base:,}"
              f"  (%{100 * _ag_base / _ag_top:.0f} - KURALIN ETKİ ALANI)"
              if _ag_top else "      (bu ay hiç atama yok)")
        if _ag_top:
            print(f"      muaf / devir agentlar  : {_ag_dis:,}"
                  f"  (%{100 * _ag_dis / _ag_top:.0f} - kural dokunamaz)")
            if _ag_dis > _ag_base:
                print(f"      !!! Base DIŞI kısım daha büyük. Kural bu vardiyanın "
                      f"yarısından fazlasını kontrol etmiyor;")
                print(f"          ağırlık artırmak korelasyonu çeviremez.")

        if len(_g) and _g["gecmis_kisi_basi_gun"].nunique() > 1:
            # ASIL METRİK: kuralın kontrol ettiği değişkenle korelasyon
            _kor_hf = _g["gecmis_kisi_basi_gun"].corr(_g["bu_ay_hafta"])
            print(f"  KORELASYON (geçmiş - base haftası) : {_kor_hf:+.2f} "
                  f"({'DOĞRU YÖN (az almışa verildi)' if _kor_hf < 0 else 'DİKKAT'})")
            print(f"     ^ kuralın kontrol ettiği değişken - BUNA BAKILACAK")
            _kor = _g["gecmis_kisi_basi_gun"].corr(_g["bu_ay_kisi_basi_gun"])
            print(f"  korelasyon (geçmiş - toplam agent-gün): {_kor:+.2f}")
            print(f"     ^ içinde muaf/devir atamaları var, kural sorumlu değil")

        print(f"  ödenen ceza (objective'e giren)  : {int(_g['odenen_ceza'].sum()):,}")
        print(f"  aylık tavanı ({int(globals().get('TAKIM_VARDIYA_DENGE_MAX_HAFTA', 0))} hafta) "
              f"aşan takım: {len(_asan)}")
        if len(_asan):
            print("    (kapasite koruması zorladı - tavan SOFT olduğu için model durmadı)")
            for _, _r in _asan.head(5).iterrows():
                print(f"      {_r['takim']}: {int(_r['bu_ay_hafta'])} hafta "
                      f"(+{int(_r['tavan_asimi_hafta'])})")

        _kol = ["gecmis_kisi_basi_gun", "ortalamadan_fark", "ceza_katsayisi",
                "bu_ay_hafta", "bu_ay_agent_gun_base",
                "bu_ay_agent_gun_base_disi", "bu_ay_agent_gun"]

        # 2026-09-11: "BU AY KİM ALDI" listesi eklendi. Önceki hâl yalnızca
        # geçmişe göre sıralıyordu; bu ayın dağıtımını görmek için tabloyu
        # Excel'de tekrar sıralamak gerekiyordu.
        print("\n  BU AY EN ÇOK ALAN 10 takım:")
        _bu_alan = _g[_g["bu_ay_agent_gun"] > 0].nlargest(10, "bu_ay_agent_gun")
        if len(_bu_alan):
            print(_bu_alan[["takim"] + _kol].to_string(index=False))
            print(f"    bunlardan CEZALI olan: "
                  f"{int((_bu_alan['ceza_katsayisi'] > 0).sum())} / {len(_bu_alan)}"
                  f"   (az olması iyi - kural çalışıyorsa cezalılar listede olmamalı)")
        else:
            print("    bu ay bu vardiyada hiçbir takım çalışmamış.")

        print("\n  GEÇMİŞTE EN ÇOK alan 5 takım (bu ay girmemeli):")
        print(_g.nlargest(5, "gecmis_kisi_basi_gun")[["takim"] + _kol].to_string(index=False))
        print("\n  GEÇMİŞTE EN AZ alan 5 takım (bu ay girmeli):")
        print(_g.nsmallest(5, "gecmis_kisi_basi_gun")[["takim"] + _kol].to_string(index=False))

    # --- DENGENİN OBJECTIVE'E FİİLEN KATTIĞI PUAN ---
    # Coverage bozulduğunda "suçlu denge mi" sorusunun tek kesin cevabı bu.
    _kd_toplam_ceza = int(takim_vardiya_denge_sonuc_df["odenen_ceza"].sum())
    _kd_asim_katki = 0
    if _kd_asim and "TAKIM_VARDIYA_DENGE_MAX_HAFTA_ASIM_W" in globals():
        _kd_asim_katki = sum(int(_kd_solver.Value(_v)) for _v in _kd_asim.values()) \
            * int(TAKIM_VARDIYA_DENGE_MAX_HAFTA_ASIM_W)
    print(f"\n=== DENGENİN OBJECTIVE'E KATKISI ===")
    print(f"  katsayı cezası     : {_kd_toplam_ceza:,}")
    print(f"  tavan aşımı cezası : {_kd_asim_katki:,}")
    print(f"  DENGE TOPLAM       : {_kd_toplam_ceza + _kd_asim_katki:,}")
    try:
        _kd_obj = _kd_solver.ObjectiveValue()
        _kd_oran = (100 * (_kd_toplam_ceza + _kd_asim_katki) / _kd_obj) if _kd_obj else 0
        print(f"  toplam objective   : {_kd_obj:,.0f}")
        print(f"  dengenin payı      : %{_kd_oran:.3f}")
        if _kd_oran > 1:
            print("  !!! Denge objective'in %1'inden fazlasını tutuyor - coverage'a "
                  "bulaşmaya başlıyor olabilir. COVERAGE_GUNLUK'u önceki run ile kıyasla.")
    except Exception:
        pass
    print(f"  KIYAS: eksik slot 50.000 (buffer içi) / 300.000 (buffer altı)")

    # --- W KALİBRASYONU: mevcut ceza büyüklüğü ve önerilen çarpan ---
    # Hedef: en pahalı takımın 1 haftası 5.000-20.000 bandında olsun
    # (tavan aşımı 5.000 ile aynı mertebe, eksik slot 50.000'in belirgin altı).
    _kd_hbc = takim_vardiya_denge_sonuc_df["hafta_basina_ceza"]
    if len(_kd_hbc) and _kd_hbc.max() > 0:
        _kd_max_hbc = int(_kd_hbc.max())
        print(f"\n=== W KALİBRASYONU ===")
        print(f"  şu anki W                      : {_kd_w}")
        print(f"  en pahalı takımın 1 haftası    : {_kd_max_hbc:,} puan")
        print(f"  hedef bant                     : 5.000 - 20.000 puan/hafta")
        print(f"  ÖNERİLEN W (hedef 10.000 için) : "
              f"{max(1, round(10000 / (_kd_max_hbc / max(1, _kd_w))))}")
    elif len(_kd_hbc):
        print(f"\n=== W KALİBRASYONU ===")
        print(f"  Bütün katsayılar 0 ya da W=0 - kural etkisiz. "
              f"W'yi artır ya da geçmiş verideki yayılımı kontrol et.")

    if "kaydet" in globals():
        kaydet("Takım vardiya aylık tavanı (denge)", _toplam_asan, tur="SOFT")

    print(f"\nDetay: takim_vardiya_denge_sonuc_df "
          f"({takim_vardiya_denge_sonuc_df.shape[0]} satır) -> Excel TAKIM_VARDIYA_DENGE")
    print("  geçmişin satır satır ham hâli        -> Excel TAKIM_VARDIYA_GECMIS_HAM")
