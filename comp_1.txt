# ===== 2026-08-12 EKLENEN - KIDEM / GECE VARDIYASI YASAGI =====
KIDEM_MIN_AY = CONFIG["KIDEM_MIN_AY"]
KIDEM_YASAK_VARDIYA_BASLANGIC = CONFIG["KIDEM_YASAK_VARDIYA_BASLANGIC"]
# ==============================================================




# %% [HÜCRE 3C] - KIDEM VERİSİ (SQL)
# NOT: Diğer SQL çekme hücreleriyle birlikte EN ÜSTTE tutulur (df_agents/df_off/df_izin/df_gecmis yanı).
# Sadece sync_engine + pd yeterli.
#
# BEKLENEN ŞEMA:
#   agent_user_code | kidem_ay      (kidem_ay = sysdate-1 eksi işe giriş tarihi, AY cinsinden)
# Ek kolonlar (name vb.) varsa dokunulmaz, olduğu gibi df_kidem'de kalır.

KIDEM_TABLE = "BNS_VP_AGENT_KIDEM"      # <-- KENDİ TABLO ADINLA DEĞİŞTİR

query = f"SELECT * FROM {KIDEM_TABLE}"
chunks = pd.read_sql(query, sync_engine, chunksize=10000)
df_kidem = pd.concat(chunks, ignore_index=True)
df_kidem.columns = [str(c).strip().lower() for c in df_kidem.columns]

print(f"df_kidem: {df_kidem.shape[0]} satır x {df_kidem.shape[1]} kolon")
print(f"kolonlar: {list(df_kidem.columns)}")

_eksik_kol = [c for c in ["agent_user_code", "kidem_ay"] if c not in df_kidem.columns]
if _eksik_kol:
    raise SystemExit(f"df_kidem'de beklenen kolon(lar) yok: {_eksik_kol}")

df_kidem["agent_user_code"] = df_kidem["agent_user_code"].astype(str).str.strip()
df_kidem["kidem_ay"] = pd.to_numeric(df_kidem["kidem_ay"], errors="coerce")

_bozuk = int(df_kidem["kidem_ay"].isna().sum())
if _bozuk:
    print(f"!!! UYARI: kidem_ay boş/sayıya çevrilemeyen {_bozuk} satır var.")

# agent -> kidem_ay
kidem_ay_map = dict(zip(df_kidem["agent_user_code"], df_kidem["kidem_ay"]))

print(f"benzersiz agent: {df_kidem['agent_user_code'].nunique()}")
print(f"kidem_ay aralığı: {df_kidem['kidem_ay'].min()} .. {df_kidem['kidem_ay'].max()}")
display(df_kidem.head())




# %% [KISIT] - KIDEMSİZ AGENT GECE VARDİYASI YASAĞI (HARD)
# Kıdemi KIDEM_MIN_AY aydan az olan agent, KIDEM_YASAK_VARDIYA_BASLANGIC saatlerinde
# başlayan vardiyalara (00:00 gece vardiyası) atanamaz. Hafta içi / hafta sonu fark etmez.
#
# TAKIM KİLİDİYLE ÇAKIŞMA YOK: team_week_base bir KARAR değişkeni, veri değil.
# Takımda kıdemsiz agent varsa model o takıma gece vardiyasını base olarak SEÇMEZ,
# uygun bir base seçer. Kilit hiç zorlanmaz, gevşetmeye gerek yok.
#
# Parametreler [CONFIG] hücresinden gelir, burada TANIMLANMAZ.
# Veri [HÜCRE 3C] - KIDEM VERİSİ (SQL) hücresinden gelir.

_eksik_cfg = [k for k in ("KIDEM_MIN_AY", "KIDEM_YASAK_VARDIYA_BASLANGIC") if k not in globals()]
if _eksik_cfg:
    raise SystemExit(f"CONFIG'de tanımlı olmalı, eksik: {_eksik_cfg} -> [CONFIG] hücresini çalıştır.")
if "kidem_ay_map" not in globals():
    raise SystemExit("kidem_ay_map yok. Önce [HÜCRE 3C] - KIDEM VERİSİ (SQL) hücresini çalıştır.")

_yasak_bas = {str(b).strip()[:5] for b in KIDEM_YASAK_VARDIYA_BASLANGIC}

# ---------------- 1) VERİ DENETİMİ: her agentin kıdemi olmalı ----------------
_agents_x = [str(a).strip() for a in AGENTS]
_kidem_yok = [a for a in _agents_x if a not in kidem_ay_map or pd.isna(kidem_ay_map.get(a))]
if _kidem_yok:
    print("!" * 70)
    print(f"!!! KIDEM VERİSİ EKSİK: {len(_kidem_yok)} / {len(_agents_x)} agentin kıdemi yok.")
    print("!!! Bu agentlar için gece yasağı UYGULANAMAZ.")
    print("!!! İlk 30:", _kidem_yok[:30])
    print("!" * 70)
    kidem_eksik_df = pd.DataFrame({"agent_user_code": _kidem_yok})
    raise SystemExit(
        f"{len(_kidem_yok)} agentin kıdem kaydı yok. Kıdem tablosunu tamamla, "
        "listesi kidem_eksik_df içinde."
    )
else:
    kidem_eksik_df = pd.DataFrame(columns=["agent_user_code"])
    print(f"Kıdem verisi tam: {len(_agents_x)}/{len(_agents_x)} agent.")

kidemsiz_agents = {a for a in _agents_x if float(kidem_ay_map[a]) < float(KIDEM_MIN_AY)}

# ---------------- 2) YASAK VARDİYALAR ----------------
_yasak_gun_vardiya = [
    (ds, v)
    for ds in PLAN_GUNLER
    for v in gun_vardiyalari.get(ds, [])
    if str(saat[(ds, v)][0]).strip()[:5] in _yasak_bas
]

kidem_gece_yasak_kisiti = 0
for a in kidemsiz_agents:
    for ds, v in _yasak_gun_vardiya:
        if (a, ds, v) in x:
            model.Add(x[(a, ds, v)] == 0).WithName(f"kidem_gece_{a}_{ds}_{v}")
            kidem_gece_yasak_kisiti += 1

print(f"KIDEM_MIN_AY: {KIDEM_MIN_AY} ay | yasak başlangıç saatleri: {sorted(_yasak_bas)}")
print(f"kıdemsiz agent: {len(kidemsiz_agents)} / {len(_agents_x)}")
print(f"yasak (gün, vardiya) kovası: {len(set(_yasak_gun_vardiya))}")
print(f"kurulan kısıt: {kidem_gece_yasak_kisiti}")

# ---------------- 3) ÖN KONTROL: gece vardiyasını kim karşılayacak? ----------------
# Takım base kilidi yüzünden gece vardiyasına ancak base'i gece olan takımlar girebilir.
# Kıdemsizi olan bir takım gece base'i alırsa o kişiler o hafta çalışamaz -> model
# o takımı seçmez. Dolayısıyla gece kapasitesi KIDEMSİZİ OLMAYAN takımlarla sınırlı.
_agent_team_k = globals().get("agent_team", {})
_takim_kidemsiz = {}
_takim_boyut = {}
for a in _agents_x:
    t = str(_agent_team_k.get(a, "")).strip()
    if not t:
        continue
    _takim_boyut[t] = _takim_boyut.get(t, 0) + 1
    if a in kidemsiz_agents:
        _takim_kidemsiz[t] = _takim_kidemsiz.get(t, 0) + 1

_temiz_kapasite = sum(v for t, v in _takim_boyut.items() if _takim_kidemsiz.get(t, 0) == 0)
_cov_low_k = globals().get("coverage_lower", {})
_gece_max_talep = 0
for ds, v in _yasak_gun_vardiya:
    if pd.to_datetime(ds).weekday() in (0, 1, 2, 3, 4):
        _gece_max_talep = max(_gece_max_talep, int(_cov_low_k.get((ds, v), talep.get((ds, v), 0))))

kidem_takim_df = pd.DataFrame([
    {"takim": t, "agent_sayisi": _takim_boyut[t], "kidemsiz": _takim_kidemsiz.get(t, 0),
     "gece_base_alabilir": _takim_kidemsiz.get(t, 0) == 0}
    for t in sorted(_takim_boyut)
]).sort_values(["gece_base_alabilir", "kidemsiz"], ascending=[True, False])

print(f"\nÖN KONTROL - gece vardiyası kapasitesi (hafta içi):")
print(f"  kıdemsizi OLMAYAN takımların toplam mevcudu : {_temiz_kapasite}")
print(f"  gece vardiyası en yüksek hafta içi talebi   : {_gece_max_talep}")
if _gece_max_talep > _temiz_kapasite:
    print("  !!! UYARI: talep, kıdemsizi olmayan takımların kapasitesini AŞIYOR.")
    print("  !!! Takım base kapasite guard'ı sıkışabilir -> INFEASIBLE riski.")
else:
    print("  marj:", _temiz_kapasite - _gece_max_talep)
if len(kidemsiz_agents):
    print("\nKıdemsizi olan takımlar:")
    display(kidem_takim_df[kidem_takim_df["kidemsiz"] > 0])




# %% [KONTROL] - KIDEMSİZ AGENT GECE VARDİYASI (HARD)
# Kıdemi KIDEM_MIN_AY aydan az olan agent gece vardiyasında çalışmamalı.
# Bu hücre modele DOKUNMAZ, mevcut çözümü denetler.

_K_MIN = globals().get("KIDEM_MIN_AY")
_K_BAS = {str(b).strip()[:5] for b in globals().get("KIDEM_YASAK_VARDIYA_BASLANGIC", ())}
_K_MAP = globals().get("kidem_ay_map", {})

if _K_MIN is None or not _K_BAS:
    print("!!! KIDEM_MIN_AY / KIDEM_YASAK_VARDIYA_BASLANGIC yok - kontrol atlandı.")
    kidem_ihlal_df = pd.DataFrame()
else:
    _agents_k = [str(a).strip() for a in AGENTS]
    _kidemsiz_k = {
        a for a in _agents_k
        if a in _K_MAP and not pd.isna(_K_MAP[a]) and float(_K_MAP[a]) < float(_K_MIN)
    }
    _kidem_yok_k = [a for a in _agents_k if a not in _K_MAP or pd.isna(_K_MAP.get(a))]

    _rows = []
    for a in _kidemsiz_k:
        for ds in PLAN_GUNLER:
            v = cozum_atama.get((a, ds))
            if not v:
                continue
            _bas, _bit = saat[(ds, v)]
            if str(_bas).strip()[:5] in _K_BAS:
                _rows.append({
                    "agent_user_code": a,
                    "kidem_ay": float(_K_MAP[a]),
                    "tarih": pd.to_datetime(ds).strftime("%Y-%m-%d"),
                    "vardiya": v,
                    "baslangic": str(_bas).strip()[:5],
                    "bitis": str(_bit).strip()[:5],
                })
    kidem_ihlal_df = pd.DataFrame(_rows)

    print(f"KIDEM_MIN_AY: {_K_MIN} ay | yasak başlangıç: {sorted(_K_BAS)}")
    print(f"kıdemsiz agent: {len(_kidemsiz_k)} / {len(_agents_k)}")
    if _kidem_yok_k:
        print(f"!!! kıdem kaydı OLMAYAN agent: {len(_kidem_yok_k)} - bunlar kontrol edilemedi")
    kaydet(f"Kıdemsiz agent gece vardiyasında değil (<{_K_MIN} ay)", len(_rows))
    if _rows:
        display(kidem_ihlal_df.sort_values(["agent_user_code", "tarih"]))
