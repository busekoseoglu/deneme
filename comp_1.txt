# %% [KARŞILAŞTIRMA 1] - EKİP PLANI (SQL)
# İş biriminin ELLE hazırladığı Ağustos planını çeker ve model planıyla aynı
# yapıya çevirir. Karşılaştırma hücresi bunu kullanır.
#
# BEKLENEN ŞEMA:
#   agent_user_code | date | vardiya
# vardiya kolonu ÜÇ tip değer taşır:
#   "08:00-17:00"  -> çalışma (saat aralığı)
#   "off"          -> çalışılmayan gün
#   "izin"         -> izinli gün
#
# Bu hücre modele DOKUNMAZ, sadece veri hazırlar.

import re

EKIP_PLANI_TABLE = "<TABLO_ADINI_YAZ>"      # <-- KENDİ TABLO ADINLA DEĞİŞTİR

query = f"SELECT * FROM {EKIP_PLANI_TABLE}"
chunks = pd.read_sql(query, sync_engine, chunksize=50000)
df_ekip_plani = pd.concat(chunks, ignore_index=True)
df_ekip_plani.columns = [str(c).strip().lower() for c in df_ekip_plani.columns]

print(f"df_ekip_plani: {df_ekip_plani.shape[0]} satır x {df_ekip_plani.shape[1]} kolon")
print(f"kolonlar: {list(df_ekip_plani.columns)}")

_bek = ["agent_user_code", "date", "vardiya"]
_eksik = [c for c in _bek if c not in df_ekip_plani.columns]
if _eksik:
    raise SystemExit(f"df_ekip_plani'nda beklenen kolon(lar) yok: {_eksik}")

df_ekip_plani["agent_user_code"] = df_ekip_plani["agent_user_code"].astype(str).str.strip()
df_ekip_plani["vardiya"] = df_ekip_plani["vardiya"].astype(str).str.strip()
df_ekip_plani["ep_date"] = pd.to_datetime(df_ekip_plani["date"], errors="coerce").dt.date


def _ep_coz(v):
    """'08:00-17:00' -> ('calisma', '08:00', '17:00') | 'off' | 'izin' | bilinmeyen"""
    t = str(v).strip()
    tl = t.lower()
    if tl in ("off", "of", "tatil", "hafta tatili"):
        return ("off", None, None)
    if tl in ("izin", "yillik izin", "yıllık izin", "idari izin", "ucretsiz izin"):
        return ("izin", None, None)
    m = re.match(r"^(\d{1,2})[:.](\d{2})\s*-\s*(\d{1,2})[:.](\d{2})$", t)
    if m:
        _b = f"{int(m.group(1)):02d}:{m.group(2)}"
        _e = f"{int(m.group(3)):02d}:{m.group(4)}"
        return ("calisma", _b, _e)
    return ("bilinmeyen", None, None)


_coz = df_ekip_plani["vardiya"].map(_ep_coz)
df_ekip_plani["tip"] = [c[0] for c in _coz]
df_ekip_plani["baslangic"] = [c[1] for c in _coz]
df_ekip_plani["bitis"] = [c[2] for c in _coz]

print("\nvardiya tipi dağılımı:")
print(df_ekip_plani["tip"].value_counts().to_string())

_bos_tarih = int(df_ekip_plani["ep_date"].isna().sum())
if _bos_tarih:
    print(f"!!! UYARI: tarihi okunamayan {_bos_tarih} satır var.")

# --- plan dönemine kırp ---
_plan_d = {pd.to_datetime(d).date() for d in PLAN_GUNLER}
_ds_of_date = {pd.to_datetime(d).date(): d for d in PLAN_GUNLER}
df_ekip_plani_ay = df_ekip_plani[df_ekip_plani["ep_date"].isin(_plan_d)].copy()

print(f"\nplan dönemine düşen satır: {len(df_ekip_plani_ay)} "
      f"({df_ekip_plani_ay['agent_user_code'].nunique()} agent)")
print(f"tarih aralığı: {df_ekip_plani['ep_date'].min()} .. {df_ekip_plani['ep_date'].max()}")

# --- aynı agent-gün için birden fazla satır var mı? (günde tek vardiya kuralı) ---
_cift = (df_ekip_plani_ay[df_ekip_plani_ay["tip"] == "calisma"]
         .groupby(["agent_user_code", "ep_date"]).size())
ekip_cift_vardiya_df = _cift[_cift > 1].reset_index(name="vardiya_sayisi")
print(f"aynı gün BİRDEN FAZLA vardiya atanmış agent-gün: {len(ekip_cift_vardiya_df)}")

# --- model planıyla aynı yapı: (agent, ds) -> 0/1 ve (agent, ds) -> (bas, bit) ---
ekip_calisma, ekip_saat = {}, {}
for _a, _d, _tip, _b, _e in zip(df_ekip_plani_ay["agent_user_code"],
                                df_ekip_plani_ay["ep_date"],
                                df_ekip_plani_ay["tip"],
                                df_ekip_plani_ay["baslangic"],
                                df_ekip_plani_ay["bitis"]):
    _ds = _ds_of_date.get(_d)
    if _ds is None:
        continue
    _a = str(_a).strip()
    if _tip == "calisma":
        ekip_calisma[(_a, _ds)] = 1
        ekip_saat[(_a, _ds)] = (_b, _e)
    else:
        ekip_calisma.setdefault((_a, _ds), 0)

ekip_agentlar = sorted({a for (a, ds) in ekip_calisma})
print(f"\nekip_calisma kaydı : {len(ekip_calisma)}")
print(f"ekip planındaki agent: {len(ekip_agentlar)}")

# --- ekip planındaki izin günleri (kendi kaydı) ---
ekip_izin_gunleri = {}
for _a, _d, _tip in zip(df_ekip_plani_ay["agent_user_code"],
                        df_ekip_plani_ay["ep_date"],
                        df_ekip_plani_ay["tip"]):
    if _tip == "izin":
        ekip_izin_gunleri.setdefault(str(_a).strip(), set()).add(_d)
print(f"ekip planında izin kaydı olan agent: {len(ekip_izin_gunleri)}")

# --- model kapsamıyla kesişim ---
_model_agentlar = {str(a).strip() for a in AGENTS}
ortak_agentlar = sorted(_model_agentlar & set(ekip_agentlar))
print(f"\nMODEL agent sayısı            : {len(_model_agentlar)}")
print(f"EKİP planı agent sayısı       : {len(ekip_agentlar)}")
print(f"ORTAK (karşılaştırılabilir)   : {len(ortak_agentlar)}")

# %% [KARŞILAŞTIRMA 2] - EKİP PLANI vs MODEL PLANI
# İki planı AYNI fonksiyondan geçirip yan yana ölçer. Aynı kod, aynı ölçüt,
# aynı veri -> "kendi kurallarınızla kendinizi ölçtünüz" eleştirisine yer yok.
#
# ÖN KOŞUL: [KARŞILAŞTIRMA 1] - EKİP PLANI (SQL)   -> ekip_calisma, ekip_saat
#           [KONTROL] - ÇÖZÜMÜ ÇIKAR               -> cozum_calisma, cozum_atama
#
# ADİLLİK KURALLARI (sunumda MUTLAKA belirtilmeli):
#   1) Karşılaştırma SADECE ortak agentlarda yapılır (ekip planında daha fazla
#      agent var; kesişim dışında kıyas elma-armut olur).
#   2) İzin / off talebi verisi İKİSİNDE DE aynı kaynaktan (tip_map) okunur.
#   3) Yeni tanımlanan kurallar (kıdem, takım bütünlüğü) ayrı grupta gösterilir -
#      ekip planı hazırlanırken bu kurallar YOKTU, "ihlal" demek haksızlık olur.
#   4) Ekip planı ay içinde revize edilmiş FİİLİ hâl olabilir; model planı ay
#      başındaki teorik plandır.

from datetime import timedelta

if "ekip_calisma" not in globals():
    raise SystemExit("ekip_calisma yok - önce [KARŞILAŞTIRMA 1] hücresini çalıştır.")
if "cozum_calisma" not in globals():
    raise SystemExit("cozum_calisma yok - önce [KONTROL] - ÇÖZÜMÜ ÇIKAR hücresini çalıştır.")

_ks_ortak = sorted(globals().get("ortak_agentlar", []))
if not _ks_ortak:
    raise SystemExit("ortak_agentlar boş - agent kodları eşleşmiyor.")

_ks_gunler = sorted(PLAN_GUNLER, key=lambda s: pd.to_datetime(s))
_ks_tarih = {ds: pd.to_datetime(ds).date() for ds in _ks_gunler}
_ks_saat = globals().get("saat", {})
_ks_tip = globals().get("tip_map", {})
_ks_gecmis = globals().get("gecmis_calisma", {})
_ks_dw = globals().get("day_week", {})
_ks_sayilmayan = set(globals().get("calisma_disi_date_set", set()))
_ks_ilk_gun = min(_ks_tarih.values())      # plan dönemi başlangıcı


def _ks_dk(t):
    h, m = str(t).strip()[:5].split(":")
    return int(h) * 60 + int(m)


# ---- model planını da (agent, ds) -> (bas, bit) yapısına çevir ----
model_calisma, model_saat = {}, {}
for _a in _ks_ortak:
    for _ds in _ks_gunler:
        _c = cozum_calisma.get((_a, _ds), 0)
        model_calisma[(_a, _ds)] = int(_c)
        if _c == 1:
            _v = cozum_atama.get((_a, _ds))
            _st = _ks_saat.get((_ds, _v))
            if _st:
                model_saat[(_a, _ds)] = (str(_st[0]).strip()[:5], str(_st[1]).strip()[:5])

# ---- agent özellikleri ----
_ks_flag = {}
for _c in ("sabah_calisir_flg", "hamile_flg", "sut_izni_flg", "mesaiye_kalamaz_flg"):
    if _c in df_tam.columns:
        _ks_flag[_c] = set(df_tam.loc[
            pd.to_numeric(df_tam[_c], errors="coerce").fillna(0).astype(int) == 1,
            "agent_user_code"].astype(str).str.strip())
    else:
        _ks_flag[_c] = set()

_ks_team = {str(a).strip(): str(t).strip()
            for a, t in globals().get("agent_team", {}).items() if t is not None}

_kd_map = globals().get("kidem_ay_map", {})
_kd_min = globals().get("KIDEM_MIN_AY")
_kd_vars = float(globals().get("KIDEM_EKSIK_VARSAYILAN_AY", 0))
_kd_bas = {str(b).strip()[:5] for b in globals().get("KIDEM_YASAK_VARDIYA_BASLANGIC", ())}
_ks_kidemsiz = set()
if _kd_min is not None:
    for _a in _ks_ortak:
        _v = _kd_map.get(_a)
        _v = _kd_vars if (_v is None or pd.isna(_v)) else float(_v)
        if _v < float(_kd_min):
            _ks_kidemsiz.add(_a)

_S_BIT = globals().get("SABAH_CALISIR_MAX_BITIS_SAATI")
_S_BAS = globals().get("SABAH_CALISIR_MAX_BASLANGIC_SAATI")
_S_GECE_Y = globals().get("SABAH_CALISIR_GECE_VARDIYA_YASAK", False)
_S_GECE_SET = globals().get("GECE_AKSAM_VARDIYA_SET", set())

_baz_map = {}
if "aylik_calisma_debug_df" in globals() and not aylik_calisma_debug_df.empty:
    _baz_map = dict(zip(
        aylik_calisma_debug_df["agent_user_code"].astype(str).str.strip(),
        aylik_calisma_debug_df["baz_gereken_calisma"].astype(int)))

_MAX_OFF = int(globals().get("MODEL_OFF_HAFTALIK_MAX", 2))


def _ks_sabah_uygun(b, e):
    if _S_BIT is None:
        return True
    bd, ed = _ks_dk(b), _ks_dk(e)
    if ed < bd:
        ed += 24 * 60
    if _S_GECE_Y and (b, e) in _S_GECE_SET:
        return False
    if _S_BAS is not None and bd > _ks_dk(_S_BAS):
        return False
    return ed <= _ks_dk(_S_BIT)


def plan_denetle(calisma, saat_atama, etiket):
    """Bir planı ölç. calisma:(a,ds)->0/1  saat_atama:(a,ds)->(bas,bit)"""
    m = {"plan": etiket}
    _izin_calisma = _offt2_calisma = _offt_karsilanmayan = 0
    _11h = _max6 = _haftalik_off = _cift_off_yok = 0
    _hamile_hs = _sabah_ihlal = _kidem_gece = 0
    _kota_eksik = _kota_fazla = _mesai_gun = 0
    _hs_sayac, _takim_hafta = {}, {}
    _detay = []

    for _a in _ks_ortak:
        _tip_a = _ks_tip.get(_a, {})
        _gun_calisilan = []
        for _ds in _ks_gunler:
            _d = _ks_tarih[_ds]
            _c = calisma.get((_a, _ds), 0)
            _t = str(_tip_a.get(_d, "")).strip().lower()
            if _c == 1:
                _gun_calisilan.append(_d)
                if _t == "izin":
                    _izin_calisma += 1
                    _detay.append({"plan": etiket, "kural": "İzin gününde çalışma",
                                   "agent_user_code": _a, "tarih": str(_d)})
                elif _t == "off_t2":
                    _offt2_calisma += 1
                    _detay.append({"plan": etiket, "kural": "Kesin off (off_t2) gününde çalışma",
                                   "agent_user_code": _a, "tarih": str(_d)})
                elif _t == "off_t":
                    _offt_karsilanmayan += 1
                _st = saat_atama.get((_a, _ds))
                if _st:
                    if _a in _ks_flag["hamile_flg"] | _ks_flag["sut_izni_flg"] and _d.weekday() in (5, 6):
                        _hamile_hs += 1
                        _detay.append({"plan": etiket, "kural": "Hamile/süt izni hafta sonu çalışma",
                                       "agent_user_code": _a, "tarih": str(_d)})
                    if _a in _ks_flag["sabah_calisir_flg"] and not _ks_sabah_uygun(*_st):
                        _sabah_ihlal += 1
                        _detay.append({"plan": etiket, "kural": "Sabah çalışan uygunsuz vardiya",
                                       "agent_user_code": _a, "tarih": str(_d),
                                       "vardiya": f"{_st[0]}-{_st[1]}"})
                    if _a in _ks_kidemsiz and _st[0] in _kd_bas:
                        _kidem_gece += 1
                        _detay.append({"plan": etiket, "kural": "Kıdemsiz agent gece vardiyası",
                                       "agent_user_code": _a, "tarih": str(_d),
                                       "vardiya": f"{_st[0]}-{_st[1]}"})
                    if _d.weekday() in (5, 6):
                        _hs_sayac[_a] = _hs_sayac.get(_a, 0) + 1
                    else:
                        _wk = _ks_dw.get(_ds)
                        _takim_hafta.setdefault((_ks_team.get(_a), _wk), set()).add(_st)

        # --- 11 saat (geçmiş gün dahil) ---
        _gc = _ks_gecmis.get(_a, {})
        for _ds in _ks_gunler:
            if calisma.get((_a, _ds), 0) != 1:
                continue
            _d = _ks_tarih[_ds]
            _st = saat_atama.get((_a, _ds))
            if not _st:
                continue
            _onceki = _d - timedelta(days=1)
            _p = None
            _p_ds = next((s for s in _ks_gunler if _ks_tarih[s] == _onceki), None)
            if _p_ds is not None and calisma.get((_a, _p_ds), 0) == 1:
                _p = saat_atama.get((_a, _p_ds))
            elif _onceki in _gc:
                _p = (str(_gc[_onceki][0]).strip()[:5], str(_gc[_onceki][1]).strip()[:5])
            if not _p:
                continue
            _pb, _pe = _ks_dk(_p[0]), _ks_dk(_p[1])
            if _pe <= _pb:
                _pe += 24 * 60
            _bit_dk = _pe - 24 * 60          # sonraki güne göre
            _bas_dk = _ks_dk(_st[0])
            if _bas_dk - _bit_dk < 11 * 60:
                _11h += 1
                _detay.append({"plan": etiket, "kural": "11 saat dinlenme ihlali",
                               "agent_user_code": _a, "tarih": str(_d),
                               "vardiya": f"önceki {_p[0]}-{_p[1]} / bugün {_st[0]}-{_st[1]}"})

        # --- ardışık 7+ gün (önceki ay dahil, AMA 7. gün plan döneminde olmalı) ---
        # Önceki ayın planı BAŞKASININ sorumluluğu. Belirleyici olan serinin
        # 7. GÜNÜNÜN nerede olduğu:
        #   25-31 Tem (7. gün 31 Tem)  -> ihlal ZATEN önceki ayda oluştu.
        #                                 Bu ayın planı ne yaparsa yapsın değişmez -> SAYILMAZ
        #   27 Tem-2 Ağu (7. gün 2 Ağu) -> 7. günü BU PLAN yarattı -> SAYILIR
        #   1-7 Ağu (7. gün 7 Ağu)      -> tamamen bu planın -> SAYILIR
        _tum = sorted(set(_gun_calisilan) | {d for d in _gc if d < _ks_ilk_gun})
        _seriler = []
        _sb = _onc = None
        for _d in _tum:
            if _onc is None or (_d - _onc).days != 1:
                if _sb is not None:
                    _seriler.append((_sb, _onc))
                _sb = _d
            _onc = _d
        if _sb is not None:
            _seriler.append((_sb, _onc))

        for _sbas, _sson in _seriler:
            _uzunluk = (_sson - _sbas).days + 1
            if _uzunluk < 7:
                continue
            _yedinci = _sbas + timedelta(days=6)      # serinin 7. günü
            if _yedinci < _ks_ilk_gun:
                continue                              # ihlal önceki ayda oluşmuş
            _max6 += 1
            _detay.append({"plan": etiket, "kural": "7+ gün aralıksız çalışma",
                           "agent_user_code": _a, "tarih": str(_yedinci),
                           "vardiya": f"{_sbas} .. {_sson} ({_uzunluk} gün), "
                                      f"7. gün {_yedinci}"})

        # --- haftalık off tavanı ---
        _hafta = {}
        for _ds in _ks_gunler:
            _d = _ks_tarih[_ds]
            if str(_tip_a.get(_d, "")).strip().lower() == "izin":
                continue
            if _d in _ks_sayilmayan and _d.weekday() in (0, 1, 2, 3, 4):
                continue
            _wk = _ks_dw.get(_ds)
            _hafta.setdefault(_wk, [0, 0])
            _hafta[_wk][0] += 1
            if calisma.get((_a, _ds), 0) != 1:
                _hafta[_wk][1] += 1
        _zorunlu = sum(1 for _ds in _ks_gunler
                       if str(_tip_a.get(_ks_tarih[_ds], "")).strip().lower() in ("off_t", "off_t2"))
        for _wk, (_say, _off) in _hafta.items():
            if _say >= 5 and _off > max(_MAX_OFF, _zorunlu):
                _haftalik_off += 1
                _detay.append({"plan": etiket, "kural": "Haftada 2'den fazla off",
                               "agent_user_code": _a, "tarih": str(_wk)})

        # --- ayda en az 1 tam Cmt-Pzr ---
        _cift = False
        for _ds in _ks_gunler:
            _d = _ks_tarih[_ds]
            if _d.weekday() != 5:
                continue
            _pz = _d + timedelta(days=1)
            _pz_ds = next((s for s in _ks_gunler if _ks_tarih[s] == _pz), None)
            if _pz_ds is None:
                continue
            if calisma.get((_a, _ds), 0) == 0 and calisma.get((_a, _pz_ds), 0) == 0:
                _cift = True
                break
        if not _cift:
            _cift_off_yok += 1
            _detay.append({"plan": etiket, "kural": "Ayda tam Cmt-Pzr çifti yok",
                           "agent_user_code": _a, "tarih": ""})

        # --- aylık kota ---
        _sayilan = sum(1 for _ds in _ks_gunler
                       if calisma.get((_a, _ds), 0) == 1
                       and _ks_tarih[_ds] not in _ks_sayilmayan)
        _baz = int(_baz_map.get(_a, 0))
        if _baz:
            if _sayilan < _baz:
                _kota_eksik += 1
            elif _sayilan > _baz:
                _kota_fazla += 1
                _mesai_gun += _sayilan - _baz

    # --- takım hafta içi bölünmesi ---
    _bolunen = sum(1 for k, v in _takim_hafta.items() if k[0] and len(v) > 1)

    # --- coverage (ortak agentlarla) ---
    _atanan = {}
    for (_a, _ds), _st in saat_atama.items():
        if calisma.get((_a, _ds), 0) == 1:
            _atanan[(_ds, _st)] = _atanan.get((_ds, _st), 0) + 1
    _eksik_kova = _eksik_kisi = _fazla_kisi = 0
    for _ds in _ks_gunler:
        for _v in gun_vardiyalari.get(_ds, []):
            if (_ds, _v) not in talep or (_ds, _v) not in _ks_saat:
                continue
            _st = (str(_ks_saat[(_ds, _v)][0]).strip()[:5], str(_ks_saat[(_ds, _v)][1]).strip()[:5])
            _req, _asg = int(talep[(_ds, _v)]), int(_atanan.get((_ds, _st), 0))
            if _asg < _req:
                _eksik_kova += 1
                _eksik_kisi += _req - _asg
            elif _asg > _req:
                _fazla_kisi += _asg - _req

    _hs = [_hs_sayac.get(a, 0) for a in _ks_ortak]
    _ort = sum(_hs) / len(_hs) if _hs else 0
    _std = (sum((x - _ort) ** 2 for x in _hs) / len(_hs)) ** 0.5 if _hs else 0

    m.update({
        "İzin gününde çalışma": _izin_calisma,
        "Kesin off gününde çalışma": _offt2_calisma,
        "11 saat dinlenme ihlali": _11h,
        "7+ gün aralıksız çalışma": _max6,
        "Haftada 2'den fazla off": _haftalik_off,
        "Tam Cmt-Pzr çifti olmayan agent": _cift_off_yok,
        "Hamile/süt hafta sonu çalışma": _hamile_hs,
        "Sabah çalışan uygunsuz vardiya": _sabah_ihlal,
        "Kıdemsiz agent gece vardiyası [YENİ KURAL]": _kidem_gece,
        "Takım hafta içi bölünmesi (takım-hafta) [YENİ KURAL]": _bolunen,
        "Kotayı tutturamayan agent": _kota_eksik,

    })
    return m, _detay
