# Pickle'daki iç içe sözlüğü tidy DataFrame'e çevir: her satır = gün × inhouse
# vardiya. Outsource / part_time hariç bırakılır.
#   talep       = Model-1 son planı (surplus DAHİL)        → mip_info["assignments"]
#   mip_ihtiyac = surplus dağıtımı ÖNCESİ MIP ihtiyacı      → mip_info["mip_info_stage1"]["assignments"]
#   surplus     = talep − mip_ihtiyac (Model-1'in bu vardiyaya eklediği kişi)
# stage1 yalnız HAFTA İÇİ var (Model-1 hafta sonuna surplus dağıtmıyor) →
# hafta sonunda / eski pickle'da mip_ihtiyac = talep, surplus = 0.
satirlar = []
for ds in sorted(results.keys()):
    r = results.get(ds)
    if not r or QUEUE not in r or not r[QUEUE]:
        continue
    mi = r[QUEUE]["mip_info"]
    sc = mi.get("shift_coverage", {})
    s1 = (mi.get("mip_info_stage1") or r[QUEUE].get("mip_info_stage1") or {}).get("assignments")
    for shift, kisi in mi.get("assignments", {}).items():
        info = sc.get(shift, {})
        if kisi <= 0 or info.get("company") != "inhouse":
            continue
        d = dt.date.fromisoformat(ds)
        mip = int(kisi) if s1 is None else max(0, min(int(kisi), int(s1.get(shift, 0) or 0)))
        satirlar.append({
            "tarih":       ds,
            "gun":         DAY_TR[d.weekday()],
            "hafta_sonu":  int(d.weekday() > 4),
            "vardiya":     shift,
            "baslangic":   info.get("start"),
            "bitis":       info.get("end"),
            "talep":       int(kisi),
            "mip_ihtiyac": mip,
            "surplus":     int(kisi) - mip,
        })

df_talep = pd.DataFrame(satirlar)
print(f"talep tablosu: {len(df_talep)} satır (gün × inhouse vardiya)")
print(df_talep.head(8).to_string(index=False))

# surplus kontrolü
_hi = df_talep[df_talep["hafta_sonu"] == 0]
if _hi["surplus"].sum() > 0:
    print(f"surplus: hafta içi {int(_hi['surplus'].sum())} kişi-gün / talep {int(_hi['talep'].sum())}"
          f"  |  surplus'lu vardiya {int((_hi['surplus'] > 0).sum())}/{len(_hi)}")
else:
    print("⚠ surplus 0 — pickle'da mip_info_stage1 yok ya da bu kuyrukta surplus dağıtılmamış")

# Ay bilgisi (H3'te izin/gün eşlemesi için)
GUNLER  = sorted(results.keys())

AY_AY   = dt.date.fromisoformat(GUNLER[0]).month

pickle'da surplus öncesi bilgisi (mip_info_stage1) yok,

Bu hücre tek başına cezayı değiştirmez. Yalnız veriyi hazırlıyor. Eksiğin surplus'tan düşmesi için modelde (bizde HÜCRE 10, coverage) eksik cezası ikiye ayrılmalı:

talep − atanan → düşük ceza (bizde 3)
mip_ihtiyac − atanan → yüksek ceza (bizde 25)
