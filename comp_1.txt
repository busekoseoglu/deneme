CREATE TABLE BSEN_SEYAHAT_2026_KITLE_EYLUL_EKIM_OOT_V2 AS
WITH BIREYSEL_CUST AS (
    SELECT A.PARTY_ID,
           MIN(A.OPEN_DATE) AS OPEN_DATE
    FROM   DM.CNF_CUSTOMER A
    INNER JOIN (
                SELECT CLIENT_NO FROM ODS_KBLIVE.CUST_PDP_PERM_BMS_KKS
                UNION
                SELECT CLIENT_NO FROM ODS_KBLIVE.CUST_PDP_PERMISSIONS
                WHERE  PERMISSION_FLAG = 'Y'
               ) KVKK
            ON A.PARTY_ID = KVKK.CLIENT_NO
    INNER JOIN (
                SELECT TO_NUMBER(C1.SBU_CODE) SBU_CODE
                FROM   DM_MIS.PRF_SBU C1
                WHERE  C1.SBU_CODE NOT LIKE '%X%'
               ) SBU
            ON SBU.SBU_CODE = A.PORTFOLIO_SBU_CODE
    WHERE  A.MVT_STATUS_ID = 1
      AND  A.CURRENT_STATUS_ID IN (1, 2, 5)
      AND  A.CUSTOMER_TYPE_ID  IN (1, 4, 47)
      AND  A.PORTFOLIO_SBU_CODE IN (410, 414, 430, 432, 440, 444, 415, 425)
    GROUP BY A.PARTY_ID
),
CAL AS (
    SELECT DISTINCT LAST_DAY_OF_MONTH AS DATA_DATE
    FROM   EDW.CMN_CALENDAR
    WHERE  LAST_DAY_OF_MONTH BETWEEN DATE '2026-08-31' AND DATE '2026-09-30'
)
SELECT B.PARTY_ID,
       C.DATA_DATE,
       ADD_MONTHS(C.DATA_DATE, 1) AS TARGET_DATE
FROM   BIREYSEL_CUST B
JOIN   CAL C
  ON   B.OPEN_DATE <= C.DATA_DATE;      -- her snapshot'ta sadece o tarihte hesabı açık olanlar
