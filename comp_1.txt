# %% [HÜCRE] - BUFFERLI COVERAGE + TALEBE YAKLAŞMA
coverage_lower = {}
coverage_upper = {}
coverage_constraints = 0
bos_kova = {}            # 2026-10-20: (ds, v) -> 1 ise talebi olan kova boş (ceza OBJECTIVE'de)
if "BOS_KOVA_W" not in globals():
    raise SystemExit("CONFIG'de tanımlı olmalı: BOS_KOVA_W -> [CONFIG] hücresini çalıştır.")
target_gap_constraints = 0

for ds in PLAN_GUNLER:
    for v in gun_vardiyalari.get(ds, []):
        if (ds, v) in globals().get("prov_vardiyalar", set()):
            continue   # 2026-10-19: provizyon coverage dışı
        required = int(talep[(ds, v)])
        lower_req = math.floor(required * (1 - BUFFER_RATE))
        upper_req = math.ceil(required * (1 + BUFFER_RATE))
        coverage_lower[(ds, v)] = lower_req
        coverage_upper[(ds, v)] = upper_req

        assigned = sum(x[(a, ds, v)] for a in AGENTS if (a, ds, v) in x)

        model.Add(assigned + under_buffer[(ds, v)] >= lower_req)
        coverage_constraints += 1
        model.Add(assigned - over_buffer[(ds, v)] <= upper_req)
        coverage_constraints += 1
        model.Add(assigned + missing_to_required[(ds, v)] >= required)
        target_gap_constraints += 1
        model.Add(assigned - excess_to_required[(ds, v)] <= required)
        target_gap_constraints += 1
        if required >= 1:
            bos_kova[(ds, v)] = model.NewBoolVar(f"bos_kova_{ds}_{v}")
            model.Add(assigned + bos_kova[(ds, v)] >= 1)

print("Coverage buffer kısıtı:", coverage_constraints)
print("Talebe göre gap kısıtı:", target_gap_constraints)
print("Boş kova kısıtı (talep >= 1):", len(bos_kova))





# %% [KONTROL 3] - COVERAGE (buffer + talep) [RAPOR/SOFT]
rows = []
buffer_alti = 0
for ds in PLAN_GUNLER:
    for v in gun_vardiyalari.get(ds, []):
        assigned = sum(V(x[(str(a).strip(), ds, v)]) for a in AGENTS if (str(a).strip(), ds, v) in x)
        req = int(talep[(ds, v)])
        lo = coverage_lower.get((ds, v), req) if "coverage_lower" in globals() else req
        hi = coverage_upper.get((ds, v), req) if "coverage_upper" in globals() else req
        durum = "ok"
        if assigned < lo: durum = "BUFFER_ALTI"; buffer_alti += 1
        elif assigned < req: durum = "talep_alti"
        elif assigned > hi: durum = "buffer_ustu"
        rows.append({"tarih": str(ds), "vardiya": v, "talep": req, "lower": lo, "upper": hi, "atanan": int(assigned), "durum": durum})
coverage_kontrol_df = pd.DataFrame(rows)
kaydet("Coverage buffer altı (soft)", buffer_alti, tur="SOFT")
kaydet("Boş kalan kova (talep >= 1, atanan 0)",   # 2026-10-20
       int(((coverage_kontrol_df["talep"] >= 1) & (coverage_kontrol_df["atanan"] == 0)).sum()), tur="SOFT")
print("Talep altı gün-vardiya:", int((coverage_kontrol_df["durum"] == "talep_alti").sum()))
print("Buffer üstü gün-vardiya:", int((coverage_kontrol_df["durum"] == "buffer_ustu").sum()))
display(coverage_kontrol_df[coverage_kontrol_df["durum"] != "ok"].head(40))
