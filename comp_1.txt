CREATE TABLE BNS_VP_GECMIS_VARDIYA_TOTAL_GUN_SAYISI AS
SELECT TEAMLEADER_NAME, WORKING_MAIN_GROUP, SHIFTS_ADJ,
       SUM(gun_sayisi) AS total_gun_sayisi
FROM (
  SELECT a.*,
         ROW_NUMBER() OVER (
           PARTITION BY hafta_bas, TEAMLEADER_NAME, WORKING_MAIN_GROUP
           ORDER BY agent_gun_sayisi DESC, gun_sayisi DESC, SHIFTS_ADJ
         ) AS rn
  FROM (
    SELECT TRUNC(WORKING_DATE, 'IW') AS hafta_bas,
           TEAMLEADER_NAME, WORKING_MAIN_GROUP, SHIFTS_ADJ,
           COUNT(DISTINCT WORKING_DATE) AS gun_sayisi,
           COUNT(*)                    AS agent_gun_sayisi
    FROM BNS_VP_YOKLAMA_LISTESI_FROM_BO_UNTIL_072026_V3
    WHERE WORKING_DAY_NAME NOT IN ('Saturday','Sunday')
    GROUP BY TRUNC(WORKING_DATE, 'IW'), TEAMLEADER_NAME,
             WORKING_MAIN_GROUP, SHIFTS_ADJ
  ) a
) WHERE rn = 1
GROUP BY TEAMLEADER_NAME, WORKING_MAIN_GROUP, SHIFTS_ADJ




SELECT TRUNC(WORKING_DATE,'IW') AS hafta, SHIFTS_ADJ,
       COUNT(DISTINCT WORKING_DATE) AS gun,
       COUNT(*) AS agent_gun
FROM BNS_VP_YOKLAMA_LISTESI_FROM_BO_UNTIL_072026_V3
WHERE WORKING_DAY_NAME NOT IN ('Saturday','Sunday')
  AND TEAMLEADER_NAME = '<BİR TAKIM LİDERİ>'
GROUP BY TRUNC(WORKING_DATE,'IW'), SHIFTS_ADJ
ORDER BY hafta, agent_gun DESC
