# %% [HÜCRE] - FAZLA ATAMA ÜST LİMİTİ
# Amaç: her gün-vardiya için talebin üstüne en fazla kaç kişi çıkabilir?
# Partial week günlerinde limit sıkılaştırılır (PARTIAL_WEEK_MAX_FAZLA_ATAMA),
# böylece model ay sonundaki yarım haftaya fazla kişi yığamaz.
#
# 2026-08-06 DÜZELTME (arife açılınca ortaya çıktı):
#   - eski izin_map yapısı kaldırıldı, tip_map kullanılıyor
#   - değişken adı fazla_atama_ust_limiti (VARDİYA TİPİ hücresinde üretiliyor)
#
# forced_count = arife özel vardiyasına ZORLANACAK kısıtlı agent sayısı.
# Bu küme [HÜCRE] - ARİFE HARD / RESMİ TATİL SOFT hücresindekiyle BİREBİR AYNI
# olmak zorunda, yoksa kapasite gevşetmesi yanlış hesaplanır:
#   izin    -> gün kapalı, x yok
#   off_t2  -> gün kapalı, x yok
#   off_t   -> x VAR ama agent ZORLANMAZ. off_t soft bir talep; hard bir kısıtla
#              çalışmaya zorlarsak talebin karşılanma ihtimali sıfırlanır ve
#              OFF_TALEP_KARSILANMAMA_W cezası anlamsızlaşır. Model isterse
#              kendi seçer, cezayı öder.
ARIFE_ZORLANMAZ_TIPLERI = {"izin", "off_t", "off_t2"}


def _arife_zorlanmaz_mi(a, ds):
    _d = pd.to_datetime(ds).date()
    _tip = str(tip_map.get(str(a).strip(), {}).get(_d, "")).strip().lower()
    return _tip in ARIFE_ZORLANMAZ_TIPLERI


# 2026-09-04: partial hafta sıkı tavanı anahtara bağlandı. Gerekçesi
# (ay sonu yarım haftaya yığılma) aylık çalışma dengesi kuralıyla ortadan
# kalktı; ayrıca sarkan haftada kişi bazlı devir kuralıyla çarpışıyordu.
if "ENABLE_PARTIAL_WEEK_FAZLA_TAVANI" not in globals():
    raise SystemExit(
        "CONFIG'de tanımlı olmalı: ENABLE_PARTIAL_WEEK_FAZLA_TAVANI "
        "-> [CONFIG] hücresini çalıştır."
    )

fazla_atama_cap_constraints = 0
arife_cap_relax_rows = []
_fa_siki_tavan_kova = 0        # sıkı partial tavanı uygulanan (gün, vardiya)

for ds in PLAN_GUNLER:
    for v in gun_vardiyalari.get(ds, []):
        required = int(talep[(ds, v)])
        assigned = sum(x[(a, ds, v)] for a in AGENTS if (a, ds, v) in x)
        wk = day_week[ds]

        if wk in partial_weeks and ENABLE_PARTIAL_WEEK_FAZLA_TAVANI:
            max_fazla = PARTIAL_WEEK_MAX_FAZLA_ATAMA
            _fa_siki_tavan_kova += 1
        else:
            max_fazla = fazla_atama_ust_limiti.get((ds, v), GENEL_MAX_FAZLA_ATAMA)

        if (ds, v) in arife_ozel_vardiyalar:
            forced_count = 0
            for a in tatil_kisitli_agents:
                a = str(a).strip()
                if _arife_zorlanmaz_mi(a, ds):
                    continue
                if (a, ds, v) in x:
                    forced_count += 1
            min_needed_extra = max(0, forced_count - required)
            if min_needed_extra > max_fazla:
                arife_cap_relax_rows.append({
                    "date": ds,
                    "shift": v,
                    "required": required,
                    "old_max_fazla": max_fazla,
                    "forced_count": forced_count,
                    "new_max_fazla": min_needed_extra,
                })
                max_fazla = min_needed_extra

        model.Add(assigned <= required + max_fazla)
        fazla_atama_cap_constraints += 1

print("Fazla atama üst limit kısıtı:", fazla_atama_cap_constraints)
print("Arife özel vardiyasına zorlanmayan tipler:", sorted(ARIFE_ZORLANMAZ_TIPLERI))
print("Genel max fazla atama:", GENEL_MAX_FAZLA_ATAMA)
print("Gece/akşam max fazla atama:", GECE_MAX_FAZLA_ATAMA)
print("ENABLE_PARTIAL_WEEK_FAZLA_TAVANI:", ENABLE_PARTIAL_WEEK_FAZLA_TAVANI)
