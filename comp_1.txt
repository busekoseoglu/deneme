/* =====================================================================
   YURT DIŞI SEYAHAT GEÇMİŞİ DEĞİŞKENLERİ
   ---------------------------------------------------------------------
   Seyahat tanımı target ile birebir aynı:
     bir müşteri bir ayda yurt dışına çıkmış sayılır, eğer o ay
       - yurt dışı fiziksel POS işlemi (terminal 11, ülke TR/TU/TUR dışı, entry 1/4/5/7) VEYA
       - NOF_INTERNATIONAL_ATM_TRX > 0
     varsa.

   Snapshot'lar (data_date) : 2025-09-30 .. 2026-09-30  (ay sonları)
   Geriye bakış             : 24 ay -> seyahat verisi 2023-10 .. 2026-09
   Zaman kuralı             : sadece data_date ve öncesindeki aylar kullanılır
                              (m = 0 -> data_date'in kendi ayı, m = 23 -> 23 ay önce).
                              Target ayı (data_date + 1) HİÇBİR değişkene girmez.

   Çıktı: BSEN_SYH_GECMIS_DEGISKENLER  (party_id, data_date)
     - Sadece son 24 ayda en az bir kez yurt dışına çıkmış müşteriler için satır üretilir.
     - Ana tabloya LEFT JOIN ON party_id, data_date.
     - Join sonrası boş gelenler = son 24 ayda hiç yurt dışına çıkmamış:
         *_flg, *_ay_say, *_tutar, *_islem  -> 0 ile doldur
         syh_son_seyahat_ay_once            -> NULL bırak (LightGBM işler) veya 99 ver
   ===================================================================== */


/* ---------------------------------------------------------------------
   1) AYLIK SEYAHAT TABLOSU: hangi müşteri hangi ay yurt dışındaydı
   --------------------------------------------------------------------- */
CREATE TABLE BSEN_SYH_AYLIK_SEYAHAT NOLOGGING AS
WITH POS AS (
    SELECT a.party_id,
           LAST_DAY(TRUNC(a.card_transaction_date)) AS seyahat_ay,
           COUNT(*) AS pos_islem,
           SUM(CASE WHEN a.transaction_amount_tl > 0
                    THEN a.transaction_amount_tl ELSE 0 END) AS pos_tutar
    FROM   edw.trx_card_transaction a
    WHERE  a.card_transaction_date >= DATE '2023-10-01'
      AND  a.card_transaction_date <  DATE '2026-10-01'            -- saat bileşenine karşı güvenli
      AND  a.transaction_terminal_tp_id = 11                        -- fiziksel POS
      AND  a.trx_card_acptor_cntry_nk NOT IN ('TR', 'TU', 'TUR')
      AND  a.transaction_entry_tp_id IN (1, 4, 5, 7)
    GROUP BY a.party_id, LAST_DAY(TRUNC(a.card_transaction_date))
),
ATM AS (
    SELECT c.party_id,
           c.data_date AS seyahat_ay,
           MAX(c.nof_international_atm_trx) AS atm_islem
    FROM   dm.cnf_card_financial_trx_aggr c
    WHERE  c.data_date BETWEEN DATE '2023-10-31' AND DATE '2026-09-30'
      AND  c.nof_international_atm_trx > 0
    GROUP BY c.party_id, c.data_date
)
SELECT party_id,
       seyahat_ay,
       SUM(pos_islem) AS pos_islem,
       SUM(pos_tutar) AS pos_tutar,
       SUM(atm_islem) AS atm_islem
FROM (
    SELECT party_id, seyahat_ay, pos_islem, pos_tutar, 0 AS atm_islem FROM pos
    UNION ALL
    SELECT party_id, seyahat_ay, 0, 0, atm_islem FROM atm
)
GROUP BY party_id, seyahat_ay;

CREATE INDEX IX_SYH_AYLIK ON BSEN_SYH_AYLIK_SEYAHAT (party_id, seyahat_ay);


/* ---------------------------------------------------------------------
   2) SNAPSHOT BAZINDA GEÇMİŞ DEĞİŞKENLERİ
      m = data_date ile seyahat ayı arasındaki ay farkı (0..23)
      Hedef ay = data_date + 1  ->  geçen yıl hedef ay = data_date - 11  (m = 11)
   --------------------------------------------------------------------- */
CREATE TABLE BSEN_SYH_GECMIS_DEGISKENLER NOLOGGING AS
WITH CAL AS (
    SELECT DISTINCT last_day_of_month AS data_date
    FROM   edw.cmn_calendar
    WHERE  last_day_of_month BETWEEN DATE '2025-09-30' AND DATE '2026-09-30'
),
X AS (
    SELECT s.party_id,
           c.data_date,
           MONTHS_BETWEEN(c.data_date, s.seyahat_ay) AS m,   -- iki ay sonu arası -> tam sayı
           s.pos_islem,
           s.pos_tutar,
           s.atm_islem
    FROM   bsen_syh_aylik_seyahat s
    JOIN   cal c
      ON   s.seyahat_ay <= c.data_date                      -- GELECEK YOK: en fazla data_date ayı
     AND   s.seyahat_ay >  ADD_MONTHS(c.data_date, -24)     -- son 24 ay
)
SELECT party_id,
       data_date,

       -- Snapshot ayında (data_date ayı) yurt dışındaydı mı
       MAX(CASE WHEN m = 0 THEN 1 ELSE 0 END)               AS syh_snap_ay_flg,

       -- Geçen yıl hedef ay (target ayının 12 ay öncesi)
       MAX(CASE WHEN m = 11 THEN 1 ELSE 0 END)              AS syh_gecen_yil_hedef_ay_flg,
       -- Geçen yıl hedef ay ±1 (3 aylık pencere; tatil tarihleri kayabilir)
       MAX(CASE WHEN m BETWEEN 10 AND 12 THEN 1 ELSE 0 END) AS syh_gecen_yil_hedef_donem_flg,
       SUM(CASE WHEN m BETWEEN 10 AND 12 THEN 1 ELSE 0 END) AS syh_gecen_yil_hedef_donem_ay_say,
       -- İki yıl önce hedef ay (target ayının 24 ay öncesi)
       MAX(CASE WHEN m = 23 THEN 1 ELSE 0 END)              AS syh_iki_yil_once_hedef_ay_flg,

       -- Sıklık: son N ayda kaç farklı ay yurt dışına çıktı
       SUM(CASE WHEN m < 3  THEN 1 ELSE 0 END)              AS syh_ay_say_3m,
       SUM(CASE WHEN m < 6  THEN 1 ELSE 0 END)              AS syh_ay_say_6m,
       SUM(CASE WHEN m < 12 THEN 1 ELSE 0 END)              AS syh_ay_say_12m,
       COUNT(*)                                             AS syh_ay_say_24m,

       -- Recency: son seyahatten bu yana geçen ay (0 = snapshot ayında)
       MIN(m)                                               AS syh_son_seyahat_ay_once,

       -- Yoğunluk: son 12 ay yurt dışı POS / ATM
       SUM(CASE WHEN m < 12 THEN pos_islem ELSE 0 END)      AS syh_yd_pos_islem_12m,
       SUM(CASE WHEN m < 12 THEN pos_tutar ELSE 0 END)      AS syh_yd_pos_tutar_12m,
       SUM(CASE WHEN m < 12 THEN atm_islem ELSE 0 END)      AS syh_yd_atm_islem_12m
FROM   x
GROUP BY party_id, data_date;

CREATE INDEX IX_SYH_GECMIS ON BSEN_SYH_GECMIS_DEGISKENLER (party_id, data_date);


/* ---------------------------------------------------------------------
   3) KONTROLLER
   --------------------------------------------------------------------- */

-- 3a) Sızıntı kontrolü: bir snapshot'ın target'ı, bir sonraki snapshot'ın
--     syh_snap_ay_flg değeriyle aynı ayı anlatır -> eşleşme ~%100 olmalı.
--     (Örn: data_date 2026-07-31'in target'ı = Ağustos seyahati
--           = data_date 2026-08-31'deki syh_snap_ay_flg)
SELECT t.data_date,
       COUNT(*)                                                    AS n_pozitif,
       SUM(CASE WHEN g.syh_snap_ay_flg = 1 THEN 1 ELSE 0 END)      AS eslesen,
       ROUND(AVG(CASE WHEN g.syh_snap_ay_flg = 1 THEN 1 ELSE 0 END), 4) AS eslesme_orani
FROM   <ANA_TABLO> t                         -- party_id, data_date, yurtdisi_travel_flg
JOIN   bsen_syh_gecmis_degiskenler g
  ON   g.party_id  = t.party_id
 AND   g.data_date = ADD_MONTHS(t.data_date, 1)
WHERE  t.yurtdisi_travel_flg = 1
GROUP BY t.data_date
ORDER BY t.data_date;

-- 3b) Ay bazında satır sayısı ve doluluk
SELECT data_date,
       COUNT(*)                          AS n,
       ROUND(AVG(syh_snap_ay_flg), 4)    AS snap_ay_oran,
       ROUND(AVG(syh_gecen_yil_hedef_ay_flg), 4) AS gecen_yil_hedef_oran
FROM   bsen_syh_gecmis_degiskenler
GROUP BY data_date
ORDER BY data_date;

-- 3c) Tekrar eden anahtar var mı (0 olmalı)
SELECT COUNT(*) FROM (
    SELECT party_id, data_date
    FROM   bsen_syh_gecmis_degiskenler
    GROUP BY party_id, data_date
    HAVING COUNT(*) > 1
);
