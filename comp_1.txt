def _sdt(dcode, bas, bit):
    base = pd.to_datetime(dcode).strftime("%Y-%m-%d")

    e = pd.to_datetime(base + " " + str(bit))

    return s, e

def _gun_11h_bloke(ds, jends):

        return False
    for v in sh:
        bas, bit = saat[(ds, v)]
