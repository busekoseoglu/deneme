from collections import defaultdict
hafta = defaultdict(list)
for ds in PLAN_GUNLER:
    hafta[str(day_week[ds])].append(ds)
ust = globals().get("fazla_atama_ust_limiti", {})
for wk in sorted(hafta):
    kap = sum(int(talep[(ds, v)]) + int(ust.get((ds, v), GENEL_MAX_FAZLA_ATAMA))
              for ds in hafta[wk] for v in gun_vardiyalari.get(ds, []))
    gerek = int(off_tavan_debug_df.loc[
        off_tavan_debug_df["hafta"].astype(str) == wk, "min_calisma"].sum())
    print(f"{wk}: kapasite {kap:>6} | zorunlu minimum {gerek:>6} | fark {kap - gerek:>6}")
