# ============ 2026-08-12 - BİREYSEL YASAK OLAN AGENT İÇİN MUAFİYET ============
# Bir agent takımının o haftaki base vardiyasında FİZİKSEL OLARAK çalışamıyorsa
# (sabah çalışan + geç base, ya da kıdemsiz + gece base) o hafta takım kilidinden
# muaf olur ve uygun başka bir vardiyada çalışır.
#
# Muafiyet KOŞULLU: base agentin yasak listesindeyse 1, değilse 0.
#   base 08:00-17:00 -> sabah agentı takımla AYNI vardiyada (muaf = 0)
#   base 13:00-22:00 -> sabah agentı muaf, takım geç base alabilir (muaf = 1)
#
# Böylece tek bir sabah agentı yüzünden 10 kişilik takım geç vardiyalardan
# mahrum kalmıyor. Arife / resmi tatilde zaten takım kilidi hiç kurulmuyor.
#
# Hamile / süt izinliler için ayrıca bir şey yapmaya gerek yok: onlarda
# sabah_calisir_flg zaten 1, yani sabah kuralı üzerinden kapsanıyorlar.
if "ENABLE_TAKIM_BASE_MUAFIYET" not in globals():

        "CONFIG'de tanımlı olmalı: ENABLE_TAKIM_BASE_MUAFIYET -> [CONFIG] hücresini çalıştır."
    )


def _tb_dk(t_):
    h, m = str(t_).strip()[:5].split(":")
    return int(h) * 60 + int(m)


# haftalık vardiya saatleri: (hafta, vardiya) -> (bas, bit)
_tb_wk_saat = {}
for _ds in PLAN_GUNLER:
    _wk_ = day_week[_ds]
    for _v in gun_vardiyalari.get(_ds, []):
        if (_wk_, _v) not in _tb_wk_saat and (_ds, _v) in saat:
            _b, _e = saat[(_ds, _v)]
            _tb_wk_saat[(_wk_, _v)] = (str(_b).strip()[:5], str(_e).strip()[:5])

# --- bireysel yasağı olan agentlar ---
_tb_sabah = set()
if "sabah_calisir_flg" in df_tam.columns:
    _tb_sabah = set(
        df_tam.loc[
            pd.to_numeric(df_tam["sabah_calisir_flg"], errors="coerce").fillna(0).astype(int) == 1,
            "agent_user_code",
        ].astype(str).str.strip()
    )

_tb_kidem_map = globals().get("kidem_ay_map", {})
_tb_kidem_min = globals().get("KIDEM_MIN_AY")
_tb_kidem_bas = {str(b).strip()[:5] for b in globals().get("KIDEM_YASAK_VARDIYA_BASLANGIC", ())}
_tb_kidemsiz = set()
if _tb_kidem_min is not None and _tb_kidem_map:
    _tb_kidemsiz = {
        str(a).strip() for a in AGENTS
        if str(a).strip() in _tb_kidem_map
        and not pd.isna(_tb_kidem_map[str(a).strip()])
        and float(_tb_kidem_map[str(a).strip()]) < float(_tb_kidem_min)
    }

_tb_sabah_bitis = globals().get("SABAH_CALISIR_MAX_BITIS_SAATI")
_tb_sabah_baslangic = globals().get("SABAH_CALISIR_MAX_BASLANGIC_SAATI")
_tb_gece_yasak = globals().get("SABAH_CALISIR_GECE_VARDIYA_YASAK", False)
_tb_gece_set = globals().get("GECE_AKSAM_VARDIYA_SET", set())


def _tb_yasak_mi(a, wk, v):
    """Agent a, wk haftasındaki v vardiyasında ÇALIŞAMAZ mı?"""
    _st = _tb_wk_saat.get((wk, v))
    if _st is None:
        return False
    b, e = _st
    if a in _tb_kidemsiz and b in _tb_kidem_bas:
        return True
    if a in _tb_sabah and _tb_sabah_bitis is not None:
        bd, ed = _tb_dk(b), _tb_dk(e)
        if ed < bd:
            ed += 24 * 60
        if _tb_gece_yasak and (b, e) in _tb_gece_set:
            return True
        if _tb_sabah_baslangic is not None and bd > _tb_dk(_tb_sabah_baslangic):
            return True
        if ed > _tb_dk(_tb_sabah_bitis):
            return True
    return False


_tb_kisitli = (_tb_sabah | _tb_kidemsiz) if ENABLE_TAKIM_BASE_MUAFIYET else set()

# ====== 2026-08-28 - SARKAN HAFTADA KİŞİ BAZLI VARDİYA DEVRİ (SOFT) ======
# Ay devri haftasında takım zaten BÖLÜNMÜŞ devrediyor: bir takımdan 2-3 kişi
# gecede, 10 kişi 15:00'ta. Takım base'i tek olduğu için gece devreden kişiler
# kilit yüzünden gece vardiyasına giremiyor; 11 saat kuralı da gündüzcüleri
# bloklayınca gece bucket'i SIFIRLANIYORDU (1 Eylül V01: talep 6, atanan 0).
#
# Bu blok, bir önceki gün takımından FARKLI vardiya devreden kişiyi o hafta
# kendi devraldığı vardiyaya bağlar. SOFT: başka kural izin vermiyorsa model
# başka vardiyaya geçer, ceza alır, DURMAZ.
#
# Bakılan gün: SADECE plan başlangıcından bir önceki gün.
# Üç katmanlı sıra:
#   1) önceki gün takımından FARKLI vardiya  -> kendi devraldığı vardiya (bu blok)
#   2) devir yok (off/izin/aynı vardiya)     -> takım kilidi (mevcut)
#   3) takım base'i agente yasak             -> muafiyet (mevcut)
ENABLE_SARKAN_KISI_DEVRI = globals().get("ENABLE_SARKAN_KISI_DEVRI", True)

SARKAN_WK = None
SARKAN_DEVIR_AGENT = {}          # agent -> devraldığı vardiya kodu
sarkan_kisi_sapma = {}           # agent -> BoolVar (devir bozulursa 1)
sarkan_devir_link_constraints = 0
_sd_rows = []

if ENABLE_SARKAN_KISI_DEVRI:
    from datetime import timedelta as _sd_td

    _sd_ilk = min(pd.to_datetime(d).date() for d in PLAN_GUNLER)
    _sd_onceki = (_sd_ilk - _sd_td(days=1)) if _sd_ilk.weekday() > 0 else None
    SARKAN_WK = day_week[min(PLAN_GUNLER, key=lambda s: pd.to_datetime(s))]
    _sd_gecmis = globals().get("gecmis_calisma", {})
    _sd_arife = globals().get("arife_ozel_vardiya_kodlari", set())
    _sd_tatil = globals().get("ozel_tatil_plan_gunleri", set())

    if _sd_onceki is None:
        print("Kişi bazlı devir: plan Pazartesi başlıyor, önceki gün yok - kurulmadı.")
    elif not _sd_gecmis:
        print("!!! Kişi bazlı devir: gecmis_calisma boş - kurulmadı.")
    else:
        # saat aralığı -> bu ayın vardiya kodu (sarkan hafta)
        _sd_saat_v = {}
        for (_wk_, _v_), _st_ in _tb_wk_saat.items():
            if _wk_ == SARKAN_WK and _v_ not in _sd_arife:

        # sarkan haftanın PLAN tarafındaki hafta içi günleri
        _sd_haftaici = [
            ds for ds in PLAN_GUNLER
            if day_week[ds] == SARKAN_WK
            and pd.to_datetime(ds).weekday() < 5
            and ds not in _sd_tatil
        ]

        # agent -> ÖNCEKİ GÜNKÜ vardiyası (o gün çalışmadıysa kayıt yok -> devir yok)
        _sd_agent_st = {}
        for _a in AGENTS:
            _a = str(_a).strip()
            _gc = _sd_gecmis.get(_a, {})
            if _sd_onceki in _gc:
                _sd_agent_st[_a] = (str(_gc[_sd_onceki][0]).strip()[:5],
                                    str(_gc[_sd_onceki][1]).strip()[:5])

        # takım -> önceki günkü baskın vardiya (agent sayısına göre)
        _sd_takim_c = {}
        for _a, _st in _sd_agent_st.items():
            _t = agent_team.get(_a)
            if _t is None or pd.isna(_t):
                continue
            _t = str(_t).strip()
            _sd_takim_c.setdefault(_t, {})
            _sd_takim_c[_t][_st] = _sd_takim_c[_t].get(_st, 0) + 1
        _sd_takim_st = {
            _t: max(_c.items(), key=lambda kv: (kv[1], kv[0]))[0]
            for _t, _c in _sd_takim_c.items()
        }

        for _a, _st in sorted(_sd_agent_st.items()):
            _t = agent_team.get(_a)
            _t = None if (_t is None or pd.isna(_t)) else str(_t).strip()
            _v = _sd_saat_v.get(_st)

            if _t is None:
                _durum = "ATLANDI - takımı yok"
            elif _a in _tb_kisitli:
                _durum = "ATLANDI - sabah/kıdemsiz (muafiyet mekanizması yönetiyor)"
            elif _v is None:
                _durum = "ATLANDI - devraldığı saat aralığı bu ayda yok"
            elif _sd_takim_st.get(_t) == _st:
                _durum = "ATLANDI - takımıyla aynı vardiya devrediyor (kilit zaten aynı)"
            elif _tb_yasak_mi(_a, SARKAN_WK, _v):
                _durum = "ATLANDI - agent bu vardiyada çalışamaz (kıdem/sabah)"
            else:
                _eksik = [ds for ds in _sd_haftaici if _v not in gun_vardiyalari.get(ds, [])]
                if _eksik:
                    _durum = f"ATLANDI - vardiya {len(_eksik)} hafta içi günde açık değil"
                else:
                    SARKAN_DEVIR_AGENT[_a] = _v
                    _durum = "KİŞİ BAZLI DEVREDİLDİ"

            _sd_rows.append({
                "agent_user_code": _a, "takim": _t,
                "onceki_gun_saat": "-".join(_st),
                "takim_baskin": "-".join(_sd_takim_st.get(_t, ("", ""))) if _t else None,
                "vardiya": _v, "durum": _durum,
            })

sarkan_kisi_devri_df = pd.DataFrame(_sd_rows)

print(f"ENABLE_SARKAN_KISI_DEVRI = {ENABLE_SARKAN_KISI_DEVRI}  |  sarkan hafta = {SARKAN_WK}")
print(f"Kişi bazlı devredilen agent: {len(SARKAN_DEVIR_AGENT)}")

# --- 2026-08-19: MUAF AGENT DE HAFTA BOYU TEK VARDIYADA KALIR ---
# Muaf agent eskiden tamamen serbest kalıyordu (x <= base + muaf; muaf=1 iken
# sağ taraf her vardiya için >=1). Artık muaf agente BİREYSEL HAFTALIK BASE
# veriliyor:  Σ_v agent_week_base[(a,wk,v)] == takim_muaf[(a,wk)]
#   muaf = 0 -> hepsi 0, agent takımıyla aynı vardiyada
#   muaf = 1 -> tam biri 1, agent hafta boyu O vardiyada
# Değişken sadece agentin ÇALIŞABİLDİĞİ vardiyalar için üretilir.
if "ENABLE_MUAF_AGENT_HAFTALIK_TEK_VARDIYA" not in globals():
    raise SystemExit(
        "CONFIG'de tanımlı olmalı: ENABLE_MUAF_AGENT_HAFTALIK_TEK_VARDIYA "
        "-> [CONFIG] hücresini çalıştır."
    )

takim_muaf = {}
takim_muaf_debug_rows = []
agent_week_base = {}          # (agent, hafta, vardiya) -> BoolVar
_awb_kurulan = set()          # (agent, hafta) - bireysel base kurulanlar
_awb_uygun_yok = []           # agentin o hafta hiç uygun vardiyası yok

for a in AGENTS:
    a = str(a).strip()
    if a not in _tb_kisitli:
        continue
    t = agent_team.get(a)
    if t is None or pd.isna(t):
        continue
    t = str(t).strip()
    for wk in WEEKS:
        _yasak_vars = [
            team_week_base[(t, wk, v)]
            for v in week_vardiyalari.get(wk, [])
            if (t, wk, v) in team_week_base and _tb_yasak_mi(a, wk, v)
        ]
        if not _yasak_vars:
            continue
        _muaf = model.NewBoolVar(f"takim_muaf_{a}_{wk}")
        model.Add(_muaf == sum(_yasak_vars))
        takim_muaf[(a, wk)] = _muaf
        takim_muaf_debug_rows.append({
            "agent_user_code": a, "takim": t, "hafta": wk,
            "yasak_base_adedi": len(_yasak_vars),
            "toplam_base_secenegi": len([v for v in week_vardiyalari.get(wk, [])
                                         if (t, wk, v) in team_week_base]),
        })

        # ---- bireysel haftalık base ----
        if not ENABLE_MUAF_AGENT_HAFTALIK_TEK_VARDIYA:
            continue
        _uygun = [
            v for v in week_vardiyalari.get(wk, [])
            if v not in arife_ozel_vardiya_kodlari and not _tb_yasak_mi(a, wk, v)
        ]
        if not _uygun:
            # hiç uygun vardiya yok -> Σ == muaf kısıtı agenti çalışamaz yapar,
            # infeasible olmasın diye ESKİ davranışa (serbest) bırakılır
            _awb_uygun_yok.append({"agent_user_code": a, "takim": t, "hafta": wk})
            continue
        _awb_vars = []
        for v in _uygun:
            _b = model.NewBoolVar(f"agent_week_base_{a}_{wk}_{v}")
            agent_week_base[(a, wk, v)] = _b
            _awb_vars.append(_b)
        model.Add(sum(_awb_vars) == _muaf)
        _awb_kurulan.add((a, wk))

takim_muaf_debug_df = pd.DataFrame(takim_muaf_debug_rows)
agent_week_base_uygun_yok_df = pd.DataFrame(_awb_uygun_yok)

print(f"ENABLE_MUAF_AGENT_HAFTALIK_TEK_VARDIYA = {ENABLE_MUAF_AGENT_HAFTALIK_TEK_VARDIYA}")
print(f"Bireysel haftalık base kurulan agent-hafta: {len(_awb_kurulan)} "
      f"({len(agent_week_base)} değişken)")
if _awb_uygun_yok:
    print(f"!!! UYARI: {len(_awb_uygun_yok)} agent-haftada agentin HİÇ uygun vardiyası yok "
          f"-> o agent-hafta eski davranışta (serbest) bırakıldı.")
    print(agent_week_base_uygun_yok_df.head(10).to_string(index=False))

# ====== 2026-08-12 - TAKIM BÜTÜNLÜĞÜ: ÇOĞUNLUK ÇALIŞAMIYORSA BASE OLMAZ ======
# Muafiyet kuralı AZINLIK için tasarlandı ("tek sabah agentı takımı engellemesin").
# Ama takımın TAMAMI kısıtlıysa (örn. uydu lokasyon, herkesin sabah_calisir_flg'i 1)
# model geç bir base seçtiği anda HERKES aynı anda muaf oluyor, kilit tümüyle
# kalkıyor ve ekip dağılıyor - biri 09:00, biri 11:00.
#
# Çözüm: bir vardiyada takımın TAKIM_BASE_MAX_YASAKLI_ORAN'dan fazlası
# çalışamıyorsa o vardiya o takıma base olarak VERİLEMEZ.
#
#   herkes sabah (%100 yasaklı) -> geç base kapalı -> model sabah base seçer
#                                  -> HERKES AYNI VARDİYADA, bölünme yok
#   karma ekip (%20 yasaklı)    -> geç base açık -> o azınlık muaf, bölünme burada
#
# GÜVENLİK: bir takım-haftada TÜM base seçenekleri yasaklanacak olursa
# (sum(vars_base) == 1 kısıtı infeasible olurdu) en düşük yasaklı orana sahip
# olan(lar) serbest bırakılır ve uyarı basılır.
if "TAKIM_BASE_MAX_YASAKLI_ORAN" not in globals():
    raise SystemExit(
        "CONFIG'de tanımlı olmalı: TAKIM_BASE_MAX_YASAKLI_ORAN -> [CONFIG] hücresini çalıştır."
    )

_tb_uyeler = {}
for _a in AGENTS:
    _a = str(_a).strip()
    _t = agent_team.get(_a)
    if _t is None or pd.isna(_t):
        continue
    _tb_uyeler.setdefault(str(_t).strip(), []).append(_a)

team_base_yasak_constraints = 0
takim_base_yasak_rows = []
_tb_tumu_yasak_uyari = []

for _t, _uy in _tb_uyeler.items():
    _n = len(_uy)
    if _n == 0:
        continue
    for _wk in WEEKS:
        _adaylar = []
        for _v in week_vardiyalari.get(_wk, []):
            if (_t, _wk, _v) not in team_week_base or _v in arife_ozel_vardiya_kodlari:
                continue
            _yasakli = sum(1 for _a in _uy if _tb_yasak_mi(_a, _wk, _v))
            _adaylar.append((_v, _yasakli, _yasakli / _n))
        if not _adaylar:
            continue

        _kapatilacak = [x for x in _adaylar if x[2] > TAKIM_BASE_MAX_YASAKLI_ORAN]
        if len(_kapatilacak) == len(_adaylar):
            # hepsi kapanacaktı -> en az yasaklı olan(lar) açık kalsın
            _min_oran = min(x[2] for x in _adaylar)
            _kapatilacak = [x for x in _adaylar if x[2] > _min_oran]
            _tb_tumu_yasak_uyari.append({
                "takim": _t, "hafta": _wk,
                "en_dusuk_yasakli_oran": round(_min_oran, 3),
                "acik_birakilan": len(_adaylar) - len(_kapatilacak),
            })

        for _v, _ys, _or in _kapatilacak:
            model.Add(team_week_base[(_t, _wk, _v)] == 0)
            team_base_yasak_constraints += 1
            _st = _tb_wk_saat.get((_wk, _v), ("", ""))
            takim_base_yasak_rows.append({
                "takim": _t, "hafta": _wk, "vardiya": _v,
                "baslangic": _st[0], "bitis": _st[1],
                "takim_mevcut": _n, "yasakli_agent": _ys,
                "yasakli_oran": round(_or, 3),
            })

takim_base_yasak_df = pd.DataFrame(takim_base_yasak_rows)
takim_base_yasak_uyari_df = pd.DataFrame(_tb_tumu_yasak_uyari)

print(f"TAKIM_BASE_MAX_YASAKLI_ORAN = {TAKIM_BASE_MAX_YASAKLI_ORAN}")
print(f"Bütünlük gereği KAPATILAN takım-hafta-vardiya base seçeneği: {team_base_yasak_constraints}")
if not takim_base_yasak_df.empty:
    _tam_kapali = takim_base_yasak_df[takim_base_yasak_df["yasakli_oran"] >= 1.0]
    print(f"  bunlardan TAMAMEN uygunsuz (oran = 1.0): {len(_tam_kapali)}")
    print("  etkilenen takım sayısı:", takim_base_yasak_df["takim"].nunique())
    print("  takım bazında kapatılan seçenek (ilk 10):")
    print(takim_base_yasak_df.groupby("takim").size().sort_values(ascending=False)
          .head(10).to_string())
if _tb_tumu_yasak_uyari:
    print(f"\n!!! UYARI: {len(_tb_tumu_yasak_uyari)} takım-haftada TÜM base seçenekleri "
          f"eşiği aşıyordu; infeasible olmasın diye en az yasaklı olan(lar) açık bırakıldı.")
    print("    Bu takımlar o hafta yine bölünecek - eşiği yükseltmek çözmez, "
          "vardiya arzı o takıma uygun değil.")
    print(takim_base_yasak_uyari_df.head(10).to_string(index=False))

team_base_constraints = 0
team_weekday_link_constraints = 0
team_muaf_link_constraints = 0
weekend_free_count = 0
tatil_team_base_skip_count = 0

# Her takım-her hafta için tek base vardiya seç
for t in TAKIMLAR:
    for wk in WEEKS:
        vars_base = [
            team_week_base[(t, wk, v)]
            for v in week_vardiyalari[wk]
            if (t, wk, v) in team_week_base and v not in arife_ozel_vardiya_kodlari
        ]
        if vars_base:
            model.Add(sum(vars_base) == 1)
            team_base_constraints += 1

# Hafta içi agent takımının base vardiyasında çalışır; hafta sonu serbest
for a in AGENTS:
    a = str(a).strip()
    t = agent_team.get(a)
    if t is None or pd.isna(t):
        continue
    t = str(t).strip()

    for ds in PLAN_GUNLER:
        if ds in ozel_tatil_plan_gunleri:
            tatil_team_base_skip_count += 1
            continue

        weekday = pd.to_datetime(ds).weekday()
        wk = day_week[ds]

        for v in gun_vardiyalari.get(ds, []):
            if (a, ds, v) not in x:
                continue
            if weekday in [0, 1, 2, 3, 4]:
                if wk == SARKAN_WK and a in SARKAN_DEVIR_AGENT:
                    # kişi bazlı devir: takım kilidi yerine kendi devraldığı vardiya.
                    # SOFT -> başka kural zorlarsa sapabilir, ceza alır, durmaz.
                    if v != SARKAN_DEVIR_AGENT[a]:
                        if a not in sarkan_kisi_sapma:
                            sarkan_kisi_sapma[a] = model.NewBoolVar(f"sarkan_kisi_sapma_{a}")
                        model.Add(x[(a, ds, v)] <= sarkan_kisi_sapma[a])
                        sarkan_devir_link_constraints += 1
                elif not TAKIM_BASE_HAFTAICI_HARD:
                    pass                      # TEŞHİS: hafta içi takım bağlantısı kapalı
                elif (t, wk, v) in team_week_base:
                    if (a, wk) in takim_muaf:
                        # base bu agente yasaksa muaf=1 -> kilit gevşer.
                        # 2026-08-19: gevşeme artık SERBEST BIRAKMIYOR - agent
                        # kendi bireysel haftalık base'ine bağlanıyor.
                        if (a, wk) in _awb_kurulan:
                            _ind = agent_week_base.get((a, wk, v))
                            if _ind is None:
                                # v bu agente yasak -> sadece takım base'i geçerli
                                model.Add(x[(a, ds, v)] <= team_week_base[(t, wk, v)])
                            else:
                                model.Add(x[(a, ds, v)]
                                          <= team_week_base[(t, wk, v)] + _ind)
                        else:
                            model.Add(x[(a, ds, v)]
                                      <= team_week_base[(t, wk, v)] + takim_muaf[(a, wk)])
                        team_muaf_link_constraints += 1
                    else:
                        model.Add(x[(a, ds, v)] <= team_week_base[(t, wk, v)])
                        team_weekday_link_constraints += 1
                else:
                    model.Add(x[(a, ds, v)] == 0)
                    team_weekday_link_constraints += 1
            else:
                weekend_free_count += 1

print("TAKIM_BASE_HAFTAICI_HARD =", TAKIM_BASE_HAFTAICI_HARD)
