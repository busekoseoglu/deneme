/* =====================================================================
   YURT DIŞI SEYAHAT GEÇMİŞİ DEĞİŞKENLERİ - CANLI (AYLIK SKORLAMA)
   ---------------------------------------------------------------------
   Çalışma zamanı : her ayın 1'i (veri t-1, yani önceki ayın son günü)
   Snapshot       : SNAP = önceki ayın son günü
                    örn. 2026-11-01'de çalışınca SNAP = 2026-10-31
                    -> model Kasım 2026'da yurt dışına çıkacakları tahmin eder
   Geriye bakış   : SNAP dahil son 24 ay  (örn. 2024-11 .. 2026-10)

   Çıktı anahtarı : PARTY_ID + DATA_DATE (= SNAP)
   - Sadece son 24 ayda en az bir kez yurt dışına çıkmış müşteriler için satır üretir.
   - Skorlama tablosuna LEFT JOIN ON party_id, data_date yapılır;
     eşleşmeyen müşterilerde TÜM değişkenler 0 ile doldurulur.

   Seyahat tanımı (target ile aynı): bir ay içinde
     - yurt dışı fiziksel POS işlemi (terminal 11, ülke TR/TU/TUR dışı, entry 1/4/5/7) VEYA
     - yurt dışı ATM işlemi (NOF_INTERNATIONAL_ATM_TRX > 0)

   Modelde kullanılan değişkenler (m = SNAP ile seyahat ayı arasındaki ay farkı):
     syh_snap_ay_flg                  : SNAP ayında yurt dışında mı           (m = 0)
     syh_gecen_yil_hedef_donem_ay_say : geçen yıl hedef ay ±1, kaç ay çıkmış  (m = 10..12)
     syh_iki_yil_once_hedef_ay_flg    : iki yıl önce hedef ayda çıkmış mı     (m = 23)
     syh_yd_pos_tutar_12m             : son 12 ay yurt dışı POS tutarı (TL)   (m = 0..11)
     syh_yd_atm_islem_12m             : son 12 ay yurt dışı ATM işlem sayısı  (m = 0..11)
   ===================================================================== */

CREATE TABLE BSEN_SYH_DEGISKENLER_CANLI NOLOGGING AS
WITH PRM AS (
    SELECT TRUNC(SYSDATE, 'MM') - 1                        AS snap,      -- önceki ayın son günü
           ADD_MONTHS(TRUNC(SYSDATE, 'MM'), -24)           AS bas_gun    -- 24 ay önceki ayın 1'i
    FROM dual
    -- Elle çalıştırmak için:
    -- SELECT DATE '2026-10-31' AS snap, DATE '2024-11-01' AS bas_gun FROM dual
),
POS AS (
    SELECT a.party_id,
           LAST_DAY(TRUNC(a.card_transaction_date)) AS seyahat_ay,
           SUM(CASE WHEN a.transaction_amount_tl > 0
                    THEN a.transaction_amount_tl ELSE 0 END) AS pos_tutar
    FROM   edw.trx_card_transaction a
    CROSS JOIN prm p
    WHERE  a.card_transaction_date >= p.bas_gun
      AND  a.card_transaction_date <  p.snap + 1
      AND  a.transaction_terminal_tp_id = 11
      AND  a.trx_card_acptor_cntry_nk NOT IN ('TR', 'TU', 'TUR')
      AND  a.transaction_entry_tp_id IN (1, 4, 5, 7)
    GROUP BY a.party_id, LAST_DAY(TRUNC(a.card_transaction_date))
),
ATM AS (
    SELECT c.party_id,
           c.data_date AS seyahat_ay,
           MAX(c.nof_international_atm_trx) AS atm_islem
    FROM   dm.cnf_card_financial_trx_aggr c
    CROSS JOIN prm p
    WHERE  c.data_date BETWEEN LAST_DAY(p.bas_gun) AND p.snap
      AND  c.nof_international_atm_trx > 0
    GROUP BY c.party_id, c.data_date
),
AYLIK AS (
    -- müşteri x seyahat ayı (aynı ay hem POS hem ATM olabilir -> birleştir)
    SELECT party_id,
           seyahat_ay,
           SUM(pos_tutar) AS pos_tutar,
           SUM(atm_islem) AS atm_islem
    FROM (
        SELECT party_id, seyahat_ay, pos_tutar, 0 AS atm_islem FROM pos
        UNION ALL
        SELECT party_id, seyahat_ay, 0, atm_islem FROM atm
    )
    GROUP BY party_id, seyahat_ay
),
X AS (
    SELECT s.party_id,
           p.snap,
           MONTHS_BETWEEN(p.snap, s.seyahat_ay) AS m,      -- 0 .. 23
           s.pos_tutar,
           s.atm_islem
    FROM   aylik s
    CROSS JOIN prm p
)
SELECT party_id,
       snap                                                   AS data_date,
       MAX(CASE WHEN m = 0             THEN 1 ELSE 0 END)     AS syh_snap_ay_flg,
       SUM(CASE WHEN m BETWEEN 10 AND 12 THEN 1 ELSE 0 END)   AS syh_gecen_yil_hedef_donem_ay_say,
       MAX(CASE WHEN m = 23            THEN 1 ELSE 0 END)     AS syh_iki_yil_once_hedef_ay_flg,
       SUM(CASE WHEN m < 12 THEN pos_tutar ELSE 0 END)        AS syh_yd_pos_tutar_12m,
       SUM(CASE WHEN m < 12 THEN atm_islem ELSE 0 END)        AS syh_yd_atm_islem_12m
FROM   x
GROUP BY party_id, snap;


/* ---------------------------------------------------------------------
   KONTROL (her çalışmadan sonra)
   Beklenen (geçmiş aylara göre):
     n            : ~1.0 - 1.1 milyon
     snap_ay_oran : ~%14 - %19 (mevsime göre)
   Bu aralığın çok dışında bir sonuç -> kaynak veride eksik/gecikme olabilir,
   skorlamayı durdur.
   --------------------------------------------------------------------- */
SELECT data_date,
       COUNT(*)                        AS n,
       ROUND(AVG(syh_snap_ay_flg), 4)  AS snap_ay_oran
FROM   bsen_syh_degiskenler_canli
GROUP BY data_date;
