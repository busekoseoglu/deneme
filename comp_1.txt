ARIFE_HAFTASONU_MESAIYE_KALAMAZ_YASAK = CONFIG["ARIFE_HAFTASONU_MESAIYE_KALAMAZ_YASAK"]



# %% [KISIT] - MESAİYE KALAMAZ AGENT: FAZLA ÇALIŞMA YOK (HARD)
# mesaiye_kalamaz_flg = 1 olan agent HİÇBİR tipte mesai yapamaz.
# Mesai tanımı AYLIK: aylık iş günü kadar çalışır, bir gün bile fazlası mesaidir.
#
#   1) aylik_mesai_gun[a] == 0
#      -> toplam_calisma tam olarak baz_gereken_calisma kadar olur
#         (izni olmayan agentte 21 gün; izinli agentte 21 - izin)
#
#   2) resmi tatil gününde work == 0
#      -> ŞART, çünkü [HAZIRLIK / KISIT] - AYLIK ÇALIŞMA DENGESİ hücresinde
#         calisma_vars resmi tatil günlerini HİÇ saymıyor. O gün çalışmak aylık
#         toplama girmediği için (1) numaralı kısıt onu göremez ve agent
#         21 gün + resmi tatil = 22 gün çalışmış olur. Bu kısıt o açığı kapatır.
#
#   3) arife günü HAFTA SONUNA denk geliyorsa work == 0
#      -> arife hafta sonuna düşerse o gün çalışmak mesai olur. Agent kotasını
#         hafta içi günlerden doldurur. HAFTA İÇİ arife günlerine dokunulmaz.
#         Parametre: ARIFE_HAFTASONU_MESAIYE_KALAMAZ_YASAK (CONFIG)
#
# NOT: hamile / süt izni agentları da tatil_kisitli_agents içinde ama onlar için
#      resmi tatil kuralı SOFT kalmaya devam eder (resmi_tatil_kisitli_ihlal).
#      Hafta sonu zaten [HÜCRE 14] ile kapalı oldukları için 3. kural onları
#      ilgilendirmiyor. Burada SADECE mesaiye_kalamaz_flg = 1 olanlar hard yapılır.
#
# Bu hücre [HAZIRLIK / KISIT] - AYLIK ÇALIŞMA DENGESİ hücresinden SONRA çalışmalıdır
# (aylik_mesai_gun değişkeni orada yaratılıyor).

if "aylik_mesai_gun" not in globals():
    raise SystemExit(
        "aylik_mesai_gun yok. Önce [HAZIRLIK / KISIT] - AYLIK ÇALIŞMA DENGESİ hücresini çalıştır."
    )
if "ARIFE_HAFTASONU_MESAIYE_KALAMAZ_YASAK" not in globals():
    raise SystemExit(
        "CONFIG'de tanımlı olmalı: ARIFE_HAFTASONU_MESAIYE_KALAMAZ_YASAK -> [CONFIG] hücresini çalıştır."
    )

_mk_flg = pd.to_numeric(df_tam["mesaiye_kalamaz_flg"], errors="coerce").fillna(0).astype(int)
mesaiye_kalamaz_agents = set(
    df_tam.loc[_mk_flg == 1, "agent_user_code"].astype(str).str.strip()
)

_mk_resmi_tatil_set = set()
if "resmi_tatil_plan_gunleri" in globals():
    _mk_resmi_tatil_set = {pd.to_datetime(d).date() for d in resmi_tatil_plan_gunleri}
elif "RESMI_TATIL_GUNLERI" in globals():
    _mk_resmi_tatil_set = {pd.to_datetime(d).date() for d in RESMI_TATIL_GUNLERI}

# --- arife günlerinden HAFTA SONUNA denk gelenler ---
_mk_arife_haftasonu = []
if ARIFE_HAFTASONU_MESAIYE_KALAMAZ_YASAK:
    _arife_gunler = globals().get("arife_plan_gunleri")
    if _arife_gunler is None:
        _arife_cfg = globals().get("ARIFE_GUNLERI", {})
        _arife_gunler = [
            ds for ds in PLAN_GUNLER
            if globals().get("ENABLE_ARIFE_KURALI", False)
            and pd.to_datetime(ds).strftime("%Y-%m-%d") in _arife_cfg
        ]
    _mk_arife_haftasonu = [ds for ds in _arife_gunler if pd.to_datetime(ds).weekday() in (5, 6)]

mk_mesai_sifir_kisiti = 0
mk_resmi_tatil_kisiti = 0
mk_arife_hs_kisiti = 0
mk_atlanan = []
mk_debug_rows = []

for a_raw in AGENTS:
    a = str(a_raw).strip()
    if a not in mesaiye_kalamaz_agents:
        continue

    # --- 1) aylık mesai sıfır ---
    if a in aylik_mesai_gun:
        model.Add(aylik_mesai_gun[a] == 0).WithName(f"mk_mesai_sifir_{a}")
        mk_mesai_sifir_kisiti += 1
        _mesai_kuruldu = True
    else:
        mk_atlanan.append((a, "aylik_mesai_gun icinde yok"))
        _mesai_kuruldu = False

    # --- 2) resmi tatilde çalışma yok ---
    _rt_kapatilan = 0
    for ds in PLAN_GUNLER:
        if pd.to_datetime(ds).date() not in _mk_resmi_tatil_set:
            continue
        if (a, ds) in work:
            model.Add(work[(a, ds)] == 0).WithName(f"mk_resmi_tatil_{a}_{ds}")
            mk_resmi_tatil_kisiti += 1
            _rt_kapatilan += 1

    # --- 3) arife hafta sonuna denk geliyorsa çalışamaz ---
    _arife_hs_kapatilan = 0
    for ds in _mk_arife_haftasonu:
        if (a, ds) in work:
            model.Add(work[(a, ds)] == 0).WithName(f"mk_arife_hs_{a}_{ds}")
            mk_arife_hs_kisiti += 1
            _arife_hs_kapatilan += 1

    mk_debug_rows.append({
        "agent_user_code": a,
        "aylik_mesai_sifir_kuruldu": _mesai_kuruldu,
        "kapatilan_resmi_tatil_gunu": _rt_kapatilan,
        "kapatilan_arife_haftasonu_gunu": _arife_hs_kapatilan,
    })

mesaiye_kalamaz_debug_df = pd.DataFrame(mk_debug_rows)

print(f"mesaiye_kalamaz_flg = 1 agent: {len(mesaiye_kalamaz_agents)} / {len(AGENTS)}")
print("plan içi resmi tatil günü:", sorted(str(d) for d in _mk_resmi_tatil_set) or "yok")
print("aylık mesai=0 kısıtı            :", mk_mesai_sifir_kisiti)
print("resmi tatilde çalışma yok kısıtı:", mk_resmi_tatil_kisiti)
print("arife hafta sonu yasağı aktif mi:", ARIFE_HAFTASONU_MESAIYE_KALAMAZ_YASAK)
print("hafta sonuna denk gelen arife günü:",
      [pd.to_datetime(d).strftime("%Y-%m-%d (%a)") for d in _mk_arife_haftasonu] or "yok")
print("arife hafta sonu çalışma yok kısıtı:", mk_arife_hs_kisiti)
if mk_atlanan:
    print("!!! ATLANAN AGENT:", mk_atlanan[:10])
if len(mesaiye_kalamaz_agents) == 0:
    print("!!! UYARI: mesaiye_kalamaz_flg = 1 olan agent yok. df_tam'da bayrak dolu mu?")




# %% [KONTROL] - MESAİYE KALAMAZ AGENT FAZLA ÇALIŞMA (HARD)
# mesaiye_kalamaz_flg = 1 olan agent aylık iş günü kadar çalışır; fazlası mesaidir.
# İki ihlal tipi ayrı ayrı sayılır:
#   - aylik_fazla_gun     : sayılan çalışma > baz_gereken_calisma
#   - resmi_tatil_calisma : resmi tatil gününde çalışma (aylık toplama girmez, ayrıca mesaidir)
#   - arife_hs_calisma    : HAFTA SONUNA denk gelen arife gününde çalışma
# Bu hücre modele DOKUNMAZ, sadece mevcut çözümü denetler.

_mk_flg_k = pd.to_numeric(df_tam["mesaiye_kalamaz_flg"], errors="coerce").fillna(0).astype(int)
_mk_agents_k = set(df_tam.loc[_mk_flg_k == 1, "agent_user_code"].astype(str).str.strip())

_mk_rt_k = set()
if "resmi_tatil_plan_gunleri" in globals():
    _mk_rt_k = {pd.to_datetime(d).date() for d in resmi_tatil_plan_gunleri}
elif "RESMI_TATIL_GUNLERI" in globals():
    _mk_rt_k = {pd.to_datetime(d).date() for d in RESMI_TATIL_GUNLERI}

# hafta sonuna denk gelen arife günleri
_mk_arife_hs_k = []
if globals().get("ARIFE_HAFTASONU_MESAIYE_KALAMAZ_YASAK", False):
    _ar_k = globals().get("arife_plan_gunleri")
    if _ar_k is None:
        _ar_cfg_k = globals().get("ARIFE_GUNLERI", {})
        _ar_k = [
            ds for ds in PLAN_GUNLER
            if globals().get("ENABLE_ARIFE_KURALI", False)
            and pd.to_datetime(ds).strftime("%Y-%m-%d") in _ar_cfg_k
        ]
    _mk_arife_hs_k = [ds for ds in _ar_k if pd.to_datetime(ds).weekday() in (5, 6)]
_mk_arife_hs_set = {pd.to_datetime(d).date() for d in _mk_arife_hs_k}

# baz_gereken_calisma: debug df varsa oradan al, yoksa aynı formülle yeniden türet
_baz_map = {}
if "aylik_calisma_debug_df" in globals() and not aylik_calisma_debug_df.empty:
    _baz_map = dict(zip(
        aylik_calisma_debug_df["agent_user_code"].astype(str).str.strip(),
        aylik_calisma_debug_df["baz_gereken_calisma"].astype(int),
    ))
else:
    _is_gunu_d = {
        pd.to_datetime(ds).date() for ds in PLAN_GUNLER
        if pd.to_datetime(ds).weekday() in (0, 1, 2, 3, 4)
        and pd.to_datetime(ds).date() not in _mk_rt_k
    }
    for a_raw in AGENTS:
        a = str(a_raw).strip()
        _izn = {d for d, t in tip_map.get(a, {}).items()
                if str(t).strip().lower() == "izin" and d in _is_gunu_d}
        _baz_map[a] = len(_is_gunu_d) - len(_izn)

mk_ihlal_rows = []
mk_tum_rows = []
for a_raw in AGENTS:
    a = str(a_raw).strip()
    if a not in _mk_agents_k:
        continue
    _calisilan = [ds for ds in PLAN_GUNLER if cozum_calisma.get((a, ds), 0) == 1]
    _rt_calisma = [ds for ds in _calisilan if pd.to_datetime(ds).date() in _mk_rt_k]
    _ar_calisma = [ds for ds in _calisilan if pd.to_datetime(ds).date() in _mk_arife_hs_set]
    _sayilan = [ds for ds in _calisilan if pd.to_datetime(ds).date() not in _mk_rt_k]
    _baz = int(_baz_map.get(a, 0))
    _fazla = max(0, len(_sayilan) - _baz)

    _row = {
        "agent_user_code": a,
        "baz_gereken_calisma": _baz,
        "sayilan_calisma": len(_sayilan),
        "aylik_fazla_gun": _fazla,
        "resmi_tatil_calisma": len(_rt_calisma),
        "resmi_tatil_gunleri": ", ".join(
            pd.to_datetime(d).strftime("%Y-%m-%d") for d in _rt_calisma
        ),
        "arife_hs_calisma": len(_ar_calisma),
        "arife_hs_gunleri": ", ".join(
            pd.to_datetime(d).strftime("%Y-%m-%d") for d in _ar_calisma
        ),
        "toplam_calisma": len(_calisilan),
    }
    mk_tum_rows.append(_row)
    if _fazla > 0 or _rt_calisma or _ar_calisma:
        mk_ihlal_rows.append(_row)

mesaiye_kalamaz_tum_df = pd.DataFrame(mk_tum_rows)
mesaiye_kalamaz_ihlal_df = pd.DataFrame(mk_ihlal_rows)

print(f"mesaiye_kalamaz_flg = 1 agent: {len(_mk_agents_k)} / {len(AGENTS)}")
print("hafta sonuna denk gelen arife günü:",
      [pd.to_datetime(d).strftime("%Y-%m-%d (%a)") for d in _mk_arife_hs_k] or "yok")
kaydet("Mesaiye kalamaz agent fazla çalışmıyor", len(mk_ihlal_rows))

if mk_ihlal_rows:
    print(f"   aylık fazla gün toplamı  : {int(mesaiye_kalamaz_ihlal_df['aylik_fazla_gun'].sum())}")
    print(f"   resmi tatilde çalışma    : {int(mesaiye_kalamaz_ihlal_df['resmi_tatil_calisma'].sum())}")
    print(f"   arife hafta sonu çalışma : {int(mesaiye_kalamaz_ihlal_df['arife_hs_calisma'].sum())}")
    display(
        mesaiye_kalamaz_ihlal_df.sort_values(
            ["aylik_fazla_gun", "resmi_tatil_calisma", "arife_hs_calisma"], ascending=False
        )
    )
elif len(_mk_agents_k) == 0:
    print("   !!! UYARI: bayrağı 1 olan agent yok, kontrol boş çalıştı.")
