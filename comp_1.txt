#   (a) provizyoncu hafta sonu sadece provizyon vardiyasında çalışır
#   (b) provizyoncu olmayan provizyon vardiyasını alamaz

from datetime import timedelta

    if _p not in globals():
        raise SystemExit(f"CONFIG'de tanımlı olmalı: {_p} -> [CONFIG] hücresini çalıştır.")

prov_gun_muaf = {}       # ds -> BoolVar (o gün provizyoncu yok)
prov_cift_kacis = {}     # (a, Cmt ds) -> BoolVar (Cmt+Pzr aynı kişi)

_pv_tarih = {pd.to_datetime(ds).date(): ds for ds, _ in prov_vardiyalar}
_pv_bos = []

for d, ds in sorted(_pv_tarih.items()):
    for a in AGENTS:
        for v in gun_vardiyalari.get(ds, []):
            if (a, ds, v) in x and (a in prov_agents) != (v == _pv):
                model.Add(x[(a, ds, v)] == 0)                       # (a) + (b)
    _gv = [x[(a, ds, _pv)] for a in sorted(prov_agents) if (a, ds, _pv) in x]
    if not _gv:
        _pv_bos.append(ds)
    prov_gun_muaf[ds] = model.NewBoolVar(f"prov_gun_muaf_{ds}")
    model.Add(sum(_gv) + prov_gun_muaf[ds] >= 1)                     # (c)
    ds2 = _pv_tarih.get(d + timedelta(days=1)) if d.weekday() == 5 else None
    if ds2 is None:
        continue
    for a in sorted(prov_agents):
        if (a, ds, _pv) in x and (a, ds2, _pv) in x:
            prov_cift_kacis[(a, ds)] = model.NewBoolVar(f"prov_cift_{a}_{ds}")
            model.Add(x[(a, ds, _pv)] + x[(a, ds2, _pv)] <= 1 + prov_cift_kacis[(a, ds)])   # (d)

print("provizyoncu:", len(prov_agents), "| hafta sonu günü:", len(prov_gun_muaf),

# %% [HÜCRE] - BUFFERLI COVERAGE + TALEBE YAKLAŞMA
coverage_lower = {}
coverage_upper = {}
coverage_constraints = 0
target_gap_constraints = 0

for ds in PLAN_GUNLER:
    for v in gun_vardiyalari.get(ds, []):
        if (ds, v) in globals().get("prov_vardiyalar", set()):
            continue   # 2026-10-19: provizyon coverage dışı
        required = int(talep[(ds, v)])
        lower_req = math.floor(required * (1 - BUFFER_RATE))
        upper_req = math.ceil(required * (1 + BUFFER_RATE))
        coverage_lower[(ds, v)] = lower_req
        coverage_upper[(ds, v)] = upper_req

        assigned = sum(x[(a, ds, v)] for a in AGENTS if (a, ds, v) in x)

        model.Add(assigned + under_buffer[(ds, v)] >= lower_req)
        coverage_constraints += 1
        model.Add(assigned - over_buffer[(ds, v)] <= upper_req)
        coverage_constraints += 1
        model.Add(assigned + missing_to_required[(ds, v)] >= required)
        target_gap_constraints += 1
        model.Add(assigned - excess_to_required[(ds, v)] <= required)
        target_gap_constraints += 1

print("Coverage buffer kısıtı:", coverage_constraints)
print("Talebe göre gap kısıtı:", target_gap_constraints)

# --------------------------- HAFTALIK NORM / TELAFİ / MESAİ HESABI (agent)
if "SARKAN_HAFTA_OFF_AYRI" not in globals():
    raise SystemExit("CONFIG'de tanımlı olmalı: SARKAN_HAFTA_OFF_AYRI -> [CONFIG] hücresini çalıştır.")
_hafta_gunleri = defaultdict(list)
for ds in PLAN_GUNLER:
    d = pd.to_datetime(ds).date()
    _pzt = d - _td(days=d.weekday())
    _hafta_gunleri[_pzt].append(ds)
for _k in _hafta_gunleri:
    _hafta_gunleri[_k] = sorted(_hafta_gunleri[_k], key=lambda s: pd.to_datetime(s))

mesai_gunleri_map = {}   # agent -> {ds}  ücretli mesai
telafi_gunleri_map = {}  # agent -> {ds}  borç kapatma
norm_gunleri_map = {}    # agent -> {ds}  kota içi hafta sonu
hafta_calisma_debug_rows = []

for a_raw in AGENTS:
    a = str(a_raw).strip()
    _gecmis_a = _gecmis_gunleri_x.get(a, set())
    _aday_a, _norm_a = [], set()   # _aday_a: haftalık kotayı aşan hafta sonu günleri (kronolojik)
    _sarkan_mesai_a = set()        # SARKAN_HAFTA_OFF_AYRI: sarkan haftada kotayı aşan -> direkt MESAİ

    for _pzt, _wk_gunler in sorted(_hafta_gunleri.items()):
        # RESMİ TATİL günleri haftalık hesaba GİRMEZ: ayrı kategori (RESMİ TATİL MESAİ),
        # aylık toplama da (sayilan_calisma) girmiyor. Haftalık ve aylık adım aynı
        # tabanı kullanmazsa resmi tatil günü aday listesinde yer tutup MESAİ slotunu
        # çalar ve gerçek mesai günü yanlışlıkla TELAFİ olarak etiketlenir.
        _calisilan = [
            ds for ds in _wk_gunler
            if cozum_calisma.get((a, ds), 0) == 1
            and pd.to_datetime(ds).date() not in _sayilmayan_set
        ]

        # aydan sarkan hafta: plan öncesi (önceki ay) çalışılan günleri de say
        _hafta_araligi = {_pzt + _td(days=k) for k in range(7)}
        _gecmis_ek = 0 if SARKAN_HAFTA_OFF_AYRI else len(
            {d for d in _gecmis_a if d in _hafta_araligi and d < _PLAN_BAS})

        _hafta_calisma = len(_calisilan) + _gecmis_ek

        # --- haftalık kota: izin ve resmi tatil günleri DÜŞÜLÜR ---
        _kota = 0
        for _k2 in range(5):                       # Pzt..Cum
            _d2 = _pzt + _td(days=_k2)
            if _d2 < _PLAN_BAS:
                # aydan sarkan hafta: önceki ayın hafta içi günleri
                if not SARKAN_HAFTA_OFF_AYRI and _d2 not in _g_izin.get(a, set()):
                    _kota += 1
            elif _d2 in _plan_date_set_x:
                if _d2 in _sayilmayan_set:
                    continue
                if str(_tip_map.get(a, {}).get(_d2, "")).strip().lower() == "izin":
                    continue
                _kota += 1
        _kota = min(_NORMAL_WORK_DAYS, _kota)

        _fazla = max(0, _hafta_calisma - _kota)

        _hs_calisilan = [ds for ds in _calisilan if pd.to_datetime(ds).weekday() in (5, 6)]
        if _fazla > 0 and _hs_calisilan:
            _aday_bu_hafta = list(_hs_calisilan[-_fazla:])
        else:
            _aday_bu_hafta = []
        _norm_bu_hafta = set(_hs_calisilan) - set(_aday_bu_hafta)

        _aday_a.extend(_aday_bu_hafta)
        if SARKAN_HAFTA_OFF_AYRI and _pzt < _PLAN_BAS:
            _sarkan_mesai_a |= set(_aday_bu_hafta)
        _norm_a |= _norm_bu_hafta

        hafta_calisma_debug_rows.append({
            "agent_user_code": a,
            "hafta_pzt": str(_pzt),
            "plan_ici_calisma": len(_calisilan),
            "oncekiaydan_gelen_calisma": _gecmis_ek,
            "hafta_toplam_calisma": _hafta_calisma,
            "kota": _kota,
            "fazla": _fazla,
            "hafta_sonu_calisilan": len(_hs_calisilan),
            "norm": len(_norm_bu_hafta),
            "kota_asan_aday": len(_aday_bu_hafta),
        })

    # ---- AYLIK ADIM: adaylardan kaçı gerçekten ÜCRETLİ mesai? ----
    _aday_a = sorted(set(_aday_a) - _sarkan_mesai_a, key=lambda d: pd.to_datetime(d))
    _sayilan_a = sum(
        1 for ds in PLAN_GUNLER
        if cozum_calisma.get((a, ds), 0) == 1
        and pd.to_datetime(ds).date() not in _sayilmayan_set
    )
    _aylik_fazla_a = max(0, _sayilan_a - int(_baz_map_x.get(a, 0)) - len(_sarkan_mesai_a))
    _mesai_sayisi = min(_aylik_fazla_a, len(_aday_a))

    mesai_gunleri_map[a] = (set(_aday_a[len(_aday_a) - _mesai_sayisi:]) if _mesai_sayisi else set()) | _sarkan_mesai_a
    telafi_gunleri_map[a] = set(_aday_a) - mesai_gunleri_map[a]
    norm_gunleri_map[a] = _norm_a
