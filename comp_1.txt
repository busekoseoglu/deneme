def _co_haftalik_mumkun(a, sat_ds, sun_ds):
    """Bu cift off alinirsa o haftanin minimum calismasi doldurulabilir mi?"""
    _gunler = _co_hafta.get(pd.to_datetime(sat_ds).date().isocalendar()[:2], [])
    _tip_a = tip_map.get(a, {})
    _sayilan, _zorunlu = [], []
    for _ds in _gunler:
        _d = pd.to_datetime(_ds).date()
        _tip = str(_tip_a.get(_d, "")).strip().lower()
        if _tip == "izin" or _d in _co_sayilmaz:
            continue
        _sayilan.append(_ds)
        if _tip in ("off_t", "off_t2") or not any(
                (a, _ds, v) in x for v in gun_vardiyalari.get(_ds, [])):
            _zorunlu.append(_ds)
    if not _sayilan:
        return True
    _tavan = max(MODEL_OFF_HAFTALIK_MAX, len(_zorunlu))
    _tavan_kalan = _tavan
    if _CO_SARKAN and pd.to_datetime(_gunler[0]).date() == _CO_ILK:

        if a in _CO_HS:
            _korunan |= {_d2 for _d2 in _sayilan

        _tavan_kalan = max(len(_korunan),
                           _tavan - len(_CO_SARKAN & set(_co_gecmis_off.get(a, set()))))
    _min_calisma = len(_sayilan) - _tavan_kalan
    if _min_calisma <= 0:
        return True

    # max-6 altinda o haftada EN FAZLA kac gun calisilabilir (cift kapali)
    _kapali = {sat_ds, sun_ds} | set(_zorunlu)
    _tohum = _co_gecmis_seri(a) if pd.to_datetime(_gunler[0]).date() == _CO_ILK else 0
    _durum = {_tohum: 0}
    for _ds in _sayilan:
        _yeni = {}

            if _ds not in _kapali and _st + 1 <= MAX_CONSECUTIVE_WORK_DAYS:
                _yeni[_st + 1] = max(_yeni.get(_st + 1, -1), _v + 1)
            _yeni[0] = max(_yeni.get(0, -1), _v)
        _durum = _yeni
    return max(_durum.values()) >= _min_calisma
