# 2026-11-12 KİTLE + GOLD (aynı hücre). english_flg = 1 agentlar ingilizceci. Her gün 00:00-24:00 her yarım saatte en az
# INGILIZCE_MIN_KISI ingilizceci çalışır (SOFT, boş yarım saat INGILIZCE_YARIM_SAAT_BOS_W). Gece yarısını
# geçen vardiya ertesi günü de kapsar; ayın 1. günü için önceki ayın son gecesi geçmişten (sabit) sayılır.
# INGILIZCE_SAAT_ARALIGI dışındaki yarım saatler aranmaz (varsayılan 00:00-24:00 = tam gün).
# Ingilizceci takım kilidinden muaf ([HÜCRE] - TAKIM HAFTALIK BASE VARDİYA), ama:
#   - hafta içi hafta boyu TEK vardiyada (HARD), arife/resmi tatil hariç
#   - takımının base'inden farklı vardiya seçerse agent-hafta başına INGILIZCE_TAKIM_DISI_W
# Max 2 hafta gece/akşam, 11 saat, max 6 gün ingilizcecilere de aynen uygulanır. Cezalar OBJECTIVE'de.
from datetime import timedelta as _ing_td

for _p in ("ENABLE_INGILIZCE", "INGILIZCE_FLAG_KOL", "INGILIZCE_MIN_KISI",
           "INGILIZCE_YARIM_SAAT_BOS_W", "INGILIZCE_TAKIM_DISI_W", "INGILIZCE_SAAT_ARALIGI"):
    if _p not in globals():
        raise SystemExit(f"CONFIG'de tanımlı olmalı: {_p} -> [CONFIG] hücresini çalıştır.")

ingilizce_agents = set()
ing_bos = {}             # (ds, yarım saat no 0-47) -> IntVar
ing_takim_disi = {}      # (a, hafta) -> BoolVar
ing_week_base = {}       # (a, hafta, v) -> BoolVar
if ENABLE_INGILIZCE:
    if INGILIZCE_FLAG_KOL not in df_tam.columns:
        raise SystemExit(f"ENABLE_INGILIZCE = True ama df_tam'da '{INGILIZCE_FLAG_KOL}' kolonu yok.")
    ingilizce_agents = set(df_tam.loc[
        pd.to_numeric(df_tam[INGILIZCE_FLAG_KOL], errors="coerce").fillna(0).astype(int) == 1,
        "agent_user_code"].astype(str).str.strip())

def _ing_dk(_t):
    _h, _m = str(_t).strip()[:5].split(":")
    return int(_h) * 60 + int(_m)

def _ing_aralik(_st):
    _b, _e = _ing_dk(_st[0]), _ing_dk(_st[1])
    return _b, (_e + 1440 if _e <= _b else _e)

if ingilizce_agents:
    _ing = sorted(ingilizce_agents)
    _gun = sorted(PLAN_GUNLER, key=lambda s: pd.to_datetime(s))
    _gc_ing = globals().get("gecmis_calisma", {})
    _bant = (_ing_dk(INGILIZCE_SAAT_ARALIGI[0]), 1440 if str(INGILIZCE_SAAT_ARALIGI[1])[:5] in ("24:00", "00:00")
             else _ing_dk(INGILIZCE_SAAT_ARALIGI[1]))
    for _i, ds in enumerate(_gun):
        _onceki = _gun[_i - 1] if _i > 0 else None
        for _k in range(48):
            _s0, _s1 = _k * 30, _k * 30 + 30
            if _s0 < _bant[0] or _s1 > _bant[1]:
                continue
            _terim, _sabit = [], 0
            for v in gun_vardiyalari.get(ds, []):
                _b, _e = _ing_aralik(saat[(ds, v)])
                if _b <= _s0 and _e >= _s1:
                    _terim += [x[(a, ds, v)] for a in _ing if (a, ds, v) in x]
            if _onceki is not None:
                for v in gun_vardiyalari.get(_onceki, []):
                    _b, _e = _ing_aralik(saat[(_onceki, v)])
                    if _b <= _s0 + 1440 and _e >= _s1 + 1440:
                        _terim += [x[(a, _onceki, v)] for a in _ing if (a, _onceki, v) in x]
            else:
                _dun = pd.to_datetime(ds).date() - _ing_td(days=1)
                for a in _ing:
                    _st = _gc_ing.get(a, {}).get(_dun)
                    if _st is not None:
                        _b, _e = _ing_aralik(_st)
                        _sabit += int(_b <= _s0 + 1440 and _e >= _s1 + 1440)
            _ihtiyac = max(0, INGILIZCE_MIN_KISI - _sabit)
            if _ihtiyac == 0:
                continue
            ing_bos[(ds, _k)] = model.NewIntVar(0, _ihtiyac, f"ing_bos_{ds}_{_k}")
            model.Add(sum(_terim) + ing_bos[(ds, _k)] >= _ihtiyac)

    # hafta içi hafta boyu tek vardiya + takımından ayrılma cezası
    _tatil_ing = set(globals().get("ozel_tatil_plan_gunleri", set()))
    for a in _ing:
        _t = agent_team.get(a)
        for ds in PLAN_GUNLER:
            if pd.to_datetime(ds).weekday() >= 5 or ds in _tatil_ing:
                continue
            wk = day_week[ds]
            for v in gun_vardiyalari.get(ds, []):
                if (a, ds, v) not in x:
                    continue
                if (a, wk, v) not in ing_week_base:
                    ing_week_base[(a, wk, v)] = model.NewBoolVar(f"ing_wb_{a}_{wk}_{v}")
                model.Add(x[(a, ds, v)] <= ing_week_base[(a, wk, v)])
        for wk in WEEKS:
            _bs = [(v, b) for (a2, w2, v), b in ing_week_base.items() if a2 == a and w2 == wk]
            if not _bs:
                continue
            model.Add(sum(b for _, b in _bs) <= 1)
            ing_takim_disi[(a, wk)] = model.NewBoolVar(f"ing_takim_disi_{a}_{wk}")
            for v, b in _bs:
                _tb = team_week_base.get((_t, wk, v))
                model.Add(ing_takim_disi[(a, wk)] >= b - (_tb if _tb is not None else 0))

print("ENABLE_INGILIZCE:", ENABLE_INGILIZCE, "| ingilizceci:", len(ingilizce_agents))
print("yarım saat kapsama kısıtı:", len(ing_bos), "| agent-hafta (takım dışı cezası):", len(ing_takim_disi))
if ENABLE_INGILIZCE and len(ingilizce_agents) < 5:
    print(f"!!! {len(ingilizce_agents)} ingilizceci 7/24 için az (gece için haftada 2 kişi + max 2 hafta gece "
          "-> en az ~5 gerekir). Boş saat kaçınılmaz olabilir.")

_ing_k = sorted(globals().get("ingilizce_agents", set()))
ingilizce_saat_df = pd.DataFrame()
if _ing_k:
    def _ik_aralik(_st):
        _b = int(str(_st[0])[:2]) * 60 + int(str(_st[0])[3:5])
        _e = int(str(_st[1])[:2]) * 60 + int(str(_st[1])[3:5])
        return _b, (_e + 1440 if _e <= _b else _e)
    _gun_k = sorted(PLAN_GUNLER, key=lambda s: pd.to_datetime(s))
    _sayac = {ds: [0] * 48 for ds in _gun_k}
    for a in _ing_k:
        for _i, ds in enumerate(_gun_k):
            v = cozum_atama.get((a, ds)) if cozum_calisma.get((a, ds), 0) == 1 else None
            if v is not None:
                _b, _e = _ik_aralik(saat[(ds, v)])
                for _k in range(48):
                    if _b <= _k * 30 and _e >= _k * 30 + 30:
                        _sayac[ds][_k] += 1
                    if _i + 1 < len(_gun_k) and _b <= _k * 30 + 1440 and _e >= _k * 30 + 1470:
                        _sayac[_gun_k[_i + 1]][_k] += 1
        _st0 = globals().get("gecmis_calisma", {}).get(a, {}).get(
            pd.to_datetime(_gun_k[0]).date() - _ik_td(days=1))
        if _st0 is not None:
            _b, _e = _ik_aralik(_st0)
            for _k in range(48):
                if _b <= _k * 30 + 1440 and _e >= _k * 30 + 1470:
                    _sayac[_gun_k[0]][_k] += 1
    _kol = [f"{_k // 2:02d}:{30 * (_k % 2):02d}" for _k in range(48)]
    _bk0 = int(str(INGILIZCE_SAAT_ARALIGI[0])[:2]) * 60 + int(str(INGILIZCE_SAAT_ARALIGI[0])[3:5])
    _bk1 = 1440 if str(INGILIZCE_SAAT_ARALIGI[1])[:5] in ("24:00", "00:00") else \
        int(str(INGILIZCE_SAAT_ARALIGI[1])[:2]) * 60 + int(str(INGILIZCE_SAAT_ARALIGI[1])[3:5])
    _bantta = [_bk0 <= _k * 30 and _k * 30 + 30 <= _bk1 for _k in range(48)]
    ingilizce_saat_df = pd.DataFrame([[ds] + _sayac[ds] for ds in _gun_k], columns=["tarih"] + _kol)
    ingilizce_saat_df.insert(1, "bos_saat", [sum(1 for _k, n in enumerate(_sayac[ds])
                                                 if _bantta[_k] and n < INGILIZCE_MIN_KISI) / 2 for ds in _gun_k])
    kaydet("İngilizce 7/24: ingilizcecisiz yarım saat", int(ingilizce_saat_df["bos_saat"].sum() * 2), tur="SOFT")
    _td_k = globals().get("ing_takim_disi", {})
    kaydet("İngilizce: takımından farklı vardiyada (agent-hafta)",
           sum(int(solver.Value(v)) for v in _td_k.values()), tur="RAPOR")

          f"boş saati olan gün: {int((ingilizce_saat_df['bos_saat'] > 0).sum())}")
    if (ingilizce_saat_df["bos_saat"] > 0).any():
        print(ingilizce_saat_df.loc[ingilizce_saat_df["bos_saat"] > 0, ["tarih", "bos_saat"]].to_string(index=False))
