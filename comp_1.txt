# %% [HÜCRE] - BUFFERLI COVERAGE + TALEBE YAKLAŞMA
coverage_lower = {}
coverage_upper = {}
coverage_constraints = 0
target_gap_constraints = 0
bos_kova = {}            # 2026-11-18: (ds, v) -> 1 ise talebi olan kova boş (ceza OBJECTIVE'de)
if "BOS_KOVA_W" not in globals():
    raise SystemExit("CONFIG'de tanımlı olmalı: BOS_KOVA_W -> [CONFIG] hücresini çalıştır.")

for ds in PLAN_GUNLER:
    for v in gun_vardiyalari.get(ds, []):
        required = int(talep[(ds, v)])
        lower_req = math.floor(required * (1 - BUFFER_RATE))
        upper_req = math.ceil(required * (1 + BUFFER_RATE))
        coverage_lower[(ds, v)] = lower_req
        coverage_upper[(ds, v)] = upper_req

        assigned = sum(x[(a, ds, v)] for a in AGENTS if (a, ds, v) in x)

        model.Add(assigned + under_buffer[(ds, v)] >= lower_req)
        coverage_constraints += 1
        model.Add(assigned - over_buffer[(ds, v)] <= upper_req)
        coverage_constraints += 1
        model.Add(assigned + missing_to_required[(ds, v)] >= required)
        target_gap_constraints += 1
        model.Add(assigned - excess_to_required[(ds, v)] <= required)
        target_gap_constraints += 1
        if required >= 1:   # 2026-11-18: talep varken kova boş kalmaz (SOFT, çok yüksek ceza)
            bos_kova[(ds, v)] = model.NewBoolVar(f"bos_kova_{ds}_{v}")
            model.Add(assigned + bos_kova[(ds, v)] >= 1)

def _tb_dk_fark(_t1, _t2):
    """İki saat arasındaki DAİRESEL dakika farkı: 23:30 ile 00:00 arası 30'dur."""
    _d = abs(_tb_dk(_t1) - _tb_dk(_t2))
    return min(_d, 1440 - _d)


def _saat_eslestir(_hedef, _aday_map, _tol):
    """(bas, bit) -> aday_map içindeki EN YAKIN vardiya kodu.

    Birebir eşleşme varsa o döner. Yoksa başlangıç ve bitiş sapmalarının
    ikisi de _tol içinde olan adaylar arasından toplam sapması en küçük
    olan seçilir; eşitlikte vardiya kodu alfabetik olarak küçük olan
    (çözüm runlar arası kaymasın diye). Aday yoksa (None, None) döner.
    Dönüş: (vardiya_kodu, toplam_sapma_dk)  - birebirse sapma 0.
    """
    if _hedef in _aday_map:
        return _aday_map[_hedef], 0
    if not _tol:
        return None, None
    _en_v, _en_s = None, None
    for (_ab, _ae), _av in _aday_map.items():
        _d1 = _tb_dk_fark(_ab, _hedef[0])
        _d2 = _tb_dk_fark(_ae, _hedef[1])
        if _d1 > _tol or _d2 > _tol:
            continue
        _s = _d1 + _d2
        if _en_s is None or _s < _en_s or (_s == _en_s and str(_av) < str(_en_v)):
            _en_v, _en_s = _av, _s
    return _en_v, _en_s

            team_base_capacity_debug_rows.append({"week": wk, "shift": v, "start": None, "end": None, "max_needed": max_needed, "status": "team_week_base_yok"})
            continue

        model.Add(sum(selected_team_capacity_terms) >= max_needed)
        team_base_capacity_constraints += 1

        start, end = None, None
        for ds in normal_weekdays:
            if (ds, v) in saat:
                start, end = saat[(ds, v)]
                break
        team_base_capacity_debug_rows.append({"week": wk, "shift": v, "start": start, "end": end, "max_needed": max_needed, "status": "constraint_added"})

    _need = max((sum(coverage_lower.get((ds, v), int(talep.get((ds, v), 0))) for v in _gvs if (ds, v) in talep)

    _terms = [int(team_size_map.get(str(t).strip(), 0)) * team_week_base[(str(t).strip(), wk, b)]
              for t in TAKIMLAR for b in _gvs if (str(t).strip(), wk, b) in team_week_base]

        # sabah 08 / akşam
        _hs = pd.to_datetime(ds).weekday() >= 5
        for _g, _vs in (("sabah_08", [v for v in _gv if str(saat[(ds, v)][0]).strip()[:5] in _ing_sabah08]),
                        ("aksam", [v for v in _gv if _ing_grup(ds, v) == "aksam"])):
            _h = _ing_hedef(ds, _g + "_hafta_sonu" if (_hs and _g == "aksam") else _g)
            if _h > 0:
                ing_hedef_eksik[(ds, _g)] = model.NewIntVar(0, _h, f"ing_eksik_{_g}_{ds}")
                model.Add(sum(_x(_vs)) + ing_hedef_eksik[(ds, _g)] >= _h)
        # gece nöbeti
        _sahip = [q for b, e, q in _tablo if b <= _d <= e]
        ing_gece_sahibi[ds] = _sahip[0] if _sahip else None
        _gece_vs = [v for v in _gv if _ing_grup(ds, v) == "gece"]
        _xg = _x(_gece_vs)
        if not _sahip:
            print(f"!!! {ds} INGILIZCE_GECE_TABLOSU'nda yok - ingilizce gece kuralı o gün kurulmadı.")
        elif _xg:
            _ben = _sahip[0] == QUEUE
            ing_gece_fazla[ds] = model.NewIntVar(0, len(_xg), f"ing_gece_fazla_{ds}")
            model.Add(sum(_xg) - ing_gece_fazla[ds] <= (1 if _ben else 0))
            if _ben:
                ing_gece_eksik[ds] = model.NewIntVar(0, 1, f"ing_gece_eksik_{ds}")
                model.Add(sum(_xg) + ing_gece_eksik[ds] >= 1)
                for a in _ing:
                    _w = int(round(GECE_KISI_W * float(_puan.get(a, 0))))
                    for v in _gece_vs:
                        if (a, ds, v) in x and _w:
                            ing_gece_kisi_ceza.append((_w, x[(a, ds, v)]))
        for a in _ing:
            for v in _gece_vs:
                if (a, ds, v) in x:
                    _k = (a, day_week[ds])
                    if _k not in _gn_hafta:
                        _gn_hafta[_k] = model.NewBoolVar(f"ing_gece_hafta_{a}_{day_week[ds]}")
                    model.Add(_gn_hafta[_k] >= x[(a, ds, v)])
    for a in _ing:                                              # arka arkaya iki gece haftası
        for _w1, _w2 in zip(WEEKS, WEEKS[1:]):
            if (a, _w1) in _gn_hafta and (a, _w2) in _gn_hafta:
                _ard = model.NewBoolVar(f"ing_gece_ard_{a}_{_w2}")
                model.Add(_ard >= _gn_hafta[(a, _w1)] + _gn_hafta[(a, _w2)] - 1)
                ing_gece_ardisik.append(_ard)

    # --- ekip bağı: hafta içi ekibin grubunda, hafta boyu tek vardiya ---
    _tatil_ing = set(globals().get("ozel_tatil_plan_gunleri", set()))
    _ing_muaf = set()                                           # sabah çalışan / hamile / süt -> grup kuralı yok
    for _c in ("sabah_calisir_flg", "hamile_flg", "sut_izni_flg"):
        if _c in df_tam.columns:
            _ing_muaf |= set(df_tam.loc[pd.to_numeric(df_tam[_c], errors="coerce").fillna(0).astype(int) == 1,
                                        "agent_user_code"].astype(str).str.strip())
    _ing_muaf &= ingilizce_agents
    _tkd = globals().get("takim_disi_hafta_v", set())
    _grp_wk = {}                                                # (hafta, v) -> grup
    for ds in PLAN_GUNLER:
        for v in gun_vardiyalari.get(ds, []):
            if (ds, v) in saat:
                _grp_wk.setdefault((day_week[ds], v), _ing_grup(ds, v))
    for a in _ing:
        _t = agent_team.get(a)
        for ds in PLAN_GUNLER:
            if pd.to_datetime(ds).weekday() >= 5 or ds in _tatil_ing:
                continue
            wk = day_week[ds]
            for v in gun_vardiyalari.get(ds, []):
                if (a, ds, v) not in x or v in _ozel_v:
                    continue
                if (a, wk, v) not in ing_week_base:
                    ing_week_base[(a, wk, v)] = model.NewBoolVar(f"ing_wb_{a}_{wk}_{v}")
                model.Add(x[(a, ds, v)] <= ing_week_base[(a, wk, v)])
                _g = _grp_wk.get((wk, v), "diger")
                if _g == "gece" or a in _ing_muaf or (wk, v) in _tkd:
                    continue                                    # gece / takım kilidi dışı vardiya / muaf ingilizceci
                _ayni = [team_week_base[(_t, wk, b)] for b in week_vardiyalari.get(wk, [])
                         if (_t, wk, b) in team_week_base and (wk, b) not in _tkd
                         and (_grp_wk.get((wk, b)) == _g if _g != "diger" else b == v)]
                if (a, wk) not in ing_grup_disi:
                    ing_grup_disi[(a, wk)] = model.NewBoolVar(f"ing_grup_disi_{a}_{wk}")
                model.Add(x[(a, ds, v)] <= sum(_ayni) + ing_grup_disi[(a, wk)])
        for wk in WEEKS:
            _bs = [(v, b) for (a2, w2, v), b in ing_week_base.items()
                   if a2 == a and w2 == wk and _grp_wk.get((wk, v)) != "gece" and (wk, v) not in _tkd
                   and (wk, v) not in globals().get("takim_bolunebilir_hafta_v", set())]   # 2026-11-18 bölünebilir: kayma yok
            if not _bs:
                continue
            model.Add(sum(b for (a2, w2, v), b in ing_week_base.items() if a2 == a and w2 == wk) <= 1)
            ing_takim_disi[(a, wk)] = model.NewBoolVar(f"ing_takim_disi_{a}_{wk}")
            for v, b in _bs:                                    # ekibin tam base'inden kayma (grup içi)
                _tb = team_week_base.get((_t, wk, v))
                model.Add(ing_takim_disi[(a, wk)] >= b - (_tb if _tb is not None else 0))

    # --- 2026-11-18: BAĞLANTI KISITI (yönlendirme) ---
    # Akşam ingilizcecisi <= akşam base'li ekiplerdeki (muaf olmayan) ingilizceci + ekibinden ayrılan. Yukarıdaki grup
    # kuralının toplamı; çözüm kümesini DEĞİŞTİRMEZ. Solver'a "akşam eksiği = ekip base'i" bağını doğrudan gösterir
    # (17 Kasım: gap %92, ingilizcecili ekibi 15:00'e taşıyan takas bulunamıyordu).
    _ing_takim_say = {}
    for a in _ing:
        if a not in _ing_muaf:
            _ing_takim_say[agent_team.get(a)] = _ing_takim_say.get(agent_team.get(a), 0) + 1
    ing_baglanti = 0
    for ds in PLAN_GUNLER:
        if pd.to_datetime(ds).weekday() >= 5 or ds in _tatil_ing:
            continue
        wk = day_week[ds]
        _sol = [x[(a, ds, v)] for a in _ing if a not in _ing_muaf for v in gun_vardiyalari.get(ds, [])
                if (a, ds, v) in x and v not in _ozel_v and _grp_wk.get((wk, v)) == "aksam" and (wk, v) not in _tkd]
        if not _sol:
            continue
        _sag = [_n * team_week_base[(t, wk, b)] for t, _n in _ing_takim_say.items() for b in week_vardiyalari.get(wk, [])
                if (t, wk, b) in team_week_base and (wk, b) not in _tkd and _grp_wk.get((wk, b)) == "aksam"]
        _gd = [ing_grup_disi[(a, wk)] for a in _ing if (a, wk) in ing_grup_disi]
        model.Add(sum(_sol) <= sum(_sag) + sum(_gd))
        ing_baglanti += 1
    print("bağlantı kısıtı (akşam ingilizcecisi <= akşam ekiplerindeki ingilizceci):", ing_baglanti)

print("ENABLE_INGILIZCE:", ENABLE_INGILIZCE, "| kuyruk:", QUEUE, "| ingilizceci:", len(ingilizce_agents),
      "| hedef:", INGILIZCE_HEDEF)
print("grup kuralından muaf ingilizceci (sabah çalışan / hamile / süt):", sorted(globals().get("_ing_muaf", set())))
print("sabah/akşam hedef kısıtı:", len(ing_hedef_eksik), "| gece günü (bu kuyruk nöbetçi):", len(ing_gece_eksik),
      "/", sum(1 for q in ing_gece_sahibi.values() if q), "| ardışık gece haftası izlenen:", len(ing_gece_ardisik))
