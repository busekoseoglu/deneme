# %% [KISIT] - GÜNDÜZ VARDİYA ROTASYONU (SOFT)
# 2026-09-04 EKLENEN. Tek başına çalışır; gereksinimi sadece team_week_base
# ([HÜCRE] - TAKIM HAFTALIK BASE VARDİYA) ve CONFIG.
#
# İş birimi: "Kişiler ay boyu aynı vardiyada kalmasın. Özellikle gündüz
# vardiyaları için. Bir hafta 9 yazıldıysa diğer hafta 10, sonra 8:30 gibi."
#
#   Σ_hafta team_week_base[(takım, hafta, gündüz_vardiyası)] <= MAX + aşım
#   ceza = GUNDUZ_ROTASYON_W × Σ aşım
#
# MAX = 2 ve ayda 5 hafta varsa takım en az 3 farklı gündüz vardiyası görür.
#
# NEDEN SOFT: HARD olsaydı HAFTALIK TAKIM BASE KAPASİTE KORUMASI ile çelişirdi.
# O kural her hafta o vardiyaya yeterli takım seçilmesini ZORUNLU kılıyor. Bir
# gündüz vardiyası her hafta kapasitenin %60'ını istiyorsa ama her takım orada
# en fazla 2/5 hafta kalabiliyorsa ayrılabilecek maksimum kapasite %40'tır ->
# infeasible. Aynı gerekçe TAKIM_VARDIYA_DENGE_MAX_HAFTA için de geçerliydi.
#
# KAPSAM DIŞI olanlar (bilinçli):
#   - 15:00 / 00:00 vardiyaları -> TAKIM_VARDIYA_DENGE kuralında dengeleniyor
#   - sarkan hafta              -> o hafta kişi bazlı devir kuralının kontrolünde
#   - arife özel vardiyaları    -> base olarak seçilemiyorlar zaten
#   - muafiyetli / devir alan agentlar -> takım base'ini takip etmiyorlar,
#     kendi bireysel base'lerinde kalıyorlar (4 Eylül: kullanıcı kabul etti)
#
# EŞLEŞTİRME SAAT ARALIĞINDAN: vardiya kodları haftadan haftaya değişebilir,
# saat aralığı sabit kalır. "09:00-18:00 ayda kaç hafta" diye sayılıyor.

for _gr_p in ("ENABLE_GUNDUZ_ROTASYON", "GUNDUZ_ILK_BASLANGIC_SAATI",
              "GUNDUZ_SON_BASLANGIC_SAATI", "GUNDUZ_AYNI_VARDIYA_MAX_HAFTA",
              "GUNDUZ_ROTASYON_W"):
    if _gr_p not in globals():
        raise SystemExit(f"CONFIG'de tanımlı olmalı: {_gr_p} -> [CONFIG] hücresini çalıştır.")

if "team_week_base" not in globals():
    raise SystemExit(
        "team_week_base yok. Önce [HÜCRE] - TAKIM HAFTALIK BASE VARDİYA hücresini çalıştır."
    )

gunduz_rotasyon_asim = {}      # (takim, "bas-bit") -> IntVar (tavanı kaç hafta aştı)
gunduz_rotasyon_rows = []
gunduz_rotasyon_kisiti = 0


def _gr_dk(_t):
    _h, _m = str(_t).strip()[:5].split(":")
    return int(_h) * 60 + int(_m)


if not ENABLE_GUNDUZ_ROTASYON:
    print("ENABLE_GUNDUZ_ROTASYON = False - gündüz rotasyonu kurulmadı.")
else:
    _gr_alt = _gr_dk(GUNDUZ_ILK_BASLANGIC_SAATI)
    _gr_ust = _gr_dk(GUNDUZ_SON_BASLANGIC_SAATI)
    _gr_arife = globals().get("arife_ozel_vardiya_kodlari", set())
    _gr_sarkan = globals().get("SARKAN_WK")

    # (hafta, vardiya) -> (bas, bit)   [bu hücre tek başına çalışsın diye burada türetilir]
    _gr_wk_saat = {}
    for _ds in PLAN_GUNLER:
        _wk = day_week[_ds]
        for _v in gun_vardiyalari.get(_ds, []):
            if (_wk, _v) not in _gr_wk_saat and (_ds, _v) in saat:
                _b, _e = saat[(_ds, _v)]
                _gr_wk_saat[(_wk, _v)] = (str(_b).strip()[:5], str(_e).strip()[:5])

    # rotasyona giren saat aralıkları
    _gr_gunduz_saat = sorted({
        _st for (_wk, _v), _st in _gr_wk_saat.items()
        if _v not in _gr_arife and _gr_alt <= _gr_dk(_st[0]) <= _gr_ust
    })

    # sayıma giren haftalar (sarkan hafta hariç)
    _gr_haftalar = [_wk for _wk in WEEKS if _wk != _gr_sarkan]

    print(f"Gündüz penceresi: {GUNDUZ_ILK_BASLANGIC_SAATI} - {GUNDUZ_SON_BASLANGIC_SAATI}"
          f"  |  tavan: {GUNDUZ_AYNI_VARDIYA_MAX_HAFTA} hafta  |  ceza: {GUNDUZ_ROTASYON_W}")
    print(f"Rotasyona giren gündüz vardiyası: {len(_gr_gunduz_saat)}"
          f"  ({', '.join('-'.join(_s) for _s in _gr_gunduz_saat) or 'YOK'})")
    print(f"Sayıma giren hafta: {len(_gr_haftalar)} / {len(WEEKS)}"
          f"  (sarkan hafta {_gr_sarkan} hariç)" if _gr_sarkan else
          f"Sayıma giren hafta: {len(_gr_haftalar)} / {len(WEEKS)}")

    if not _gr_gunduz_saat:
        print("!!! UYARI: pencereye giren gündüz vardiyası YOK - kural boş çalışıyor.")
        print("!!! GUNDUZ_ILK/SON_BASLANGIC_SAATI değerlerini kontrol et.")

    for _t in TAKIMLAR:
        _t = str(_t).strip()
        for _st in _gr_gunduz_saat:
            _vars = []
            for _wk in _gr_haftalar:
                for _v in week_vardiyalari.get(_wk, []):
                    if _gr_wk_saat.get((_wk, _v)) != _st:
                        continue
                    if (_t, _wk, _v) in team_week_base:
                        _vars.append(team_week_base[(_t, _wk, _v)])
            # tavan aşılamayacaksa kısıt üretme (o vardiya takıma zaten az açık)
            if len(_vars) <= GUNDUZ_AYNI_VARDIYA_MAX_HAFTA:
                continue
            _ad = "-".join(_st)
            _asim = model.NewIntVar(0, len(_vars) - GUNDUZ_AYNI_VARDIYA_MAX_HAFTA,
                                    f"gunduz_rot_asim_{_t}_{_ad}")
            model.Add(sum(_vars) - GUNDUZ_AYNI_VARDIYA_MAX_HAFTA <= _asim)
            gunduz_rotasyon_asim[(_t, _ad)] = _asim
            gunduz_rotasyon_kisiti += 1
            gunduz_rotasyon_rows.append({
                "takim": _t, "vardiya_saati": _ad,
                "secilebilir_hafta": len(_vars),
                "tavan": GUNDUZ_AYNI_VARDIYA_MAX_HAFTA,
            })

gunduz_rotasyon_df = pd.DataFrame(gunduz_rotasyon_rows)

print("Gündüz rotasyon kısıtı:", gunduz_rotasyon_kisiti)
if not gunduz_rotasyon_df.empty:
    print("Kural kurulan takım:", gunduz_rotasyon_df["takim"].nunique())
    print("Takım başına gündüz vardiyası seçeneği (dağılım):")
    print(gunduz_rotasyon_df.groupby("takim").size().value_counts()
          .rename_axis("vardiya_secenegi").to_frame("takim_sayisi").to_string())
    # Bir takımın seçebileceği gündüz vardiyası sayısı tavandan azsa rotasyon
    # o takım için fiilen imkansız - ceza sürekli ödenir, plan değişmez.
    _gr_dar = (gunduz_rotasyon_df.groupby("takim").size()
               <= GUNDUZ_AYNI_VARDIYA_MAX_HAFTA)
    if _gr_dar.any():
        print(f"\n!!! {int(_gr_dar.sum())} takımın gündüz vardiya seçeneği tavandan az "
              f"-> o takımlar dönemez, ceza boşuna birikir.")
        print("!!! Bu takımlarda ya vardiya arzı dar ya da takım bütünlüğü kuralı "
              "(TAKIM_BASE_MAX_YASAKLI_ORAN) seçenekleri kapatmış olabilir.")
        print(_gr_dar[_gr_dar].head(10).to_string())





# %% [HÜCRE] - OBJECTIVE
# 2026-08-05: off_t talepleri SOFT. Karşılanmayan her off_t günü
# OFF_TALEP_KARSILANMAMA_W ile cezalandırılır ([KISIT] - OFF_T SOFT hücresi).
# izin ve off_t2 HARD kaldığı için objective'de karşılıkları yoktur.

objective_terms = []

for ds in PLAN_GUNLER:
    for v in gun_vardiyalari.get(ds, []):
        objective_terms.append(UNDER_BUFFER_W * under_buffer[(ds, v)])
        objective_terms.append(MISSING_TO_REQUIRED_W * missing_to_required[(ds, v)])

        if gece_aksam_vardiyasi_mi[(ds, v)]:
            excess_weight = EXCESS_GECE_AKSAM_W
        elif mola_destek_vardiyasi_mi[(ds, v)]:
            excess_weight = EXCESS_MOLA_DESTEK_W
        else:
            excess_weight = EXCESS_NORMAL_W

        objective_terms.append(excess_weight * excess_to_required[(ds, v)])
        objective_terms.append(OVER_BUFFER_W * over_buffer[(ds, v)])

# İkinci mesai ekstra cezası
for a in AGENTS:
    objective_terms.append(IKINCI_MESAI_W * ikinci_mesai_aylik[a])

# Hafta sonu çalışma adil dağıtım cezası
for a in hafta_sonu_calisma_sayisi.keys():
    objective_terms.append(HAFTA_SONU_FAZLA_CALISMA_W * fazla_hafta_sonu_calisma[a])

# Arife / resmi tatil mesai ve ihlal cezaları
if "arife_mesai" in globals() and ARIFE_MESAI_W != 0:
    for (a, ds, v), var in arife_mesai.items():
        objective_terms.append(ARIFE_MESAI_W * var)

if "resmi_tatil_mesai" in globals() and RESMI_TATIL_MESAI_W != 0:
    for (a, ds, v), var in resmi_tatil_mesai.items():
        objective_terms.append(RESMI_TATIL_MESAI_W * var)

if "resmi_tatil_kisitli_ihlal" in globals():
    for (a, ds, v), var in resmi_tatil_kisitli_ihlal.items():
        objective_terms.append(RESMI_TATIL_KISITLI_IHLAL_W * var)

print("ARIFE_MESAI_W:", ARIFE_MESAI_W)
print("RESMI_TATIL_MESAI_W:", RESMI_TATIL_MESAI_W)
print("RESMI_TATIL_KISITLI_IHLAL_W:", RESMI_TATIL_KISITLI_IHLAL_W)

# Haftalık hedef sapma cezası - ÖLÜ BLOK (2026-08-10 tespit)
# weekly_under / weekly_over değişkenlerini üreten hiçbir hücre YOK, hiçbir
# sürümde de olmamış. Bu iki if hiçbir zaman True olmuyor, WEEKLY_UNDER_W ve
# WEEKLY_OVER_W objective'e hiç girmiyor. Ağırlıkları değiştirmek bir şey
# değiştirmez. Aynı işi [KISIT] - HAFTALIK OFF TAVANI yapıyor.
if "weekly_under" in globals():
    for (a, wk), var in weekly_under.items():
        objective_terms.append(WEEKLY_UNDER_W * var)
if "weekly_over" in globals():
    for (a, wk), var in weekly_over.items():
        objective_terms.append(WEEKLY_OVER_W * var)

print("WEEKLY UNDER W:", WEEKLY_UNDER_W)
print("WEEKLY OVER W:", WEEKLY_OVER_W)

# Günlük fazla atama yığılma cezası
if "daily_excess_over" in globals():
    for ds, var in daily_excess_over.items():
        objective_terms.append(DAILY_EXCESS_OVER_W * var)
print("DAILY EXCESS SOFT LIMIT:", DAILY_EXCESS_SOFT_LIMIT)
print("DAILY_EXCESS_OVER_W:", DAILY_EXCESS_OVER_W)

# Partial end week tamamen OFF kalma cezası
if "partial_end_week_all_off" in globals():
    for key, var in partial_end_week_all_off.items():
        objective_terms.append(PARTIAL_END_WEEK_ALL_OFF_W * var)
print("PARTIAL END WEEK ALL OFF W:", PARTIAL_END_WEEK_ALL_OFF_W)

# off_t soft ceza - değişkenler varsa
if "off_t_ihlal" in globals():
    for (a, ds), var in off_t_ihlal.items():
        objective_terms.append(OFF_TALEP_KARSILANMAMA_W * var)
    print("off_t soft ceza terimi:", len(off_t_ihlal))
else:
    print("off_t_ihlal yok -> off_t cezası objective'e girmedi")
print("OFF_TALEP_KARSILANMAMA_W:", OFF_TALEP_KARSILANMAMA_W)

# haftalık off tavanı aşımı - sadece SOFT modda değişken üretilir
if "model_off_tavan_asim" in globals() and model_off_tavan_asim:
    for (a, wk), var in model_off_tavan_asim.items():
        objective_terms.append(MODEL_OFF_TAVAN_ASIM_W * var)
    print("haftalık off tavanı soft ceza terimi:", len(model_off_tavan_asim))

# ===== 2026-08-12 EKLENEN - TAKIM VARDIYA GECMIS DENGESI =====
# Değişkenler [KISIT] - TAKIM VARDİYA GEÇMİŞ DENGESİ (SOFT) hücresinde üretilir.
# ENABLE_TAKIM_VARDIYA_DENGE = False iken sözlükler BOŞ kalır, buraya hiçbir
# terim girmez - bu blok o durumda sessizce atlanır.
if "takim_vardiya_denge_sapma" in globals() and takim_vardiya_denge_sapma:
    for (_t, _b), (_arti, _eksi) in takim_vardiya_denge_sapma.items():
        objective_terms.append(TAKIM_VARDIYA_DENGE_W * _arti)
        objective_terms.append(TAKIM_VARDIYA_DENGE_W * _eksi)
    print("takım vardiya denge sapma terimi:", 2 * len(takim_vardiya_denge_sapma))
    print("TAKIM_VARDIYA_DENGE_W:", TAKIM_VARDIYA_DENGE_W)

if "takim_vardiya_denge_asim" in globals() and takim_vardiya_denge_asim:
    for (_t, _b), _var in takim_vardiya_denge_asim.items():
        objective_terms.append(TAKIM_VARDIYA_DENGE_MAX_HAFTA_ASIM_W * _var)
    print("takım vardiya aylık tavan aşımı terimi:", len(takim_vardiya_denge_asim))
    print("TAKIM_VARDIYA_DENGE_MAX_HAFTA:", TAKIM_VARDIYA_DENGE_MAX_HAFTA,
          "| aşım W:", TAKIM_VARDIYA_DENGE_MAX_HAFTA_ASIM_W)
# =============================================================

# ===== 2026-08-28 EKLENEN - AY GECISI SAPMA CEZALARI =====
# Ay geçişi kuralları SOFT: sapma cezalandırılır ama yasaklanmaz.
# Ağırlıklar coverage'ın ALTINDA -> model önce talebi karşılar, sonra sürekliliği.
if "sarkan_base_sapma" in globals() and sarkan_base_sapma:
    for _t, _var in sarkan_base_sapma.items():
        objective_terms.append(SARKAN_BASE_SAPMA_W * _var)
    print("ay geçişi - takım süreklilik sapma terimi:", len(sarkan_base_sapma))

if "sarkan_agent_sapma" in globals() and sarkan_agent_sapma:
    for _a, _var in sarkan_agent_sapma.items():
        objective_terms.append(SARKAN_KISI_SAPMA_W * _var)
    print("ay geçişi - muaf agent süreklilik sapma terimi:", len(sarkan_agent_sapma))

if "sarkan_kisi_sapma" in globals() and sarkan_kisi_sapma:
    for _a, _var in sarkan_kisi_sapma.items():
        objective_terms.append(SARKAN_KISI_SAPMA_W * _var)
    print("ay geçişi - kişi bazlı devir sapma terimi:", len(sarkan_kisi_sapma))
    print("SARKAN_BASE_SAPMA_W:", SARKAN_BASE_SAPMA_W,
          "| SARKAN_KISI_SAPMA_W:", SARKAN_KISI_SAPMA_W)
# =========================================================

# ===== 2026-09-04 EKLENEN - GUNDUZ VARDIYA ROTASYONU =====
# Değişkenler [KISIT] - GÜNDÜZ VARDİYA ROTASYONU (SOFT) hücresinde üretilir.
# ENABLE_GUNDUZ_ROTASYON = False iken sözlük BOŞ kalır, buraya terim girmez.
if "gunduz_rotasyon_asim" in globals() and gunduz_rotasyon_asim:
    for (_t, _sa), _var in gunduz_rotasyon_asim.items():
        objective_terms.append(GUNDUZ_ROTASYON_W * _var)
    print("gündüz rotasyon aşım terimi:", len(gunduz_rotasyon_asim))
    print("GUNDUZ_ROTASYON_W:", GUNDUZ_ROTASYON_W,
          "| tavan:", GUNDUZ_AYNI_VARDIYA_MAX_HAFTA, "hafta")
# =========================================================

model.Minimize(sum(objective_terms))

print("Objective term sayısı:", len(objective_terms))
print("UNDER_BUFFER_W:", UNDER_BUFFER_W)
print("MISSING_TO_REQUIRED_W:", MISSING_TO_REQUIRED_W)
print("OVERTIME_W:", OVERTIME_W)
print("EXCESS_GECE_AKSAM_W:", EXCESS_GECE_AKSAM_W)
print("EXCESS_NORMAL_W:", EXCESS_NORMAL_W)
print("EXCESS_MOLA_DESTEK_W:", EXCESS_MOLA_DESTEK_W)
print("OVER_BUFFER_W:", OVER_BUFFER_W)

# hücresi (cozum_calisma / cozum_atama). Diğer df'ler yoksa o sheet atlanır, hücre patlamaz.
#
# NORM / TELAFİ / MESAİ tanımı:
#
# 1) HAFTALIK ADIM - hangi hafta sonu günleri "fazla"?
#      hafta_kota    = o haftadaki hafta içi (Pzt-Cum) günlerden İZİN ve RESMİ TATİL
#                      OLMAYANLARIN sayısı. Aydan sarkan haftalarda ÖNCEKİ AY'ın
#                      hafta içi günleri de sayılır.
#                      SABİT 5 DEĞİL: izin bir iş gününün yerine geçer, o hafta
#                      agentin normal çalışması 5 değil 4'tür.
#      hafta_calisma = o ISO haftada çalışılan gün sayısı (RESMİ TATİL HARİÇ)
#      fazla         = max(0, hafta_calisma - hafta_kota)
#      o haftanın çalışılan hafta sonu günleri tarih sırasına dizilir;
#      son `fazla` tanesi ADAY, kalanı NORM çalışma.
#      Aydan sarkan haftalarda önceki ayda çalışılan günler de sayılır.
#
# 2) AYLIK ADIM - adaylardan hangileri gerçekten ÜCRETLİ mesai?
#      aylik_fazla = max(0, sayilan_calisma - baz_gereken_calisma)   (resmi tatil hariç)
#      adaylar kronolojik sırada; son `aylik_fazla` tanesi -> MESAİ
#                                  kalanı                  -> TELAFİ
#
# Neden: agent bir hafta 3 gün off alıp (normali 2) 4 gün çalışırsa 1 gün borçlanır;
# başka bir hafta 6 gün çalışıp borcu kapatır. Aylık toplam yine baz'dır, yani
# ÜCRETLİ MESAİ YOKTUR - o gün TELAFİ'dir. Ama agent borcunu kapatıp ÜSTÜNE de
# fazladan çalışırsa aylik_fazla > 0 olur ve o günler MESAİ olarak kalır.
#
# Sonuç: MESAİ gün sayısı = AGENT_OZET sayfasındaki kota_farki ile BİREBİR aynıdır.

    ("MESAİ + vardiya",   "mesai",       "Hafta sonu çalışma, AYLIK kotayı aşan (ücretli mesai)"),
    ("TELAFİ + vardiya",  "telafi",      "Hafta sonu çalışma, başka haftadaki fazla off'un telafisi (ücretli mesai DEĞİL)"),
    ("kırmızı hücre",     "off_t_ihlal", "off_t talebi vardı ama KARŞILANMADI - agent o gün çalışıyor"),
    ("dolgusuz",          None,          "Normal hafta içi çalışma"),
    ("--- ÖNCEKİ AY (geçmiş, plan dışı, salt okunur) ---", None, ""),
    ("saat aralığı",      "g_calisma",   "Önceki ayda çalışılmış gün"),
    ("İZİN",              "g_izin",      "Önceki ayda yıllık/idari izin"),
    ("H.İZİN",            "g_h_izin",    "Önceki ayda haftalık izin günü (süt izni düzeni), off DEĞİL"),
    ("OFF",               "g_off",       "Önceki ayda off"),
    ("boş",               None,          "Önceki ayda o güne ait kayıt yok"),
]

if "cozum_calisma" not in globals() or "cozum_atama" not in globals():

_saat = globals().get("saat", {})
_tip_map = globals().get("tip_map", {})
_gecmis_gunleri_x = globals().get("gecmis_calisma_gunleri", {})

_g_off       = globals().get("gecmis_off_gunleri", {})
_g_izin      = globals().get("gecmis_izin_gunleri", {})

_g_kapsam    = globals().get("gecmis_kapsam", {})

_plan_dates = sorted(pd.to_datetime(ds).date() for ds in PLAN_GUNLER)
_PLAN_BAS = min(_plan_dates)
_PLAN_BIT = max(_plan_dates)
_plan_date_set_x = set(_plan_dates)

_resmi_tatil_set = set()
if "resmi_tatil_plan_gunleri" in globals():
    _resmi_tatil_set = {pd.to_datetime(d).date() for d in resmi_tatil_plan_gunleri}
elif "RESMI_TATIL_GUNLERI" in globals():
    _resmi_tatil_set = {pd.to_datetime(d).date() for d in RESMI_TATIL_GUNLERI}

# --- ARİFE: resmi tatille aynı kefede, aylık iş günü sayılmaz ---
_arife_set = set()
if "arife_plan_gunleri" in globals():
    _arife_set = {pd.to_datetime(d).date() for d in arife_plan_gunleri}
elif globals().get("ENABLE_ARIFE_KURALI", False) and "ARIFE_GUNLERI" in globals():
    _arife_set = {
        pd.to_datetime(ds).date() for ds in PLAN_GUNLER
        if pd.to_datetime(ds).strftime("%Y-%m-%d") in ARIFE_GUNLERI
    }

# aylık ve haftalık hesaptan çıkarılacak günler (resmi tatil + arife)
_sayilmayan_set = _resmi_tatil_set | _arife_set

# arifede 09:00-13:00 çalışan grup: hamile / süt izni / mesaiye kalamaz
_tatil_kisitli_x = globals().get("tatil_kisitli_agents")
if _tatil_kisitli_x is None:
    _mk1 = pd.to_numeric(df_tam.get("hamile_flg", 0), errors="coerce").fillna(0).astype(int)
    _mk2 = pd.to_numeric(df_tam.get("sut_izni_flg", 0), errors="coerce").fillna(0).astype(int)
    _mk3 = pd.to_numeric(df_tam.get("mesaiye_kalamaz_flg", 0), errors="coerce").fillna(0).astype(int)
    _tatil_kisitli_x = set(
        df_tam.loc[(_mk1 == 1) | (_mk2 == 1) | (_mk3 == 1), "agent_user_code"]
        .astype(str).str.strip()
    )
_tatil_kisitli_x = {str(a).strip() for a in _tatil_kisitli_x}

_GUN_ADI = ["Pzt", "Sal", "Çar", "Per", "Cum", "Cmt", "Paz"]

# 2026-08-30: geçmiş kolon başlıkları ay adını tarihten alır - "Tem" sabiti kalktı.
_AY_KISA = ["Oca", "Şub", "Mar", "Nis", "May", "Haz",
            "Tem", "Ağu", "Eyl", "Eki", "Kas", "Ara"]


def _vardiya_metni(ds, v):
    """Vardiya kodunu okunur saat aralığına çevirir."""
    if v is None:
        return ""
    st = _saat.get((ds, v))
    if st:
        return f"{str(st[0]).strip()[:5]}-{str(st[1]).strip()[:5]}"
    return str(v)


# ---------------------------------- AYLIK KOTA (baz_gereken_calisma) - agent
_baz_map_x = {}
if "aylik_calisma_debug_df" in globals() and not aylik_calisma_debug_df.empty:
    _baz_map_x = dict(zip(
        aylik_calisma_debug_df["agent_user_code"].astype(str).str.strip(),
        aylik_calisma_debug_df["baz_gereken_calisma"].astype(int),
    ))
    _izin_map_x = dict(zip(
        aylik_calisma_debug_df["agent_user_code"].astype(str).str.strip(),
        aylik_calisma_debug_df["izin_is_gunu_sayisi"].astype(int),
    ))
    _aylik_is_gunu_x = int(aylik_calisma_debug_df["aylik_is_gunu"].iloc[0])
else:
    _is_gunu_d = {
        pd.to_datetime(ds).date() for ds in PLAN_GUNLER
        if pd.to_datetime(ds).weekday() in (0, 1, 2, 3, 4)
        and pd.to_datetime(ds).date() not in _sayilmayan_set
    }
    _aylik_is_gunu_x = len(_is_gunu_d)
    _baz_map_x, _izin_map_x = {}, {}
    for a_raw in AGENTS:
        a = str(a_raw).strip()
        _izn = {d for d, t in _tip_map.get(a, {}).items()
                if str(t).strip().lower() == "izin" and d in _is_gunu_d}
        _izin_map_x[a] = len(_izn)
        _baz_map_x[a] = _aylik_is_gunu_x - len(_izn)

# --------------------------- HAFTALIK NORM / TELAFİ / MESAİ HESABI (agent)
_hafta_gunleri = defaultdict(list)
for ds in PLAN_GUNLER:
    d = pd.to_datetime(ds).date()
    _pzt = d - _td(days=d.weekday())
    _hafta_gunleri[_pzt].append(ds)
for _k in _hafta_gunleri:
    _hafta_gunleri[_k] = sorted(_hafta_gunleri[_k], key=lambda s: pd.to_datetime(s))

mesai_gunleri_map = {}   # agent -> {ds}  ücretli mesai
telafi_gunleri_map = {}  # agent -> {ds}  borç kapatma
norm_gunleri_map = {}    # agent -> {ds}  kota içi hafta sonu
hafta_calisma_debug_rows = []

for a_raw in AGENTS:
    a = str(a_raw).strip()
    _gecmis_a = _gecmis_gunleri_x.get(a, set())
    _aday_a, _norm_a = [], set()   # _aday_a: haftalık kotayı aşan hafta sonu günleri (kronolojik)

    for _pzt, _wk_gunler in sorted(_hafta_gunleri.items()):
        # RESMİ TATİL günleri haftalık hesaba GİRMEZ: ayrı kategori (RESMİ TATİL MESAİ),
        # aylık toplama da (sayilan_calisma) girmiyor. Haftalık ve aylık adım aynı
        # tabanı kullanmazsa resmi tatil günü aday listesinde yer tutup MESAİ slotunu
        # çalar ve gerçek mesai günü yanlışlıkla TELAFİ olarak etiketlenir.
        _calisilan = [
            ds for ds in _wk_gunler
            if cozum_calisma.get((a, ds), 0) == 1
            and pd.to_datetime(ds).date() not in _sayilmayan_set
        ]

        # aydan sarkan hafta: plan öncesi (önceki ay) çalışılan günleri de say
        _hafta_araligi = {_pzt + _td(days=k) for k in range(7)}
        _gecmis_ek = len({d for d in _gecmis_a if d in _hafta_araligi and d < _PLAN_BAS})

        _hafta_calisma = len(_calisilan) + _gecmis_ek

        # --- haftalık kota: izin ve resmi tatil günleri DÜŞÜLÜR ---
        _kota = 0
        for _k2 in range(5):                       # Pzt..Cum
            _d2 = _pzt + _td(days=_k2)
            if _d2 < _PLAN_BAS:
                # aydan sarkan hafta: önceki ayın hafta içi günleri
                if _d2 not in _g_izin.get(a, set()):
                    _kota += 1
            elif _d2 in _plan_date_set_x:
                if _d2 in _sayilmayan_set:
                    continue
                if str(_tip_map.get(a, {}).get(_d2, "")).strip().lower() == "izin":
                    continue
                _kota += 1
        _kota = min(_NORMAL_WORK_DAYS, _kota)

        _fazla = max(0, _hafta_calisma - _kota)

        _hs_calisilan = [ds for ds in _calisilan if pd.to_datetime(ds).weekday() in (5, 6)]
        if _fazla > 0 and _hs_calisilan:
            _aday_bu_hafta = list(_hs_calisilan[-_fazla:])
        else:
            _aday_bu_hafta = []
        _norm_bu_hafta = set(_hs_calisilan) - set(_aday_bu_hafta)

        _aday_a.extend(_aday_bu_hafta)
        _norm_a |= _norm_bu_hafta

        hafta_calisma_debug_rows.append({
            "agent_user_code": a,
            "hafta_pzt": str(_pzt),
            "plan_ici_calisma": len(_calisilan),
            "oncekiaydan_gelen_calisma": _gecmis_ek,
            "hafta_toplam_calisma": _hafta_calisma,
            "kota": _kota,
            "fazla": _fazla,
            "hafta_sonu_calisilan": len(_hs_calisilan),
            "norm": len(_norm_bu_hafta),
            "kota_asan_aday": len(_aday_bu_hafta),
        })

    # ---- AYLIK ADIM: adaylardan kaçı gerçekten ÜCRETLİ mesai? ----
    _aday_a = sorted(set(_aday_a), key=lambda d: pd.to_datetime(d))
    _sayilan_a = sum(
        1 for ds in PLAN_GUNLER
        if cozum_calisma.get((a, ds), 0) == 1
        and pd.to_datetime(ds).date() not in _sayilmayan_set
    )
    _aylik_fazla_a = max(0, _sayilan_a - int(_baz_map_x.get(a, 0)))
    _mesai_sayisi = min(_aylik_fazla_a, len(_aday_a))

    mesai_gunleri_map[a] = set(_aday_a[len(_aday_a) - _mesai_sayisi:]) if _mesai_sayisi else set()
    telafi_gunleri_map[a] = set(_aday_a) - mesai_gunleri_map[a]
    norm_gunleri_map[a] = _norm_a

hafta_calisma_debug_df = pd.DataFrame(hafta_calisma_debug_rows)

# --------------------------------------------------- TAKVİM TABLOSU KURULUMU
# Takım bölünmesini filtrelemek için ilgili bayraklar kimlik kolonlarına dahil.
_kimlik_kaynak = ["agent_user_code", "agent_name",
                  "working_main_group", "takim", "lokasyon",
                  "sabah_calisir_flg", "hamile_flg", "sut_izni_flg",
                  "mesaiye_kalamaz_flg", "kismi_calisan_flg"]
_kimlik_df_kolonlari = [c for c in _kimlik_kaynak if c in df_tam.columns]
_kimlik_df = df_tam[_kimlik_df_kolonlari].copy()
_kimlik_df["agent_user_code"] = _kimlik_df["agent_user_code"].astype(str).str.strip()
_kimlik_lookup = _kimlik_df.set_index("agent_user_code").to_dict("index")

# --- kıdem: df_tam'da değil, kidem_ay_map'ten gelir ---
_kd_map_e = globals().get("kidem_ay_map", {})
_kd_min_e = globals().get("KIDEM_MIN_AY")
_kd_vars_e = float(globals().get("KIDEM_EKSIK_VARSAYILAN_AY", 0))
_kidem_kolonlari = ["kidem_ay", "kidem_kaydi_var", "kidemsiz"] if _kd_map_e else []

# --- takım base muafiyeti: agent kaç haftada kullanmış ---
_muaf_hafta_sayaci = {}
_muaf_map_e = globals().get("takim_muaf", {})
if _muaf_map_e and "solver" in globals():
    try:
        for (_ma, _mwk), _mvar in _muaf_map_e.items():
            if solver.Value(_mvar) == 1:
                _muaf_hafta_sayaci[_ma] = _muaf_hafta_sayaci.get(_ma, 0) + 1
    except Exception as _e:
        print("takim_muaf okunamadı:", _e)
_muaf_kolonlari = ["takim_muaf_hafta"] if _muaf_map_e else []

_kimlik_kolonlari = _kimlik_df_kolonlari + _kidem_kolonlari + _muaf_kolonlari

_ozet_kolonlari = ["toplam_calisma", "hafta_ici_calisma", "norm_hafta_sonu",
                   "telafi_hafta_sonu", "mesai_hafta_sonu", "resmi_tatil_calisma",
                   "arife_calisma", "arife_mesai", "izin_gun",
                   "off_gun", "off_t_talep", "off_t_karsilanmayan", "off_t2_talep",
                   "gecmis_calisma_gun"]

# --- GEÇMİŞ (önceki ay) kolonları: plan başlangıcından ÖNCEKİ günler, salt okunur ---
# 2026-08-30: ne ay adı ne gün sayısı sabit. Günler geçmiş verisinin kendisinden
# (gecmis_kapsam) gelir, en fazla GECMIS_KOLON_MAX_GUN tanesi gösterilir.
# Veri yoksa plan başlangıcından geriye sayılır (eski davranış).
_gecmis_tarihleri = sorted({
    d for _s in _g_kapsam.values() for d in _s if d < _PLAN_BAS
})[-GECMIS_KOLON_MAX_GUN:]
if not _gecmis_tarihleri:
    _gecmis_tarihleri = [_PLAN_BAS - _td(days=k)
                         for k in range(GECMIS_KOLON_MAX_GUN, 0, -1)]
    print(f"! Geçmiş kaydı yok - TAKVIM'e {GECMIS_KOLON_MAX_GUN} boş geçmiş kolonu kondu.")
_gecmis_kolonlari = []
_gecmis_kolon_date = {}
for d in _gecmis_tarihleri:
    _kol = f"{d.day} {_AY_KISA[d.month - 1]} {_GUN_ADI[d.weekday()]}"
    _gecmis_kolonlari.append(_kol)
    _gecmis_kolon_date[_kol] = d
print(f"TAKVIM geçmiş kolonu: {len(_gecmis_kolonlari)} gün "
      f"({_gecmis_tarihleri[0]} .. {_gecmis_tarihleri[-1]})")

# --- PLAN AYI kolonları ---
_gun_kolonlari = []
_gun_kolon_ds = {}
for ds in PLAN_GUNLER:
    d = pd.to_datetime(ds).date()
    _kol = f"{d.day} {_GUN_ADI[d.weekday()]}"
    _gun_kolonlari.append(_kol)
    _gun_kolon_ds[_kol] = ds

takvim_rows = []
hucre_renk = {}   # (satir_no_0bazli, kolon_adi) -> RENK anahtari

for _i, a_raw in enumerate(AGENTS):
    a = str(a_raw).strip()
    _k = _kimlik_lookup.get(a, {})
    row = {c: _k.get(c, "") for c in _kimlik_df_kolonlari}
    row["agent_user_code"] = a
    if _kidem_kolonlari:
        _kv = _kd_map_e.get(a)
        _kv_var = not (_kv is None or pd.isna(_kv))
        _kv = _kd_vars_e if not _kv_var else float(_kv)
        row["kidem_ay"] = _kv
        row["kidem_kaydi_var"] = _kv_var
        row["kidemsiz"] = bool(_kd_min_e is not None and _kv < float(_kd_min_e))
    if _muaf_kolonlari:
        row["takim_muaf_hafta"] = int(_muaf_hafta_sayaci.get(a, 0))

    _tip_dates = _tip_map.get(a, {})
    _mesai_a = mesai_gunleri_map.get(a, set())
    _telafi_a = telafi_gunleri_map.get(a, set())
    _norm_a = norm_gunleri_map.get(a, set())

    _n_calisma = _n_hafta_ici = _n_norm = _n_mesai = _n_telafi = _n_rt = 0
    _n_arife = _n_arife_mesai = 0
    _n_izin = _n_off = _n_off_t = _n_off_t_ihlal = _n_off_t2 = 0
    _n_gecmis_calisma = 0

    # ---- GEÇMİŞ (önceki ay, plan dışı) ----
    _gc_a = _g_calisma.get(a, {})
    for _kol in _gecmis_kolonlari:
        d = _gecmis_kolon_date[_kol]
        if d in _gc_a:
            _bas, _bit = _gc_a[d]
            row[_kol] = f"{str(_bas).strip()[:5]}-{str(_bit).strip()[:5]}"
            hucre_renk[(_i, _kol)] = "g_calisma"
            _n_gecmis_calisma += 1
        elif d in _g_izin.get(a, set()):
            row[_kol] = "İZİN"
            hucre_renk[(_i, _kol)] = "g_izin"
        elif d in _g_h_izin.get(a, set()):
            # süt izni düzeni: off DEĞİL, ayrı kategori
            row[_kol] = "H.İZİN"
            hucre_renk[(_i, _kol)] = "g_h_izin"
        elif d in _g_off.get(a, set()):
            row[_kol] = "OFF"
            hucre_renk[(_i, _kol)] = "g_off"
        else:
            row[_kol] = ""   # kayıt yok

    # ---- AĞUSTOS (plan) ----
    for _kol in _gun_kolonlari:
        ds = _gun_kolon_ds[_kol]
        d = pd.to_datetime(ds).date()
        tip = str(_tip_dates.get(d, "")).strip().lower()
        calisiyor = cozum_calisma.get((a, ds), 0) == 1
        vardiya = _vardiya_metni(ds, cozum_atama.get((a, ds))) if calisiyor else ""

        if tip == "izin":
            _n_izin += 1
            row[_kol] = "İZİN"
            hucre_renk[(_i, _kol)] = "izin"
            continue

        if tip == "off_t":
            _n_off_t += 1
        elif tip == "off_t2":
            _n_off_t2 += 1

        if not calisiyor:
            if tip == "off_t":
                row[_kol] = "OFF_T"
                hucre_renk[(_i, _kol)] = "off_t"
            elif tip == "off_t2":
                row[_kol] = "OFF_T2"
                hucre_renk[(_i, _kol)] = "off_t2"
            else:
                _n_off += 1
                row[_kol] = "OFF"
                hucre_renk[(_i, _kol)] = "off"
            continue

        # --- buradan sonrası: agent o gün ÇALIŞIYOR ---
        _n_calisma += 1
        if d in _resmi_tatil_set:
            _n_rt += 1
            row[_kol] = f"RESMİ TATİL MESAİ\n{vardiya}"
            _renk = "resmi_tatil"
        elif d in _arife_set:
            # tatil kısıtlı agent arifede 09:00-13:00 çalışır -> normal, mesai değil
            if a in _tatil_kisitli_x:
                _n_arife += 1
                row[_kol] = f"ARİFE\n{vardiya}"
                _renk = "arife"
            else:
                _n_arife_mesai += 1
                row[_kol] = f"ARİFE MESAİ\n{vardiya}"
                _renk = "arife_mesai"
        elif ds in _mesai_a:
            _n_mesai += 1
            row[_kol] = f"MESAİ\n{vardiya}"
            _renk = "mesai"
        elif ds in _telafi_a:
            _n_telafi += 1
            row[_kol] = f"TELAFİ\n{vardiya}"
            _renk = "telafi"
        elif ds in _norm_a:
            _n_norm += 1
            row[_kol] = f"NORM\n{vardiya}"
            _renk = "norm"
        else:
            _n_hafta_ici += 1
            row[_kol] = vardiya
            _renk = None

        # off_t talebi vardı ama karşılanmadı -> rengi ezer
        if tip == "off_t":
            _n_off_t_ihlal += 1
            _renk = "off_t_ihlal"

        if _renk:
            hucre_renk[(_i, _kol)] = _renk

    row["toplam_calisma"] = _n_calisma
    row["hafta_ici_calisma"] = _n_hafta_ici
    row["norm_hafta_sonu"] = _n_norm
    row["mesai_hafta_sonu"] = _n_mesai
    row["telafi_hafta_sonu"] = _n_telafi
    row["resmi_tatil_calisma"] = _n_rt
    row["arife_calisma"] = _n_arife
    row["arife_mesai"] = _n_arife_mesai
    row["izin_gun"] = _n_izin
    row["off_gun"] = _n_off
    row["off_t_talep"] = _n_off_t
    row["off_t_karsilanmayan"] = _n_off_t_ihlal
    row["off_t2_talep"] = _n_off_t2
    row["gecmis_calisma_gun"] = _n_gecmis_calisma

    takvim_rows.append(row)

takvim_df = pd.DataFrame(
    takvim_rows,
    columns=_kimlik_kolonlari + _ozet_kolonlari + _gecmis_kolonlari + _gun_kolonlari,
)

print("  TELAFİ hafta sonu    :", int(takvim_df["telafi_hafta_sonu"].sum()))
print("  MESAİ hafta sonu     :", int(takvim_df["mesai_hafta_sonu"].sum()))
print("  resmi tatil çalışma  :", int(takvim_df["resmi_tatil_calisma"].sum()))

_atanan_sayac = {}
for (a, ds), v in cozum_atama.items():
    if cozum_calisma.get((a, ds), 0) == 1:
        _atanan_sayac[(ds, v)] = _atanan_sayac.get((ds, v), 0) + 1

_cov_lower = globals().get("coverage_lower", {})
_cov_upper = globals().get("coverage_upper", {})
_gece_mi = globals().get("gece_aksam_vardiyasi_mi", {})
_ust_limit = globals().get("fazla_atama_ust_limiti", {})
_day_week_x = globals().get("day_week", {})

_cov_rows = []
for ds in PLAN_GUNLER:
    d = pd.to_datetime(ds).date()
    for v in gun_vardiyalari.get(ds, []):
        _req = int(talep[(ds, v)])
        _asg = int(_atanan_sayac.get((ds, v), 0))
        _st = _saat.get((ds, v), ("", ""))
        _cov_rows.append({
            "tarih": d.strftime("%Y-%m-%d"),
            "gun": _GUN_ADI[d.weekday()],
            "hafta": _day_week_x.get(ds, ""),
            "vardiya": v,
            "baslangic": str(_st[0]).strip()[:5],
            "bitis": str(_st[1]).strip()[:5],
            "gece_aksam_mi": bool(_gece_mi.get((ds, v), False)),
            "talep": _req,
            "atanan": _asg,
            "fark": _asg - _req,
            # 2026-08-30: fark tek kolonda eksik ve fazlayı karıştırıyordu.
            # Ayrıştırılmış hâli COVERAGE_GUNLUK toplamlarıyla birebir tutar.
            "eksik": max(0, _req - _asg),
            "fazla": max(0, _asg - _req),
            "karsilanma_yuzde": round(100 * _asg / _req, 1) if _req else None,
            "buffer_alt": _cov_lower.get((ds, v)),
            "buffer_ust": _cov_upper.get((ds, v)),
            "buffer_altinda": (_asg < _cov_lower[(ds, v)]) if (ds, v) in _cov_lower else None,
            "buffer_ustunde": (_asg > _cov_upper[(ds, v)]) if (ds, v) in _cov_upper else None,
            "fazla_atama_ust_limiti": _ust_limit.get((ds, v)),
        })

# --------------------------------------------------- COVERAGE - GÜNLÜK TOPLAM
# 2026-08-30 EKLENEN. COVERAGE vardiya (saat) kırılımında; burada aynı veri gün
# bazında toplanır. eksik_toplam = Σ max(0, talep - atanan): aynı günün başka bir
# vardiyasındaki FAZLA atama, eksiği maskelemez - ikisi ayrı kolonda durur.
_cg_rows = []
for ds in PLAN_GUNLER:

    _alt = coverage_df[coverage_df["tarih"] == d.strftime("%Y-%m-%d")]
    if _alt.empty:
        continue
    _talep_g = int(_alt["talep"].sum())
    _atanan_g = int(_alt["atanan"].sum())
    _eksik_g = int(_alt["eksik"].sum())
    _fazla_g = int(_alt["fazla"].sum())
    _kotu = _alt.sort_values("eksik", ascending=False).iloc[0]
    _cg_rows.append({
        "tarih": d.strftime("%Y-%m-%d"),
        "gun": _GUN_ADI[d.weekday()],
        "hafta": _day_week_x.get(ds, ""),
        "hafta_sonu": int(d.weekday() > 4),
        "vardiya_sayisi": len(_alt),
        "toplam_talep": _talep_g,
        "toplam_atanan": _atanan_g,
        "eksik_toplam": _eksik_g,
        "fazla_toplam": _fazla_g,
        "net_fark": _atanan_g - _talep_g,
        "eksik_kova": int((_alt["eksik"] > 0).sum()),
        "fazla_kova": int((_alt["fazla"] > 0).sum()),
        # talebi olup HİÇ kimse atanmayan vardiya - coverage tamamen SOFT olduğu
        # için mümkün; sıfırdan farklıysa o gün mutlaka incelenmeli.
        "bos_kova": int(((_alt["talep"] > 0) & (_alt["atanan"] == 0)).sum()),
        "karsilanma_yuzde": (round(100 * (_talep_g - _eksik_g) / _talep_g, 1)
                             if _talep_g else None),
        "en_kotu_vardiya": (f"{_kotu['baslangic']}-{_kotu['bitis']}"
                            if int(_kotu["eksik"]) > 0 else ""),
        "en_kotu_eksik": int(_kotu["eksik"]),
