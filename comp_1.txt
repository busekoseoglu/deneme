def _dk(t):
    s = str(t).strip()[:5]; h, m = s.split(":"); return int(h) * 60 + int(m)

def _shift_dt(ds, bas, bit):
    s = pd.to_datetime(f"{pd.to_datetime(ds).strftime('%Y-%m-%d')} {bas}")
    e = pd.to_datetime(f"{pd.to_datetime(ds).strftime('%Y-%m-%d')} {bit}")
    if e <= s: e += pd.Timedelta(days=1)
    return s, e

def kaydet(kural, adet, tur="HARD"):

# %% [KONTROL 3] - COVERAGE (buffer + talep) [RAPOR/SOFT]
rows = []

for ds in PLAN_GUNLER:
    for v in gun_vardiyalari.get(ds, []):
        assigned = sum(V(x[(str(a).strip(), ds, v)]) for a in AGENTS if (str(a).strip(), ds, v) in x)

        rows.append({"tarih": str(ds), "vardiya": v, "talep": req, "lower": lo, "upper": hi, "atanan": int(assigned), "durum": durum})

# %% [KONTROL 6] - MAX 6 GÜN ÜST ÜSTE (TEMMUZ GEÇMİŞİ DAHİL)
from datetime import timedelta
ihlal = []
for a_raw in AGENTS:
    a = str(a_raw).strip()
    worked = set(_gecmis_gunleri.get(a, set()))  # Temmuz gerçek çalışma
    for ds in PLAN_GUNLER:
        if cozum_calisma[(a, ds)] == 1:
            worked.add(pd.to_datetime(ds).date())
    if not worked:
        continue
    tum = sorted(worked)
    lo, hi = tum[0], tum[-1]
    d = lo
    while d <= hi - timedelta(days=WINDOW_DAYS - 1):
        pencere = [d + timedelta(days=k) for k in range(WINDOW_DAYS)]
        c = sum(1 for x_ in pencere if x_ in worked)
        if c > MAX_CONSECUTIVE_WORK_DAYS:
            ihlal.append((a, str(d), "pencere_calisma", c))
        d += timedelta(days=1)
kaydet("Max 6 gün üst üste (Temmuz dahil)", len(ihlal))

# %% [KONTROL 7] - MİN 11 SAAT DİNLENME (TEMMUZ GEÇMİŞİ DAHİL)

        if rest < MIN_REST_MINUTES:

# Gerçek OFF: izin DEĞİL; (off_t/off_t2) veya (work=0). work=1 -> OFF değil.
def _gercek_off(a, ds):
    d = pd.to_datetime(ds).date()
    tip = str(tip_map.get(a, {}).get(d, "")).strip().lower()
    if tip == "izin": return False
    if tip in ("off_t", "off_t2"): return True
    return cozum_calisma.get((a, ds), 0) == 0

    tum_izinli = len(PLAN_GUNLER) > 0 and all(pd.to_datetime(ds).date() in plan_izin for ds in PLAN_GUNLER)
    if tum_izinli:
