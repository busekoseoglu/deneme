# %% [TEŞHİS] - BİR (GÜN, VARDİYA) KOVASI NEDEN EKSİK KALDI
# 2026-08-31 EKLENEN. Tek başına çalışır. Gereksinim: [KONTROL] - ÇÖZÜMÜ ÇIKAR
# (cozum_calisma / cozum_atama) + solver. MODELE DOKUNMAZ, hiçbir kısıt eklemez.
#
# Soru: "1 Eylül V06 - talep 126, atanan 106. Eksik 20 kişi nerede?"
#
# Her agent için "bu vardiyada neden değilsin" sorusu TEK bir sebebe indirgenir.
# Kurallar burada YENİDEN türetilir (hücreler arası bağımlılık yok). Bir agent
# birden çok sebeple engelli olabilir; SIRA önemlidir, ilk yakalayan yazılır.
#
# Sebepler İKİ SINIFA ayrılır - ayrım bu hücrenin bütün amacı:
#
#   [SABİT]  Veriden / geçmişten gelir, model DEĞİŞTİREMEZ.
#            izin, off_t2, hamile-süt, sabah penceresi, kıdem, 11 saat (önceki ay),
#            max-6 (önceki ay). Bunlar her runda AYNI kalır.
#
#   [SEÇİM]  Model başka türlü de seçebilirdi. Takım base kilidi, kişi devri,
#            11 saat (plan içi), karşılanmış off_t, o gün off verilmesi.
#            Bunlar RUNDAN RUNA DEĞİŞİR.
#
# Okuma kuralı:
#   teorik havuz (= toplam - SABİT engelli)  <  talep  -> YAPISAL açık. Kural
#       gevşetilmeden kapanmaz; ağırlıkla oynamak işe yaramaz. Her run tekrarlar.
#   teorik havuz >= talep                              -> DAĞITIM / ARAMA sorunu.
#       Kişi vardı, model başka yere koydu ya da 300 saniyede bulamadı.
#       Runlar arasında oynaması BEKLENİR.

# ---------------------------------------------------------- TEŞHİS GİRDİSİ
# Model parametresi DEĞİL - sadece hangi kovaya bakılacağı. CONFIG'e girmez.
# İkisi de None ise en çok eksiği olan kova otomatik seçilir.
TESHIS_GUN = None          # örn "2026-09-01"
TESHIS_VARDIYA = None      # örn "V06"

# Bu hücre TEK BAŞINA çalışır - başka hücrenin import'una güvenmez.
from datetime import timedelta as _ts_td

# ------------------------------------------------------- ÖN KOŞUL KONTROLÜ
if "cozum_calisma" not in globals() or "cozum_atama" not in globals():
    raise SystemExit("cozum_calisma / cozum_atama yok. Önce [KONTROL] - ÇÖZÜMÜ ÇIKAR'ı çalıştır.")

_ts_saat = globals().get("saat", {})
_ts_tip = globals().get("tip_map", {})


def _ts_dk(t):
    _s = str(t).strip()[:5]
    _h, _m = _s.split(":")
    return int(_h) * 60 + int(_m)


def _ts_vt(ds, v):
    _st = _ts_saat.get((ds, v))
    return f"{str(_st[0]).strip()[:5]}-{str(_st[1]).strip()[:5]}" if _st else str(v)


def _ts_dt(ds, bas, bit):
    _s = pd.to_datetime(f"{pd.to_datetime(ds).strftime('%Y-%m-%d')} {str(bas).strip()[:5]}")
    _e = pd.to_datetime(f"{pd.to_datetime(ds).strftime('%Y-%m-%d')} {str(bit).strip()[:5]}")
    if _e <= _s:
        _e += pd.Timedelta(days=1)
    return _s, _e


# ---------------------------------------------- 1) HEDEF KOVA + SOLVER DURUMU
_ts_atanan = {}
for (_a0, _ds0), _v0 in cozum_atama.items():
    if cozum_calisma.get((_a0, _ds0), 0) == 1:
        _ts_atanan[(_ds0, _v0)] = _ts_atanan.get((_ds0, _v0), 0) + 1

if TESHIS_GUN is None or TESHIS_VARDIYA is None:
    _en_kotu, _en_eksik = None, 0
    for _ds0 in PLAN_GUNLER:
        for _v0 in gun_vardiyalari.get(_ds0, []):
            _e0 = int(talep[(_ds0, _v0)]) - int(_ts_atanan.get((_ds0, _v0), 0))
            if _e0 > _en_eksik:
                _en_kotu, _en_eksik = (_ds0, _v0), _e0
    if _en_kotu is None:
        raise SystemExit("Hiçbir kovada eksik yok - teşhise gerek kalmadı.")
    TESHIS_GUN, TESHIS_VARDIYA = _en_kotu
    print(f"Hedef kova OTOMATİK seçildi (en büyük eksik): {TESHIS_GUN} / {TESHIS_VARDIYA}")

DS_T, V_T = TESHIS_GUN, TESHIS_VARDIYA
if (DS_T, V_T) not in talep:
    raise SystemExit(f"({DS_T}, {V_T}) plan içinde yok. gun_vardiyalari'na bak.")

_D_T = pd.to_datetime(DS_T).date()
_TALEP = int(talep[(DS_T, V_T)])
_ATANAN = int(_ts_atanan.get((DS_T, V_T), 0))
_EKSIK = _TALEP - _ATANAN
_BAS_T, _BIT_T = _ts_saat.get((DS_T, V_T), ("", ""))
_START_T, _END_T = _ts_dt(DS_T, _BAS_T, _BIT_T)
_HAFTA_ICI_T = _D_T.weekday() in (0, 1, 2, 3, 4)

print("=" * 78)
print(f"KOVA: {DS_T} ({['Pzt','Sal','Çar','Per','Cum','Cmt','Paz'][_D_T.weekday()]}) "
      f"| {V_T} | {_ts_vt(DS_T, V_T)}")
print(f"talep {_TALEP} | atanan {_ATANAN} | EKSİK {_EKSIK}")

_alt_t = globals().get("coverage_lower", {}).get((DS_T, V_T))
if _alt_t is not None:
    _ub = max(0, _alt_t - _ATANAN) * int(globals().get("UNDER_BUFFER_W", 0))
    _mr = max(0, _TALEP - _ATANAN) * int(globals().get("MISSING_TO_REQUIRED_W", 0))
    print(f"bu kovanın objective maliyeti: under_buffer {_ub:,} + missing {_mr:,} "
          f"= {_ub + _mr:,}".replace(",", "."))

# --- solver durumu: yorumu tamamen değiştirir ---
print("-" * 78)
if "solver" in globals() and "status" in globals():
    try:
        _sn = solver.StatusName(status)
        print(f"Solver durumu: {_sn} | süre: {solver.WallTime():.1f} sn")
        if _sn != "OPTIMAL":
            _ob, _bb = solver.ObjectiveValue(), solver.BestObjectiveBound()
            print(f"  objective {_ob:,.0f} | alt sınır {_bb:,.0f} | "
                  f"kanıtlanmamış boşluk {_ob - _bb:,.0f}")
            print("  !!! OPTIMAL DEĞİL. Bu eksik, HARD bir blok olmayabilir - model")
            print("  !!! süre dolduğu için daha iyisini BULAMAMIŞ da olabilir.")
            print("  !!! 8 worker + süre limiti + sabit seed yok => HER RUN FARKLI PLAN.")
            print("  !!! Aynı kovanın bir sonraki runda dolması normaldir.")
    except Exception as _e:
        print("solver durumu okunamadı:", _e)
else:
    print("solver / status yok - durum bilgisi atlandı.")
print("=" * 78)

# ------------------------------------------------- 2) KURALLARI YENİDEN TÜRET
_ts_flag = {}
for _c in ["sabah_calisir_flg", "hamile_flg", "sut_izni_flg", "mesaiye_kalamaz_flg"]:
    if _c in df_tam.columns:
        _ts_flag[_c] = dict(zip(
            df_tam["agent_user_code"].astype(str).str.strip(),
            pd.to_numeric(df_tam[_c], errors="coerce").fillna(0).astype(int),
        ))


def _bayrak(a, c):
    return _ts_flag.get(c, {}).get(a, 0) == 1


# sabah çalışan penceresi
_S_BIT = globals().get("SABAH_CALISIR_MAX_BITIS_SAATI")
_S_BAS = globals().get("SABAH_CALISIR_MAX_BASLANGIC_SAATI")
_S_GY = globals().get("SABAH_CALISIR_GECE_VARDIYA_YASAK", False)
_S_GSET = globals().get("GECE_AKSAM_VARDIYA_SET", set())
_sabah_yasak_t = False
if _S_BIT is not None and _BAS_T:
    _b, _e = str(_BAS_T).strip()[:5], str(_BIT_T).strip()[:5]
    _bd, _ed = _ts_dk(_b), _ts_dk(_e)
    if _ed < _bd:
        _ed += 24 * 60
    _sabah_yasak_t = (
        (_S_GY and (_b, _e) in _S_GSET)
        or (_S_BAS is not None and _bd > _ts_dk(_S_BAS))
        or _ed > _ts_dk(_S_BIT)
    )

# kıdemsizlik
_kd_map = globals().get("kidem_ay_map", {})
_kd_min = globals().get("KIDEM_MIN_AY")
_kd_var = float(globals().get("KIDEM_EKSIK_VARSAYILAN_AY", 0))
_kd_bas = {str(_b2).strip()[:5] for _b2 in globals().get("KIDEM_YASAK_VARDIYA_BASLANGIC", ())}
_kidem_yasak_t = str(_BAS_T).strip()[:5] in _kd_bas if _kd_bas else False


def _kidemsiz(a):
    if _kd_min is None:
        return False
    _v2 = _kd_map.get(a)
    _v2 = _kd_var if (_v2 is None or pd.isna(_v2)) else float(_v2)
    return _v2 < float(_kd_min)


# 11 saat - ÖNCEKİ AY (geçmiş sabit, gerçek HARD blok)
_g11 = globals().get("gecmis_11h_bloklu_agent", {})
_g11_set = {a for a, bl in _g11.items() if (DS_T, V_T) in bl}

# max-6 - limit=0 penceresi (geçmişten gelen zorunlu off)
_m6_set = set()
for _r6 in globals().get("max6_sifir_limit", []):
    try:
        _a6, _ws6 = str(_r6[0]).strip(), pd.to_datetime(_r6[1]).date()
        _wd6 = int(globals().get("WINDOW_DAYS", 7))
        if _ws6 <= _D_T <= _ws6 + _ts_td(days=_wd6 - 1):
            _m6_set.add(_a6)
    except Exception:
        pass

# takım base kilidi
_dw_t = globals().get("day_week", {})
_WK_T = _dw_t.get(DS_T)
if _WK_T is None:
    _iso = pd.to_datetime(DS_T).isocalendar()
    _WK_T = f"{int(_iso.year)}-W{int(_iso.week):02d}"
_at_t = globals().get("agent_team") or dict(zip(
    df_tam["agent_user_code"].astype(str).str.strip(),
    df_tam["takim"].astype(str).str.strip()))
_secilen_base = {}
if "team_week_base" in globals() and "solver" in globals():
    try:
        for (_t3, _wk3, _v3), _var3 in team_week_base.items():
            if solver.Value(_var3) == 1:
                _secilen_base[(_t3, _wk3)] = _v3
    except Exception as _e:
        print("team_week_base okunamadı:", _e)
_muaf_aktif = set()
if globals().get("takim_muaf") and "solver" in globals():
    try:
        _muaf_aktif = {k for k, var in takim_muaf.items() if solver.Value(var) == 1}
    except Exception:
        pass
_devir_t = dict(globals().get("SARKAN_DEVIR_AGENT", {}) or {})
_SWK_T = globals().get("SARKAN_WK")

# 11 saat - PLAN İÇİ (çözümdeki önceki gün ataması)
_MINREST = int(globals().get("MIN_REST_MINUTES", 660))
_onceki_ds = None
for _ds4 in PLAN_GUNLER:
    if pd.to_datetime(_ds4).date() == _D_T - _ts_td(days=1):
        _onceki_ds = _ds4
        break

_ozel_tatil_t = set(globals().get("ozel_tatil_plan_gunleri", []))

# ------------------------------------------------------- 3) AGENT ELEME HUNİSİ
SABIT_SEBEPLER = {
    "izin (HARD)", "off_t2 talebi (HARD)", "hamile/süt - hafta sonu (HARD)",
    "mesaiye kalamaz - tatil/arife günü (HARD)", "sabah çalışan - vardiya penceresi dışı (HARD)",
    "kıdemsiz - gece vardiyası (HARD)", "11 saat - ÖNCEKİ AY vardiyası (HARD)",
    "max-6 - önceki aydan gelen zorunlu off (HARD)",
}

_ts_rows = []
for _a_raw in AGENTS:
    a = str(_a_raw).strip()
    _tip = str(_ts_tip.get(a, {}).get(_D_T, "")).strip().lower()
    _t_a = _at_t.get(a)
    _t_a = None if (_t_a is None or (isinstance(_t_a, float) and pd.isna(_t_a))) else str(_t_a).strip()
    _base_a = _secilen_base.get((_t_a, _WK_T)) if _t_a else None

    if _tip == "izin":
        _s = "izin (HARD)"
    elif _tip == "off_t2":
        _s = "off_t2 talebi (HARD)"
    elif (not _HAFTA_ICI_T) and (_bayrak(a, "hamile_flg") or _bayrak(a, "sut_izni_flg")):
        _s = "hamile/süt - hafta sonu (HARD)"
    elif _bayrak(a, "mesaiye_kalamaz_flg") and DS_T in _ozel_tatil_t:
        _s = "mesaiye kalamaz - tatil/arife günü (HARD)"
    elif _sabah_yasak_t and _bayrak(a, "sabah_calisir_flg"):
        _s = "sabah çalışan - vardiya penceresi dışı (HARD)"
    elif _kidem_yasak_t and _kidemsiz(a):
        _s = "kıdemsiz - gece vardiyası (HARD)"
    elif a in _g11_set:
        _s = "11 saat - ÖNCEKİ AY vardiyası (HARD)"
    elif a in _m6_set:
        _s = "max-6 - önceki aydan gelen zorunlu off (HARD)"
    elif _HAFTA_ICI_T and _SWK_T == _WK_T and a in _devir_t and _devir_t[a] != V_T:
        _s = f"kişi devri - {_ts_vt(DS_T, _devir_t[a])} devraldı [SEÇİM]"
    elif (_HAFTA_ICI_T and _t_a is not None and _base_a is not None
          and _base_a != V_T and (a, _WK_T) not in _muaf_aktif):
        _s = f"takım base kilidi - takım base'i {_ts_vt(DS_T, _base_a)} [SEÇİM]"
    else:
        # 11 saat - plan içi (çözümdeki önceki gün atamasına BAĞLI)
        _11_plan = False
        if _onceki_ds is not None and cozum_calisma.get((a, _onceki_ds), 0) == 1:
            _vp = cozum_atama.get((a, _onceki_ds))
            _stp = _ts_saat.get((_onceki_ds, _vp)) if _vp else None
            if _stp:
                _, _endp = _ts_dt(_onceki_ds, _stp[0], _stp[1])
                if _START_T < _endp + pd.Timedelta(minutes=_MINREST):
                    _11_plan = True
        if _11_plan:
            _s = "11 saat - önceki gün PLAN İÇİ vardiyası [SEÇİM]"
        elif cozum_atama.get((a, DS_T)) == V_T and cozum_calisma.get((a, DS_T), 0) == 1:
            _s = ">>> BU VARDİYADA ÇALIŞIYOR"
        elif cozum_calisma.get((a, DS_T), 0) == 1:
            _s = f"o gün BAŞKA vardiyada - {_ts_vt(DS_T, cozum_atama.get((a, DS_T)))} [SEÇİM]"
        elif _tip == "off_t":
            _s = "off_t talebi karşılandı [SEÇİM]"
        else:
            _s = "o gün OFF verilmiş [SEÇİM]"

    _ts_rows.append({
        "agent_user_code": a, "takim": _t_a,
        "takim_base": _ts_vt(DS_T, _base_a) if _base_a else "",
        "tip": _tip, "sebep": _s,
        "sinif": ("SABİT" if _s in SABIT_SEBEPLER
                  else ("ÇALIŞIYOR" if _s.startswith(">>>") else "SEÇİM")),
    })

teshis_kova_df = pd.DataFrame(_ts_rows)

# ------------------------------------------------------------------- 4) ÇIKTI
print(f"\n--- {len(AGENTS)} agentın tamamı, tek sebebe indirgenmiş ---")
_ozet_t = (teshis_kova_df.groupby(["sinif", "sebep"]).size()
           .reset_index(name="agent").sort_values(["sinif", "agent"], ascending=[True, False]))
print(_ozet_t.to_string(index=False))

_sabit = int((teshis_kova_df["sinif"] == "SABİT").sum())
_havuz = len(AGENTS) - _sabit
print("\n" + "=" * 78)
print(f"toplam agent                              : {len(AGENTS)}")
print(f"[SABİT] engelli (model değiştiremez)      : {_sabit}")
print(f"TEORİK HAVUZ (bu vardiyaya girebilecek)   : {_havuz}")
print(f"talep                                     : {_TALEP}")
print("-" * 78)
if _havuz < _TALEP:
    print(f"SONUÇ: YAPISAL AÇIK. Havuz talebin {_TALEP - _havuz} kişi ALTINDA.")
    print("       Ağırlık değiştirmek işe yaramaz - bu eksik HER RUN tekrarlar.")
    print("       Kural gevşetmek ya da o vardiyanın talebini düşürmek gerekir.")
else:
    print(f"SONUÇ: YAPISAL AÇIK YOK. Havuz talebi {_havuz - _TALEP} kişi AŞIYOR.")
    print("       Eksik, HARD bir bloktan değil - [SEÇİM] satırlarından geliyor.")
    print("       Bu eksik RUNDAN RUNA DEĞİŞİR; başka bir runda kapanabilir.")
    _sec = (teshis_kova_df[teshis_kova_df["sinif"] == "SEÇİM"]
            .groupby("sebep").size().sort_values(ascending=False))
    if len(_sec):
        print("\n       En çok kişi hangi SEÇİM'de kayboldu:")
        for _sb, _n in _sec.head(5).items():
            print(f"         {_n:>4}  {_sb}")
print("=" * 78)
print("Detay: teshis_kova_df  (agent bazında tek satır, sebep + sınıf)")
