# %% [KISIT] - OFF_T SOFT: KARŞILANMAYAN OFF TALEBİ
# off_t artık HARD değil. Model istenen günü off verebilir, veremezse ceza öder.
# izin ve off_t2 HARD kalır -> [HÜCRE 9] o günlerde zaten hiç x yaratmaz,
# dolayısıyla burada da ihlal değişkenleri oluşmaz.
#
# Ayrı bir ihlal değişkeni YARATILMAZ: off_t gününde ihlal, o günün
# work[(a, ds)] değişkeninin kendisidir (work=1 -> off talebi karşılanmadı).
# Bu yüzden bu hücre modele hiç kısıt eklemez, sadece objective'in kullanacağı
# sözlüğü kurar.

off_t_ihlal = {}
off_t_talep_gunleri = {}
off_t_kapali_gun_sayisi = 0

for a_raw in AGENTS:
    a = str(a_raw).strip()
    tip_dates = tip_map.get(a, {})
    off_t_gunleri = {d for d, tip in tip_dates.items() if str(tip).strip().lower() == "off_t"}
    if not off_t_gunleri:
        continue
    for ds in PLAN_GUNLER:
        ds_date = pd.to_datetime(ds).date()

# ===== 2026-09-18 EKLENEN - OFF TALEP ONCELIGI =====
# Karşılanmayan off_t cezası agent bazlı ağırlıklanır: kıdemine oranla az off
# talep etmiş agentin talebini reddetmek daha pahalı. Oran [HÜCRE 3E]'den gelir.
# Sıralama YALNIZCA bu ay off_t talebi olan agentlar arasında yapılır; band
# tamamen yarışan kişilere yayılsın diye. Modele kısıt eklenmez, sadece sözlük.
off_t_agent_ceza_w = {}
if globals().get("ENABLE_OFF_TALEP_ONCELIGI", False) and off_t_talep_gunleri:
    _oran_map = globals().get("off_kidem_ratio_map", {})
    _orta = (OFF_ONCELIK_MIN_CARPAN + OFF_ONCELIK_MAX_CARPAN) / 2
    _oran = pd.Series({a: _oran_map.get(a) for a in off_t_talep_gunleri},
                      dtype="float64").dropna()
    _carpan = pd.Series(_orta, index=list(off_t_talep_gunleri), dtype="float64")
    if len(_oran) > 1:
        # düşük oran (az talep etmiş) -> MAX çarpan. Eşit oran eşit çarpan alır.
        _p = _oran.rank(method="average", pct=True)
        _p = ((_p - _p.min()) / (_p.max() - _p.min())
              if _p.max() > _p.min() else _p * 0 + 0.5)
        _carpan.loc[_oran.index] = (
            OFF_ONCELIK_MAX_CARPAN - _p * (OFF_ONCELIK_MAX_CARPAN - OFF_ONCELIK_MIN_CARPAN))
    off_t_agent_ceza_w = {a: int(round(OFF_TALEP_KARSILANMAMA_W * c))
                          for a, c in _carpan.items()}
    print(f"off önceliği: oran kaydı olan {len(_oran)}/{len(off_t_talep_gunleri)} agent"
          f" | ceza {min(off_t_agent_ceza_w.values())} .. {max(off_t_agent_ceza_w.values())}")
else:
    print("off önceliği KAPALI -> tüm off_t talepleri eşit ağırlıkta")

# %% [HÜCRE] - OBJECTIVE
# 2026-08-05: off_t talepleri SOFT. Karşılanmayan her off_t günü

objective_terms = []

for ds in PLAN_GUNLER:
    for v in gun_vardiyalari.get(ds, []):
        objective_terms.append(UNDER_BUFFER_W * under_buffer[(ds, v)])
        objective_terms.append(MISSING_TO_REQUIRED_W * missing_to_required[(ds, v)])

        if gece_aksam_vardiyasi_mi[(ds, v)]:
            excess_weight = EXCESS_GECE_AKSAM_W
        elif mola_destek_vardiyasi_mi[(ds, v)]:
            excess_weight = EXCESS_MOLA_DESTEK_W
        else:
            excess_weight = EXCESS_NORMAL_W

        objective_terms.append(excess_weight * excess_to_required[(ds, v)])
        objective_terms.append(OVER_BUFFER_W * over_buffer[(ds, v)])

# İkinci mesai ekstra cezası
for a in AGENTS:
    objective_terms.append(IKINCI_MESAI_W * ikinci_mesai_aylik[a])

# Hafta sonu çalışma adil dağıtım cezası
for a in hafta_sonu_calisma_sayisi.keys():

# Arife / resmi tatil mesai ve ihlal cezaları
if "arife_mesai" in globals() and ARIFE_MESAI_W != 0:
    for (a, ds, v), var in arife_mesai.items():
        objective_terms.append(ARIFE_MESAI_W * var)

if "resmi_tatil_mesai" in globals() and RESMI_TATIL_MESAI_W != 0:
    for (a, ds, v), var in resmi_tatil_mesai.items():
        objective_terms.append(RESMI_TATIL_MESAI_W * var)

if "resmi_tatil_kisitli_ihlal" in globals():
    for (a, ds, v), var in resmi_tatil_kisitli_ihlal.items():
        objective_terms.append(RESMI_TATIL_KISITLI_IHLAL_W * var)

print("ARIFE_MESAI_W:", ARIFE_MESAI_W)
print("RESMI_TATIL_MESAI_W:", RESMI_TATIL_MESAI_W)
print("RESMI_TATIL_KISITLI_IHLAL_W:", RESMI_TATIL_KISITLI_IHLAL_W)
