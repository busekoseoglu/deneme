from collections import defaultdict
proto = model.Proto()
s0 = {i for i in range(len(proto.variables))
      if list(proto.variables[i].domain)[:2] == [0, 0]}
for k in range(len(proto.constraints)):
    c = proto.constraints[k]
    if len(c.enforcement_literal) == 0 and len(c.linear.vars) == 1 \
       and list(c.linear.domain)[:2] == [0, 0]:
        s0.add(c.linear.vars[0])
idx = {proto.variables[i].name: i for i in range(len(proto.variables))}

hafta = defaultdict(list)
for ds in PLAN_GUNLER:

                   for v in gun_vardiyalari.get(ds, []) if (a, ds, v) in x):
                n += 1
        acik[(a, wk)] = n
