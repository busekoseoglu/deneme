SELECT EXTRACT(YEAR FROM h.data_date) AS yil,
       COUNT(DISTINCT h.party_id)     AS musteri_sayisi
FROM   dm.cnf_customer_hst h
WHERE  h.data_date IN (DATE '2023-12-31', DATE '2024-12-31',
                       DATE '2025-12-31', DATE '2026-10-31')   -- 2026 için son ay sonu
  AND  h.mvt_status_id = 1                                       -- aktif müşteri
  AND  h.current_status_id IN (1, 2, 5)
  AND  h.customer_type_id  IN (1, 4, 47)                         -- bireysel
  AND  h.portfolio_sbu_code IN (400, 500)                        -- istediğin SBU'lar
GROUP BY EXTRACT(YEAR FROM h.data_date)
ORDER BY yil;
