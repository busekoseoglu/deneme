# %% [HÜCRE] - BUFFERLI COVERAGE + TALEBE YAKLAŞMA
coverage_lower = {}
coverage_upper = {}
coverage_constraints = 0
bos_kova = {}            # 2026-10-20: (ds, v) -> 1 ise talebi olan kova boş (ceza OBJECTIVE'de)
if "BOS_KOVA_W" not in globals():
    raise SystemExit("CONFIG'de tanımlı olmalı: BOS_KOVA_W -> [CONFIG] hücresini çalıştır.")
target_gap_constraints = 0

for ds in PLAN_GUNLER:
    for v in gun_vardiyalari.get(ds, []):
        if (ds, v) in globals().get("prov_vardiyalar", set()):
            continue   # 2026-10-19: provizyon coverage dışı
        required = int(talep[(ds, v)])
        lower_req = math.floor(required * (1 - BUFFER_RATE))
        upper_req = math.ceil(required * (1 + BUFFER_RATE))
        coverage_lower[(ds, v)] = lower_req
        coverage_upper[(ds, v)] = upper_req

        assigned = sum(x[(a, ds, v)] for a in AGENTS if (a, ds, v) in x)

        model.Add(assigned + under_buffer[(ds, v)] >= lower_req)
        coverage_constraints += 1
        model.Add(assigned - over_buffer[(ds, v)] <= upper_req)
        coverage_constraints += 1
        model.Add(assigned + missing_to_required[(ds, v)] >= required)
        target_gap_constraints += 1
        model.Add(assigned - excess_to_required[(ds, v)] <= required)
        target_gap_constraints += 1
        if required >= 1:
            bos_kova[(ds, v)] = model.NewBoolVar(f"bos_kova_{ds}_{v}")
            model.Add(assigned + bos_kova[(ds, v)] >= 1)

print("Coverage buffer kısıtı:", coverage_constraints)

def _tb_dk_fark(_t1, _t2):
    """İki saat arasındaki DAİRESEL dakika farkı: 23:30 ile 00:00 arası 30'dur."""
    _d = abs(_tb_dk(_t1) - _tb_dk(_t2))
    return min(_d, 1440 - _d)


def _saat_eslestir(_hedef, _aday_map, _tol):
    """(bas, bit) -> aday_map içindeki EN YAKIN vardiya kodu.

    Birebir eşleşme varsa o döner. Yoksa başlangıç ve bitiş sapmalarının
    ikisi de _tol içinde olan adaylar arasından toplam sapması en küçük
    olan seçilir; eşitlikte vardiya kodu alfabetik olarak küçük olan
    (çözüm runlar arası kaymasın diye). Aday yoksa (None, None) döner.
    Dönüş: (vardiya_kodu, toplam_sapma_dk)  - birebirse sapma 0.
    """
    if _hedef in _aday_map:
        return _aday_map[_hedef], 0
    if not _tol:
        return None, None
    _en_v, _en_s = None, None
    for (_ab, _ae), _av in _aday_map.items():
        _d1 = _tb_dk_fark(_ab, _hedef[0])
        _d2 = _tb_dk_fark(_ae, _hedef[1])
        if _d1 > _tol or _d2 > _tol:
            continue
        _s = _d1 + _d2
        if _en_s is None or _s < _en_s or (_s == _en_s and str(_av) < str(_en_v)):
            _en_v, _en_s = _av, _s
    return _en_v, _en_s

# ===== 2026-08-31 - DEVIR AGENTI DA HAFTA BOYU TEK VARDIYADA =====
# 28 Agustos'taki hata: devir agenti icin takim kilidi kaldirildi ve yerine
# `x[(a,ds,v)] <= sarkan_kisi_sapma[a]` konuldu. Sapma degiskeni AGENT BASINA
# TEK oldugu icin bir kez 1 olunca kisit `x <= 1`e donusuyor, yani sarkan hafta
# boyunca agent icin HICBIR kisit kalmiyordu - her gun baska vardiyada
# calisabiliyor ve ceza sadece BIR KEZ (20.000) odeniyordu. 31 Agustos'ta
# gercek planda goruldu: 1 Eylul 10:30, 2 Eylul 09:30, 3 Eylul 11:00.
#
# Duzeltme, muaf agent mekanizmasinin (2026-08-19) aynisi:
#   Sigma_v devir_week_base[(a, v)] == 1        -> HARD, hafta boyu TEK vardiya
#   x[(a, ds, v)] <= devir_week_base[(a, v)]    -> hafta ici gunlerde
#   devir_week_base[(a, devraldigi)] + sapma >= 1 -> SOFT, tercihen devraldigi
#
# "Hafta boyu tek vardiya" HARD, "hangi vardiya" SOFT. Ay gecisi kurallarinin
# HARD olmama ilkesi korunur: model mecbur kalirsa sapar, ama TEK bir baska
# vardiyaya sapar - her gun baskasina degil.
devir_week_base = {}          # (agent, vardiya) -> BoolVar
_devir_base_kurulan = set()   # bireysel haftalik base kurulan devir agentlari
_devir_uygun_yok = []

if SARKAN_DEVIR_AGENT and SARKAN_WK is not None:
    _dwb_uygun_havuz = [
        v for v in week_vardiyalari.get(SARKAN_WK, [])
        if v not in arife_ozel_vardiya_kodlari
    ]
    for _a7, _dv7 in sorted(SARKAN_DEVIR_AGENT.items()):
        _uygun7 = [v for v in _dwb_uygun_havuz if not _tb_yasak_mi(_a7, SARKAN_WK, v)]
        if not _uygun7 or _dv7 not in _uygun7:
            # havuz bos ya da devraldigi vardiya havuzda yok -> Sigma==1 kisiti
            # agenti calisamaz hale getirebilir. Infeasible olmasin diye bu agent
            # devir kapsamindan CIKARILIR, normal takim kilidine tabi olur.
            _devir_uygun_yok.append({
                "agent_user_code": _a7, "devralinan_vardiya": _dv7,
                "sebep": ("hafta içinde uygun vardiya yok" if not _uygun7
                          else "devraldığı vardiya agente yasak"),
            })
            continue
        _dwb_vars = []
        for v in _uygun7:
            _b7 = model.NewBoolVar(f"devir_week_base_{_a7}_{v}")
            devir_week_base[(_a7, v)] = _b7
            _dwb_vars.append(_b7)
        model.Add(sum(_dwb_vars) == 1)
        if SARKAN_DEVIR_HARD:
            # 2026-09-04: devraldığı vardiyada kalmak ZORUNLU. Sapma değişkeni
            # üretilmez -> objective'e devir cezası hiç girmez.
            model.Add(devir_week_base[(_a7, _dv7)] == 1)
        else:
            # devraldığı vardiyada kalması TERCİH; kalamazsa sapma=1, ceza objective'de
            if _a7 not in sarkan_kisi_sapma:
                sarkan_kisi_sapma[_a7] = model.NewBoolVar(f"sarkan_kisi_sapma_{_a7}")
            model.Add(devir_week_base[(_a7, _dv7)] + sarkan_kisi_sapma[_a7] >= 1)
        _devir_base_kurulan.add(_a7)

devir_week_base_uygun_yok_df = pd.DataFrame(_devir_uygun_yok)

# GÜVENLİK: bir takım-haftada TÜM base seçenekleri yasaklanacak olursa

if "TAKIM_BASE_MAX_YASAKLI_ORAN" not in globals():

                    _dwb = devir_week_base.get((a, v))

                elif not TAKIM_BASE_HAFTAICI_HARD:

                elif (t, wk, v) in team_week_base:
                    if (a, wk) in takim_muaf:

                        # 2026-08-19: gevşeme artık SERBEST BIRAKMIYOR - agent

# 2026-09-29: hafta sonu coverage cezasi CARPAN ile ucuzlatilir. Gerekcesi CONFIG'de.

_hs_under_w = int(round(UNDER_BUFFER_W * HAFTA_SONU_COVERAGE_CARPANI))
_hs_missing_w = int(round(MISSING_TO_REQUIRED_W * HAFTA_SONU_COVERAGE_CARPANI))
print(f"coverage ağırlığı -> hafta içi {UNDER_BUFFER_W}/{MISSING_TO_REQUIRED_W} | "

for ds in PLAN_GUNLER:

    _u_w = _hs_under_w if _hafta_sonu_mu else UNDER_BUFFER_W

    _kg_c = globals().get("kritik_gun_carpan", {}).get(ds, 1.0)   # 2026-10-21 kritik gün
    _u_w, _m_w = int(round(_u_w * _kg_c)), int(round(_m_w * _kg_c))
    for v in gun_vardiyalari.get(ds, []):

    print("hafta sonu 1. kademe terimi:", len(fazla_hafta_sonu_kademe1),

        # aydan sarkan hafta: plan öncesi (önceki ay) çalışılan günleri de say
        _hafta_araligi = {_pzt + _td(days=k) for k in range(7)}

# Kaynak: [HÜCRE] - TAKIM HAFTALIK BASE VARDİYA -> SARKAN_DEVIR_AGENT. Sarkan hafta dışı boş.
_dv_map = globals().get("SARKAN_DEVIR_AGENT", {})

for _dv_v in _dv_map.values():
    _dv_kisi[_dv_v] += 1
_dv_calisan = defaultdict(int)       # (gün, vardiya) -> devraldığı vardiyada çalışan
for (a, ds), v in cozum_atama.items():
    if cozum_calisma.get((a, ds), 0) == 1 and _dv_map.get(a) == v:
        _dv_calisan[(ds, v)] += 1

    for v in gun_vardiyalari.get(ds, []):
        _req = int(talep[(ds, v)])
        _asg = int(_atanan_sayac.get((ds, v), 0))
        _st = _saat.get((ds, v), ("", ""))
        _cov_rows.append({
            "tarih": d.strftime("%Y-%m-%d"),
            "gun": _GUN_ADI[d.weekday()],
            "hafta": _day_week_x.get(ds, ""),
            "vardiya": v,

            "bitis": str(_st[1]).strip()[:5],

            "talep": _req,
            "atanan": _asg,
            "fark": _asg - _req,
            # 2026-08-30: fark tek kolonda eksik ve fazlayı karıştırıyordu.
            # Ayrıştırılmış hâli COVERAGE_GUNLUK toplamlarıyla birebir tutar.
            "eksik": max(0, _req - _asg),
            "fazla": max(0, _asg - _req),
            "devralan_kisi": _dv_kisi.get(v, 0) if _dv_map and _day_week_x.get(ds) == _dv_wk else None,

            "devir_disi_atanan": (_asg - _dv_calisan.get((ds, v), 0)
                                  if _dv_map and _day_week_x.get(ds) == _dv_wk else None),
            "karsilanma_yuzde": round(100 * _asg / _req, 1) if _req else None,
            "buffer_alt": _cov_lower.get((ds, v)),
            "buffer_ust": _cov_upper.get((ds, v)),
            "buffer_altinda": (_asg < _cov_lower[(ds, v)]) if (ds, v) in _cov_lower else None,
            "buffer_ustunde": (_asg > _cov_upper[(ds, v)]) if (ds, v) in _cov_upper else None,

            _wc.cell(row=_j + 2, column=_fk_idx).fill = PatternFill("solid", fgColor=RENK["mesai"])

    # COVERAGE_GUNLUK: eksiği olan günleri boya
    _wg = writer.sheets["COVERAGE_GUNLUK"]
    _eg_idx = list(coverage_gunluk_df.columns).index("eksik_toplam") + 1
    _bk_idx = list(coverage_gunluk_df.columns).index("bos_kova") + 1
    for _j in range(len(coverage_gunluk_df)):
        if int(coverage_gunluk_df["eksik_toplam"].iloc[_j]) > 0:
            _wg.cell(row=_j + 2, column=_eg_idx).fill = PatternFill(
                "solid", fgColor=RENK["off_t_ihlal"])
        if int(coverage_gunluk_df["bos_kova"].iloc[_j]) > 0:
            _wg.cell(row=_j + 2, column=_bk_idx).fill = PatternFill(
                "solid", fgColor=RENK["off_t_ihlal"])

    # ---------------- KURAL_KONTROL renklendirme
    if "KURAL_KONTROL" in writer.sheets:
        ws_k = writer.sheets["KURAL_KONTROL"]
        ws_k.column_dimensions["A"].width = 55
        ws_k.column_dimensions["B"].width = 10
        ws_k.column_dimensions["C"].width = 10
        _kk = _kural_df.reset_index(drop=True)
        for _j in range(len(_kk)):
            _hard = str(_kk.loc[_j, "tur"]).strip().upper() == "HARD"
            _ihl = int(_kk.loc[_j, "ihlal"]) > 0
            if _hard and _ihl:
                for _cc in range(1, len(_kk.columns) + 1):
                    ws_k.cell(row=_j + 2, column=_cc).fill = PatternFill(
                        "solid", fgColor=RENK["hard_ihlal"]
                    )
