# %% [HÜCRE 6] — Aylık çalıştır + cache
YEAR = 2026
MONTH = 10
# Cache adı AYARLARI da içerir (2026-10-12): eskiden yalnız yıl/ay vardı →
# mod/ceza/gap değişince ESKİ plan sessizce yükleniyordu. Ayar değişince yeni
# dosya oluşur. Pipeline KODU değişirse bunu yakalamaz → o durumda dosyayı sil.
_hv = CONFIG.get('haftalik_vardiya', {}) or {}
_cz = CONFIG.get('ceza', {}) or {}
_gap = (CONFIG.get('solver', {}).get('gap_rel_by_queue', {}) or {}).get(
    'kitle', CONFIG.get('solver', {}).get('gap_rel'))

# ---- DEVRİK HAFTA (2026-10-14) ----
# Ayın 1'i Sal-Cum ise ilk haftanın hafta içi inhouse'u önceki ayın SON
# gününün Model-2 planıyla (df_onceki, HÜCRE 2) devam eder. Pzt/Cmt/Pzr → yok.
DEVRIK = devrik_dagilim(globals().get('df_onceki'), YEAR, MONTH, config=CONFIG,
                        queues=('kitle', 'kurumsal', 'gold'))

# PT vardiya SAYISI da ada girer (2026-10-14): 3 vardiya → _pt3, PT kapalı → _pt0.
# Vardiya eklenince/çıkınca yeni pickle oluşur. Saatlerin kendisi CONFIG sayfasında.
_ptc = CONFIG.get('part_time', {}) or {}
_pt_etiket = len(_ptc.get('shifts', [])) if _ptc.get('enabled', False) else 0
CACHE_FILE = (f"monthly_unified_v2_{YEAR}_{MONTH:02d}"
              f"_{_hv.get('inhouse_mod', 'sabit')}"
              f"_sur{int(bool(_hv.get('inhouse_surplus', True)))}"
              f"_esit{int(bool(_hv.get('outsource_gunluk_esit', True)))}"
              f"_rrg{int(bool((CONFIG.get('rr_gun_dengesi') or {}).get('enabled', False)))}"
              f"_k{int(_cz.get('kapsama_hafta_ici', 0))}-{int(_cz.get('kapsama_hafta_sonu', 0))}"
              f"_gap{_gap}"
              f"_pt{_pt_etiket}"
              f"{'_eg1' if _hv.get('ertesi_gun_bakis', False) else ''}"   # Pazar→Pzt bakışı
              f"_sc{'kap' if CONFIG.get('slot_cap_olcu', 'erlang') == 'kap_rr' else 'erl'}"   # slot_cap ölçüsü
              f"{'_ia1' if (CONFIG.get('iki_asama') or {}).get('enabled') else ''}"   # 2. aşama giriş dağıtımı
              f"{'_dv1' if DEVRIK else ''}.pkl")     # devrik hafta uygulandıysa ayrı pickle
print(f"cache dosyası: {CACHE_FILE}")

# ---- ÖNCEKİ AYIN SON GÜNÜ — gece sarkanı (2026-10-14) ----
# Ayın 1'inin gece slotlarında sahada olan kişi ÖNCEKİ ayın son akşamının
# vardiyasıdır (ör. 30 Eylül 17:00-01:00). Verilmezse 1'i gece EKSİK görünür.
#   inhouse / PT : df_onceki'den OTOMATİK (HÜCRE 2)
#   outsource    : df_onceki'de yok → aşağıya ELLE. Inhouse YAZMA (çift sayılır).
# SADECE gece yarısını aşan vardiyalar; satır: (baş, bitiş, 'outsource', kişi)
ONCEKI_AY_GECE = {
    'kitle':    [],      # örn: ('19:00', '03:30', 'outsource', 14),
    'kurumsal': [],
    'gold':     [],
}

_onc_son = (dt.date(YEAR, MONTH, 1) - dt.timedelta(days=1)).isoformat()
_gece = onceki_gece_df(globals().get('df_onceki'), YEAR, MONTH, config=CONFIG,
                       queues=('kitle', 'kurumsal', 'gold'))
for _q, _l in ONCEKI_AY_GECE.items():
    _gece.setdefault(_q, []).extend(_l)
ONCEKI_AY_SON_GUN = gece_sarkan_gun(_gece) if any(_gece.values()) else None
if ONCEKI_AY_SON_GUN:
    for _q, _l in _gece.items():
        if _l:
            print(f"▶ {_onc_son} gece sarkanı {_q}: "
                  + ', '.join(f"{a}-{b} {c} {k}" for a, b, c, k in _l))
else:
    print(f"⚠ {_onc_son} gece sarkanı YOK (df_onceki'de gece yarısını aşan vardiya yok, "
          f"ONCEKI_AY_GECE boş) → ayın 1'i gece slotlarında sarkan kadro sayılmaz")

if os.path.exists(CACHE_FILE):
    with open(CACHE_FILE, 'rb') as f:
        results = pickle.load(f)
    print(f"Cache'ten yüklendi: {len(results)} gün ({CACHE_FILE})")
else:
    results = run_month_unified(
        YEAR, MONTH, df_forecast, df_shifts_dict, config=CONFIG,
        queues=('kitle', 'kurumsal', 'gold'),
        mip1=True,                   # False → tek çözüm, süre yarıya iner
        # sarkan_devret → CONFIG['haftalik_vardiya']['sarkan_devret'] (2026-10-14 taşındı)
        onceki_ay_son_gun=ONCEKI_AY_SON_GUN,
        verbose=True,
        devrik=DEVRIK,
    )
    with open(CACHE_FILE, 'wb') as f:
        pickle.dump(results, f)
    print(f"\nKaydedildi: {CACHE_FILE}")
