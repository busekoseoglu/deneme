       ROUND(SUM(syh_yd_pos_tutar_12m)) pos_tutar_toplam
FROM   BSEN_SYH_PILOT_EKIM_2026

SELECT 'eski', COUNT(*), SUM(syh_snap_ay_flg), ROUND(SUM(syh_yd_pos_tutar_12m))
FROM   BSEN_SYH_GECMIS_DEGISKENLER
WHERE  data_date = DATE '2026-09-30';
