# Hafta içi her gün, bir ekipte çalışmayan agent sayısı EKIP_GUNLUK_MODEL_OFF_MAX'ı geçmesin.
# Sayılan: model off + off_t. Sayılmayan: EKIP_OFF_HARIC_TIPLER (izin, off_t2),
# hafta içine denk gelen arife / resmi tatil, agentın o gün hiç vardiyası olmayan gün.
# Aşan her kişi EKIP_GUNLUK_OFF_ASIM_W ceza öder ([HÜCRE] - OBJECTIVE).
# 2026-11-10: KADEMELİ - ilk aşan kişi W, 2. ve sonrakiler EKIP_GUNLUK_OFF_ASIM_W2 (tek güne 6 off yığılıyordu).
_eksik_cfg = [k for k in ("ENABLE_EKIP_GUNLUK_OFF_TAVANI", "EKIP_GUNLUK_MODEL_OFF_MAX",
                          "EKIP_GUNLUK_OFF_ASIM_W", "EKIP_OFF_HARIC_TIPLER",
                          "EKIP_GUNLUK_OFF_ASIM_W2") if k not in globals()]
if _eksik_cfg:
    raise SystemExit(f"CONFIG'de tanımlı olmalı, eksik: {_eksik_cfg} -> [CONFIG] hücresini çalıştır.")

ekip_off_asim = {}      # (takim, ds) -> IntVar
ekip_off_asim2 = {}     # (takim, ds) -> IntVar: ilk aşan kişiden sonrakiler
if ENABLE_EKIP_GUNLUK_OFF_TAVANI:
    _haric_e = {str(t).strip().lower() for t in EKIP_OFF_HARIC_TIPLER}
    _cd_e = set(globals().get("calisma_disi_date_set", set()))
    for ds in PLAN_GUNLER:
        _d = pd.to_datetime(ds).date()
        if _d.weekday() >= 5 or _d in _cd_e:
            continue
        _ekip_w = {}
        for a_raw in AGENTS:
            a = str(a_raw).strip()
            if str(tip_map.get(a, {}).get(_d, "")).strip().lower() in _haric_e:
                continue
            if not any((a, ds, v) in x for v in gun_vardiyalari.get(ds, [])):
                continue
            _ekip_w.setdefault(agent_team.get(a), []).append(work[(a, ds)])
        for t, _ws in _ekip_w.items():
            if len(_ws) <= EKIP_GUNLUK_MODEL_OFF_MAX:
                continue
            _v = model.NewIntVar(0, len(_ws), f"ekip_off_asim_{t}_{ds}")
            model.Add(len(_ws) - sum(_ws) - EKIP_GUNLUK_MODEL_OFF_MAX <= _v)
            ekip_off_asim[(t, ds)] = _v
            _v2 = model.NewIntVar(0, len(_ws), f"ekip_off_asim2_{t}_{ds}")
            model.Add(_v - 1 <= _v2)
            ekip_off_asim2[(t, ds)] = _v2

print(f"ekip günlük off tavanı: {'AÇIK' if ENABLE_EKIP_GUNLUK_OFF_TAVANI else 'KAPALI'} | "
      f"max {EKIP_GUNLUK_MODEL_OFF_MAX} | W {EKIP_GUNLUK_OFF_ASIM_W} / 2.+ kişi {EKIP_GUNLUK_OFF_ASIM_W2} | "
      f"kurulan ekip-gün: {len(ekip_off_asim)}")
