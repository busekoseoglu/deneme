# mesaiye_kalamaz_flg = 1 olan agent HİÇBİR tipte mesai yapamaz.
# Mesai tanımı AYLIK: aylık iş günü kadar çalışır, bir gün bile fazlası mesaidir.
#
#   1) aylik_mesai_gun[a] == 0

#         calisma_vars resmi tatil günlerini HİÇ saymıyor. O gün çalışmak aylık
#         toplama girmediği için (1) numaralı kısıt onu göremez ve agent
#         21 gün + resmi tatil = 22 gün çalışmış olur. Bu kısıt o açığı kapatır.
#
# NOT: hamile / süt izni agentları da tatil_kisitli_agents içinde ama onlar için
#      resmi tatil kuralı SOFT kalmaya devam eder (resmi_tatil_kisitli_ihlal).
#      Burada SADECE mesaiye_kalamaz_flg = 1 olanlar hard yapılır.

# Bu hücre [HAZIRLIK / KISIT] - AYLIK ÇALIŞMA DENGESİ hücresinden SONRA çalışmalıdır

elif "RESMI_TATIL_GUNLERI" in globals():

    # --- 2) resmi tatilde çalışma yok ---
    _rt_kapatilan = 0

        if pd.to_datetime(ds).date() not in _mk_resmi_tatil_set:

            mk_resmi_tatil_kisiti += 1
            _rt_kapatilan += 1

        "agent_user_code": a,
        "aylik_mesai_sifir_kuruldu": _mesai_kuruldu,
        "kapatilan_resmi_tatil_gunu": _rt_kapatilan,
    })

def _vardiya_metni(ds, v):
    """Vardiya kodunu okunur saat aralığına çevirir."""
    if v is None:
        return ""
    st = _saat.get((ds, v))
    if st:
        return f"{str(st[0]).strip()[:5]}-{str(st[1]).strip()[:5]}"
    return str(v)

_hafta_gunleri = defaultdict(list)
for ds in PLAN_GUNLER:
    d = pd.to_datetime(ds).date()
    _pzt = d - _td(days=d.weekday())
    _hafta_gunleri[_pzt].append(ds)
for _k in _hafta_gunleri:
    _hafta_gunleri[_k] = sorted(_hafta_gunleri[_k], key=lambda s: pd.to_datetime(s))

mesai_gunleri_map = {}   # agent -> {ds}
norm_gunleri_map = {}    # agent -> {ds}
hafta_calisma_debug_rows = []

for a_raw in AGENTS:
    a = str(a_raw).strip()
    _gecmis_a = _gecmis_gunleri_x.get(a, set())
    _mesai_a, _norm_a = set(), set()

    for _pzt, _wk_gunler in _hafta_gunleri.items():
        _calisilan = [ds for ds in _wk_gunler if cozum_calisma.get((a, ds), 0) == 1]

        # aydan sarkan hafta: plan öncesi (Temmuz) çalışılan günleri de say
        _hafta_araligi = {_pzt + _td(days=k) for k in range(7)}
        _temmuz_ek = len({d for d in _gecmis_a if d in _hafta_araligi and d < _PLAN_BAS})

        _hafta_calisma = len(_calisilan) + _temmuz_ek
        _fazla = max(0, _hafta_calisma - _NORMAL_WORK_DAYS)

        _hs_calisilan = [ds for ds in _calisilan if pd.to_datetime(ds).weekday() in (5, 6)]
        if _fazla > 0 and _hs_calisilan:
            _mesai_bu_hafta = set(_hs_calisilan[-_fazla:])
        else:
            _mesai_bu_hafta = set()
        _norm_bu_hafta = set(_hs_calisilan) - _mesai_bu_hafta

        _mesai_a |= _mesai_bu_hafta
        _norm_a |= _norm_bu_hafta

        hafta_calisma_debug_rows.append({
            "agent_user_code": a,
            "hafta_pzt": str(_pzt),
            "plan_ici_calisma": len(_calisilan),
            "temmuzdan_gelen_calisma": _temmuz_ek,
            "hafta_toplam_calisma": _hafta_calisma,
            "kota": _NORMAL_WORK_DAYS,
            "fazla": _fazla,

            "norm": len(_norm_bu_hafta),
            "mesai": len(_mesai_bu_hafta),
