# =====================================================================
# 1) AYARLAR VE VERİ
#    Gerekli değişkenler notebook'ta hazır olmalı:
#    X_train, y_train, X_val, y_val  (syh_ değişkenleri eklenmiş, NULL'lar doldurulmuş)
#    X_eylul, y_eylul, X_ekim, y_ekim  (OOT - sadece en sonda değerlendirme için)
# =====================================================================
import lightgbm as lgb
import optuna
import numpy as np
import pandas as pd
import joblib
from sklearn.metrics import (average_precision_score, precision_score,
                             recall_score, fbeta_score, confusion_matrix)

DROP = ["syh_ay_say_24m"]
feats = [c for c in X_train.columns if c not in DROP]
print("Değişken sayısı:", len(feats))

X_tr, y_tr = X_train[feats], y_train
X_v, y_v = X_val[feats], y_val

K = int(len(y_v) * 0.016)   # val'de, OOT'deki ~0.6 eşiğine denk hacim (bilgi amaçlı)

FIXED_PARAMS = {
    "objective": "binary",
    "metric": "average_precision",   # early stopping bunu izler
    "boosting_type": "gbdt",
    "subsample_freq": 1,
    "n_estimators": 5000,            # üst sınır; early stopping keser
    "random_state": 42,
    "verbosity": -1,
    "n_jobs": -1,
}


def precision_at_k(y, p, k):
    return np.asarray(y)[np.argsort(-p)[:k]].mean()


# =====================================================================
# 2) OPTUNA
#    Hedef: val PR-AUC (eşikten bağımsız). Eşik en sonda OOT'de seçilir.
# =====================================================================
def objective(trial):
    max_depth = trial.suggest_int("max_depth", 3, 8)
    params = {
        "max_depth": max_depth,
        "num_leaves": trial.suggest_int("num_leaves", 8, min(127, 2 ** max_depth)),
        "learning_rate": trial.suggest_float("learning_rate", 0.02, 0.1, log=True),
        "min_child_samples": trial.suggest_int("min_child_samples", 500, 50000, log=True),
        "subsample": trial.suggest_float("subsample", 0.7, 1.0),
        "colsample_bytree": trial.suggest_float("colsample_bytree", 0.6, 1.0),
        "reg_alpha": trial.suggest_float("reg_alpha", 1e-3, 50, log=True),
        "reg_lambda": trial.suggest_float("reg_lambda", 1e-3, 50, log=True),
        "scale_pos_weight": trial.suggest_float("scale_pos_weight", 1.0, 3.0),
    }
    model = lgb.LGBMClassifier(**FIXED_PARAMS, **params)
    model.fit(X_tr, y_tr,
              eval_set=[(X_v, y_v)],
              callbacks=[lgb.early_stopping(100, verbose=False)])

    p = model.predict_proba(X_v)[:, 1]
    ap = average_precision_score(y_v, p)
    pk = precision_at_k(y_v, p, K)
    trial.set_user_attr("best_iter", int(model.best_iteration_))
    trial.set_user_attr("prec_at_k", float(pk))
    print(f"Trial {trial.number} | PR-AUC: {ap:.4f} | P@K: {pk:.4f} "
          f"| iter: {model.best_iteration_}")
    return ap


study = optuna.create_study(direction="maximize",
                            sampler=optuna.samplers.TPESampler(seed=42))
study.optimize(objective, n_trials=20)
# Devam etmek istersen: study.optimize(objective, n_trials=10)

bt = study.best_trial
print("\nEn iyi trial:", bt.number)
print("PR-AUC:", round(bt.value, 4), "| P@K:", round(bt.user_attrs["prec_at_k"], 4),
      "| iter:", bt.user_attrs["best_iter"])
print("Parametreler:", bt.params)

# Tüm trial'ları tablo olarak gör
trials_df = study.trials_dataframe().sort_values("value", ascending=False)
print(trials_df.head(10).to_string(index=False))


# =====================================================================
# 3) FİNAL MODEL: train + val ile
# =====================================================================
best_params = {**FIXED_PARAMS, **bt.params}
best_params["n_estimators"] = int(bt.user_attrs["best_iter"] * 1.1)  # veri büyüdü, ~%10 fazla ağaç
best_params.pop("metric", None)

final_model = lgb.LGBMClassifier(**best_params)
final_model.fit(pd.concat([X_train[feats], X_val[feats]]),
                pd.concat([y_train, y_val]))
print("Final model eğitildi. Ağaç sayısı:", best_params["n_estimators"])


# =====================================================================
# 4) OOT DEĞERLENDİRME (Eylül ve Ekim) - eşik tablosu
# =====================================================================
def esik_tablosu(model, X, y, esikler=np.arange(0.3, 0.85, 0.05)):
    p = model.predict_proba(X[feats])[:, 1]
    y = np.asarray(y)
    rows = []
    for t in esikler:
        pred = (p >= t).astype(int)
        tn, fp, fn, tp = confusion_matrix(y, pred, labels=[0, 1]).ravel()
        prec = tp / (tp + fp) if (tp + fp) else 0
        rows.append({
            "esik": round(float(t), 2),
            "precision": prec,
            "recall": tp / y.sum(),
            "f05": fbeta_score(y, pred, beta=0.5, zero_division=0),
            "tp": tp, "fp": fp,
            "secilen": tp + fp,
            "lift": prec / y.mean() if prec else 0,
        })
    return pd.DataFrame(rows)


print("\n=== EYLÜL OOT (Ağustos verisi -> Eylül seyahati) ===")
print(esik_tablosu(final_model, X_eylul, y_eylul).round(4).to_string(index=False))

print("\n=== EKİM OOT (Eylül verisi -> Ekim seyahati) ===")
print(esik_tablosu(final_model, X_ekim, y_ekim).round(4).to_string(index=False))
